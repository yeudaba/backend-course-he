הפלט הוא מחרוזת או null — זה חלק מההגדרה, ומי שקורא לפונקציה צריך לדעת.

✅ פתרון
  1. קלט: מחרוזת (מה שהוקלד).
  2. פלט: מחרוזת בפורמט 050-1234567, או null.
  3. פעלים ← פונקציות: digitsOnly(text) — מנקה; isValidMobile(digits) — בודקת 10 ספרות ו־05; formatMobile(input) — מחברת: מנקה, בודקת, ומעצבת.
  4. דוגמאות: " 054 123-4567 " ← "054-1234567"; "0501234567" ← "050-1234567"; "03-1234567" ← null.
function digitsOnly(text) {
  let result = "";
  for (const ch of text) {
    if (ch >= "0" && ch <= "9") {
      result += ch;
    }
  }
  return result;
}
const isValidMobile = (digits) => digits.length === 10 && digits.startsWith("05");
function formatMobile(input) {
  const digits = digitsOnly(input);
  if (!isValidMobile(digits)) {
    return null;
  }
  return `${digits.slice(0, 3)}-${digits.slice(3)}`;
}
console.log(formatMobile(" 054 123-4567 "), formatMobile("03-1234567"));
054-1234567 null

function clampScore(input) {

P05-E18 · ציון לסיסמה 🤖 💻 כתיבת קוד 🤖 רמה 3

💡 רמז 2

אפשר להשתמש ב־countMatching מ־P05-E16 בצורה: hasDigit = (t) => countMatching(t, isDigit) > 0, או לולאה עם return true מוקדם.

✅ פתרון
// P05-E18: ציון חוזק סיסמה
function hasMatching(text, predicate) {
  for (const ch of text) {
    if (predicate(ch)) {
      return true;
    }
  }
  return false;
}

const hasDigit = (text) => hasMatching(text, (ch) => "0123456789".includes(ch));
const hasUpper = (text) => hasMatching(text, (ch) => ch !== ch.toLowerCase());
const hasSpecial = (text) => hasMatching(text, (ch) => "!@#$%".includes(ch));

function passwordScore(password) {
  let score = 0;
  if (password.length >= 8) score++;
  if (hasDigit(password)) score++;
  if (hasUpper(password)) score++;
  if (hasSpecial(password)) score++;
  return score;
}

function passwordLabel(password) {
  const score = passwordScore(password);
  if (score <= 1) {
    return "חלשה";
  }
  if (score <= 3) {
    return "בינונית";
  }
  return "חזקה";
}

console.log(passwordScore("abc"), passwordLabel("Secret123"), passwordLabel("Secret123!"));

module.exports = { hasDigit, hasUpper, hasSpecial, passwordScore, passwordLabel };
0 בינונית חזקה

(if (cond) score++; בלי סוגריים מסולסלים — מותר לשורה אחת, ומקובל כשהגוף קצר מאוד.) hasMatching עוצרת ב־return true הראשון — כמו break, אבל מתוך פונקציה.

⚠️ טעויות נפוצות

P05-E74 · פס התקדמות 🤖 💻 כתיבת קוד 🤖 רמה 3

💡 רמז 1

clamp — Math.min(max, Math.max(min, n)). percent משתמשת ב־clamp. progressBar משתמשת ב־percent.

💡 רמז 2

סולמיות: Math.floor((p * width) / 100). מקפים: השאר. "".padEnd(filled, "#").padEnd(width, "-") — או לולאה.

✅ פתרון

function clamp(n, min, max) {
  return Math.min(max, Math.max(min, n));
}

function percent(part, total) {
  if (total === 0) {
    return 0;
  }
  return clamp(Math.round((part / total) * 100), 0, 100);
}

function progressBar(part, total, width = 10) {
  const p = percent(part, total);
  const filled = Math.floor((p * width) / 100);
  const bar = "".padEnd(filled, "#").padEnd(width, "-");
  return `[${bar}] ${p}%`;
}

console.log(progressBar(4, 10), progressBar(1, 3), progressBar(12, 10));

module.exports = { clamp, percent, progressBar };
[####------] 40% [###-------] 33% [##########] 100%

שלוש פונקציות, כל אחת בונה על הקודמת. כשהבודק נכשל ב־progressBar, הוא בודק גם את clamp ו־percent בנפרד — כך יודעים איפה הבעיה.

⚠️ טעויות נפוצות
  • Math.round לסולמיות — 33% ברוחב 10 נותן 3 (נכון), אבל 95% נותן 10 סולמיות למרות שלא הגענו ל־100.

function normalize(text) {
  let result = "";
  for (const ch of text.toLowerCase()) {
    if ((ch >= "a" && ch <= "z") || (ch >= "0" && ch <= "9")) {
      result += ch;
    }
  }
  return result;
}

function reverse(text) {
  let result = "";
  for (const ch of text) {
    result = ch + result;
  }
  return result;
}

function isPalindrome(text) {
  const clean = normalize(text);
  return clean === reverse(clean);
}

console.log(normalize("Was it a car?"), isPalindrome("Was it a car or a cat I saw"));

module.exports = { normalize, reverse, isPalindrome };

function digitsOnly(text) {
  let result = "";
  for (const ch of text) {
    if (ch >= "0" && ch <= "9") {
      result += ch;
    }
  }
  return result;
}

function maskAllButLast(digits, visible = 4) {
  return digits.slice(-visible).padStart(digits.length, "*");
}

function groupBy4(text) {
  let result = "";
  for (let i = 0; i < text.length; i++) {
    if (i > 0 && i % 4 === 0) {
      result += " ";
    }
    result += text[i];
  }
  return result;
}

function maskCard(input) {
  const digits = digitsOnly(input);
  if (digits.length !== 16) {
    return null;
  }
  return groupBy4(maskAllButLast(digits));
}

function getUnitPrice(code) {
  switch (code.trim().toUpperCase()) {
    case "KB-101":
      return 199;
    case "KB-102":
      return 249;
    case "MS-201":
      return 89;
    default:
      return null;
  }
}

function validateQuantity(input, max = 5) {
  const quantity = Number(input);
  if (input.trim() === "" || Number.isNaN(quantity)) {
    return "כמות חייבת להיות מספר";
  }
  if (!Number.isInteger(quantity) || quantity < 1 || quantity > max) {
    return `כמות חייבת להיות מספר שלם בין 1 ל־${max}`;
  }
  return null;
}

function couponDiscount(coupon, subtotal) {
  const code = coupon.trim().toUpperCase();
  if (code === "") {
    return 0;
  }
  if (code === "VIP20") {
    return subtotal > 500 ? subtotal * 0.2 : null;
  }
  if (code === "WELCOME") {
    return Math.min(30, subtotal);
  }
  return null;
}

function checkout(code, quantityInput, coupon, isLoggedIn) {
  if (!isLoggedIn) {
    return "שגיאה: יש להתחבר כדי להזמין";
  }
  const unitPrice = getUnitPrice(code);
  if (unitPrice === null) {
    return "שגיאה: מוצר לא קיים";
  }
  const qtyError = validateQuantity(quantityInput);
  if (qtyError !== null) {
    return `שגיאה: ${qtyError}`;
  }
  const subtotal = unitPrice * Number(quantityInput);
  const discount = couponDiscount(coupon, subtotal);
  if (discount === null) {
    return "שגיאה: הקופון לא קיים או לא תקף להזמנה הזו";
  }
  return `לתשלום: ${(subtotal - discount).toFixed(2)}`;
}

// P05-E25: עיצוב מספר עם פסיקים
function addCommas(digits) {
  let result = "";
  let count = 0;
  for (let i = digits.length - 1; i >= 0; i--) {
    if (count > 0 && count % 3 === 0) {
      result = "," + result;
    }
    result = digits[i] + result;
    count++;
  }
  return result;
}

function formatNumber(n, decimals = 2) {
  const sign = n < 0 ? "-" : "";
  const fixed = Math.abs(n).toFixed(decimals);
  const dot = fixed.indexOf(".");
  const intPart = dot === -1 ? fixed : fixed.slice(0, dot);
  const decPart = dot === -1 ? "" : fixed.slice(dot);
  return sign + addCommas(intPart) + decPart;
}

function splitBill(total, people, tipPercent = 10) {
  if (!(total > 0)) {
    return null;
  }
  if (!Number.isInteger(people) || people < 1) {
    return null;
  }
  const withTip = total * (1 + tipPercent / 100);
  return Math.ceil(withTip / people);
}

function billSummary(total, people, tipPercent = 10) {
  const each = splitBill(total, people, tipPercent);
  if (each === null) {
    return "קלט לא תקין";
  }
  return `${people} סועדים × ${each} ₪ = ${people * each} ₪ (כולל טיפ ${tipPercent}%)`;
}
