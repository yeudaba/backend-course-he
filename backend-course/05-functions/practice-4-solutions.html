P05-E18 · ציון לסיסמה 🤖 💻 כתיבת קוד 🤖 רמה 3

💡 רמז 1

ספרה: "0123456789".includes(ch). אות גדולה: ch !== ch.toLowerCase() (עובד רק לאותיות — בעברית ובספרות אין גדולה וקטנה).

💡 רמז 2

אפשר להשתמש ב־countMatching מ־P05-E16 בצורה: hasDigit = (t) => countMatching(t, isDigit) > 0, או לולאה עם return true מוקדם.

✅ פתרון
// P05-E18: ציון חוזק סיסמה
function hasMatching(text, predicate) {
  for (const ch of text) {
    if (predicate(ch)) {
      return true;
    }
  }
  return false;
}

const hasDigit = (text) => hasMatching(text, (ch) => "0123456789".includes(ch));
const hasUpper = (text) => hasMatching(text, (ch) => ch !== ch.toLowerCase());
const hasSpecial = (text) => hasMatching(text, (ch) => "!@#$%".includes(ch));

function passwordScore(password) {
  let score = 0;
  if (password.length >= 8) score++;
  if (hasDigit(password)) score++;
  if (hasUpper(password)) score++;
  if (hasSpecial(password)) score++;
  return score;
}

function passwordLabel(password) {
  const score = passwordScore(password);
  if (score <= 1) {
    return "חלשה";
  }
  if (score <= 3) {
    return "בינונית";
  }
  return "חזקה";
}

console.log(passwordScore("abc"), passwordLabel("Secret123"), passwordLabel("Secret123!"));

module.exports = { hasDigit, hasUpper, hasSpecial, passwordScore, passwordLabel };
0 בינונית חזקה

(if (cond) score++; בלי סוגריים מסולסלים — מותר לשורה אחת, ומקובל כשהגוף קצר מאוד.) hasMatching עוצרת ב־return true הראשון — כמו break, אבל מתוך פונקציה.

⚠️ טעויות נפוצות
  • hasUpper עם ch === ch.toUpperCase() — ספרות וסימנים "שווים לעצמם באותיות גדולות", ויסומנו בטעות.

// P05-E19: קבלה עם פונקציות
const AMOUNT_WIDTH = 8;

function formatLine(label, amount, width = 28) {
  return label.toUpperCase().padEnd(width - AMOUNT_WIDTH, ".") + amount.toFixed(2).padStart(AMOUNT_WIDTH);
}

function calcTotals(subtotal, couponPercent, vatRate) {
  const afterDiscount = subtotal * (1 - couponPercent / 100);
  const total = afterDiscount * (1 + vatRate);
  return Math.round(total * 100) / 100;
}

console.log(formatLine("mouse", 89.9));
console.log(formatLine("total", calcTotals(358.4, 10, 0.18)));

module.exports = { formatLine, calcTotals };

⚠️ טעויות נפוצות

P05-E23 · בדיקת גיל — שוב, כפונקציה 🤖 💻 כתיבת קוד 🤖 רמה 4

💡 רמז 1

סדר ה־guard clauses. ו: מה ברירת המחדל של minAge בפועל, כשקוראים לפונקציה בלי ארגומנט שני?

💡 רמז 2

בדוק את הקלט "0" — הוא חוקי (תינוק). ואת " ".

✅ פתרון
// P05-E23: קבוצת גיל — guard clauses בסדר הנכון
function ageGroup(input, minAge = 0) {
  if (input.trim() === "") {
    return "שגיאה: חובה להזין גיל";
  }
  const age = Number(input);
  if (Number.isNaN(age)) {
    return "שגיאה: גיל חייב להיות מספר";
  }
  if (age < minAge || age > 120) {
    return "שגיאה: גיל לא הגיוני";
  }
  if (age < 18) {
    return "קבוצה: קטין";
  }
  if (age < 65) {
    return "קבוצה: מבוגר";
  }
  return "קבוצה: גמלאי";
}

console.log(ageGroup("0"), "|", ageGroup("   "), "|", ageGroup("70"));

module.exports = { ageGroup };
קבוצה: קטין | שגיאה: חובה להזין גיל | קבוצה: גמלאי

(1) בדיקת "ריק" נעשתה עם !age — אבל Number("0") הוא 0, falsy, ולכן תינוק בן 0 נדחה כ"ריק". (2) ברירת המחדל minAge = 1 — אז גם "0" נכשל בטווח. כאן שני באגים מסתירים זה את זה: מתקנים אחד, ועדיין "0" נכשל — וזה הרגע לא לוותר ולבדוק שוב.

⚠️ טעויות נפוצות
  • להסתפק בבדיקה ידנית של 2–3 קלטים. בשביל זה יש בודק עם 10.

✅ פתרון
// P05-E24: תהליך הזמנה מפורק לפונקציות
function getUnitPrice(code) {
  switch (code.trim().toUpperCase()) {
    case "KB-101":
      return 199;
    case "KB-102":
      return 249;
    case "MS-201":
      return 89;
    default:
      return null;
  }
}

function validateQuantity(input, max = 5) {
  const quantity = Number(input);
  if (input.trim() === "" || Number.isNaN(quantity)) {
    return "כמות חייבת להיות מספר";
  }
  if (!Number.isInteger(quantity) || quantity < 1 || quantity > max) {
    return `כמות חייבת להיות מספר שלם בין 1 ל־${max}`;
  }
  return null;
}

function couponDiscount(coupon, subtotal) {
  const code = coupon.trim().toUpperCase();
  if (code === "") {
    return 0;
  }
  if (code === "VIP20") {
    return subtotal > 500 ? subtotal * 0.2 : null;
  }
  if (code === "WELCOME") {
    return Math.min(30, subtotal);
  }
  return null;
}

function checkout(code, quantityInput, coupon, isLoggedIn) {
  if (!isLoggedIn) {
    return "שגיאה: יש להתחבר כדי להזמין";
  }
  const unitPrice = getUnitPrice(code);
  if (unitPrice === null) {
    return "שגיאה: מוצר לא קיים";
  }
  const qtyError = validateQuantity(quantityInput);
  if (qtyError !== null) {
    return `שגיאה: ${qtyError}`;
  }
  const subtotal = unitPrice * Number(quantityInput);
  const discount = couponDiscount(coupon, subtotal);
  if (discount === null) {
    return "שגיאה: הקופון לא קיים או לא תקף להזמנה הזו";
  }
  return `לתשלום: ${(subtotal - discount).toFixed(2)}`;
}

console.log(checkout(" kb-102 ", "3", "VIP20", true));
console.log(checkout("KB-101", "2", "VIP20", true));

module.exports = { getUnitPrice, validateQuantity, couponDiscount, checkout };
לתשלום: 597.60
שגיאה: הקופון לא קיים או לא תקף להזמנה הזו

השווה ל־P03-E24: אותה התנהגות, אבל כל חלק נבדק לבד, ו־checkout נקרא כמו רשימת צעדים. זה בדיוק המבנה של "service" בשרת (פרק 20).

⚠️ טעויות נפוצות
  • if (!discount) — הנחה של 0 (בלי קופון) תיחשב כשגיאה.

P05-E25 · אתגר: פסיקים במספרים 🤖 💻 כתיבת קוד 🤖 רמה 5

💡 רמז 1

n.toFixed(decimals) ← פצל לחלק שלם ולחלק עשרוני לפי indexOf("."). סימן מינוס — טפל בו בנפרד.

💡 רמז 2

הוספת פסיקים: עבור על הספרות מהסוף (לולאה מ־length - 1 ל־0) וספור; כל 3 ספרות (חוץ מהראשונה בצד שמאל) הוסף פסיק.

✅ פתרון
// P05-E25: עיצוב מספר עם פסיקים
function addCommas(digits) {
  let result = "";
  let count = 0;
  for (let i = digits.length - 1; i >= 0; i--) {
    if (count > 0 && count % 3 === 0) {
      result = "," + result;
    }
    result = digits[i] + result;
    count++;
  }
  return result;
}

function formatNumber(n, decimals = 2) {
  const sign = n < 0 ? "-" : "";
  const fixed = Math.abs(n).toFixed(decimals);
  const dot = fixed.indexOf(".");
  const intPart = dot === -1 ? fixed : fixed.slice(0, dot);
  const decPart = dot === -1 ? "" : fixed.slice(dot);
  return sign + addCommas(intPart) + decPart;
}

console.log(formatNumber(1234567.891));
console.log(formatNumber(-1234.5, 1));
console.log(formatNumber(1000000, 0));

module.exports = { formatNumber, addCommas };

בעולם האמיתי: n.toLocaleString("he-IL") עושה את זה (ויודע על מוסכמות של כל מדינה). אבל זה תרגיל מצוין בפירוק, בלולאה הפוכה ובמקרי קצה — ובראיונות מבקשים בדיוק "בלי פונקציות מובנות".

⚠️ טעויות נפוצות
  • להפעיל addCommas על המספר כולל המינוס — יוצא -,123.
