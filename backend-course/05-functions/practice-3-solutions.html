💡 רמז 2

C: יש const hour בהמשך הקובץ... וקריאה לפני ההגדרה שלו. מה קורה עם typeof?

✅ פתרון

A: קפה הפינה נפתח ב־8. B: קפה הפינה. C: שגיאה — ReferenceError: Cannot access 'hour' before initialization, כי hour החיצוני מוגדר בהמשך הקובץ (אפילו typeof לא מציל מזה). התוכנית נעצרת כאן, ו־D ו־E לא רצים.

⚠️ טעויות נפוצות
  • לחשוב ש־C ידפיס undefined — זה נכון לשם שלא קיים בכלל (D), אבל לא לשם שמוגדר מאוחר יותר עם const.

P05-E12 · הפעל פעמיים 🤖 💻 כתיבת קוד 🤖 רמה 2

💡 רמז 1

fn הוא פונקציה — קוראים לה כמו לכל פונקציה: fn(value).

💡 רמז 2

let result = value; for (...) { result = fn(result); } return result;

✅ פתרון
// P05-E12: פונקציה שמקבלת פונקציה
function applyTwice(fn, value) {
  return fn(fn(value));
}

function applyN(fn, value, times) {
  let result = value;
  for (let i = 0; i < times; i++) {
    result = fn(result);
  }
  return result;
}

console.log(applyTwice((x) => x + 3, 10));
console.log(applyN((s) => s + "!", "היי", 3));

module.exports = { applyTwice, applyN };
16
היי!!!
⚠️ טעויות נפוצות
  • fn(value) + fn(value) — זה מפעיל פעמיים על אותו ערך ומחבר; לא "על התוצאה".

P05-E16 · ספירת תווים 🤖 💻 כתיבת קוד 🤖 רמה 3

💡 רמז 1

countMatching היא לולאת for...of עם if (predicate(ch)) count++.

💡 רמז 2

countVowels = (text) => countMatching(text.toLowerCase(), (ch) => "aeiou".includes(ch));

✅ פתרון
// P05-E16: ספירת תווים
function countMatching(text, predicate) {
  let count = 0;
  for (const ch of text) {
    if (predicate(ch)) {
      count++;
    }
  }
  return count;
}

function countChar(text, ch) {
  return countMatching(text, (c) => c === ch);
}

function countVowels(text) {
  return countMatching(text.toLowerCase(), (ch) => "aeiou".includes(ch));
}

console.log(countChar("banana", "a"), countVowels("Backend Developers Love Coffee"));

module.exports = { countChar, countMatching, countVowels };
3 11

גם countChar השתמשה ב־countMatching — לולאה אחת בכל הקובץ. זה הכוח של callbacks: את איך עוברים כותבים פעם אחת, ואת מה בודקים מעבירים מבחוץ.

⚠️ טעויות נפוצות
  • "aeiou".includes(ch) כש־ch הוא "" — מחזיר true! (כאן זה לא קורה, כי for...of לא מייצר תו ריק.)

// P05-E17: slug וקיצור טקסט
function toSlug(title) {
  const clean = title.trim().toLowerCase();
  let slug = "";
  for (const ch of clean) {
    if (ch === " ") {
      if (slug[slug.length - 1] !== "-") {
        slug += "-";
      }
    } else {
      slug += ch;
    }
  }
  return slug;
}

function truncate(text, maxLength) {
  if (text.length <= maxLength) {
    return text;
  }
  return text.slice(0, maxLength - 3) + "...";
}

console.log(toSlug("  Learn   Node JS  "));
console.log(truncate("Backend development with Node.js", 15));

P05-E21 · הסכום שגדל לבד 🤖 💻 כתיבת קוד 🤖 רמה 4

💡 רמז 1

איפה מוגדר total? מה קורה לו בין קריאה לקריאה?

💡 רמז 2

✅ פתרון
// P05-E21: סכום עגלה — פונקציה טהורה
function sumPrices(pricesText) {
  let total = 0;
  let current = "";
  for (const ch of pricesText + ",") {
    if (ch === ",") {
      total += Number(current);
      current = "";
    } else {
      current += ch;
    }
  }
  return total;
}

console.log(sumPrices("10,20,30"));
console.log(sumPrices("10,20,30"));

module.exports = { sumPrices };
60
60

הגרסה המקורית צברה ל־total חיצוני — תופעת לוואי — ולכן כל קריאה המשיכה מהסכום של הקודמת (60, ואז 120). בשרת זה באג מסוכן במיוחד: הוא מופיע רק בבקשה השנייה, ורק כשהשרת לא הופעל מחדש.

⚠️ טעויות נפוצות
  • "לאפס" את total בסוף הפונקציה — עובד, אבל שביר: אם הפונקציה תצא מוקדם (שגיאה), האיפוס לא יקרה.

P05-E22 · callback שהופעל מוקדם מדי 🤖 💻 כתיבת קוד 🤖 רמה 4

💡 רמז 1

הפרמטר השלישי צריך להיות פונקציה. double(5) הוא פונקציה או מספר?

💡 רמז 2

סדר הארגומנטים: (from, to, fn).

✅ פתרון
// P05-E22: העברת פונקציות כארגומנט — מתוקן
function transformAll(from, to, fn) {
  let result = "";
  for (let i = from; i <= to; i++) {
    result += fn(i) + " ";
  }
  return result.trim();
}

const double = (n) => n * 2;
const label = (n) => `#${n}`;

const doubles = transformAll(1, 3, double);
const labels = transformAll(1, 3, label);
const squares = transformAll(1, 3, (n) => n * n);

console.log(doubles);
console.log(labels);
console.log(squares);

module.exports = { transformAll, doubles, labels, squares };
2 4 6
#1 #2 #3
1 4 9

(1) double(5) הפעיל את הפונקציה מיד והעביר את המספר 10 — ואז fn(i) נכשל: TypeError: fn is not a function. מעבירים את הפונקציה עצמה: double. (2) transformAll(label, 1, 3) — סדר שגוי. (3) (n) => { n * n } — בלי return, יצא undefined undefined undefined.

⚠️ טעויות נפוצות
  • "לתקן" את transformAll כך שתקבל מספר — מפספס את הרעיון.

P05-E26 · מונה עם זיכרון (closure) 🤖 💻 כתיבת קוד 🤖 רמה 2

💡 רמז 1

בתוך makeCounter: משתנה let count = start;, ואחריו return של פונקציה (arrow מתאימה כאן).

💡 רמז 2

הפונקציה הפנימית עושה שני דברים: מגדילה את count ומחזירה אותו. return () => { ... };

✅ פתרון
// P05-E26: מונה עם זיכרון
function makeCounter(start = 0) {
  let count = start;
  return () => {
    count++;
    return count;
  };
}

const visits = makeCounter();
const tickets = makeCounter(100);
console.log(visits(), visits(), tickets(), visits());

module.exports = { makeCounter };
1 2 101 3

count הוא משתנה מקומי של makeCounter. בדרך כלל משתנה מקומי "נעלם" כשהפונקציה מסתיימת. כאן הוא נשאר, כי הפונקציה שהחזרנו עדיין משתמשת בו. כל קריאה ל־makeCounter יוצרת count חדש, ולכן visits ו־tickets לא מפריעים זה לזה.

בשרת תראה את זה הרבה: middleware שמקבל הגדרות ומחזיר פונקציה (פרק 15), ו־rate limiter שזוכר כמה בקשות הגיעו (פרק 12).

⚠️ טעויות נפוצות
  • let count מחוץ ל־makeCounter — כל המונים חולקים משתנה אחד, ו־tickets() מקלקל את visits. הבודק תופס את זה.
  • return count++; — מחזיר את הערך לפני ההגדלה (0 בקריאה הראשונה). כתוב ++count, או שתי שורות.
  • להחזיר count במקום פונקציה — visits() נכשל עם TypeError: visits is not a function.
