  return x + 1;
}

  console.log(x + 1);
}

  x + 1;
}

P05-E03 · איזה return?

רמה 1⏱ 10 דק׳ידע נדרש: שיעור 2

לכל קריאה — מה יחזור, ואיזו שורת return החזירה אותו?

function classify(temp) {
  if (typeof temp !== "number") {
    return "לא מספר";
  }
  if (temp < 0) {
    return "קפוא";
  }
  if (temp < 20) {
    return "קר";
  }
  return "נעים";
}
console.log(classify(-5));
console.log(classify(0));
console.log(classify(25));
console.log(classify("25"));

P05-E04 · מי רואה מה?

רמה 1⏱ 10 דק׳ידע נדרש: שיעור 3

לכל שורת console.log (מסומנות A–E) — מה תדפיס, או איזו שגיאה תתקבל?

const shop = "קפה הפינה";
function open(hour) {
  const message = `${shop} נפתח ב־${hour}`;
  return message;
}
console.log(open(8));       // A
console.log(shop);          // B
console.log(typeof hour);   // C
console.log(typeof message);// D
const hour = 10;
console.log(open(hour));    // E

P05-E12 · הפעל פעמיים 🤖
