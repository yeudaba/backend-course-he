P05-E79 · ביקורת קוד: פונקציה ארוכה עם משתנים בחוץ

let total = 0;
let count = 0;
function processPrices(from, to) {
  for (let price = from; price <= to; price += 10) {

      total += price * 0.9;
    } else {
      total += price;
    }
    count++;
  }
  console.log(`${count} מחירים, סה״כ ${total}`);
}
processPrices(10, 50);
processPrices(10, 50);

splitBill(100, 0);       // null
billSummary(100, 4);     // "4 סועדים × 28 ₪ = 112 ₪ (כולל טיפ 10%)"
billSummary(100, 2.5);   // "קלט לא תקין"

שים לב: בשורת הסיכום הסכום הוא מה שנאסף בפועל (4 × 28), לא החשבון המקורי.
