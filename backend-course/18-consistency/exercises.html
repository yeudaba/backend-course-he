-- א
BEGIN;
UPDATE accounts SET balance = balance - 100 WHERE id = 1;
UPDATE accounts SET balance = balance + 100 WHERE id = 3;
ROLLBACK;
-- ב
UPDATE accounts SET balance = balance - 100 WHERE id = 1;
UPDATE accounts SET balance = balance + 100 WHERE id = 99;
-- ג
BEGIN;
UPDATE accounts SET balance = balance - 2000 WHERE id = 1;
UPDATE accounts SET balance = balance + 2000 WHERE id = 3;
COMMIT;

P18-E03 · איזה כלי?

רמה 1⏱ 10 דק׳ידע נדרש: שיעור 2

לכל תרחיש — איזה פתרון מתאים: עדכון אטומי עם תנאי, FOR UPDATE, אילוץ (UNIQUE/CHECK), או Serializable + ניסיון חוזר? למה?

  1. הורדת מלאי בקנייה.
  2. הרשמה עם אימייל — שני אנשים נרשמים עם אותו אימייל באותו רגע.
  3. העברה בין חשבונות, עם בדיקות: חשבון קפוא? מגבלה יומית? הנחה לפי סוג לקוח?
  4. "לכל רופא לפחות רופא תורן אחד בכל משמרת" — שני רופאים מבטלים תורנות באותו רגע.
  5. מונה צפיות בסרטון.

P18-E05 · כמה שאילתות?

רמה 1⏱ 10 דק׳ידע נדרש: שיעור 4

כמה שאילתות שולח הקוד לרשימה של 50 הזמנות, שבכל אחת 3 פריטים בממוצע? כתוב גרסה עם 3 שאילתות קבועות (לא תלוי בכמות).

const orders = (await pool.query("SELECT * FROM orders ORDER BY id DESC LIMIT 50")).rows;
for (const o of orders) {
  o.customer = (await pool.query("SELECT * FROM customers WHERE id = $1", [o.customer_id])).rows[0];
  o.items = (await pool.query("SELECT * FROM order_items WHERE order_id = $1", [o.id])).rows;
  for (const item of o.items) {
    item.product = (await pool.query("SELECT * FROM products WHERE id = $1", [item.product_id])).rows[0];
  }
}

P18-E13 · העברה עם נעילה 🤖

רמה 3⏱ 30 דק׳ידע נדרש: שיעור 2

בקובץ 18-consistency/starter/P18-E13.ts השלם transferSafe(pool, fromId, toId, amount), עם בדיקות לפני כל שינוי:

  • fromId === toId ← AppError("SAME_ACCOUNT"); amount לא חיובי ← AppError("INVALID_AMOUNT") (בלי לגשת למסד).
  • ב־transaction: נעל את שני החשבונות עם SELECT ... FOR UPDATE, לפי סדר id. חסר אחד ← AppError("ACCOUNT_NOT_FOUND"). יתרה לא מספיקה ← AppError("INSUFFICIENT_FUNDS"). אחרת — העדכונים, רישום ב־transfers, COMMIT. מחזירה { transferId, fromBalance, toBalance } (מספרים).

הבדיקה מריצה 40 העברות במקביל בין Dana ל־Avi, בשני הכיוונים — בלי סדר נעילה קבוע זה מייצר deadlocks.

P18-E19 · ניסיון חוזר ב־Serializable 🤖

רמה 3⏱ 30 דק׳ידע נדרש: שיעור 2

בקובץ 18-consistency/starter/P18-E19.ts השלם withSerializableRetry(pool, work, { retries = 5 } = {}):

  • מריצה את work(client) בתוך BEGIN ISOLATION LEVEL SERIALIZABLE ... COMMIT, ומחזירה את מה ש־work החזיר.
  • שגיאה עם code 40001 (serialization) או 40P01 (deadlock) — גם ב־COMMIT עצמו! — ← ROLLBACK, המתנה קצרה (אקראית, 5–25ms), וניסיון חוזר של כל העבודה. אחרי retries ניסיונות חוזרים — השגיאה האחרונה נזרקת.
  • כל שגיאה אחרת ← ROLLBACK ונזרקת מיד. החיבור משתחרר תמיד.

הבדיקה: 15 "העלאות מונה" במקביל (קרא ← הוסף 1 ← כתוב) ב־Serializable — בלי ניסיון חוזר רובן נכשלות; עם — המונה מגיע ל־15.

P18-E24 · אתגר: checkout אמיתי 🤖

רמה 5⏱ 60 דק׳ידע נדרש: כל הפרק

בקובץ 18-consistency/starter/P18-E24.ts השלם checkout(pool, { customerId, items, idempotencyKey }) מעל catalog, shop_orders, shop_order_lines. items: [{ productId, quantity }].

  • items ריק ← AppError("EMPTY_CART"). אותו מוצר פעמיים ← מאחדים כמויות.
  • הכול ב־transaction אחד: נעילת המוצרים (FOR UPDATE, לפי id), מוצר חסר ← AppError("PRODUCT_NOT_FOUND"), מלאי לא מספיק לאחד מהם ← AppError("OUT_OF_STOCK") עם productId על השגיאה — ואז אף מלאי לא משתנה.
  • מחיר נשמר ב־unit_price מהקטלוג ברגע ההזמנה; total = סכום השורות. מחזירה { orderId, total, replayed: false } (total כמספר).
  • Idempotency: אם כבר יש הזמנה עם אותו idempotencyKey — מחזירה אותה (replayed: true) בלי לשנות כלום. גם כששתי בקשות עם אותו מפתח מגיעות יחד.

P18-E25 · אתגר: שלוש שאילתות, שני indexes 🤖

רמה 5⏱ 30 דק׳ידע נדרש: כל הפרק

הדשבורד מריץ שלוש שאילתות על big_orders, וכולן סורקות את כל הטבלה. בקובץ 18-consistency/starter/P18-E25.sql צור לכל היותר שני indexes, כך שכל השלוש ישתמשו ב־index:

-- 1: ההזמנות האחרונות של לקוח
SELECT id, total FROM big_orders WHERE customer_id = 42 ORDER BY created_at DESC LIMIT 10;
-- 2: הזמנות ממתינות מהשבוע האחרון
SELECT id FROM big_orders WHERE status = 'pending' AND created_at >= '2025-12-24' ORDER BY created_at;
-- 3: כמה בוטלו מאז דצמבר
SELECT COUNT(*) FROM big_orders WHERE status = 'cancelled' AND created_at >= '2025-12-01';

🏁 מבחן מעבר

  • לכתוב העברת כספים ב־transaction מתוך Node, עם ROLLBACK בכל שגיאה ושחרור החיבור.
  • להראות בשני חלונות psql איך שתי מכירות של הפריט האחרון יוצרות מלאי שלילי — ולתקן בשתי דרכים.
  • להסביר מתי index עוזר, מתי לא, ולמה לא שמים index על כל עמודה.
  • להוסיף index לשאילתה איטית ולהראות ב־EXPLAIN את ההבדל.
  • להסביר מה זה N+1 ואיך מזהים אותו ביומן השאילתות.
