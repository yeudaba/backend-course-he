P20-E52 · ‏class-validator מול Zod

🧠 הבנהרמה 2⏱ 10 דק׳ידע נדרש: פרק 15, שיעור 2

אותה דרישה בשתי הדרכים: "name 2–80 תווים, price חיובי עם עד 2 ספרות, tags — עד 5 מחרוזות, אופציונלי (ברירת מחדל [])".

// Zod (פרק 15)
const CreateProduct = z.object({ name: z.string().min(2).max(80), price: z.number().positive().multipleOf(0.01), tags: z.array(z.string()).max(5).default([]) });
type CreateProduct = z.infer<typeof CreateProduct>;

// class-validator (השיעור)
export class CreateProductDto {
  @IsString() @Length(2, 80) name: string;
  @IsNumber({ maxDecimalPlaces: 2 }) @IsPositive() price: number;
  @IsOptional() @IsArray() @ArrayMaxSize(5) @IsString({ each: true }) tags: string[] = [];
}
