P20-E60 · מה ה־interceptor רואה, ומתי?

🧠 הבנהרמה 2⏱ 10 דק׳ידע נדרש: שיעור 3

מה יודפס, ובאיזה סדר, ב־GET /x? ומה יקבל הלקוח?

@Injectable()
class LogInterceptor implements NestInterceptor {
  intercept(ctx: ExecutionContext, next: CallHandler) {
    console.log("before");
    return next.handle().pipe(
      map((data) => { console.log("after", data); return { data }; }),
    );
  }
}

@Controller("x") @UseInterceptors(LogInterceptor)
class XController {
  @Get() get() { console.log("handler"); return [1, 2]; }
  @Get("fail") fail() { console.log("handler"); throw new NotFoundException("אין"); }
}

// א
@Injectable() class TasksService {
