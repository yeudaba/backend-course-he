P20-E66 · מה Swagger יודע — ומאיפה? 🧠 הבנה רמה 1

💡 רמז 1

Swagger של Nest קורא את מה ש־Nest כבר יודע: controllers, decorators של routes, ו־classes.

💡 רמז 2

מה ש"רק בקוד של ה־handler" (throw new NotFoundException) — Swagger לא מריץ.

✅ פתרון
  1. כן — מה־@Controller וה־@Get/@Post.
  2. כן — מ־@Param("id").
  3. חלקית — עם @ApiProperty() על כל שדה (או ה־CLI plugin של Swagger, שמוסיף אותם בזמן build). בלי — ה־DTO מופיע ריק.
  4. לא לבד — @ApiProperty עם minLength, ‏maxLength ו־example: "Buy milk". (ה־plugin יודע לקרוא חלק מכללי class-validator.)
  5. לא — Swagger לא יודע מה הקוד זורק. @ApiNotFoundResponse() (או @ApiResponse({ status: 404 })) על המתודה.
  6. לא — Zod הוא ערך בזמן ריצה, לא class. צריך לתאר ביד (@ApiQuery) או ספרייה שמתרגמת Zod ל־OpenAPI.
⚠️ טעויות נפוצות
  • להסתפק בתיעוד שנוצר "לבד" — לקוח של ה־API לא יודע מה חובה, ומה החזרות האפשריות.

💡 רמז 2

✅ פתרון
  1. הבדיקות רצות בלי ה־pipe וה־filter: בדיקה של "400 על קלט לא תקין" או של מבנה השגיאה תיכשל — או גרוע, מישהו "יתקן" את הקוד כך שיעבור בלי ValidationPipe, ובייצור ההתנהגות תהיה אחרת. וההפך: באג ב־filter עצמו לא ייבדק לעולם.

P20-E70 · ‏Location ו־@Res({ passthrough: true }) 🧠 הבנה רמה 2

💡 רמז 1

💡 רמז 2

P20-E71 · ‏@Global() ל־PrismaModule — ולמה לא לכל דבר 🧠 הבנה רמה 2

💡 רמז 1

module גלובלי — כל ה־modules יכולים להזריק את מה שהוא מייצא בלי imports.

💡 רמז 2

כמה modules צריכים גישה למסד? וכמה צריכים את הלוגיקה של משימות?

✅ פתרון
  1. לא צריך imports: [PrismaModule] בכל module שיש לו repository — PrismaService זמין בכל מקום (הוא עדיין צריך exports).
  2. ‏PrismaService הוא תשתית שכמעט כל module צריך — כמו שעון או logger. ‏TasksService שייך לתחום אחד; אם module אחר צריך אותו — זה קשר שכדאי שיהיה גלוי ב־imports (P20-E58).

⚠️ טעויות נפוצות
  • לסמן הכול @Global() — אף module לא מצהיר במה הוא תלוי, ו"מה ישבר אם אשנה את X" הופכת לשאלה בלי תשובה.

P20-E19 · Swagger 🤖 💻 כתיבת קוד 🤖 רמה 3

💡 רמז 1

הקוד בשיעור 4, סעיף 1.

💡 רמז 2

בלי @ApiProperty, Swagger לא "רואה" את השדות: TypeScript נמחק, ו־class-validator לא מדבר עם Swagger לבד (בלי ה־plugin של ה־CLI).

✅ פתרון
// P20-E19
import { Body, Controller, Get, Module, Post, type INestApplication } from "@nestjs/common";
import { ApiCreatedResponse, ApiProperty, ApiTags, DocumentBuilder, SwaggerModule } from "@nestjs/swagger";
import { IsInt, IsOptional, IsString, Length, Max, Min } from "class-validator";

export class CreateBookDto {
  @ApiProperty({ example: "Dune", minLength: 1, maxLength: 200 })
  @IsString()
  @Length(1, 200)
  title: string;

  @ApiProperty({ example: 1965, required: false })
  @IsOptional()
  @IsInt()
  @Min(0)
  @Max(2100)
  year?: number;
}

export class BookDto extends CreateBookDto {
  @ApiProperty({ example: 1 })
  id: number;
}

@ApiTags("books")
@Controller("books")
export class BooksController {
  private readonly books: BookDto[] = [];

  @Get()
  list() {
    return this.books;
  }

  @Post()
  @ApiCreatedResponse({ type: BookDto })
  create(@Body() dto: CreateBookDto): BookDto {
    const book = { id: this.books.length + 1, ...dto };
    this.books.push(book);
    return book;
  }
}

@Module({ controllers: [BooksController] })
export class BooksModule {}

export function setupDocs(app: INestApplication) {
  const config = new DocumentBuilder().setTitle("Library API").setDescription("ספרים").setVersion("1.0").build();
  SwaggerModule.setup("docs", app, () => SwaggerModule.createDocument(app, config));
}

מה הבודק מצפה לראות ב־/docs-json: components.schemas.CreateBookDto עם properties של title ו־year, required: ["title"] (בזכות required: false על year), וה־tag books על POST /books. מזה Swagger UI בונה טופס, ו־Frontend יכול לייצר לקוח.

⚠️ טעויות נפוצות
  • לשכוח את required: false על שדה אופציונלי ← בתיעוד הוא חובה, והלקוח מבולבל.
  • לכתוב את התיעוד ביד (קובץ YAML) — הוא מתיישן אחרי השינוי הראשון בקוד. כאן התיעוד נוצר מה־DTOs עצמם.

// P20-E72: configureApp — prefix, pipe ו־filter
import { ArgumentsHost, Body, Catch, Controller, ExceptionFilter, Get, HttpException, INestApplication, Module, NotFoundException, Param, ParseIntPipe, Post, ValidationPipe } from "@nestjs/common";
import { IsString, Length } from "class-validator";
import type { Response } from "express";

export class CreateNoteDto {
  @IsString()
  @Length(1, 200)
  text: string;
}

@Controller("notes")
export class NotesController {
  private notes: { id: number; text: string }[] = [];

  @Post()
  create(@Body() dto: CreateNoteDto) {
    const note = { id: this.notes.length + 1, text: dto.text };
    this.notes.push(note);
    return note;
  }

  @Get(":id")
  findOne(@Param("id", ParseIntPipe) id: number) {
    const note = this.notes.find((n) => n.id === id);
    if (!note) throw new NotFoundException(`פתק ${id} לא נמצא`);
    return note;
  }
}

@Module({ controllers: [NotesController] })
export class NotesModule {}

const CODES: Record<number, string> = { 400: "BAD_REQUEST", 404: "NOT_FOUND" };

@Catch(HttpException)
export class ApiErrorFilter implements ExceptionFilter {
  catch(exception: HttpException, host: ArgumentsHost) {
    const res = host.switchToHttp().getResponse<Response>();
    const status = exception.getStatus();
    const body = exception.getResponse();
    const msg = typeof body === "object" && body !== null ? (body as { message?: unknown }).message : body;
    const error = Array.isArray(msg)
      ? { code: CODES[status] ?? "ERROR", message: "Validation failed", details: msg }
      : { code: CODES[status] ?? "ERROR", message: exception.message };
    res.status(status).json({ error });
  }
}

export function configureApp(app: INestApplication) {
  app.setGlobalPrefix("api");
  app.useGlobalPipes(new ValidationPipe({ whitelist: true, forbidNonWhitelisted: true, transform: true }));
  app.useGlobalFilters(new ApiErrorFilter());
}

‏main.ts: ‏

const app = await NestFactory.create(NotesModule); configureApp(app); await app.listen(3000);

הבדיקות — אותה שורה בדיוק, בין createNestApplication ל־init.

⚠️ טעויות נפוצות
  • לקרוא ל־configureApp אחרי app.init() בבדיקה — ה־prefix כבר לא חל על ה־routes שנרשמו.

// P20-E74: Location ב־POST
import { Body, Controller, Get, Module, NotFoundException, Param, ParseIntPipe, Post, Res } from "@nestjs/common";
import type { Response } from "express";

@Controller("links")
export class LinksController {
  private links: { id: number; url: string }[] = [];

  create(@Body() body: { url: string }, @Res({ passthrough: true }) res: Response) {

  }
}

P20-E75 · ‏filter לכל שגיאה — גם JSON שבור ו־500 🤖 💻 כתיבת קוד 🤖 רמה 3

💡 רמז 1

@Catch() בלי פרמטרים — וה־catch(exception: unknown, host) בודק exception instanceof HttpException.

💡 רמז 2

(‏הרמז של השיעור).

✅ פתרון
// P20-E75: filter לכל שגיאה — גם JSON שבור ו־500
import { ArgumentsHost, BadRequestException, Body, Catch, Controller, ExceptionFilter, Get, HttpException, Module, NotFoundException, Param, Post } from "@nestjs/common";
import { APP_FILTER } from "@nestjs/core";
import type { Response } from "express";

const CODES: Record<number, string> = { 400: "BAD_REQUEST", 401: "UNAUTHORIZED", 403: "FORBIDDEN", 404: "NOT_FOUND", 409: "CONFLICT" };

@Catch()
export class AllErrorsFilter implements ExceptionFilter {
  catch(exception: unknown, host: ArgumentsHost) {
    const res = host.switchToHttp().getResponse<Response>();
    if (exception instanceof BadRequestException && /JSON/i.test(exception.message)) {
      return res.status(400).json({ error: { code: "INVALID_JSON", message: "גוף הבקשה אינו JSON תקין" } });
    }
    if (exception instanceof HttpException) {
      const status = exception.getStatus();
      return res.status(status).json({ error: { code: CODES[status] ?? "ERROR", message: exception.message } });
    }
    res.status(500).json({ error: { code: "INTERNAL", message: "שגיאה פנימית" } });
  }
}

@Controller("items")
export class ItemsController {
  @Get("boom/now")
  boom() {
    throw new Error("secret: db password");
  }

  @Get(":id")
  findOne(@Param("id") id: string) {
    if (id !== "1") throw new NotFoundException(`פריט ${id} לא נמצא`);
    return { id: 1 };
  }

  @Post()
  create(@Body() body: unknown) {
    return body;
  }
}

@Module({ controllers: [ItemsController], providers: [{ provide: APP_FILTER, useClass: AllErrorsFilter }] })
export class ItemsModule {}

שלוש קטגוריות, ממוינות מהספציפית לכללית: JSON שבור (גם הוא HttpException, ולכן נבדק קודם), שגיאות HTTP, והשאר. ב־500 — שום פרט מבפנים ללקוח (ובאפליקציה אמיתית — לוג מלא, פרק 24).

⚠️ טעויות נפוצות
  • @Catch(HttpException) — ה־Error הרגיל לא נתפס, ו־Nest מחזיר את ה־500 שלו, במבנה אחר.
  • להחזיר exception.message גם ב־500 — הסיסמה מההודעה מגיעה ללקוח.

// P20-E76: ZodValidationPipe עם נתיבים
import { ArgumentMetadata, BadRequestException, Body, Controller, Get, Module, Param, PipeTransform, Post, Query } from "@nestjs/common";
import { z } from "zod";

export class ZodValidationPipe implements PipeTransform {
  constructor(private readonly schema: z.ZodType) {}

  transform(value: unknown, metadata: ArgumentMetadata) {
    const r = this.schema.safeParse(value);
    if (r.success) return r.data;
    const type = metadata.type === "param" ? "params" : metadata.type;
    const details = r.error.issues.map((issue) => ({
      path: [type, ...(metadata.data ? [metadata.data] : []), ...issue.path.map(String)].join("."),
      message: issue.message,
    }));
    throw new BadRequestException({ error: { code: "VALIDATION_ERROR", message: "קלט לא תקין", details } });

export const CLOCK = "CLOCK";
export interface Clock {
  now(): Date;

// ---------- שגיאות: { error: { code, message, details? } } ----------
const apiError = (code: string, message: string, details?: unknown) => ({ code, message, ...(details ? { details } : {}) });
const CODES: Record<number, string> = { 400: "BAD_REQUEST", 404: "NOT_FOUND", 409: "CONFLICT" };

@Catch(HttpException)
export class ApiExceptionFilter implements ExceptionFilter {
  catch(exception: HttpException, host: ArgumentsHost) {
    const res = host.switchToHttp().getResponse<Response>();
    const status = exception.getStatus();
    const body = exception.getResponse();
    const error = typeof body === "object" && body !== null && "code" in body ? body : apiError(CODES[status] ?? "ERROR", exception.message);
    res.status(status).json({ error });
  }
}

export function createValidationPipe() {
  return new ValidationPipe({
    whitelist: true,
    forbidNonWhitelisted: true,
    transform: true,
    exceptionFactory: (errors: ValidationError[]) =>
      new BadRequestException(apiError("VALIDATION_ERROR", "קלט לא תקין", errors.map((e) => ({ field: e.property, messages: Object.values(e.constraints ?? {}) })))),
  });
}

  @IsOptional() @IsIn(["true", "false"]) available?: string;
  @IsOptional() @Type(() => Number) @IsInt() @Min(1) page: number = 1;
  @IsOptional() @Type(() => Number) @IsInt() @Min(1) @Max(50) limit: number = 10;

// ---------- service ----------
export interface Book {
  id: number;
  title: string;
  author: string;
  copies: number;
  available: number;
}
export interface Loan {
  id: number;
  bookId: number;
  member: string;
  loanedAt: string;
  dueAt: string;
  returnedAt: string | null;
}

const LOAN_DAYS = 14;

@Injectable()
export class LibraryService {
  private readonly books: Book[] = [];
  private readonly loans: Loan[] = [];

  constructor(@Inject(CLOCK) private readonly clock: Clock) {}

  createBook(dto: CreateBookDto): Book {
    const book = { id: this.books.length + 1, title: dto.title, author: dto.author, copies: dto.copies, available: dto.copies };
    this.books.push(book);
    return book;
  }

  listBooks({ author, available, page, limit }: ListBooksQuery) {
    let books = this.books;
    if (author) books = books.filter((b) => b.author.toLowerCase().includes(author.toLowerCase()));
    if (available === "true") books = books.filter((b) => b.available > 0);
    return { data: books.slice((page - 1) * limit, page * limit), meta: { page, limit, total: books.length, totalPages: Math.ceil(books.length / limit) } };
  }

  getBook(id: number): Book {
    const book = this.books.find((b) => b.id === id);
    if (!book) throw new NotFoundException(apiError("BOOK_NOT_FOUND", `ספר ${id} לא נמצא`));
    return book;
  }

  lend(bookId: number, member: string): Loan {
    const book = this.getBook(bookId);
    if (this.loans.some((l) => l.bookId === bookId && l.member === member && l.returnedAt === null)) {
      throw new ConflictException(apiError("ALREADY_BORROWED", `${member} כבר מחזיק/ה את הספר`));
    }
    if (book.available === 0) throw new ConflictException(apiError("NO_COPIES", "אין עותק פנוי"));
    const now = this.clock.now();
    const loan = {
      id: this.loans.length + 1,
      bookId,
      member,
      loanedAt: now.toISOString(),
      dueAt: new Date(now.getTime() + LOAN_DAYS * 24 * 60 * 60 * 1000).toISOString(),
      returnedAt: null,
    };
    this.loans.push(loan);
    book.available--;
    return loan;
  }

  returnLoan(id: number): Loan {
    const loan = this.loans.find((l) => l.id === id);
    if (!loan) throw new NotFoundException(apiError("LOAN_NOT_FOUND", `השאלה ${id} לא נמצאה`));
    if (loan.returnedAt) throw new ConflictException(apiError("ALREADY_RETURNED", "ההשאלה כבר הוחזרה"));
    loan.returnedAt = this.clock.now().toISOString();
    this.getBook(loan.bookId).available++;
    return loan;
  }

  overdue(): Loan[] {
    const now = this.clock.now().getTime();
    return this.loans.filter((l) => l.returnedAt === null && new Date(l.dueAt).getTime() < now);
  }
}

// P20-E82: לקרוא מסמך OpenAPI בקוד
const ORDER = ["get", "post", "put", "patch", "delete"];

export function listEndpoints(doc: any): string[] {
  const rows = Object.entries(doc.paths ?? {}).flatMap(([path, ops]) =>
    Object.keys(ops as object)
      .filter((m) => ORDER.includes(m))
      .map((m) => ({ path, m })),
  );
  rows.sort((a, b) => (a.path === b.path ? ORDER.indexOf(a.m) - ORDER.indexOf(b.m) : a.path < b.path ? -1 : 1));
  return rows.map((r) => `${r.m.toUpperCase()} ${r.path}`);
}

export function requiredFields(doc: any, schema: string): string[] {
  const s = doc.components?.schemas?.[schema];
  if (!s) throw new Error(`אין סכמה ${schema}`);
  return [...(s.required ?? [])].sort();
}

@Controller("health")

  @Get()
  async check() {
    let timer: NodeJS.Timeout | undefined;
    const timeout = new Promise<never>((_, reject) => {
      timer = setTimeout(() => reject(new Error("timeout")), 200);
    });
    try {
      await Promise.race([this.ping(), timeout]);
      return { status: "ok", checks: { db: "up" } };
    } catch {
      throw new HttpException({ status: "error", checks: { db: "down" } }, 503);
    } finally {
      clearTimeout(timer);
    }
  }
}
