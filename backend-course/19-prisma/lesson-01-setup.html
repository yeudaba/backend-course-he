1. אותה שאילתה, פעמיים

// pg (פרק 18): מחרוזת SQL, ותוצאה מסוג any
const { rows } = await pool.query(
  "SELECT id, name, price FROM products WHERE is_active AND price < $1 ORDER BY price", [50]);
rows[0].nmae; // שגיאת כתיב — TypeScript לא יודע. undefined בזמן ריצה

// Prisma: שאילתה כקוד, ותוצאה מוקלדת
const products = await prisma.product.findMany({
  where: { isActive: true, price: { lt: 50 } },
  select: { id: true, name: true, price: true },
  orderBy: { price: "asc" },
});
products[0].nmae; // ✘ שגיאת TypeScript מיד ב־VS Code
מה ORM נותןמה המחיר
טיפוסים מלאים, השלמה אוטומטיתשכבה נוספת ללמוד ולדבג
migrations מנוהלים מהסכמהלא כל שאילתה מורכבת נוחה לכתיבה (ואז — SQL ישיר)
קשרים (include) בלי JOIN ידני, ובלי N+1קל לכתוב שאילתה לא יעילה בלי לראות את ה־SQL
הגנה מ־SQL injection כברירת מחדלתלות בספרייה ובגרסה שלה

בגלל השורה האחרונה בצד ימין, למדת SQL קודם: כשהשאילתה של Prisma איטית, תדע לקרוא מה הוא שלח למסד.

prisma.config.ts בשורש הפרויקט (את הקובץ הזה כותבים בעצמנו; npx prisma init בגרסה 7.10 יוצר אותו בשם prisma7.config.ts — אפשר לשנות שם, או פשוט לכתוב):

prisma.config.ts 
import "dotenv/config";                       // טוען את .env (Prisma 7 לא עושה זאת לבד)
import { defineConfig } from "prisma/config";

export default defineConfig({
  schema: "prisma/schema.prisma",
  migrations: { path: "prisma/migrations" },
  datasource: { url: process.env.DATABASE_URL },
});

שים לב: ב־Prisma 7, migrate dev לא מריץ generate בשבילך. שינית סכמה? שתי הפקודות. (ותוסיף src/generated ל־.gitignore — זה קוד שנוצר, כמו node_modules.)

וה־client עצמו:

src/db.ts 
import { PrismaPg } from "@prisma/adapter-pg";
import { PrismaClient } from "./generated/prisma/client.ts";

const adapter = new PrismaPg({
  connectionString: process.env.DATABASE_URL,
  options: "-c timezone=UTC",   // ← ראה סעיף 4
});
export const prisma = new PrismaClient({ adapter });

ב־Prisma 7 החיבור למסד עובר דרך driver adapter — כאן @prisma/adapter-pg, שמבוסס על אותה ספריית pg מפרק 18. אובייקט prisma אחד לכל האפליקציה (יש לו pool של חיבורים בפנים), לא אחד לכל בקשה.

4. ⚠️ מלכודת אזור הזמן

ניסוי אמיתי, על מסד שה־TimeZone שלו Asia/Jerusalem (ברירת המחדל אצלך, אם ה־Mac מוגדר לישראל): הזמנה שנוצרה ב־SQL ב־09:12+02 (כלומר 07:12 UTC), ונקראה דרך Prisma:

בלי options:                    order.createdAt → 2026-01-05T09:12:00.000Z   ✘ (שעתיים הפרש)
עם options: "-c timezone=UTC":  order.createdAt → 2026-01-05T07:12:00.000Z   ✔

בגרסה הזו, Prisma מפרש את הזמנים שהמסד מחזיר כאילו הם ב־UTC — וכשה־session של המסד באזור זמן אחר, כל זמן זז. (גם בכתיבה: תאריך שנשמר דרך Prisma יוצא שגוי כשקוראים אותו ב־SQL.) הפתרון: לבקש מהמסד לעבוד ב־UTC בכל חיבור — זה ה־options. וזה גם כלל טוב בכלל: שרתים ומסדי נתונים עובדים ב־UTC, ורק התצוגה למשתמש מתורגמת לשעון המקומי (פרק 24).
