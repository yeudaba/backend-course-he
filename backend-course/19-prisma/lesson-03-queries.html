2. include: קשרים בלי N+1

const orders = await prisma.order.findMany({
  where: { customerId: 1 },
  orderBy: { createdAt: "desc" },
  include: {
    items: { include: { product: { select: { name: true } } } },
  },
});
// orders[0].items[0].product.name  ← מוקלד

Prisma מריץ שאילתה לכל "רמה" (הזמנות; כל הפריטים שלהן בבת אחת; כל המוצרים בבת אחת) ומחבר בזיכרון — בדיוק "אפשרות 2" משיעור N+1 בפרק 18. אבל לולאה עם findUnique לכל פריט — זה עדיין N+1, גם ב־Prisma.

ולראות מה Prisma שולח: new PrismaClient({ adapter, log: ["query"] }) — כל שאילתה מודפסת. שווה להפעיל מדי פעם בפיתוח.

5. עדכון אטומי ו־transactions

// הורדת מלאי בלי מרוץ — updateMany עם תנאי (כמו P18-E10)
const { count } = await prisma.product.updateMany({
  where: { id: productId, stock: { gte: qty } },
  data: { stock: { decrement: qty } },
});
if (count === 0) throw new AppError("OUT_OF_STOCK");

// interactive transaction — כמה פעולות, הכול או כלום
const order = await prisma.$transaction(async (tx) => {
  const p = await tx.product.updateMany({ where: { id: 1, stock: { gte: 1 } }, data: { stock: { decrement: 1 } } });
  if (p.count === 0) throw new AppError("OUT_OF_STOCK");  // זריקה ← ROLLBACK
  return tx.order.create({ data: { customerId: 1, items: { create: [{ productId: 1, quantity: 1, unitPrice: 89.9 }] } } });
});
⚠️ tx, לא prisma

בתוך ה־callback — כל הפעולות דרך tx. שאילתה דרך prisma רצה מחוץ ל־transaction (חיבור אחר!), ולא תתבטל בכישלון — אותו באג כמו pool.query("BEGIN") בפרק 18.

6. SQL ישיר

const rows = await prisma.$queryRaw<{ category: string; revenue: number }[]>`
  SELECT c.name AS category, SUM(oi.quantity * oi.unit_price)::float AS revenue
  FROM order_items oi
  JOIN products p ON p.id = oi.product_id
  JOIN categories c ON c.id = p.category_id
  WHERE p.price > ${minPrice}
  GROUP BY c.name`;

דוחות עם GROUP BY על כמה טבלאות, חלונות, CTE — ב־SQL זה פשוט יותר. $queryRaw עם template literal (הגרש ההפוך) הופך כל ${...} לפרמטר — בטוח מ־injection. $queryRawUnsafe("..." + x) — לא בטוח, ולא צריך כמעט אף פעם.

7. עוד כמה כלים: upsert, ‏aggregate, ‏_count

19-prisma/files/play/more.ts 
import { config } from "dotenv";
import { PrismaClient, Prisma } from "../generated/prisma/client.ts";
import { PrismaPg } from "@prisma/adapter-pg";

config({ path: "../../.env", quiet: true }); // SHOP_DATABASE_URL — כמו prisma.config.ts
const prisma = new PrismaClient({ adapter: new PrismaPg({ connectionString: process.env.SHOP_DATABASE_URL, options: "-c timezone=UTC" }) });

// upsert — "צור, ואם כבר קיים — עדכן". where חייב להיות שדה ייחודי
const tag = await prisma.tag.upsert({
  where: { name: "organic" },
  create: { name: "organic" },
  update: {},
});
const again = await prisma.tag.upsert({ where: { name: "organic" }, create: { name: "organic" }, update: {} });
console.log(tag.id === again.id, await prisma.tag.count({ where: { name: "organic" } }));

// aggregate — צבירות על כל השורות שמתאימות, בלי להביא אותן
const stats = await prisma.product.aggregate({
  where: { isActive: true },
  _count: true,
  _avg: { price: true },
  _max: { price: true },
});
console.log(stats);

// _count של קשר — כמה מוצרים בכל קטגוריה, בתוך select
const categories = await prisma.category.findMany({
  select: { name: true, _count: { select: { products: true } } },
  orderBy: { id: "asc" },
});
console.log(categories);

// findUniqueOrThrow — כמו findUnique, אבל זורק P2025 במקום null
try {
  await prisma.product.findUniqueOrThrow({ where: { id: 999 } });
} catch (err) {
  if (err instanceof Prisma.PrismaClientKnownRequestError) console.log(err.code);
}

// where שנבנה בהדרגה — עם הטיפוס ש־Prisma מייצר לכל מודל
const where: Prisma.ProductWhereInput = { isActive: true };
const maxPrice: number | undefined = 100;
if (maxPrice !== undefined) where.price = { lte: maxPrice };
console.log(await prisma.product.count({ where }));
await prisma.$disconnect();

6
  • upsert — INSERT ... ON CONFLICT DO UPDATE של Prisma. הקריאה השנייה לא יצרה תגית נוספת (true 1): היא מצאה את הקיימת לפי where, והריצה את update (כאן ריק — לא משנים כלום). הכלי ל־seed שאפשר להריץ שוב ושוב.
  • aggregate — COUNT, ‏AVG, ‏MAX... במסד, ושורה אחת חוזרת. _avg ו־_max של שדה Decimal חוזרים כ־Decimal (שיעור 2), לכן Number(stats._avg.price) לפני חשבון. (צבירה לכל קבוצה — groupBy, בשיעור 4.)
  • _count בתוך select (או include) — כמה רשומות קשורות לכל שורה, בלי להביא אותן. "gift cards" עם 0 — כמו LEFT JOIN ו־COUNT בפרק 17.
  • findUniqueOrThrow (וגם findFirstOrThrow) — כשאין רשומה, זורק P2025 במקום להחזיר null. נוח כשאחרי זה ממילא מתרגמים ל־404 (סעיף 4).
  • טיפוסי הקלט: לכל מודל Prisma מייצר טיפוסים כמו Prisma.ProductWhereInput, ‏Prisma.ProductOrderByWithRelationInput ו־Prisma.ProductUncheckedUpdateInput (עדכון עם categoryId ישירות) — כדי לבנות where/orderBy/data בהדרגה, כמו כאן.
