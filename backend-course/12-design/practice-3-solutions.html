// P12-E11
export class InMemoryRepository<T extends { id: number }> {
  #items = new Map<number, T>();

  async findById(id: number): Promise<T | null> {
    const item = this.#items.get(id);
    return item ? { ...item } : null;
  }

  async findAll(): Promise<T[]> {
    return [...this.#items.values()].sort((a, b) => a.id - b.id).map((x) => ({ ...x }));
  }

  async save(item: T): Promise<void> {
    this.#items.set(item.id, { ...item });
  }

  async delete(id: number): Promise<boolean> {
    return this.#items.delete(id);
  }

  async count(): Promise<number> {
    return this.#items.size;
  }
}

// P12-E15
export type User = { id: number; email: string; name: string };

export interface UserRepository {
  findById(id: number): Promise<User | null>;
  findByEmail(email: string): Promise<User | null>;
  count(): Promise<number>;
  save(user: User): Promise<void>;
}

export class InMemoryUserRepository implements UserRepository {
  #users = new Map<number, User>();
  async findById(id: number) {
    const u = this.#users.get(id);
    return u ? { ...u } : null;
  }
  async findByEmail(email: string) {
    for (const u of this.#users.values()) if (u.email === email) return { ...u };
    return null;
  }
  async count() {
    return this.#users.size;
  }
  async save(user: User) {
    this.#users.set(user.id, { ...user });
  }
}

export class NotFoundError extends Error {
  constructor(entity: string, id: number) {
    super(`${entity} ${id} לא נמצא`);
    this.name = "NotFoundError";
  }
}
export class ConflictError extends Error {
  constructor(message: string) {
    super(message);
    this.name = "ConflictError";
  }
}
export class ValidationError extends Error {
  readonly field: string;
  constructor(field: string, message: string) {
    super(message);
    this.name = "ValidationError";
    this.field = field;
  }
}

export class UserService {
  #users: UserRepository;

  constructor(users: UserRepository) {
    this.#users = users;
  }

  async register(input: { email: string; name: string }): Promise<User> {
    const email = input.email.trim().toLowerCase();
    if (await this.#users.findByEmail(email)) {
      throw new ConflictError("האימייל כבר רשום");
    }
    const user: User = { id: (await this.#users.count()) + 1, email, name: input.name.trim() };
    await this.#users.save(user);
    return user;
  }

  async getById(id: number): Promise<User> {
    const user = await this.#users.findById(id);
    if (!user) {
      throw new NotFoundError("משתמש", id);
    }
    return user;
  }

  async rename(id: number, name: string): Promise<User> {
    if (name.trim() === "") {
      throw new ValidationError("name", "שם חובה");
    }
    const user = await this.getById(id);
    const updated = { ...user, name: name.trim() };
    await this.#users.save(updated);
    return updated;
  }
}

console.log(await service.rename(1, "דנה כהן"));

// P12-E16
export type Product = { id: number; name: string; price: number; stock: number };
export type OrderLine = { productId: number; qty: number; unitPrice: number };
export type Order = { id: number; userId: number; createdAt: Date; lines: OrderLine[]; total: number };

export interface Repo<T extends { id: number }> {
  findById(id: number): Promise<T | null>;
  save(item: T): Promise<void>;
  count(): Promise<number>;
}

export class MemoryRepo<T extends { id: number }> implements Repo<T> {
  #items = new Map<number, T>();
  async findById(id: number) {
    const x = this.#items.get(id);
    return x ? structuredClone(x) : null;
  }
  async save(item: T) {
    this.#items.set(item.id, structuredClone(item));
  }
  async count() {
    return this.#items.size;
  }
}

export class NotFoundError extends Error {}
export class ConflictError extends Error {}
export class ValidationError extends Error {
  readonly field: string;
  constructor(field: string, message: string) {
    super(message);
    this.field = field;
  }
}

export class OrderService {
  #products: Repo<Product>;
  #orders: Repo<Order>;
  #now: () => Date;

  constructor(deps: { products: Repo<Product>; orders: Repo<Order>; now: () => Date }) {
    this.#products = deps.products;
    this.#orders = deps.orders;
    this.#now = deps.now;
  }

  async placeOrder(userId: number, items: { productId: number; qty: number }[]): Promise<Order> {
    if (items.length === 0) {
      throw new ValidationError("items", "הזמנה ריקה");
    }
    const products: Product[] = [];
    for (const { productId, qty } of items) {
      const product = await this.#products.findById(productId);
      if (!product) {
        throw new NotFoundError(`מוצר ${productId} לא נמצא`);
      }
      if (product.stock < qty) {
        throw new ConflictError(`אין מספיק מלאי למוצר ${productId}`);
      }
      products.push(product);
    }

    const lines: OrderLine[] = items.map((item, i) => ({ ...item, unitPrice: products[i].price }));
    for (let i = 0; i < items.length; i++) {
      await this.#products.save({ ...products[i], stock: products[i].stock - items[i].qty });
    }
    const total = Math.round(lines.reduce((s, l) => s + l.unitPrice * l.qty, 0) * 100) / 100;
    const order: Order = { id: (await this.#orders.count()) + 1, userId, createdAt: this.#now(), lines, total };
    await this.#orders.save(order);
    return order;
  }
}

const products = new MemoryRepo<Product>();
await products.save({ id: 1, name: "עט", price: 5, stock: 10 });
const service = new OrderService({ products, orders: new MemoryRepo<Order>(), now: () => new Date("2026-11-01T10:00:00Z") });
const order = await service.placeOrder(7, [{ productId: 1, qty: 3 }]);
console.log(order.total, order.lines[0].unitPrice, (await products.findById(1))?.stock);

// P12-E17
export interface Repo<T extends { id: number }> {
  findById(id: number): Promise<T | null>;
  save(item: T): Promise<void>;
}

export class CachedRepository<T extends { id: number }> implements Repo<T> {
  #inner: Repo<T>;
  #cache = new Map<number, T | null>();
  #hits = 0;
  #misses = 0;

  constructor(inner: Repo<T>) {
    this.#inner = inner;
  }

  get hits(): number {
    return this.#hits;
  }
  get misses(): number {
    return this.#misses;
  }

  async findById(id: number): Promise<T | null> {
    if (this.#cache.has(id)) {
      this.#hits++;
      const cached = this.#cache.get(id) ?? null;
      return cached ? { ...cached } : null;
    }
    this.#misses++;
    const item = await this.#inner.findById(id);
    this.#cache.set(id, item ? { ...item } : null);
    return item;
  }

  async save(item: T): Promise<void> {
    await this.#inner.save(item);
    this.#cache.set(item.id, { ...item });
  }
}

const calls: number[] = [];
const slow: Repo<{ id: number; v: string }> = {
  async findById(id) {
    calls.push(id);
    return id === 1 ? { id: 1, v: "a" } : null;
  },
  async save() {},
};

}

export class PricingService {
  #repo: ProductRepo;
  constructor(repo: ProductRepo) {
    this.#repo = repo;

P12-E28 · דוח הלקוחות הגדולים 🤖 💻 כתיבת קוד 🤖 רמה 3

💡 רמז 1

שלושה שלבים בתוך topCustomers: (1) await this.#orders.findAll() ← filter ← קיבוץ ל־Map<number, number> של userId ← סכום. (2) [...totals] (מערך של זוגות) ← sort ← slice(0, n). (3) findByIds על ה־ids שנבחרו, ו־Map מ־id לשם.

(כמו שראית בשיעור 1: את השדות כותבים במפורש — #orders: OrderRepository; ובתוך ה־constructor ‏this.#orders = orders;.)

💡 רמז 2

מיון זוגות: .sort(([idA, a], [idB, b]) => b - a || idA - idB) — אם ההפרש 0, ה־|| עובר להשוואה השנייה. בסוף: top.map(([id, total]) => ({ name: names.get(id) ?? "לא ידוע", total })).

✅ פתרון
// P12-E28: דוח הלקוחות הגדולים
export type Order = { id: number; userId: number; total: number; status: "paid" | "cancelled" };
export type User = { id: number; name: string };

export interface OrderRepository {
  findAll(): Promise<Order[]>;
}
export interface UserRepository {
  findByIds(ids: number[]): Promise<User[]>;
}

export class ReportService {
  #orders: OrderRepository;
  #users: UserRepository;

  constructor(orders: OrderRepository, users: UserRepository) {
    this.#orders = orders;
    this.#users = users;
  }

  async topCustomers(n: number): Promise<{ name: string; total: number }[]> {
    const orders = await this.#orders.findAll();

    const totals = new Map<number, number>();
    for (const o of orders.filter((o) => o.status === "paid")) {
      totals.set(o.userId, (totals.get(o.userId) ?? 0) + o.total);
    }

    const top = [...totals].sort(([idA, a], [idB, b]) => b - a || idA - idB).slice(0, n);

    const users = await this.#users.findByIds(top.map(([id]) => id));
    const names = new Map(users.map((u) => [u.id, u.name]));

    return top.map(([id, total]) => ({ name: names.get(id) ?? "לא ידוע", total }));
  }
}

  findAll: async () => [
    { id: 1, userId: 7, total: 100, status: "paid" },
    { id: 2, userId: 8, total: 300, status: "paid" },
    { id: 3, userId: 7, total: 250, status: "paid" },
    { id: 4, userId: 9, total: 999, status: "cancelled" },

ה־service לא יודע מאיפה מגיעים הנתונים: ממסד, מקובץ או מאובייקט מזויף בבדיקה. הוא יודע רק את ה־interface. בזכות זה הבודק יכול להזריק repository שסופר קריאות, ולוודא שלא נוצרה בעיית N+1: קריאה למסד לכל לקוח בנפרד, במקום קריאה אחת לכולם (תפגוש אותה שוב בפרקים 17 ו־19).

[...totals] הופך Map למערך של זוגות [userId, total], ומשם זה מיון רגיל (פרק 6). ‏new Map(users.map(...)) הוא הדרך הקצרה לבנות מילון מ־id לשם, כדי שהחיפוש בסוף לא יסרוק את המערך כל פעם מחדש.

⚠️ טעויות נפוצות
  • for (const [id] of top) { await this.#users.findByIds([id]); } — ‏N קריאות במקום אחת. הבודק סופר.
  • לשלוף שמות לכל הלקוחות לפני ה־slice — עובד, אבל מבקש מהמסד אלפי שמות כדי להציג חמישה.
  • .sort((a, b) => b[1] - a[1]) בלי שובר שוויון — הסדר בין לקוחות עם אותו סכום תלוי בסדר ההזמנות.
  • constructor(private orders: OrderRepository) — Node לא מריץ את הכתיב המקוצר הזה ישירות מקובץ .ts (שיעור 1).
