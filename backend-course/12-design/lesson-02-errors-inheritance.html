1. שגיאות משלך

class AppError extends Error {
  readonly status: number;
  readonly code: string;
  constructor(message: string, status: number, code: string) {
    super(message);
    this.name = new.target.name;
    this.status = status;
    this.code = code;
  }
}

class NotFoundError extends AppError {
  constructor(entity: string, id: number | string) {
    super(`${entity} ${id} לא נמצא`, 404, "NOT_FOUND");
  }
}

class ValidationError extends AppError {
  readonly field: string;
  constructor(field: string, message: string) {
    super(message, 400, "VALIDATION");
    this.field = field;
  }
}

function handle(err: unknown): string {
  if (err instanceof ValidationError) {
    return `400 — ${err.field}: ${err.message}`;
  }
  if (err instanceof AppError) {
    return `${err.status} — ${err.message}`;
  }
  return "500 — שגיאה פנימית";
}

console.log(handle(new NotFoundError("הזמנה", 77)));
console.log(handle(new ValidationError("email", "אימייל לא תקין")));
console.log(handle(new TypeError("x is undefined")));

זה בדיוק toHttpResponse מפרק 8 — אבל במקום לבדוק שדות, כל סוג שגיאה "יודע" את ה־status שלו. NestJS מגיע עם מחלקות כאלה מוכנות (NotFoundException, BadRequestException).

שים לב לסדר ב־handle: הבדיקה הספציפית (ValidationError) לפני הכללית (AppError), כי כל ValidationError היא גם AppError — בדיוק כמו בסדר של else if בפרק 3.

2. ירושה: כלי חזק, להשתמש בזהירות

הבעיה עם ירושה עמוקה (AdminUser extends PremiumUser extends User): שינוי במחלקת אב שובר את כל הילדים, וקשה לדעת מאיפה מגיעה כל התנהגות. הכלל המקובל: "העדף הרכבה על ירושה". ירושה — בעיקר להיררכיות קטנות ויציבות, כמו שגיאות. כל השאר — אובייקט שמחזיק אובייקטים אחרים (שיעור 3).
