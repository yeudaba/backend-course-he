function parse(input: unknown) {
  if (typeof input !== "object" || input === null) throw new Error("לא אובייקט");
  const { productId, qty, note } = input as Record<string, unknown>;
  const double = qty * 2;
  const upper = note.toUpperCase();
  if (typeof qty === "number" && typeof note === "string") {
    return { productId, qty: qty * 2, note: note.toUpperCase() };
  }
  throw new Error("שדות לא תקינים");
}

  • resolve(name) — יוצר פעם אחת בלבד (singleton) ומחזיר אותו מופע בכל קריאה. שם לא רשום ← Error(השירות ${name} לא רשום).
  • זיהוי מעגל (A צריך את B שצריך את A) ← Error("תלות מעגלית: A -> B -> A").

🔗 איך זה מתחבר: Map מפרק 7 לרישום ולמטמון, רקורסיה מפרק 5 (factory שקוראת ל־resolve), ו־Set לזיהוי תלות מעגלית. מהפרק: DI (שיעור 3, P12-E57) — בדיוק מה ש־NestJS עושה בשבילך בפרק 20.
