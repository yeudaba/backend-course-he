class Notifier {
  send(to: string): string {
    return `הודעה ל־${to}`;

}

P12-E49 · שגיאה משלך מתוך async

🧠 הבנהרמה 2⏱ 10 דק׳ידע נדרש: פרק 10, שיעור 2

בלי להריץ — מה יודפס?

class NotFoundError extends Error {
  constructor(what: string) {
    super(`${what} לא נמצא`);
    this.name = new.target.name;
  }
}

async function findUser(id: number): Promise<string> {
  if (id !== 1) throw new NotFoundError(`משתמש ${id}`);
  return "דנה";
}

async function main() {
  const results = await Promise.allSettled([findUser(1), findUser(2)]);
  for (const r of results) {
    if (r.status === "fulfilled") console.log("נמצא:", r.value);
    else console.log(r.reason instanceof NotFoundError, r.reason.message);
  }
  try {
    await findUser(3);
  } catch (err) {
    console.log(err instanceof NotFoundError ? "404" : "500");
  }
}
main();

🔗 איך זה מתחבר: פרק 10: throw בתוך async הופך ל־Promise שנכשל, ו־allSettled שם את השגיאה ב־reason. השיעור: השגיאה שומרת על המחלקה שלה גם "בדרך" — ו־instanceof עובד בצד השני.

P12-E50 · switch על code מול instanceof

🧠 הבנהרמה 2⏱ 15 דק׳ידע נדרש: פרק 3, פרק 8, שיעור 2

שתי דרכים לטפל בשגיאות: לפי code (כמו ב־Node: ENOENT) או לפי מחלקה. בלי להריץ — מה יודפס?

class AppError extends Error {
  readonly code: string;
  constructor(message: string, code: string) {
    super(message);
    this.code = code;
  }
}
class ConflictError extends AppError {
  constructor(message: string) {
    super(message, "CONFLICT");
  }
}

function byCode(err: unknown): number {
  const code = (err as { code?: string }).code;
  switch (code) {
    case "NOT_FOUND": return 404;
    case "CONFLICT": return 409;
    case "ENOENT": return 404;
    default: return 500;
  }
}

function byClass(err: unknown): number {
  if (err instanceof ConflictError) return 409;
  if (err instanceof AppError) return 400;
  return 500;
}

const errors: unknown[] = [
  new ConflictError("אימייל קיים"),
  new AppError("x", "NOT_FOUND"),
  Object.assign(new Error("no file"), { code: "ENOENT" }),
  new TypeError("bug"),
];
console.log(errors.map(byCode));
console.log(errors.map(byClass));

🔗 איך זה מתחבר: switch מפרק 3, ו־err.code מפרק 8 ומפרק 10 (ENOENT של fs). השיעור: instanceof על היררכיה. בקוד אמיתי תפגוש את שתיהן — שגיאות של ספריות מגיעות בדרך כלל עם code.

class CounterMap extends Map<string, number> {
  get(key: string): number {
    return super.get(key) ?? 0;
  }
  inc(key: string): void {
    this.set(key, this.get(key) + 1);
  }
}

const m = new CounterMap();
m.inc("a");
m.inc("a");
console.log(m.get("a"), m.get("zzz"));
console.log(m.has("zzz"), m.size);

  • class NothingToUndoError ו־class NothingToRedoError — שתיהן extends Error, עם הודעות "אין מה לבטל" ו־"אין מה לבצע שוב", ו־name נכון.
  • class History:
    • push(action: string): void — פעולה חדשה. מוחקת את כל מה שאפשר היה לבצע שוב.
    • undo(): string — מבטלת את הפעולה האחרונה ומחזירה אותה. אין ← NothingToUndoError.
    • redo(): string — מחזירה את הפעולה האחרונה שבוטלה ומחזירה אותה. אין ← NothingToRedoError.
    • getter ‏done — עותק של הפעולות שבתוקף, מהישנה לחדשה.
