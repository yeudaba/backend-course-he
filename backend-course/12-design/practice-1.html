class Counter {
  #count = 0;
  readonly name: string;
  constructor(name: string) {
    this.name = name;
  }
  get value(): number {
    return this.#count;
  }
  inc(): this {
    this.#count++;
    return this;
  }
}
const a = new Counter("a");
const b = new Counter("b");
a.inc().inc();
b.inc();
const c = a;
c.inc();
console.log(a.value, b.value, c.value, a === c, a.name);

class Ticket {
  static issued = 0;
  readonly id: number;
  #used = false;

  constructor() {
    Ticket.issued++;
    this.id = Ticket.issued;
  }

    if (this.#used) return `כרטיס ${this.id} כבר נוצל`;
    this.#used = true;
    return `כרטיס ${this.id} נוצל`;
  }
}

class Cart {

P12-E38 · this בתוך setTimeout

🧠 הבנהרמה 2⏱ 10 דק׳ידע נדרש: פרק 10, שיעור 1

בלי להריץ — מה יודפס? (שים לב להבדל בין שתי המתודות.)

class Reminder {
  readonly text: string;
  constructor(text: string) {
    this.text = text;
  }
  withArrow(): void {
    setTimeout(() => {
      console.log("arrow:", this.text);
    }, 10);
  }
  withFunction(): void {
    setTimeout(function (this: any) {
      console.log("function:", this.text);
    }, 20);
  }
}

const r = new Reminder("לקנות חלב");
r.withArrow();
r.withFunction();
console.log("נרשמו שני טיימרים");

🔗 איך זה מתחבר: פרק 10: ה־callback של setTimeout רץ אחר כך, מתוך ה־event loop — לא "מתוך" המתודה. השיעור: this הוא "המופע הזה" — אבל רק אם הפונקציה יודעת מי המופע. פונקציית חץ לוקחת את ה־this מהמקום שבו היא נכתבה; function רגילה — מי שקרא לה.

P12-E27 · Money: אובייקט ערך 🤖

💻 כתיבת קוד 🤖🔁 חזרהרמה 3⏱ 20 דק׳ידע נדרש: שיעור 1

🔁 חזרה על אגורות (פרק 2, ‏P08-E26) ועל פסיקים במספרים (P05-E25), הפעם כמחלקה. בשיעור 1 ראית Money קטנה. כתוב גרסה מלאה, ששומרת אגורות שלמות ואי אפשר לשנות אותה (immutable):

  • static fromAgorot(agorot: number): Money — אם agorot אינו מספר שלם, זורקת RangeError("סכום באגורות חייב להיות מספר שלם").

const price = Money.fromShekels(1234.5);
String(price);                         // "₪1,234.50"
String(price.times(3));                // "₪3,703.50"
Money.fromShekels(0.1).plus(Money.fromShekels(0.2)).equals(Money.fromAgorot(30)); // true

P12-E40 · תור משימות אסינכרוני 🤖

💻 כתיבת קוד 🤖✍️ מאפסרמה 3⏱ 20 דק׳ידע נדרש: פרק 10, שיעור 1

כתוב וייצא מחלקה TaskQueue שמריצה משימות אסינכרוניות אחת אחרי השנייה, גם כשמוסיפים אותן בבת אחת (למשל כדי שכתיבות לקובץ לא יתערבבו):

  • add<T>(task: () => Promise<T>): Promise<T> — מכניסה לתור, ומחזירה Promise שמסתיים עם התוצאה של המשימה (או נכשל עם השגיאה שלה).
  • משימה מתחילה רק אחרי שהקודמת הסתיימה — גם אם הקודמת נכשלה.
  • getter ‏pending — כמה משימות עוד לא הסתיימו (כולל זו שרצה עכשיו).
const q = new TaskQueue();
const a = q.add(async () => { await sleep(30); return "א"; });
const b = q.add(async () => "ב");   // מתחיל רק אחרי שא הסתיים
