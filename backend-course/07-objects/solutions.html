P07-E26 · mapValues עם Object.fromEntries 🤖 🔥 חימום רמה 2

💡 רמז 1

שלושה שלבים: Object.entries(obj) ← .map(...) ← Object.fromEntries(...).

💡 רמז 2

ב־map מקבלים זוג ומחזירים זוג: ([key, value]) => [key, fn(value)].

✅ פתרון
// P07-E26: mapValues
function mapValues(obj, fn) {
  return Object.fromEntries(Object.entries(obj).map(([key, value]) => [key, fn(value)]));
}

const prices = { keyboard: 100, mouse: 50 };
console.log(mapValues(prices, (p) => p * 2));
console.log(prices);

module.exports = { mapValues };
{ keyboard: 200, mouse: 100 }
{ keyboard: 100, mouse: 50 }

ה־map מקבל כל זוג ומפרק אותו מיד בפרמטר: ([key, value]). זה destructuring של מערך, כמו ב־for (const [item, qty] of Object.entries(stock)) משיעור 3. הוא מחזיר זוג חדש, ו־fromEntries מרכיב ממנו אובייקט חדש. בשום שלב לא נגענו ב־obj, ולכן הוא לא השתנה.

את אותו דפוס אפשר להפעיל גם עם filter באמצע. למשל, כך מסירים שדות ריקים מ־query string: Object.fromEntries(Object.entries(q).filter(([, v]) => v !== "")).

⚠️ טעויות נפוצות
  • .map((pair) => fn(pair)) — fn מקבלת את כל הזוג (["keyboard", 100]), ולא את הערך.
  • .map(([k, v]) => ({ [k]: fn(v) })) — יוצא מערך של אובייקטים קטנים, ו־fromEntries לא יודעת מה לעשות איתם.
  • for (const k in obj) obj[k] = fn(obj[k]) — עובד, אבל משנה את המקור.

P07-E27 · מלאי פרטי (closure) 🤖 🔁 חזרה רמה 3

💡 רמז 1

const items = new Map(); בתוך createStock, ואחר כך return { add, take, count }; — אובייקט ששלוש הפונקציות שבו "זוכרות" את items.

💡 רמז 2

count היא items.get(item) ?? 0, ושתי האחרות יכולות להשתמש בה. take מתחילה ב־guard: if (count(item) < qty) return false;.

✅ פתרון
// P07-E27: מלאי פרטי
function createStock() {
  const items = new Map();

  function count(item) {
    return items.get(item) ?? 0;
  }

  function add(item, qty) {
    items.set(item, count(item) + qty);
  }

  function take(item, qty) {
    if (count(item) < qty) return false;
    items.set(item, count(item) - qty);
    return true;
  }

  return { add, take, count };
}

const stock = createStock();
stock.add("pen", 5);
console.log(stock.take("pen", 2), stock.take("pen", 10), stock.count("pen"), stock.count("cup"));
console.log(stock.items);

module.exports = { createStock };

זה אותו רעיון כמו המונה בפרק 5, עם שלוש פונקציות במקום אחת. כולן נוצרו בתוך אותה קריאה ל־createStock, ולכן כולן רואות את אותו items. השורה stock.items מחזירה undefined: מבחוץ אין שום דרך לגשת ל־Map. הדרך היחידה לשנות את המלאי היא דרך add ו־take, וכך אף אחד לא יכול להכניס כמות שלילית "בטעות".

בפרק 12 תכתוב את אותו דבר כ־class עם שדה פרטי (#items). זה אותו רעיון בתחביר אחר.

⚠️ טעויות נפוצות
  • const items = new Map() מחוץ ל־createStock — כל המלאים חולקים את אותו Map. הבודק תופס את זה.
  • items.get(item) + qty לפריט חדש — undefined + 5 הוא NaN. בגלל זה ?? 0.
  • ב־take: להוריד קודם ולבדוק אחר כך — הכמות יורדת מתחת ל־0 גם כשמחזירים false.

P07-E28 · יתרה מאירועים 🤖 🔁 חזרה רמה 3

💡 רמז 1

events.reduce((balance, e) => { switch (e.type) { ... } }, 0). כל case מחזיר את היתרה החדשה.

💡 רמז 2

case "withdraw": return e.amount <= balance ? balance - e.amount : balance; ו־default: return balance;.

✅ פתרון
// P07-E28: יתרה מאירועים
const FEE = 5;

function balanceOf(events) {
  return events.reduce((balance, e) => {
    switch (e.type) {
      case "deposit":
        return balance + e.amount;
      case "withdraw":
        return e.amount <= balance ? balance - e.amount : balance;
      case "fee":
        return balance - FEE;
      default:
        return balance;
    }
  }, 0);
}

console.log(
  balanceOf([
    { type: "deposit", amount: 100 },
    { type: "withdraw", amount: 30 },
    { type: "withdraw", amount: 500 },
    { type: "fee" },
    { type: "bonus", amount: 1000 },
  ])
);

module.exports = { balanceOf };
65

ה־reducer מקבל "היתרה עד עכשיו" ואירוע אחד, ומחזיר את היתרה אחרי האירוע. ה־switch מחליט איך. כל case מסתיים ב־return, ולכן אין צורך ב־break (כמו ב־P05-E27). ה־default חשוב במיוחד כאן: בלעדיו, אירוע לא מוכר מחזיר undefined, וכל האירועים שאחריו מחושבים על undefined ונותנים NaN.

הדפוס הזה נקרא event sourcing: שומרים את מה שקרה, ומחשבים את המצב ממנו. כך עובדים הרבה מערכות פיננסיות, כי ההיסטוריה המלאה נשמרת תמיד.

⚠️ טעויות נפוצות
  • לשכוח את default — אירוע לא מוכר הופך את היתרה ל־undefined, ואחר כך ל־NaN.
  • לשכוח את ה־0 בסוף ה־reduce — האירוע הראשון (אובייקט) הופך ליתרה ההתחלתית.
  • e.amount < balance — משיכה של כל היתרה (בדיוק) נדחית.

P07-E12 · query string 🤖 רמה 2

💡 רמז 1

💡 רמז 2

// P07-E12: אובייקט ⇄ query string
function toQueryString(params) {
  return Object.entries(params)
    .filter(([, value]) => value !== null && value !== undefined)
    .map(([key, value]) => `${key}=${value}`)
    .join("&");
}

function parseQueryString(text) {
  const clean = text.startsWith("?") ? text.slice(1) : text;
  const result = {};
  if (clean === "") {
    return result;
  }
  for (const pair of clean.split("&")) {
    const [key, value = ""] = pair.split("=");
    result[key] = value;
  }
  return result;
}

console.log(toQueryString({ page: 2, sort: "price", q: null }));
console.log(parseQueryString("?page=2&sort=price"));

module.exports = { toQueryString, parseQueryString };

{ page: '2', sort: 'price' }

בעולם האמיתי יש URLSearchParams מובנה, שגם מטפל בתווים מיוחדים ובעברית (קידוד %D7%A9...). נשתמש בו בפרק 14. וכאן רואים שוב: פרמטרים מה־URL הם תמיד מחרוזות.

⚠️ טעויות נפוצות

P07-E13 · סכום לפי קטגוריה 🤖 רמה 3

// P07-E13: קיבוץ (group by)
function totalsByCategory(sales) {
  const result = {};
  for (const { category, amount } of sales) {
    result[category] = (result[category] ?? 0) + amount;
  }
  return result;
}

function countByStatus(orders) {
  const result = {};
  for (const order of orders) {
    result[order.status] = (result[order.status] ?? 0) + 1;
  }
  return result;
}

function topCategory(sales) {
  let best = null;
  let bestTotal = -Infinity;
  for (const [category, total] of Object.entries(totalsByCategory(sales))) {
    if (total > bestTotal) {
      best = category;
      bestTotal = total;
    }
  }
  return best;
}

const sales = [
  { category: "ספרים", amount: 80 },
  { category: "מחשבים", amount: 1200 },
  { category: "ספרים", amount: 45 },
];
console.log(totalsByCategory(sales), topCategory(sales));

function applyOrder(inventory, order) {
  const next = { ...inventory };
  for (const { code, qty } of order.items) {
    if (!(code in next)) {
      return { ok: false, inventory, error: `מוצר ${code} לא קיים` };
    }
    if (next[code] < qty) {
      return { ok: false, inventory, error: `אין מספיק מלאי של ${code} (נשארו ${next[code]})` };
    }
    next[code] -= qty;

}

// P07-E25: twoSum ב־O(n) ואנגרמות
function twoSumFast(nums, target) {
  const seen = new Map();
  for (let j = 0; j < nums.length; j++) {
    const need = target - nums[j];
    if (seen.has(need)) {
      return [seen.get(need), j];
    }
    if (!seen.has(nums[j])) {
      seen.set(nums[j], j);
    }
  }
  return null;
}

function groupAnagrams(words) {
  const groups = new Map();
  for (const w of words) {
    const key = w.split("").sort().join("");
    if (!groups.has(key)) {
      groups.set(key, []);
    }
    groups.get(key).push(w);
  }
  return [...groups.values()];
}
