P07-E47 · שלוש דרכים לספור 🧠 הבנה רמה 2

💡 רמז 1

גרסה b: מה undefined + 1? (וב־++ זה בדיוק מה שקורה.)

💡 רמז 2

גרסה c: + חזק יותר מ־?? — איך JavaScript שמה סוגריים?

✅ פתרון
const orders = [{ status: "paid" }, { status: "new" }, { status: "paid" }];

const a = {};
for (const o of orders) {
  a[o.status] = (a[o.status] ?? 0) + 1;
}

const b = {};
for (const o of orders) {
  b[o.status]++;
}

const c = {};
for (const o of orders) {
  c[o.status] = c[o.status] ?? 0 + 1;
}

console.log(a, b, c);
{ paid: 2, new: 1 } { paid: NaN, new: NaN } { paid: 1, new: 1 }

רק a נכונה. ב־b הערך הראשון הוא undefined, ו־undefined + 1 הוא NaN — ומשם NaN לתמיד. ב־c, ‏c[o.status] ?? 0 + 1 נקרא כ־c[o.status] ?? (0 + 1): בפעם הראשונה 1, ובפעם השנייה — הערך הקיים (1), בלי להוסיף. הסוגריים ב־a הם לא קישוט.

⚠️ טעויות נפוצות
  • לבדוק את הספירה רק עם סטטוס שמופיע פעם אחת — גם c נראית תקינה.

P07-E13 · סכום לפי קטגוריה 🤖 💻 כתיבת קוד 🤖 רמה 3

💡 רמז 1

result[key] = (result[key] ?? 0) + amount;

💡 רמז 2

topCategory: Object.entries(totals) ← מקסימום לפי הערך.

✅ פתרון
// P07-E13: קיבוץ (group by)
function totalsByCategory(sales) {
  const result = {};
  for (const { category, amount } of sales) {
    result[category] = (result[category] ?? 0) + amount;
  }
  return result;
}

function countByStatus(orders) {
  const result = {};
  for (const order of orders) {
    result[order.status] = (result[order.status] ?? 0) + 1;
  }
  return result;
}

function topCategory(sales) {
  let best = null;
  let bestTotal = -Infinity;
  for (const [category, total] of Object.entries(totalsByCategory(sales))) {
    if (total > bestTotal) {
      best = category;
      bestTotal = total;
    }
  }
  return best;
}

const sales = [
  { category: "ספרים", amount: 80 },
  { category: "מחשבים", amount: 1200 },
  { category: "ספרים", amount: 45 },
];
console.log(totalsByCategory(sales), topCategory(sales));

module.exports = { totalsByCategory, countByStatus, topCategory };
{ 'ספרים': 125, 'מחשבים': 1200 } מחשבים

(destructuring ישר בתוך ה־for...of: for (const { category, amount } of sales).)

⚠️ טעויות נפוצות
  • result[category] += amount בפעם הראשונה ← undefined + 80 ← NaN.

// P07-E14: סיכום עגלה
function cartSummary(cart) {
  const itemCount = cart.reduce((s, it) => s + it.qty, 0);
  const subtotal = cart.reduce((s, it) => s + it.price * it.qty, 0);
  const top = cart.reduce((best, it) => (best === null || it.price > best.price ? it : best), null);
  const lines = cart.map((it) => `${it.name} × ${it.qty} = ${it.price * it.qty}`);
  return { itemCount, subtotal, mostExpensive: top?.name ?? null, lines };
}

P07-E51 · מלאי נמוך 🤖 💻 כתיבת קוד 🤖 רמה 3

function lowStock(products, threshold = 3) {
  return products
    .filter((p) => p.stock <= threshold)
    .map((p) => ({ name: p.name, stock: p.stock }))
    .sort((a, b) => a.stock - b.stock || a.name.localeCompare(b.name));
}

const products = [

// P07-E28: יתרה מאירועים
const FEE = 5;

function balanceOf(events) {
  return events.reduce((balance, e) => {
    switch (e.type) {
      case "deposit":
        return balance + e.amount;
      case "withdraw":
        return e.amount <= balance ? balance - e.amount : balance;
      case "fee":
        return balance - FEE;
      default:
        return balance;
    }
  }, 0);
}

console.log(
  balanceOf([
    { type: "deposit", amount: 100 },
    { type: "withdraw", amount: 30 },
    { type: "withdraw", amount: 500 },
    { type: "fee" },
    { type: "bonus", amount: 1000 },
  ])
);

module.exports = { balanceOf };

ה־reducer מקבל "היתרה עד עכשיו" ואירוע אחד, ומחזיר את היתרה אחרי האירוע. ה־switch מחליט איך. כל case מסתיים ב־return, ולכן אין צורך ב־break (כמו ב־P05-E27). ה־default חשוב במיוחד כאן: בלעדיו, אירוע לא מוכר מחזיר undefined, וכל האירועים שאחריו מחושבים על undefined ונותנים NaN.

הדפוס הזה נקרא event sourcing: שומרים את מה שקרה, ומחשבים את המצב ממנו. כך עובדים הרבה מערכות פיננסיות, כי ההיסטוריה המלאה נשמרת תמיד.

⚠️ טעויות נפוצות
  • לשכוח את default — אירוע לא מוכר הופך את היתרה ל־undefined, ואחר כך ל־NaN.
  • לשכוח את ה־0 בסוף ה־reduce — האירוע הראשון (אובייקט) הופך ליתרה ההתחלתית.
  • e.amount < balance — משיכה של כל היתרה (בדיוק) נדחית.

function productErrors(p) {
  const errors = [];
  if ((p.name ?? "").trim() === "") {
    errors.push("חסר שם");
  }
  if (typeof p.price !== "number" || Number.isNaN(p.price) || p.price <= 0) {
    errors.push("מחיר לא תקין");
  }
  if (!Number.isInteger(p.stock) || p.stock < 0) {
    errors.push("מלאי לא תקין");
  }
  return errors;
}

function validateAll(products) {
  const messages = [];
  for (let i = 0; i < products.length; i++) {
    for (const error of productErrors(products[i])) {
      messages.push(`שורה ${i + 1}: ${error}`);
    }
  }
  return messages;
}

console.log(validateAll([
  { name: "עט", price: 5, stock: 10 },
  { name: "  ", price: "5", stock: 1 },
  { name: "תיק", price: 120, stock: -2 },
]));
