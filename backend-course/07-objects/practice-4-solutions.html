P07-E18 · ספירת מילים 🤖 💻 כתיבת קוד 🤖 רמה 3

💡 רמז 1

ניקוי: text.toLowerCase().replaceAll(".", "")... ואז split ו־filter של ריקים.

💡 רמז 2

topWords: [...map] נותן מערך של זוגות; מיון: (a, b) => b[1] - a[1] || a[0].localeCompare(b[0]).

✅ פתרון
// P07-E18: תדירות מילים עם Map
function words(text) {
  let clean = text.toLowerCase();
  for (const mark of [".", ",", "!", "?"]) {
    clean = clean.replaceAll(mark, " ");
  }
  return clean.split(" ").filter((w) => w !== "");
}

function wordFrequencies(text) {
  const counts = new Map();
  for (const w of words(text)) {
    counts.set(w, (counts.get(w) ?? 0) + 1);
  }
  return counts;
}

function topWords(text, n) {
  return [...wordFrequencies(text)]
    .sort((a, b) => b[1] - a[1] || a[0].localeCompare(b[0]))
    .slice(0, n);
}

console.log(topWords("Node is fast. Node is fun! JS is everywhere, is it?", 3));

module.exports = { wordFrequencies, topWords };
[ [ 'is', 4 ], [ 'node', 2 ], [ 'everywhere', 1 ] ]

b[1] - a[1] || ...: אם ההפרש 0 (תיקו, falsy) — עוברים להשוואה השנייה. דפוס נפוץ למיון לפי כמה שדות.

⚠️ טעויות נפוצות
  • להחליף סימני פיסוק ב־"" — "fast.Node" (בלי רווח) יהפוך למילה אחת. עדיף רווח.

💡 רמז 2

totalsByCustomer: Map של id ← סכום, ואז המר למערך עם השמות.

✅ פתרון
// P07-E19: חיבור נתונים לפי מזהה
const indexById = (list) => new Map(list.map((x) => [x.id, x]));

function attachCustomers(orders, customers) {
  const byId = indexById(customers);
  return orders.map((o) => ({ ...o, customerName: byId.get(o.customerId)?.name ?? null }));
}

function totalsByCustomer(orders, customers) {
  const byId = indexById(customers);
  const totals = new Map();
  for (const o of orders) {
    if (byId.has(o.customerId)) {
      totals.set(o.customerId, (totals.get(o.customerId) ?? 0) + o.total);
    }
  }
  return [...totals]
    .map(([id, total]) => ({ name: byId.get(id).name, total }))
    .sort((a, b) => b.total - a.total);
}

const customers = [{ id: 1, name: "דנה" }, { id: 2, name: "עומר" }];
const orders = [
  { id: 7, customerId: 2, total: 320 },
  { id: 8, customerId: 1, total: 150 },
  { id: 9, customerId: 2, total: 80 },
  { id: 10, customerId: 3, total: 40 },
];
console.log(attachCustomers(orders, customers)[3], totalsByCustomer(orders, customers));

module.exports = { attachCustomers, totalsByCustomer };
{ id: 10, customerId: 3, total: 40, customerName: null } [ { name: 'עומר', total: 400 }, { name: 'דנה', total: 150 } ]

P07-E23 · Map שלא מוצא 🤖 💻 כתיבת קוד 🤖 רמה 4

💡 רמז 1

איך מכניסים ל־Map? map[key] = value או map.set(key, value)?

💡 רמז 2

המזהים במלאי הם מספרים; getStock מקבלת מזהה מ־URL — מחרוזת. ב־Map, 2 ו־"2" הם מפתחות שונים.

✅ פתרון
// P07-E23: מתוקן
function stockLookup(items) {
  const map = new Map();
  for (const item of items) {
    map.set(item.id, item.stock);
  }
  return map;
}

function getStock(lookup, idFromUrl) {
  return lookup.get(Number(idFromUrl)) ?? 0;
}

const lookup = stockLookup([{ id: 1, stock: 5 }, { id: 2, stock: 0 }]);
console.log(getStock(lookup, "1"), getStock(lookup, "2"), getStock(lookup, "9"), lookup.size);

module.exports = { stockLookup, getStock };

(1) map[item.id] = ... יוצר שדה רגיל על אובייקט ה־Map, לא רשומה בו — size נשאר 0 ו־get לא מוצא. (2) המרה מפורשת של המזהה מה־URL. את הבאג הזה תפגוש באמת: req.params.id הוא תמיד מחרוזת (פרק 15).

⚠️ טעויות נפוצות
  • לפתור את (2) ע״י המרת המפתחות ל־String בבנייה — עובד, אבל עדיף לשמור מספרים כמספרים ולהמיר את הקלט בגבול.

P07-E27 · מלאי פרטי (closure) 🤖 💻 כתיבת קוד 🤖 רמה 3

💡 רמז 1

const items = new Map(); בתוך createStock, ואחר כך return { add, take, count }; — אובייקט ששלוש הפונקציות שבו "זוכרות" את items.

💡 רמז 2

count היא items.get(item) ?? 0, ושתי האחרות יכולות להשתמש בה. take מתחילה ב־guard: if (count(item) < qty) return false;.

✅ פתרון
// P07-E27: מלאי פרטי
function createStock() {
  const items = new Map();

  function count(item) {
    return items.get(item) ?? 0;
  }

  function add(item, qty) {
    items.set(item, count(item) + qty);
  }

  function take(item, qty) {
    if (count(item) < qty) return false;
    items.set(item, count(item) - qty);
    return true;
  }

  return { add, take, count };
}

const stock = createStock();
stock.add("pen", 5);
console.log(stock.take("pen", 2), stock.take("pen", 10), stock.count("pen"), stock.count("cup"));
console.log(stock.items);

module.exports = { createStock };
true false 3 0
undefined

זה אותו רעיון כמו המונה בפרק 5, עם שלוש פונקציות במקום אחת. כולן נוצרו בתוך אותה קריאה ל־createStock, ולכן כולן רואות את אותו items. השורה stock.items מחזירה undefined: מבחוץ אין שום דרך לגשת ל־Map. הדרך היחידה לשנות את המלאי היא דרך add ו־take, וכך אף אחד לא יכול להכניס כמות שלילית "בטעות".

בפרק 12 תכתוב את אותו דבר כ־class עם שדה פרטי (#items). זה אותו רעיון בתחביר אחר.

⚠️ טעויות נפוצות
  • const items = new Map() מחוץ ל־createStock — כל המלאים חולקים את אותו Map. הבודק תופס את זה.
  • items.get(item) + qty לפריט חדש — undefined + 5 הוא NaN. בגלל זה ?? 0.
  • ב־take: להוריד קודם ולבדוק אחר כך — הכמות יורדת מתחת ל־0 גם כשמחזירים false.

P07-E24 · אתגר: מלאי והזמנות 🤖 💻 כתיבת קוד 🤖 רמה 5

💡 רמז 1

applyOrder: קודם בודקים את כל הפריטים מול עותק, ורק אם הכול תקין — מחזירים אותו. עותק: { ...inventory } (ערכים מספריים — רדוד מספיק).

💡 רמז 2

עבוד על העותק: לכל פריט — אם !(code in copy) ← שגיאה; אם copy[code] < qty ← שגיאה; אחרת copy[code] -= qty. כך פריט כפול נבדק מול הכמות שכבר הורדה.

✅ פתרון
// P07-E24: מלאי והזמנות — הכול או כלום
function applyOrder(inventory, order) {
  const next = { ...inventory };
  for (const { code, qty } of order.items) {
    if (!(code in next)) {
      return { ok: false, inventory, error: `מוצר ${code} לא קיים` };
    }
    if (next[code] < qty) {
      return { ok: false, inventory, error: `אין מספיק מלאי של ${code} (נשארו ${next[code]})` };
    }
    next[code] -= qty;
  }
  return { ok: true, inventory: next };
}

function processOrders(inventory, orders) {
  let current = inventory;
  const accepted = [];
  const rejected = [];
  for (const order of orders) {
    const result = applyOrder(current, order);
    if (result.ok) {
      current = result.inventory;
      accepted.push(order.id);
    } else {
      rejected.push({ id: order.id, error: result.error });
    }
  }
  return { inventory: current, accepted, rejected };
}

const inventory = { A: 5, B: 1 };
const orders = [
  { id: 1, items: [{ code: "A", qty: 2 }, { code: "B", qty: 1 }] },
  { id: 2, items: [{ code: "A", qty: 2 }, { code: "B", qty: 1 }] },
  { id: 3, items: [{ code: "A", qty: 2 }, { code: "A", qty: 2 }] },
  { id: 4, items: [{ code: "Z", qty: 1 }] },
];
console.log(processOrders(inventory, orders), inventory);

module.exports = { applyOrder, processOrders };

  inventory: { A: 3, B: 0 },

    { id: 3, error: 'אין מספיק מלאי של A (נשארו 1)' },
    { id: 4, error: 'מוצר Z לא קיים' }

(code in next — האם השדה קיים באובייקט, גם אם ערכו 0.) "הכול או כלום" — זה בדיוק מה שtransaction עושה במסד נתונים (פרק 18), ומה שפרויקט הדגל שלך יצטרך: הזמנה של 3 פריטים שאחד מהם אזל — לא מורידים חלקית.

⚠️ טעויות נפוצות
  • להוריד מהמלאי תוך כדי בדיקה על המקור — אם פריט שלישי נכשל, שני הראשונים כבר הורדו.
  • if (!next[code]) לבדיקת קיום — מוצר עם מלאי 0 ייחשב "לא קיים".

P07-E25 · אתגר: twoSum מהיר ואנגרמות 🤖 💻 כתיבת קוד 🤖 רמה 5

💡 רמז 1

twoSumFast: Map של ערך ← אינדקס. לכל nums[j]: אם target - nums[j] כבר במפה — מצאנו. אחרת מוסיפים את nums[j].

💡 רמז 2

אנגרמות: "חתימה" של מילה = האותיות ממוינות: w.split("").sort().join(""). Map של חתימה ← מערך מילים.

✅ פתרון
// P07-E25: twoSum ב־O(n) ואנגרמות
function twoSumFast(nums, target) {
  const seen = new Map();
  for (let j = 0; j < nums.length; j++) {
    const need = target - nums[j];
    if (seen.has(need)) {
      return [seen.get(need), j];
    }
    if (!seen.has(nums[j])) {
      seen.set(nums[j], j);
    }
  }
  return null;
}

function groupAnagrams(words) {
  const groups = new Map();
  for (const w of words) {
    const key = w.split("").sort().join("");
    if (!groups.has(key)) {
      groups.set(key, []);
    }
    groups.get(key).push(w);
  }
  return [...groups.values()];
}

console.log(twoSumFast([2, 7, 11, 15], 9), twoSumFast([3, 3], 6));
console.log(groupAnagrams(["eat", "tea", "tan", "ate", "nat", "bat"]));

module.exports = { twoSumFast, groupAnagrams };
[ 0, 1 ] [ 0, 1 ]
[ [ 'eat', 'tea', 'ate' ], [ 'tan', 'nat' ], [ 'bat' ] ]

הדפוס "Map של מה שכבר ראיתי" פותר עשרות שאלות ראיון. ובראיון: "הפתרון הראשון הוא O(n²) בזמן ו־O(1) בזיכרון; השני O(n) בזמן אבל O(n) בזיכרון — זו הפשרה."

⚠️ טעויות נפוצות
  • לבנות את כל ה־Map מראש ואז לחפש — [3, 3] עם target 6 עלול להחזיר [0, 0] או [1, 1].
