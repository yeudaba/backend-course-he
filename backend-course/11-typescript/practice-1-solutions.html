// P11-E06
export function formatPrice(amount: number, currency = "₪"): string {
  return `${currency}${amount.toFixed(2)}`;
}

export function isAdult(age: number): boolean {
  return age >= 18;
}

export function average(nums: number[]): number {
  if (nums.length === 0) {
    return 0;
  }
  return nums.reduce((s, n) => s + n, 0) / nums.length;
}

console.log(formatPrice(12.5), isAdult(17), average([1, 2, 3]));

chunk<T>(arr: T[], size: number): T[][] — אפשר גם בלי generic (number[]), אבל עם generic זה עובד לכל מערך.

✅ פתרון
// P11-E07: אותה לוגיקה מפרק 6, עם טיפוסים
export function cleanTags(input: string, max = 5): string[] {
  return input
    .split(",")
    .map((tag) => tag.trim().toLowerCase())
    .filter((tag) => tag !== "")
    .filter((tag, i, all) => all.indexOf(tag) === i)
    .slice(0, max);
}

export function chunk<T>(arr: T[], size: number): T[][] {
  const result: T[][] = [];
  for (let i = 0; i < arr.length; i += size) {
    result.push(arr.slice(i, i + size));
  }
  return result;
}

export function median(nums: number[]): number | null {
  if (nums.length === 0) {
    return null;
  }
  const sorted = [...nums].sort((a, b) => a - b);
  const mid = Math.floor(sorted.length / 2);
  return sorted.length % 2 === 1 ? sorted[mid] : (sorted[mid - 1] + sorted[mid]) / 2;
}

console.log(cleanTags(" A, b ,a"), chunk([1, 2, 3], 2), median([3, 1, 2]));
[ 'a', 'b' ] [ [ 1, 2 ], [ 3 ] ] 2

שים לב ל־median: ההחזרה היא number | null — TypeScript יכריח את מי שקורא לה לטפל במקרה הריק. וב־callbacks של map/filter לא צריך טיפוסים — הם מוסקים.

⚠️ טעויות נפוצות
  • להחזיר number מ־median ולהסתיר את ה־null עם as.

✅ פתרון
// P11-E20: מתוקן
export type LineItem = { name: string; unitPrice: number; quantity: number };

export function lineTotal(item: LineItem): number {
  return item.unitPrice * item.quantity;
}

export function orderTotal(items: LineItem[], shipping: number): number {
  return items.reduce((s, it) => s + lineTotal(it), 0) + shipping;
}

export function parseShipping(input: string): number {
  return Number(input);
}

console.log(orderTotal([{ name: "עט", unitPrice: 5, quantity: 2 }], parseShipping("20")));
30

הבאגים: orderTotal(items, "20") — מחרוזת, ו־+ הפך לשרשור ("1020"); ו־item.qty במקום quantity — undefined ← NaN. שאר השגיאות היו "רק" טיפוסים חסרים — אבל כל אחת מהן הסתירה מקום שבו באג כזה יכול לגור.

⚠️ טעויות נפוצות
  • לפתור את השגיאה של המחרוזת עם shipping: any.
