💡 רמז 2

// P11-E15: מכונת מצבים של הזמנה
export type OrderStatus = "pending" | "paid" | "shipped" | "delivered" | "cancelled";
export type Order = { id: number; status: OrderStatus };

const TRANSITIONS: Record<OrderStatus, OrderStatus[]> = {
  pending: ["paid", "cancelled"],
  paid: ["shipped", "cancelled"],
  shipped: ["delivered"],
  delivered: [],
  cancelled: [],
};

export function allowedNext(status: OrderStatus): OrderStatus[] {
  return TRANSITIONS[status];
}

export function canTransition(from: OrderStatus, to: OrderStatus): boolean {
  return TRANSITIONS[from].includes(to);
}

export function transition(order: Order, to: OrderStatus): Order {
  if (!canTransition(order.status, to)) {
    throw new Error(`לא ניתן לעבור מ־${order.status} ל־${to}`);
  }
  return { ...order, status: to };
}

console.log(transition({ id: 1, status: "pending" }, "paid"), allowedNext("shipped"));

"מכונת מצבים" — דפוס שתמצא בכל מערכת הזמנות, תשלומים ומשלוחים. בפרויקט הדגל תשתמש בו.

// P11-E16
export type Shape =
  | { kind: "circle"; radius: number }
  | { kind: "rect"; width: number; height: number }
  | { kind: "triangle"; base: number; height: number };

function unreachable(x: never): never {
  throw new Error(`צורה לא מוכרת: ${JSON.stringify(x)}`);
}

export function area(shape: Shape): number {
  let a: number;
  switch (shape.kind) {
    case "circle":
      a = Math.PI * shape.radius ** 2;
      break;
    case "rect":
      a = shape.width * shape.height;
      break;
    case "triangle":
      a = (shape.base * shape.height) / 2;
      break;
    default:
      return unreachable(shape);
  }
  return Math.round(a * 100) / 100;
}

export function describe(shape: Shape): string {
  switch (shape.kind) {
    case "circle":
      return `עיגול ברדיוס ${shape.radius}`;
    case "rect":
      return `מלבן ${shape.width}×${shape.height}`;
    case "triangle":
      return `משולש בסיס ${shape.base} גובה ${shape.height}`;
    default:
      return unreachable(shape);
  }
}

console.log(area({ kind: "circle", radius: 2 }), describe({ kind: "rect", width: 3, height: 4 }));

הפונקציה unreachable — דפוס מקובל: מקום אחד שגם תופס שכחה בזמן כתיבה, וגם זורק אם בכל זאת הגיע ערך מוזר בזמן ריצה.

// P11-E17
export type Result<T> = { ok: true; value: T } | { ok: false; error: string };

export function ok<T>(value: T): Result<T> {
  return { ok: true, value };
}

export function fail<T = never>(error: string): Result<T> {
  return { ok: false, error };
}

export function parseAge(input: string): Result<number> {
  const n = Number(input);
  if (input.trim() === "" || !Number.isInteger(n)) {
    return fail("לא מספר");
  }
  if (n < 0 || n > 120) {
    return fail("מחוץ לטווח");
  }
  return ok(n);
}

export function sumResults(results: Result<number>[]): Result<number> {
  let total = 0;
  for (const r of results) {
    if (!r.ok) {
      return r;
    }
    total += r.value;
  }
  return ok(total);
}

console.log(parseAge("30"), sumResults([parseAge("30"), parseAge("abc"), parseAge("500")]));


{ ok: true, value: 30 } { ok: false, error: 'לא מספר' }


זה אותו רעיון כמו attempt מ־P08-E10 — עכשיו עם טיפוסים, כך שאי אפשר לגשת ל־value בלי לבדוק ok קודם.

💡 רמז 1


ב־parseSize: const t = text.trim().toUpperCase(); ואז switch (t) עם case "S": case "M": case "L": case "XL": return t; — בתוך ה־case, TypeScript יודע ש־t הוא אחת המידות.




    
💡 רמז 2

// P11-E62: מידות בגדים
export type Size = "S" | "M" | "L" | "XL";

export function parseSize(text: string): Size | null {
  const t = text.trim().toUpperCase();
  switch (t) {
    case "S":
    case "M":
    case "L":
    case "XL":
      return t;
    default:
      return null;
  }
}

export function surcharge(s: Size): number {
  switch (s) {
    case "S":
    case "M":
      return 0;
    case "L":
      return 5;
    case "XL":
      return 10;
    default: {
      const unhandled: never = s;
      throw new Error(`מידה לא מוכרת: ${unhandled}`);
    }
  }
}

for (const input of [" m ", "xl", "XXL"]) {
  const size = parseSize(input);
  console.log(JSON.stringify(input), size, size === null ? "-" : surcharge(size));
}


" m " M 0
"xl" XL 10
"XXL" null -


parseSize היא "שער": נכנס string כלשהו, יוצא Size בטוח או null. מכאן והלאה הקוד עובד עם Size בלבד — ואם יתווסף "XXL" לטיפוס, ה־never ב־surcharge יצביע על המקום שצריך לעדכן.




    
⚠️ טעויות נפוצות


  • return t as Size בלי לבדוק — כל טקסט הופך ל"מידה".
  • לבדוק עם ["S", "M", "L", "XL"].includes(t) ואז return t — TypeScript לא מצמצם עם includes, ו־t נשאר string.

// P11-E27: משך זמן
export function parseDuration(text: string): number | null {
  const words = text.split(" ").filter((w) => w !== "");
  if (words.length === 0) return null;

  let total = 0;
  for (const w of words) {
    const unit = w.slice(-1);
    const digits = w.slice(0, -1);
    const n = Number(digits);
    if (digits === "" || !Number.isInteger(n) || n < 0) return null;
    if (unit === "h") {
      total += n * 60;
    } else if (unit === "m") {
      total += n;
    } else {
      return null;
    }
  }
  return total;
}

export function formatDuration(minutes: number): string {
  const h = Math.floor(minutes / 60);
  const m = minutes % 60;
  const parts: string[] = [];
  if (h > 0) parts.push(`${h}h`);
  if (m > 0 || h === 0) parts.push(`${m}m`);
  return parts.join(" ");
}

// P11-E67: יתרה מאירועים — עם union
export type AccountEvent =
  | { type: "deposit"; amount: number }
  | { type: "withdraw"; amount: number }
  | { type: "fee"; amount: number; reason: string }
  | { type: "freeze" };

export function balance(events: AccountEvent[]): number {
  let total = 0;
  for (const e of events) {
    switch (e.type) {
      case "deposit":
        total += e.amount;
        break;
      case "withdraw":
      case "fee":
        total -= e.amount;
        break;
      case "freeze":
        return total;
    }
  }
  return total;
}

export function feeReasons(events: AccountEvent[]): string[] {
  const reasons: string[] = [];
  for (const e of events) {
    if (e.type === "fee") reasons.push(e.reason);
  }
  return reasons;
}

  { type: "deposit", amount: 1000 },
];

// P11-E68: פקודות CLI כ־union
export type Command =
  | { kind: "add"; name: string; amount: number }
  | { kind: "list" }
  | { kind: "remove"; id: number }
  | { kind: "error"; message: string };

const error = (message: string): Command => ({ kind: "error", message });

export function parseCommand(line: string): Command {
  const [cmd, ...args] = line.trim().split(" ").filter((w) => w !== "");
  switch (cmd) {
    case "add": {
      if (args.length !== 2) return error("שימוש: add <name> <amount>");
      const amount = Number(args[1]);
      if (!Number.isFinite(amount) || amount <= 0) return error("סכום לא תקין");
      return { kind: "add", name: args[0], amount };
    }
    case "list":
      return args.length === 0 ? { kind: "list" } : error("שימוש: list");
    case "remove": {
      if (args.length !== 1) return error("שימוש: remove <id>");
      const id = Number(args[0]);
      if (!Number.isInteger(id) || id <= 0) return error("מזהה לא תקין");
      return { kind: "remove", id };
    }
    default:
      return error("פקודה לא מוכרת");
  }
}
