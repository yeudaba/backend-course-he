// P08-E06: פונקציות עיצוב
function formatPrice(n) {
  return `₪${n.toFixed(2)}`;
}

function formatLine(name, price) {
  return `${name}: ${formatPrice(price)}`;
}

module.exports = { formatPrice, formatLine };
// P08-E06: קבלה שמשתמשת במודול העיצוב
const { formatPrice, formatLine } = require("./P08-E06-format");

function receipt(items) {
  const total = items.reduce((s, i) => s + i.price, 0);
  return [...items.map((i) => formatLine(i.name, i.price)), `סה״כ: ${formatPrice(total)}`].join("\n");
}

P08-E07 · אותו דבר ב־ES Modules 🤖 💻 כתיבת קוד 🤖 רמה 2

✅ פתרון
// P08-E07: ES Modules
const words = (text) => text.trim().split(" ").filter((w) => w !== "");

export function slugify(text) {
  return words(text.toLowerCase()).join("-");
}

export function wordCount(text) {
  return words(text).length;
}

export default function summary(text) {
  return `${wordCount(text)} מילים: ${slugify(text)}`;
}

words לא יוצא — היא פרטית. (בבודק, הייצוא ה־default מגיע בשם default; לכן הוא בודק mod.default.)

⚠️ טעויות נפוצות
  • לערבב: export function וגם module.exports באותו קובץ .mjs — module לא קיים ב־ESM.

// P08-E20: כלי עזר — מתוקן
function capitalize(word) {

P08-E26 · מודול כסף ב־ES Modules 🤖 💻 כתיבת קוד 🤖 רמה 2

💡 רמז 1

ב־money.mjs: Math.round(shekels * 100), ו־(agorot / 100).toFixed(2). בראש P08-E26.mjs: import { toAgorot, formatILS } from "./P08-E26-money.mjs"; — עם ./ ועם הסיומת.

💡 רמז 2

prices.map(toAgorot) נותן מערך של מספרים שלמים. מחברים עם reduce, ומעצבים את התוצאה עם formatILS.

✅ פתרון
// P08-E26: מודול כסף
export function toAgorot(shekels) {
  return Math.round(shekels * 100);
}

export function formatILS(agorot) {
  return `₪${(agorot / 100).toFixed(2)}`;
}
// P08-E26: סכום עגלה, באגורות
import { toAgorot, formatILS } from "./P08-E26-money.mjs";

export function cartTotal(prices) {
  const total = prices.map(toAgorot).reduce((sum, a) => sum + a, 0);
  return formatILS(total);
}

console.log(cartTotal([19.9, 0.1, 5]));
console.log(cartTotal([0.1, 0.2]));
% node 08-modules-errors/starter/P08-E26.mjs

החלוקה לשני קבצים היא לא רק סדר. money.mjs יודע על כסף ולא יודע על עגלות, ולכן אפשר להשתמש בו גם בדוח, בקבלה ובפרויקט 1. cartTotal לא יודעת איך מעגלים או מעצבים — היא רק מחברת.

prices.map(toAgorot): מעבירים את הפונקציה עצמה, בלי סוגריים, ו־map קוראת לה על כל מחיר (פרק 5, "פונקציה כערך"). החיבור קורה על מספרים שלמים, ולכן 0.1 + 0.2 לא מייצר שארית.

⚠️ טעויות נפוצות
  • from "./P08-E26-money" בלי .mjs — ב־ES Modules הסיומת חובה: ERR_MODULE_NOT_FOUND.
  • shekels * 100 בלי Math.round — ‏1989.9999999999998 אגורות.
  • לחבר בשקלים ולהמיר בסוף — עם [0.1, 0.2] זה עדיין עובד בזכות העיגול, אבל בסכומים גדולים השגיאה מצטברת. ההמרה קורה מיד, לכל מחיר.
