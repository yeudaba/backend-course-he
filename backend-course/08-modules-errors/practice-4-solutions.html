// P08-E17: ייבוא עם דוח שגיאות לכל שורה
function parseLine(line) {
  const parts = line.split(",").map((p) => p.trim());
  if (parts.length < 3 || parts.some((p) => p === "")) {
    throw new Error("חסרים שדות");
  }
  const [date, category, amountText] = parts;
  const amount = Number(amountText);
  if (Number.isNaN(amount) || amount <= 0) {
    throw new Error("סכום לא תקין");
  }
  return { date, category, amount };
}

function parseExpenses(text) {
  const expenses = [];
  const errors = [];
  const lines = text.split("\n");
  for (let i = 0; i < lines.length; i++) {
    if (lines[i].trim() === "") {
      continue;
    }
    try {
      expenses.push(parseLine(lines[i]));
    } catch (err) {
      errors.push({ line: i + 1, message: err.message });
    }
  }
  return { expenses, errors };
}

2026-10-02, תחבורה

2026-10-03, קפה, abc
2026-10-04, ספרים, 89`;
console.log(parseExpenses(input));

// P08-E18: טעינת הגדרות — להיכשל מוקדם
const REQUIRED = ["DATABASE_URL", "JWT_SECRET"];

function loadConfig(env) {
  const missing = REQUIRED.filter((key) => !env[key]);
  if (missing.length > 0) {
    throw new Error(`חסרות הגדרות: ${missing.join(", ")}`);
  }
  const port = Number(env.PORT ?? 3000);
  if (!Number.isInteger(port) || port < 1 || port > 65535) {
    throw new Error("PORT לא תקין");
  }
  return { port, databaseUrl: env.DATABASE_URL, jwtSecret: env.JWT_SECRET };
}

// P08-E19: מנגנון בדיקות מינימלי
function assertEqual(actual, expected, label) {
  if (actual !== expected) {
    throw new Error(`${label}: ציפינו ל־${expected} וקיבלנו ${actual}`);
  }
}

function runChecks(checks) {
  const passed = [];
  const failed = [];
  for (const { name, fn } of checks) {
    try {
      fn();
      passed.push(name);
    } catch (err) {
      failed.push({ name, error: err.message });
    }
  }
  return { passed, failed };
}

const double = (n) => n + n;

    { name: "string", fn: () => assertEqual(double("a"), 2, "double('a')") },

💡 רמז 2

transfer: const newFrom = withdraw(from, amount); const newTo = deposit(to, amount); return [newFrom, newTo]; — אם withdraw זורק, deposit לא קורה. "הכול או כלום" בחינם.

✅ פתרון
// P08-E24: מודול בנק
function bankError(code, message) {
  const err = new Error(message);
  err.code = code;
  return err;
}

function assertAmount(amount) {
  if (typeof amount !== "number" || !Number.isFinite(amount) || amount <= 0) {
    throw bankError("INVALID_AMOUNT", "סכום לא תקין");
  }
}

function deposit(account, amount) {
  assertAmount(amount);
  return { ...account, balance: account.balance + amount };
}

function withdraw(account, amount) {
  assertAmount(amount);
  if (account.balance < amount) {
    throw bankError("INSUFFICIENT_FUNDS", "אין מספיק יתרה");
  }
  return { ...account, balance: account.balance - amount };
}

function transfer(from, to, amount) {
  if (from.id === to.id) {
    throw bankError("SAME_ACCOUNT", "לא ניתן להעביר לאותו חשבון");
  }
  const newFrom = withdraw(from, amount);
  const newTo = deposit(to, amount);
  return [newFrom, newTo];
}

function safeTransfer(from, to, amount) {
  try {
    const [newFrom, newTo] = transfer(from, to, amount);
    return { ok: true, from: newFrom, to: newTo };
  } catch (err) {
    return { ok: false, code: err.code, message: err.message, from, to };
  }
}

const a = { id: 1, owner: "דנה", balance: 100 };
const b = { id: 2, owner: "עומר", balance: 20 };
console.log(safeTransfer(a, b, 30));
console.log(safeTransfer(a, b, 500).code);

module.exports = { bankError, deposit, withdraw, transfer, safeTransfer };

P08-E25 · אתגר: מחשבון בטוח 🤖 💻 כתיבת קוד 🤖 רמה 5

💡 רמז 1

שלב 1 — tokenize: הפוך את הטקסט למערך כמו [2, "+", 3, "*", 4]. עבור תו־תו: ספרה או נקודה ← צבור למספר; אופרטור ← "סגור" את המספר והוסף את האופרטור; רווח ← דלג; אחר ← שגיאה.

💡 רמז 2

שלב 2 — שני מעברים: קודם עבור על הטוקנים וחשב כל * ו־/ מיד (מחליף שלושה טוקנים בתוצאה), ואז חבר/חסר את מה שנשאר משמאל לימין. בדוק תקינות: טוקנים במקומות זוגיים — מספרים; במקומות אי־זוגיים — אופרטורים; האורך אי־זוגי.

✅ פתרון
// P08-E25: מחשבון בלי eval
const OPERATORS = ["+", "-", "*", "/"];

function tokenize(expression) {
  const tokens = [];
  let current = "";
  const flush = () => {
    if (current !== "") {
      const n = Number(current);
      if (Number.isNaN(n)) throw new Error("ביטוי לא תקין");
      tokens.push(n);
      current = "";
    }
  };
  for (const ch of expression) {
    if ("0123456789.".includes(ch)) {
      current += ch;
    } else if (OPERATORS.includes(ch)) {
      flush();
      tokens.push(ch);
    } else if (ch === " ") {
      flush();
    } else {
      throw new Error(`תו לא חוקי: ${ch}`);
    }
  }
  flush();
  return tokens;
}

function validate(tokens) {
  const ok =
    tokens.length % 2 === 1 &&
    tokens.every((t, i) => (i % 2 === 0 ? typeof t === "number" : OPERATORS.includes(t)));
  if (!ok) throw new Error("ביטוי לא תקין");
}

function calculate(expression) {
  const tokens = tokenize(expression);
  validate(tokens);

  const stack = [tokens[0]];
  for (let i = 1; i < tokens.length; i += 2) {
    const op = tokens[i];
    const n = tokens[i + 1];
    if (op === "*") {
      stack.push(stack.pop() * n);
    } else if (op === "/") {
      if (n === 0) throw new Error("חילוק באפס");
      stack.push(stack.pop() / n);
    } else {
      stack.push(op, n);
    }
  }

  let result = stack[0];
  for (let i = 1; i < stack.length; i += 2) {
    result = stack[i] === "+" ? result + stack[i + 1] : result - stack[i + 1];
  }
  return result;
}

console.log(calculate("2 + 3 * 4"), calculate("10 / 4 - 1"), calculate("7"));

  } catch (err) {
    console.log(err.message);
  }
}
