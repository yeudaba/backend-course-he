// P08-E22: הרצף הארוך ביותר — מתוקן
function longestStreak(days) {
  let best = 0;
  let current = 0;
  for (const visited of days) {
    current = visited ? current + 1 : 0;
    best = Math.max(best, current);
  }
  return best;
}

console.log(longestStreak([true, false, true, true, true]));

module.exports = { longestStreak };

P08-E60 · פירוק שורת at 🤖 💻 כתיבת קוד 🤖 רמה 3

💡 רמז 1

אחרי trim ו־slice(3) (בלי "at "): אם השורה נגמרת ב־")" — השם הוא מה שלפני " (", והמיקום — מה שבתוך הסוגריים. אחרת — אין שם, וכל השורה היא המיקום.

💡 רמז 2

✅ פתרון
08-modules-errors/starter/P08-E60.js 
function parseFrame(line) {
  const text = line.trim().slice(3);
  let fn = null;
  let location = text;
  if (text.endsWith(")")) {
    const open = text.indexOf(" (");
    fn = text.slice(0, open);
    location = text.slice(open + 2, -1);
  }
  const lastColon = location.lastIndexOf(":");
  const prevColon = location.lastIndexOf(":", lastColon - 1);
  const lineNo = Number(location.slice(prevColon + 1, lastColon));
  const colNo = Number(location.slice(lastColon + 1));
  if (lastColon === -1 || prevColon === -1 || Number.isNaN(lineNo) || Number.isNaN(colNo)) {
    return { fn, file: location, line: null, col: null };
  }
  return { fn, file: location.slice(0, prevColon), line: lineNo, col: colNo };
}

console.log(parseFrame("    at lineTotal (/p/lines.js:2:15)"));
console.log(parseFrame("    at Array.map (<anonymous>)"));

module.exports = { parseFrame };

lastIndexOf(":", from) מחפש אחורה החל ממקום מסוים — כך מוצאים את שני הנקודתיים האחרונים בלי לגעת בנתיב. זה מה ש־VS Code עושה כשאתה לוחץ ⌘ על נתיב בטרמינל: מפרק את הקובץ, השורה והעמודה.

⚠️ טעויות נפוצות
  • location.split(":") — שובר נתיבים שיש בהם : (כמו node:internal/... או C:\).
  • להשאיר את line כמחרוזת "2" — הבודק מצפה למספר.

function ownFrames(stack, projectDir) {
  const result = [];
  for (const raw of stack.split("\n")) {
    const text = raw.trim();
    if (!text.startsWith("at ")) {
      continue;
    }
    const location = text.endsWith(")") ? text.slice(text.lastIndexOf("(") + 1, -1) : text.slice(3);
    const lastColon = location.lastIndexOf(":");
    const prevColon = location.lastIndexOf(":", lastColon - 1);
    const file = location.slice(0, prevColon);
    if (!file.startsWith(projectDir + "/") || file.includes("/node_modules/")) {
      continue;
    }
    result.push(`${file.slice(projectDir.length + 1)}:${location.slice(prevColon + 1, lastColon)}`);
  }
  return result;
}

function explainError(err) {
  const msg = err.message;
  switch (err.name) {
    case "TypeError": {
      const prefix = "Cannot read properties of ";
      if (msg.startsWith(prefix)) {
        const value = msg.slice(prefix.length, msg.indexOf(" ("));
        const field = msg.slice(msg.indexOf("'") + 1, msg.lastIndexOf("'"));
        return `ניסית לקרוא את "${field}" מתוך ${value} — מה שמשמאל לנקודה ריק`;
      }
      const notFn = " is not a function";
      if (msg.endsWith(notFn)) {
        return `"${msg.slice(0, -notFn.length)}" הוא לא פונקציה — בדוק את השם ואת הייבוא`;
      }
      break;
    }
    case "ReferenceError": {
      const notDefined = " is not defined";
      if (msg.endsWith(notDefined)) {
        return `"${msg.slice(0, -notDefined.length)}" לא מוגדר — שגיאת כתיב, או שחסר ייבוא?`;
      }
      break;
    }
    case "SyntaxError":
      return `טקסט לא תקין: ${msg}`;
  }
  return `${err.name}: ${msg}`;
}

try {
  null.price;
} catch (err) {
  console.log(explainError(err));
}

module.exports = { explainError };
ניסית לקרוא את "price" מתוך null — מה שמשמאל לנקודה ריק

ה־break בתוך case "TypeError" יוצא מה־switch כשאף מקרה לא התאים — וממשיכים לשורת ברירת המחדל שאחריו. (הסוגריים המסולסלים אחרי case מאפשרים להגדיר const בתוכו.)

⚠️ טעויות נפוצות
  • לבדוק רק את err.message בלי err.name — הודעה של throw new Error("x is not defined") שלך תתפרש כ־ReferenceError.
