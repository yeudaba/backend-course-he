const JWT_SECRET = "secret";

@Post("register")
async register(@Body() body: any) {
  const user = await this.prisma.user.create({ data: { ...body, password: body.password } });
  return user;
}

@Post("login")
async login(@Body() { email, password }: any) {
  console.log("login attempt", email, password);
  const user = await this.prisma.user.findUnique({ where: { email } });
  if (!user) throw new NotFoundException("אין משתמש עם האימייל הזה");
  if (user.password !== password) throw new UnauthorizedException("סיסמה שגויה");
  return { token: jwt.sign({ sub: user.id, role: user.role, email, password }, JWT_SECRET) };
}

@Get("orders/:id")
async order(@Param("id") id: string) {
  return this.prisma.order.findUnique({ where: { id: Number(id) } });
}

P21-E25 · אתגר: מפתחות API 🤖

💻 כתיבת קוד 🤖רמה 5⏱ 60 דק׳ידע נדרש: שיעור 4

P21-E75 · ‏401, 403 או 404 — בקוד 🤖

💻 כתיבת קוד 🤖רמה 3⏱ 20 דק׳ידע נדרש: שיעור 4

בקובץ 21-auth/starter/P21-E75.ts כתוב וייצא פונקציית מדיניות אחת, שמרכזת את כל ההחלטות של השיעור:

export type User = { id: number; role: "user" | "admin" };
export type Task = { id: number; ownerId: number; archived: boolean };
export type Action = "read" | "update" | "delete" | "list-all";
export type Decision = "OK" | "UNAUTHORIZED" | "FORBIDDEN" | "NOT_FOUND";
export function decide(user: User | null, action: Action, task?: Task): Decision
export function statusOf(d: Decision): 200 | 401 | 403 | 404

הכללים, לפי הסדר:

  1. אין משתמש ← UNAUTHORIZED.
  2. list-all ← OK ל־admin, ‏FORBIDDEN לכל אחר.
  3. אין משימה, או שהיא לא של המשתמש והוא לא admin ← NOT_FOUND (לא מגלים שהיא קיימת).
  4. update על משימה ב־archive ← FORBIDDEN (גם לבעלים וגם ל־admin).
  5. אחרת ← OK.

P21-E81 · ‏CORS עם רשימה מפורשת ב־Express 🤖

export function corsAllowlist(allowed: string[]): (req, res, next) => void
  • תמיד: Vary: Origin (כי התשובה תלויה ב־Origin).
  • ‏Origin של הבקשה ב־allowed (התאמה מדויקת) ← Access-Control-Allow-Origin: <origin> ו־Access-Control-Allow-Credentials: true. אף פעם לא *.
  • ‏OPTIONS (preflight) מ־origin מותר ← גם Access-Control-Allow-Methods: GET, POST, PATCH, DELETE ו־Access-Control-Allow-Headers: Authorization, Content-Type, ותשובה 204 בלי להמשיך ל־routes.
  • OPTIONS מ־origin לא מותר ← 403, בלי headers של CORS. בקשה רגילה מ־origin לא מותר (או בלי Origin, כמו curl) ← ממשיכים כרגיל, בלי headers של CORS (הדפדפן יחסום את הקריאה).
