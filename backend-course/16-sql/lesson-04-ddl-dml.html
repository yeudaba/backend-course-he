CHECK על כמה עמודות, CHECK עם NULL ו־regex

CREATE TABLE shipments (
  id           integer GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  tracking     text CHECK (tracking ~ '^[A-Z]{2}[0-9]{6}$'),
  weight_kg    numeric(6, 2) CHECK (weight_kg > 0),
  sent_on      date NOT NULL DEFAULT CURRENT_DATE,
  delivered_on date,
  CHECK (delivered_on >= sent_on)
);

ERROR:  new row for relation "shipments" violates check constraint "shipments_check"
DETAIL:  Failing row contains (3, null, null, 2026-05-01, 2026-04-30).
  • אילוץ על כמה עמודות נכתב בסוף הטבלה, אחרי העמודות, כשורה משלו: CHECK (delivered_on >= sent_on).
  • CHECK עם NULL עובר. weight_kg > 0 כש־weight_kg הוא NULL נותן NULL — ו־CHECK נכשל רק על false. לכן "לא חובה, אבל אם יש — חיובי" זה פשוט CHECK (weight_kg > 0) בלי NOT NULL. (ההפך מ־WHERE, שמחזיר רק true — שיעור 3.)
  • ~ — התאמה ל־regex (כמו test ב־JavaScript, פרק 8 שיעור 5). '^[A-Z]{2}[0-9]{6}$' — בדיוק 2 אותיות גדולות ו־6 ספרות.
  • CURRENT_DATE — התאריך של היום (לפי השעון של המסד). כאן כברירת מחדל.
  • num_nonnulls(a, b, ...) — כמה מהערכים אינם NULL. "בדיוק אחד מ־a ו־b": CHECK (num_nonnulls(a, b) = 1).

זו "שכבת ההגנה האחרונה": גם אם יש באג בקוד, או מישהו מריץ SQL ידנית — המסד לא יסכים לנתונים לא תקינים. Zod בגבול ה־API + אילוצים במסד = שתי שכבות.

⚠️ UPDATE / DELETE בלי WHERE

DELETE FROM customers; מוחק את כל הלקוחות. בלי אישור, בלי "פח מחזור". UPDATE products SET price = 0; — כל המוצרים בחינם. זה קורה גם לאנשים מנוסים. הרגל שמציל: קודם מריצים SELECT עם אותו WHERE ובודקים כמה שורות חוזרות — ורק אז מחליפים ל־UPDATE/DELETE. (ובפרק 18: לעטוף ב־transaction, ולבטל אם טעית.)

ועוד שינויים נפוצים: ALTER TABLE customers ALTER COLUMN city SET NOT NULL — מעכשיו חובה. זה נכשל אם כבר יש בעמודה NULL, ולכן קודם ממלאים (UPDATE ... WHERE city IS NULL), ורק אז SET NOT NULL. ו־upper(text) / lower(text) — אותיות גדולות / קטנות, למשל כדי למלא עמודה חדשה מעמודה קיימת:

ALTER TABLE customers ALTER COLUMN phone SET NOT NULL;
UPDATE customers SET city = upper(city) WHERE city IS NOT NULL;
UPDATE customers SET city = 'UNKNOWN' WHERE city IS NULL;
ALTER TABLE customers ALTER COLUMN city SET NOT NULL;
