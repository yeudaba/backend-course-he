1. CREATE TABLE

CREATE TABLE tasks (
  id          integer GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  title       text NOT NULL CHECK (length(title) BETWEEN 1 AND 120),
  status      text NOT NULL DEFAULT 'todo' CHECK (status IN ('todo', 'in_progress', 'done')),
  priority    smallint NOT NULL DEFAULT 2,
  due_date    date,
  created_at  timestamptz NOT NULL DEFAULT now()
);

⚠️ UPDATE / DELETE בלי WHERE

DELETE FROM customers; מוחק את כל הלקוחות. בלי אישור, בלי "פח מחזור". UPDATE products SET price = 0; — כל המוצרים בחינם. זה קורה גם לאנשים מנוסים. הרגל שמציל: קודם מריצים SELECT עם אותו WHERE ובודקים כמה שורות חוזרות — ורק אז מחליפים ל־UPDATE/DELETE. (ובפרק 18: לעטוף ב־transaction, ולבטל אם טעית.)
