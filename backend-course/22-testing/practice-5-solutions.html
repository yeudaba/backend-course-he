P22-E89 · ‏TestClock 🤖 💻 כתיבת קוד 🤖 רמה 2

💡 רמז 1

שמור מספר (מילישניות): #ms = Date.parse(iso), ובדוק Number.isNaN. ‏now() ← new Date(this.#ms).

💡 רמז 2

מחרוזת: קודם /^(\d+[dhms])+$/.test(s), ואז for (const [, n, unit] of s.matchAll(/(\d+)([dhms])/g)).

✅ פתרון
// P22-E89: TestClock — שעון שהבדיקה שולטת בו
const UNIT = { d: 86_400_000, h: 3_600_000, m: 60_000, s: 1_000 } as const;

function parseIso(iso: string): number {
  const ms = Date.parse(iso);
  if (Number.isNaN(ms)) throw new RangeError(`תאריך לא תקין: ${iso}`);
  return ms;
}

export class TestClock {
  #ms: number;

  constructor(iso = "2026-11-04T10:00:00.000Z") {
    this.#ms = parseIso(iso);
  }

  now(): Date {
    return new Date(this.#ms);
  }

  set(iso: string): void {
    this.#ms = parseIso(iso);
  }

  advance(duration: number | string): void {
    if (typeof duration === "number") {
      this.#ms += duration;
      return;
    }
    if (!/^(\d+[dhms])+$/.test(duration)) throw new RangeError(`משך לא תקין: ${duration}`);
    for (const [, n, unit] of duration.matchAll(/(\d+)([dhms])/g)) {
      this.#ms += Number(n) * UNIT[unit as keyof typeof UNIT];
    }
  }
}

עם advance("14m59s") ו־advance("1s") בדיקת "תוקף של 15 דקות" נקראת כמו המשפט באפיון. ו־now() מחזיר עותק: קוד ש"מעגל" את ה־Date שקיבל (‏setHours(0)) לא יזיז את השעון לכל שאר הבדיקה.

⚠️ טעויות נפוצות
  • להחזיר את אותו Date בכל פעם — now().setDate(...) במקום אחד משנה את הזמן לכולם.
  • /(\d+)([dhms])/g בלי הבדיקה המלאה — ‏"5x15m" "עובר" כ־15 דקות.

P22-E90 · ‏signUp ו־auth — עזרי e2e 🤖 💻 כתיבת קוד 🤖 רמה 2

💡 רמז 1

פונקציה פנימית post(path, body) שמחזירה את התשובה.

💡 רמז 2

בשגיאה: await res.text() — כך רואים גם גוף שאינו JSON.

✅ פתרון
// P22-E90: עזרי e2e — הרשמה + כניסה בשורה אחת
export type Tokens = { accessToken: string; refreshToken: string };

export async function signUp(baseUrl: string, email: string, password = "password-123"): Promise<Tokens> {
  const post = (path: string, body: unknown) =>
    fetch(baseUrl + path, { method: "POST", headers: { "content-type": "application/json" }, body: JSON.stringify(body) });
  const reg = await post("/auth/register", { email, password });
  if (reg.status !== 201) throw new Error(`signUp(${email}): ${reg.status} ${await reg.text()}`);
  const login = await post("/auth/login", { email, password });
  if (login.status !== 200) throw new Error(`signUp(${email}): ${login.status} ${await login.text()}`);
  const { accessToken, refreshToken } = await login.json();
  return { accessToken, refreshToken };
}

export function auth(token: string): { authorization: string } {
  return { authorization: `Bearer ${token}` };
}

בלי ה־helper, כל בדיקה של בעלות פותחת בשמונה שורות של הרשמה וכניסה. איתו: const dana = await signUp(base, "dana@example.com"). וההודעה המפורטת חוסכת חצי שעה כשמשהו בהרשמה נשבר: "signUp(dana@example.com): 400 {...}" במקום "Cannot read properties of undefined".

⚠️ טעויות נפוצות
  • לא לבדוק את ה־status של ההרשמה — הכניסה נכשלת ב־401, וההודעה מטעה.

P22-E91 · בדיקות ל־AuthService — שבעה באגים 🤖 💻 כתיבת קוד 🤖 רמה 4

💡 רמז 1

שעון שאפשר להזיז:

let now = new Date("2026-11-04T10:00:00Z"); const clock = { now: () => now };

ואז now = new Date(now.getTime() + 15 * 60_000).

💡 רמז 2

עבור על הרשימה למעלה, ולכל משפט שאל: "אם זה היה שבור, איזו בדיקה הייתה נכשלת?" — ובמיוחד: גבולות (14:59 / 15:00, ‏7 ימים), השוואה בין שתי שגיאות, ומה לא יוצא בתשובה.

✅ פתרון
// P22-E91 — בדיקות ל־AuthService
import { test } from "node:test";
import assert from "node:assert/strict";
import { AuthService } from "./P22-E91.ts";

const MIN = 60_000;
function setup() {
  const clock = { t: new Date("2026-11-04T10:00:00Z").getTime(), now() { return new Date(this.t); } };
  return { clock, auth: new AuthService(clock) };
}

test("register — מנורמל, ובלי hash בתשובה", async () => {
  const { auth } = setup();
  assert.deepEqual(await auth.register("  Dana@Example.com ", "password-1"), { id: 1, email: "dana@example.com", role: "user" });
  await assert.rejects(auth.register("DANA@example.com", "other-pass"), { status: 409, code: "EMAIL_TAKEN" });
});

test("login — אותה שגיאה לאימייל לא קיים ולסיסמה שגויה", async () => {
  const { auth } = setup();
  await auth.register("dana@example.com", "password-1");
  const errors: unknown[] = [];
  for (const [email, pw] of [["nobody@example.com", "password-1"], ["dana@example.com", "wrong-pass"]]) {
    await auth.login(email, pw).catch((e) => errors.push({ status: e.status, code: e.code, message: e.message }));
  }
  assert.equal(errors.length, 2);
  assert.deepEqual(errors[0], { status: 401, code: "INVALID_CREDENTIALS", message: "אימייל או סיסמה שגויים" });
  assert.deepEqual(errors[1], errors[0]);
  await auth.login("DANA@example.com", "password-1");
});

test("access token — 15 דקות בדיוק", async () => {
  const { clock, auth } = setup();
  await auth.register("dana@example.com", "password-1");
  const { accessToken } = await auth.login("dana@example.com", "password-1");
  assert.deepEqual(await auth.verifyAccessToken(accessToken), { id: 1, role: "user" });
  clock.t += 15 * MIN - 1000;
  assert.deepEqual(await auth.verifyAccessToken(accessToken), { id: 1, role: "user" });
  clock.t += 1000;
  await assert.rejects(auth.verifyAccessToken(accessToken), { status: 401, code: "INVALID_TOKEN" });
});

test("refresh — rotation", async () => {
  const { auth } = setup();
  await auth.register("dana@example.com", "password-1");
  const first = await auth.login("dana@example.com", "password-1");
  const second = await auth.refresh(first.refreshToken);
  assert.notEqual(second.refreshToken, first.refreshToken);
  assert.deepEqual(await auth.verifyAccessToken(second.accessToken), { id: 1, role: "user" });
  await assert.rejects(auth.refresh(first.refreshToken), { status: 401, code: "INVALID_REFRESH_TOKEN" });
});

test("שימוש חוזר מבטל הכול — גם את הטוקן החדש", async () => {
  const { auth } = setup();
  await auth.register("dana@example.com", "password-1");
  const first = await auth.login("dana@example.com", "password-1");
  const second = await auth.refresh(first.refreshToken);
  await assert.rejects(auth.refresh(first.refreshToken));
  await assert.rejects(auth.refresh(second.refreshToken), { status: 401 });
});

test("refresh token — 7 ימים", async () => {
  const { clock, auth } = setup();
  await auth.register("dana@example.com", "password-1");
  const a = await auth.login("dana@example.com", "password-1");
  const b = await auth.login("dana@example.com", "password-1");
  clock.t += 7 * 24 * 60 * MIN - 1000;
  await auth.refresh(a.refreshToken);
  clock.t += 1000;
  await assert.rejects(auth.refresh(b.refreshToken), { status: 401, code: "INVALID_REFRESH_TOKEN" });
});

שים לב לבדיקת ה־login: לא כותבים את ההודעה פעמיים, אלא משווים את שתי השגיאות זו לזו. כך הבדיקה תופסת כל הבדל, גם בשדה שלא חשבת עליו. ובבדיקת 7 הימים — שני טוקנים: אחד לבדוק "עוד עובד" ברגע לפני, ואחד (שלא נגעו בו) לבדוק "כבר לא" ברגע עצמו.

⚠️ טעויות נפוצות
  • לבדוק רק "+20 דקות ← 401". טוקן של 16 דקות (באג) עדיין נכשל ב־20. הגבול עצמו תופס.
  • בבדיקת 7 ימים להשתמש באותו טוקן פעמיים — הראשון כבר בוטל ב־rotation, והשני נכשל מסיבה אחרת.

import fs from "node:fs";
import path from "node:path";

export async function applyMigrations(client: { query(sql: string): Promise<unknown> }, dir: string): Promise<string[]> {
  const names = fs
    .readdirSync(dir, { withFileTypes: true })
    .filter((e) => e.isDirectory() && fs.existsSync(path.join(dir, e.name, "migration.sql")))
    .map((e) => e.name)
    .sort();
  if (names.length === 0) throw new Error(`אין migrations ב־${dir}`);
  for (const name of names) {
    await client.query(fs.readFileSync(path.join(dir, name, "migration.sql"), "utf8"));
  }
  return names;
}

// P22-E100: fake fetch — routes קבועים, ורישום של כל קריאה
export type FakeReply = { status: number; body?: unknown };
type Req = { method: string; path: string; headers: Record<string, string>; body: unknown };
export type Route = FakeReply | ((req: Req) => FakeReply);

export function createFakeFetch(routes: Record<string, Route>) {
  const calls: Req[] = [];
  const fake = async (input: string | URL | Request, init?: RequestInit): Promise<Response> => {
    const url = new URL(String(input), "http://fake");
    const method = (init?.method ?? "GET").toUpperCase();
    const headers = Object.fromEntries(new Headers(init?.headers));
    const body = typeof init?.body === "string" ? JSON.parse(init.body) : undefined;
    const req = { method, path: url.pathname, headers, body };
    calls.push(req);
    const key = `${method} ${url.pathname}`;
    const route = routes[key];
    const reply: FakeReply = route === undefined ? { status: 404, body: { error: `אין route ל־${key}` } } : typeof route === "function" ? route(req) : route;
    if (reply.status === 204 || reply.body === undefined) return new Response(null, { status: reply.status });
    return new Response(JSON.stringify(reply.body), { status: reply.status, headers: { "content-type": "application/json" } });
  };
  return Object.assign(fake as typeof fetch, { calls });
}

עם fake כזה, בדיקה של "refresh אוטומטי על 401" היא: route ל־GET /tasks שמחזיר 401 בפעם הראשונה ו־200 בשנייה (פונקציה עם מונה), route ל־POST /auth/refresh, ואז

import { test } from "node:test";
import assert from "node:assert/strict";
import type { AddressInfo } from "node:net";
import type { Server } from "node:http";
import { createApp } from "./P22-E101.ts";

async function start(t: { after(fn: () => void): void }) {
  const server = await new Promise<Server>((resolve) => {
    const s = createApp().listen(0, () => resolve(s));
  });
  t.after(() => server.close());
  const base = `http://localhost:${(server.address() as AddressInfo).port}`;
  return (method: string, path: string, raw?: string) =>
    fetch(base + path, { method, headers: { "content-type": "application/json" }, body: raw });
}

test("HttpError ← ה־status וה־code שלו", async (t) => {
  const call = await start(t);
  const r = await call("GET", "/http-error");
  assert.equal(r.status, 409);
  assert.deepEqual(await r.json(), { code: "EMAIL_TAKEN", message: "האימייל כבר רשום" });
});

test("ZodError ← 400 עם details", async (t) => {
  const call = await start(t);
  const r = await call("POST", "/validate", JSON.stringify({ name: 5 }));
  assert.equal(r.status, 400);
  const body = await r.json();
  assert.equal(body.code, "VALIDATION_ERROR");
  assert.equal(typeof body.message, "string");
  assert.deepEqual(body.details.map((d: { path: string }) => d.path), ["name"]);
  const ok = await call("POST", "/validate", JSON.stringify({ name: "x" }));
  assert.equal(ok.status, 200);
});

test("JSON שבור ← 400 INVALID_JSON", async (t) => {
  const call = await start(t);
  const r = await call("POST", "/validate", "{oops");
  assert.equal(r.status, 400);
  assert.equal((await r.json()).code, "INVALID_JSON");
});

test("שגיאה לא צפויה ← 500 כללי, בלי פרטים", async (t) => {
  const call = await start(t);
  const r = await call("GET", "/boom");
  assert.equal(r.status, 500);
  const body = await r.json();
  assert.deepEqual(body, { code: "INTERNAL", message: "שגיאה פנימית" });
  assert.doesNotMatch(JSON.stringify(body), /db password|at /);
});

ה־500 נבדק עם deepEqual על כל הגוף: אם מישהו מוסיף "רק לדיבאג" שדה stack, הבדיקה נכשלת. זו בדיוק בדיקת האבטחה של API8.
