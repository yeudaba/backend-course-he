// P30-E53: נקודה בקורות החיים — בדיקה מהירה
const VERBS = new Set(["built", "designed", "implemented", "added", "cut", "reduced", "improved", "prevented", "migrated", "wrote", "tested", "deployed", "automated", "replaced", "fixed", "measured"]);

export function bulletIssues(bullet: string): string[] {
  const issues: string[] = [];
  const text = bullet.trim();
  const words = text.split(/\s+/).filter(Boolean);
  if (/^(responsible for|worked on|worked with|helped|participated in|assisted)\b/i.test(text)) issues.push("weak_start");
  else if (!VERBS.has((words[0] ?? "").toLowerCase().replace(/[^a-z]/g, ""))) issues.push("no_verb");
  if (!/\d/.test(text)) issues.push("no_number");
  if (words.length > 30) issues.push("too_long");
  return issues;
}

// P30-E54: סעיף Skills מרשימה מבולגנת
export function skillsSection(raw: string[], groups: Record<string, string[]>, aliases: Record<string, string>): string {
  const seen = new Set<string>();
  const skills: string[] = [];
  for (const item of raw) {
    const t = item.trim();
    if (!t) continue;
    const name = Object.hasOwn(aliases, t.toLowerCase()) ? aliases[t.toLowerCase()] : t;
    if (seen.has(name.toLowerCase())) continue;
    seen.add(name.toLowerCase());
    skills.push(name);
  }
  const used = new Set<string>();
  const lines: string[] = [];
  for (const [group, members] of Object.entries(groups)) {
    const inGroup = skills.filter((s) => members.includes(s));
    inGroup.forEach((s) => used.add(s));
    if (inGroup.length) lines.push(`${group}: ${inGroup.join(", ")}`);
  }
  const other = skills.filter((s) => !used.has(s));
  if (other.length) lines.push(`Other: ${other.join(", ")}`);
  return lines.join("\n");
}
⚠️ טעויות נפוצות

// P30-E57: סדר הסעיפים
const ORDER = ["Summary", "Skills", "Projects", "Experience", "Education", "Languages"];
const REQUIRED = ["Summary", "Skills", "Projects", "Education"];

export function sectionCheck(md: string) {
  const names = (md.match(/^## .*$/gm) ?? []).map((l) => l.slice(3).trim());
  const lower = ORDER.map((s) => s.toLowerCase());
  const known: number[] = [];
  const unknown: string[] = [];
  for (const n of names) {
    const i = lower.indexOf(n.toLowerCase());
    if (i === -1) unknown.push(n);
    else known.push(i);
  }
  const missing = REQUIRED.filter((r) => !known.includes(ORDER.indexOf(r)));
  const outOfOrder = known.some((v, i) => i > 0 && v <= known[i - 1]);
  return { missing, outOfOrder, unknown };
}

// P30-E58: טווח תאריכים לקורות חיים
const MONTHS = ["Jan", "Feb", "Mar", "Apr", "May", "Jun", "Jul", "Aug", "Sep", "Oct", "Nov", "Dec"];

function parse(s: string) {
  const m = /^(\d{4})-(0[1-9]|1[0-2])$/.exec(s);
  if (!m) throw new RangeError(`תאריך לא תקין: ${s}`);
  return { year: m[1], month: MONTHS[Number(m[2]) - 1] };
}

export function dateRange(start: string, end?: string): string {
  const a = parse(start);
  if (end === undefined) return `${a.month} ${a.year} – Present`;
  const b = parse(end);
  if (end < start) throw new RangeError(`${end} לפני ${start}`);
  if (start === end) return `${a.month} ${a.year}`;
  if (a.year === b.year) return `${a.month} – ${b.month} ${a.year}`;
  return `${a.month} ${a.year} – ${b.month} ${b.year}`;
}

// P30-E62: נקודה ממדידה אמיתית
function pct(xs: number[], p: number): number {
  const sorted = [...xs].sort((a, b) => a - b);
  return Math.round(sorted[Math.ceil((p / 100) * sorted.length) - 1]);
}

export function latencyClaim(before: number[], after: number[], what: string) {
  if (!before.length || !after.length) throw new RangeError("צריך מדידות לפני ואחרי");
  const p95Before = pct(before, 95);
  const p95After = pct(after, 95);
  if (p95After === 0) throw new RangeError("p95 אחרי הוא 0 — המדידה לא אמינה");
  const speedup = Math.round((p95Before / p95After) * 10) / 10;
  const bullet = p95After < p95Before ? `Cut p95 latency of ${what} from ${p95Before} ms to ${p95After} ms (${speedup}× faster)` : null;
  return { p50Before: pct(before, 50), p95Before, p50After: pct(after, 50), p95After, speedup, bullet };
}

הנקודה נבנית מהמדידה, לא להפך. ואם מראיין שואל "p95 של מה?" — יש לך את המערכים.

// P30-E63: כל מספר — שאלת ראיון
const NUM = /(\d{1,3}(?:,\d{3})+|\d+(?:\.\d+)?)(?: ?(%|ms|x|×|\+|k))?/g;

export function numbersToExplain(text: string) {
  const out: { value: string; unit: string; sentence: string }[] = [];
  const sentences = text
    .split(/\n|(?<=[.!?])\s+/)
    .map((s) => s.trim().replace(/^[-*]\s+/, ""))
    .filter(Boolean);
  for (const sentence of sentences) {
    for (const m of sentence.matchAll(NUM)) {
      const value = m[1];
      const unit = m[2] ?? "";
      if (!unit && /^(19|20)\d{2}$/.test(value)) continue;
      out.push({ value, unit, sentence });
    }
  }
  return out;
}

// P30-E64: קורות חיים ל־JSON
type Cv = { name: string; title: string; contacts: { email?: string; github?: string; linkedin?: string }; sections: Record<string, string[]> };

export function parseCv(md: string): Cv {
  const lines = md.split("\n");
  const nameIdx = lines.findIndex((l) => l.startsWith("# "));
  if (nameIdx === -1) throw new Error("אין שורת שם (# )");
  const name = lines[nameIdx].slice(2).trim();
  const next = lines.slice(nameIdx + 1).find((l) => l.trim() !== "") ?? "";
  const title = next.startsWith("#") || /@|http/.test(next) ? "" : next.trim();

  const head = lines.slice(0, 10).join("\n");
  const contacts: Cv["contacts"] = {};
  const email = head.match(/[\w.+-]+@[\w-]+(\.[\w-]+)+/);
  if (email) contacts.email = email[0];
  const gh = head.match(/(?:www\.)?(github\.com\/[\w-]+)/);
  if (gh) contacts.github = gh[1];
  const li = head.match(/(?:www\.)?(linkedin\.com\/in\/[\w-]+)/);
  if (li) contacts.linkedin = li[1];

  const sections: Record<string, string[]> = {};
  let current: string | null = null;
  for (const l of lines) {
    if (l.startsWith("## ")) {
      current = l.slice(3).trim();
      sections[current] = [];
    } else if (current && /^[-*] /.test(l)) {
      sections[current].push(l.slice(2).trim());
    }
  }
  return { name, title, contacts, sections };
}
