  5. cache: חוזרים למסד, רק לאט יותר. תור: מייל האישור נשלח מיד בלי תור; ביטול אוטומטי לא קורה עד ש־Redis חוזר — מחיר שהחלטתי לקבל, ונכתב ב־README.

  7. המשתמש מה־JWT, לא מה־URL; כל שאילתה מסוננת לפי userId; של אחר ← 404 ולא 403.
  8. PR ← CI (בדיקות, typecheck, build) ← merge ← migrations ← deploy hook ← בדיקת עשן מול /version.
  9. הייתי מודד: request ID ולוגים עם זמנים, ו־EXPLAIN ANALYZE. כרגע — חיפוש עם ILIKE, שלא משתמש באינדקס; לכן ה־cache.
  10. outbox לאירועים אחרי תשלום; ואולי להתחיל מבדיקת עומס מוקדם יותר.

// לפני שה־repo הופך לציבורי (ומראיינים פותחים אותו): סריקה לסודות. סוד שעלה ל־GitHub — נחשב דלוף,
// גם אם מחקת אותו ב־commit הבא (הוא בהיסטוריה). מבטלים אותו ומייצרים חדש
export interface Finding {
  path: string;
  line: number;
  kind: "env-file" | "anthropic-key" | "aws-key" | "private-key" | "db-password" | "jwt-secret";
}

const PLACEHOLDER = /^(\$\{.*\}|<.*>|your[_-]?\w*|password|changeme|secret|xxx+|\*+|)$/i;

export function findSecrets(files: readonly { path: string; content: string }[]): Finding[] {
  const out: Finding[] = [];
  for (const f of files) {
    if (/(^|\/)node_modules\//.test(f.path)) continue;
    const base = f.path.split("/").pop()!;
    const isExample = /\.example$|\.sample$|\.template$/.test(base);
    if (/^\.env(\..+)?$/.test(base) && !isExample) out.push({ path: f.path, line: 0, kind: "env-file" });
    f.content.split("\n").forEach((text, i) => {
      const line = i + 1;
      if (/sk-ant-[A-Za-z0-9_-]{10,}/.test(text)) out.push({ path: f.path, line, kind: "anthropic-key" });
      if (/\bAKIA[0-9A-Z]{16}\b/.test(text)) out.push({ path: f.path, line, kind: "aws-key" });
      if (/-----BEGIN [A-Z ]*PRIVATE KEY-----/.test(text)) out.push({ path: f.path, line, kind: "private-key" });
      const db = /postgres(?:ql)?:\/\/[^:\s/@]+:([^@\s]+)@/.exec(text);
      if (db && !PLACEHOLDER.test(db[1])) out.push({ path: f.path, line, kind: "db-password" });
      const jwt = /^\s*JWT_SECRET\s*=\s*["']?([^"'\s#]*)/.exec(text);
      if (jwt && !isExample && !PLACEHOLDER.test(jwt[1])) out.push({ path: f.path, line, kind: "jwt-secret" });
    });
  }
  return out;
}

// P30-E09
// היסטוריית commits היא חלק מתיק העבודות: מראיין שפותח את ה־repo רואה "fix", "wip", "asdf" — או סיפור.
// הכללים המקובלים: שורת נושא קצרה שאומרת מה השתנה, בלי נקודה בסוף; שורה ריקה; גוף שמסביר למה
const VAGUE = /^(wip|fix(es|ed)?|update[sd]?|changes?|stuff|misc|asdf+|test|tmp|temp|\.+|commit|final|done|more)$/i;

export function lintCommit(message: string): string[] {
  const problems: string[] = [];
  const lines = message.replace(/\r\n/g, "\n").replace(/\n+$/, "").split("\n");
  const subject = (lines[0] ?? "").trim();
  if (!subject) return ["שורת נושא ריקה"];
  if (subject.length > 72) problems.push(`שורת נושא ארוכה (${subject.length} תווים, עד 72)`);
  if (/[.。]$/.test(subject)) problems.push("נקודה בסוף שורת הנושא");
  if (VAGUE.test(subject) || subject.split(/\s+/).length < 2) problems.push("שורת נושא לא אומרת מה השתנה");
  if (lines.length > 1 && lines[1].trim() !== "") problems.push("חסרה שורה ריקה בין הנושא לגוף");
  lines.slice(2).forEach((l, i) => {
    if (l.length > 100) problems.push(`שורה ${i + 3} בגוף ארוכה מ־100 תווים`);
  });
  return problems;
}

// p95 ב־nearest-rank: ממיינים, ולוקחים את האיבר במקום ceil(0.95 × n)
export interface EndpointStats {
  endpoint: string;
  count: number;
  p95: number;
  errorRate: number;
}

const LINE = /^(\S+)\s+(GET|POST|PUT|PATCH|DELETE)\s+(\S+)\s+(\d{3})\s+(\d+)ms$/;

export function normalizePath(path: string): string {
  return path
    .split("?")[0]
    .split("/")
    .map((seg) => (/^\d+$/.test(seg) ? ":id" : seg))
    .join("/");
}

export function latencyReport(lines: readonly string[]): { endpoints: EndpointStats[]; skipped: number } {
  const groups = new Map<string, { ms: number[]; errors: number }>();
  let skipped = 0;
  for (const raw of lines) {
    const m = LINE.exec(raw.trim());
    if (!m) {
      if (raw.trim()) skipped++;
      continue;
    }
    const key = `${m[2]} ${normalizePath(m[3])}`;
    const g = groups.get(key) ?? groups.set(key, { ms: [], errors: 0 }).get(key)!;
    g.ms.push(Number(m[5]));
    if (Number(m[4]) >= 500) g.errors++;
  }
  const endpoints = [...groups].map(([endpoint, g]) => {
    const sorted = [...g.ms].sort((a, b) => a - b); // מספרים! בלי פונקציה — מיון של מחרוזות
    return {
      endpoint,
      count: sorted.length,
      p95: sorted[Math.ceil(0.95 * sorted.length) - 1],
      errorRate: Math.round((g.errors / sorted.length) * 1000) / 1000,
    };
  });
  endpoints.sort((a, b) => b.p95 - a.p95 || (a.endpoint < b.endpoint ? -1 : 1));
  return { endpoints, skipped };
}

Union-Find: find עם דחיסת מסלול, union של שני שורשים. ו־Map מערך ← הרשומה הראשונה שראתה אותו.

// P30-E14
// שאלת קוד מראיון: איחוד רשומות לקוח כפולות. אותו אימייל *או* אותו טלפון ← אותו אדם — גם בשרשרת
// (A ו־B חולקים אימייל, B ו־C חולקים טלפון ← A, B, C ביחד). Union-Find: כמעט O(n)
export interface CustomerRecord {
  id: number;
  email?: string | null;
  phone?: string | null;
}

export const normEmail = (e?: string | null) => (e ?? "").trim().toLowerCase() || null;
export function normPhone(p?: string | null): string | null {
  let d = (p ?? "").replace(/\D/g, "");
  if (d.startsWith("972")) d = "0" + d.slice(3);
  return d.length >= 9 ? d : null;
}

export function mergeCustomers(records: readonly CustomerRecord[]): number[][] {
  const parent = new Map<number, number>();
  const find = (x: number): number => {
    let r = x;
    while (parent.get(r) !== r) r = parent.get(r)!;
    while (parent.get(x) !== r) {
      const next = parent.get(x)!;
      parent.set(x, r); // דחיסת מסלול
      x = next;
    }
    return r;
  };
  const union = (a: number, b: number) => {
    const ra = find(a);
    const rb = find(b);
    if (ra !== rb) parent.set(Math.max(ra, rb), Math.min(ra, rb));
  };
  for (const r of records) parent.set(r.id, r.id);
  const owner = new Map<string, number>(); // "e:..."/"p:..." ← הרשומה הראשונה עם הערך
  for (const r of records) {
    for (const key of [normEmail(r.email) && `e:${normEmail(r.email)}`, normPhone(r.phone) && `p:${normPhone(r.phone)}`]) {
      if (!key) continue;
      const first = owner.get(key);
      if (first === undefined) owner.set(key, r.id);
      else union(first, r.id);
    }
  }
  const groups = new Map<number, number[]>();
  for (const r of records) {
    const root = find(r.id);
    (groups.get(root) ?? groups.set(root, []).get(root)!).push(r.id);
  }
  return [...groups.values()].map((g) => g.sort((a, b) => a - b)).sort((a, b) => a[0] - b[0]);
}

  stage: Stage;

export function pipelineStats(apps: readonly Application[], now: Date, { followUpDays = 7 }: { followUpDays?: number } = {}) {
  const byStage: Record<Stage, number> = { applied: 0, screen: 0, interview: 0, offer: 0, rejected: 0 };
  for (const a of apps) byStage[a.stage]++;

const LINE = /^(\S+)\s+(GET|POST|PUT|PATCH|DELETE)\s+(\S+)\s+(\d{3})\s+(\d+)ms$/;

export function latencyReport(lines: readonly string[]) {
  const groups = new Map<string, { ms: number[]; errors: number }>();
  for (const raw of lines) {
    const m = LINE.exec(raw.trim());
    if (!m) continue;
    const path = m[3].split("?")[0].replace(/\/\d+(?=\/|$)/g, "/:id");
    const key = `${m[2]} ${path}`;
    const g = groups.get(key) ?? groups.set(key, { ms: [], errors: 0 }).get(key)!;
    g.ms.push(Number(m[5]));
    if (Number(m[4]) >= 500) g.errors++;
  }
  return [...groups].map(([endpoint, g]) => {
    const sorted = [...g.ms].sort((a, b) => a - b); // תקלה 1: היה sort() ← מיון כמחרוזות: 1000 לפני 200
    const p95 = sorted[Math.ceil(0.95 * sorted.length) - 1]; // תקלה 2: היה floor(0.95 * n) ← באורך 20 זה האיבר האחרון (המקסימום, לא p95), ובאורך 1 — undefined
    return { endpoint, count: sorted.length, p95, errorRate: Math.round((g.errors / sorted.length) * 1000) / 1000 }; // תקלה 3: errors / lines.length ← שיעור שגיאות ביחס לכל הלוג, לא ל־endpoint
  });

// מטלת בית טיפוסית: "בודק זמינות" לרשימת כתובות. מה בודקים במטלה כזו: מקביליות מוגבלת, timeout,
// retry רק כשזה הגיוני, תוצאות בסדר הקלט, ושגיאה של אחד לא מפילה את כולם
export type Check = (url: string, signal: AbortSignal) => Promise<{ status: number }>;
export interface UrlResult {
  url: string;
  ok: boolean;
  status?: number;
  error?: string;
  attempts: number;
  ms: number;
}

export async function checkUrls(
  urls: readonly string[],
  check: Check,
  { concurrency = 5, timeoutMs = 3000, retries = 1, now = () => Date.now() }: { concurrency?: number; timeoutMs?: number; retries?: number; now?: () => number } = {},
): Promise<{ results: UrlResult[]; up: number; down: number }> {
  const results: UrlResult[] = new Array(urls.length);

  async function once(url: string): Promise<{ status?: number; error?: string }> {
    const ac = new AbortController();
    let timer: ReturnType<typeof setTimeout> | undefined;
    const timeout = new Promise<never>((_, reject) => {
      timer = setTimeout(() => {
        ac.abort();
        reject(new Error("timeout"));
      }, timeoutMs);
    });
    try {
      const r = await Promise.race([check(url, ac.signal), timeout]);
      return { status: r.status };
    } catch (err) {
      return { error: (err as Error).message };
    } finally {
      clearTimeout(timer);
    }
  }

  async function run(i: number) {
    const url = urls[i];
    const start = now();
    let attempts = 0;
    let last: { status?: number; error?: string } = {};
    while (attempts <= retries) {
      attempts++;
      last = await once(url);
      const retryable = last.error !== undefined || (last.status ?? 0) >= 500;
      if (!retryable) break; // 2xx/3xx/4xx — תשובה סופית; 404 לא יהפוך ל־200 בניסיון שני
    }
    const ok = last.status !== undefined && last.status < 400;
    results[i] = { url, ok, ...(last.status !== undefined ? { status: last.status } : {}), ...(last.error ? { error: last.error } : {}), attempts, ms: now() - start };
  }

  let next = 0;
  await Promise.all(Array.from({ length: Math.min(concurrency, urls.length) }, async () => {
    while (next < urls.length) await run(next++);
  }));
  const up = results.filter((r) => r.ok).length;
  return { results, up, down: results.length - up };
}

// P30-E25
// שאלת "בית" מתקדמת: event sourcing בקטן. במקום לשמור רק את המצב הנוכחי של הזמנה, שומרים את כל האירועים,
// ומשחזרים את המצב מהם — עם אותה מכונת מצבים מפרויקט 3. אירוע לא חוקי ← שגיאה שאומרת איפה
export type OrderEvent =
  | { orderId: number; type: "created"; at: string; totalCents: number }
  | { orderId: number; type: "paid"; at: string; amountCents: number }
  | { orderId: number; type: "shipped"; at: string }
  | { orderId: number; type: "cancelled"; at: string }
  | { orderId: number; type: "refunded"; at: string; amountCents: number };

export type Status = "pending_payment" | "paid" | "shipped" | "cancelled";

export interface OrderState {
  orderId: number;
  status: Status;
  totalCents: number;
  paidCents: number;
  refundedCents: number;
  history: { status: Status; at: string }[];
}

export class InvalidEvent extends Error {
  index: number;
  constructor(index: number, message: string) {
    super(`אירוע ${index}: ${message}`);
    this.index = index;
  }
}

const NEXT: Record<Status, Partial<Record<OrderEvent["type"], Status>>> = {
  pending_payment: { paid: "paid", cancelled: "cancelled" },
  paid: { shipped: "shipped", cancelled: "cancelled" },
  shipped: {},
  cancelled: {},
};

export function replay(events: readonly OrderEvent[]): OrderState {
  if (events.length === 0 || events[0].type !== "created") throw new InvalidEvent(0, "הזמנה חייבת להתחיל ב־created");
  const first = events[0];
  const s: OrderState = { orderId: first.orderId, status: "pending_payment", totalCents: first.totalCents, paidCents: 0, refundedCents: 0, history: [{ status: "pending_payment", at: first.at }] };
  for (let i = 1; i < events.length; i++) {
    const e = events[i];
    if (e.orderId !== s.orderId) throw new InvalidEvent(i, `שייך להזמנה ${e.orderId}`);
    if (Date.parse(e.at) < Date.parse(events[i - 1].at)) throw new InvalidEvent(i, "לפני האירוע הקודם");
    if (e.type === "refunded") {
      if (s.status !== "cancelled") throw new InvalidEvent(i, "החזר רק להזמנה מבוטלת");
      if (s.refundedCents + e.amountCents > s.paidCents) throw new InvalidEvent(i, "החזר גדול ממה ששולם");
      s.refundedCents += e.amountCents;
      continue;
    }
    if (e.type === "created") throw new InvalidEvent(i, "created כפול");
    const to = NEXT[s.status][e.type];
    if (!to) throw new InvalidEvent(i, `${e.type} לא אפשרי במצב ${s.status}`);
    if (e.type === "paid") {
      if (e.amountCents !== s.totalCents) throw new InvalidEvent(i, `שולם ${e.amountCents}, הסכום ${s.totalCents}`);
      s.paidCents = e.amountCents;
    }
    s.status = to;
    s.history.push({ status: to, at: e.at });
  }
  return s;
}

// כל ההזמנות מזרם אחד; הזמנה עם אירוע לא חוקי — נכנסת לרשימת השגיאות ולא מפילה את השאר
export function replayAll(events: readonly OrderEvent[]) {
  const byOrder = new Map<number, OrderEvent[]>();
  for (const e of events) (byOrder.get(e.orderId) ?? byOrder.set(e.orderId, []).get(e.orderId)!).push(e);
  const orders: OrderState[] = [];
  const errors: { orderId: number; message: string }[] = [];
  for (const [orderId, list] of [...byOrder].sort((a, b) => a[0] - b[0])) {
    try {
      orders.push(replay(list));
    } catch (err) {
      errors.push({ orderId, message: (err as Error).message });
    }
  }
  return { orders, errors };
}

P30-E28 · לנקות סודות לפני שמדביקים 🤖 רמה 3

💡 רמז 1

רשימה של כללים { type, re } עם הדגל g, ו־text.replace(re, fn) לכל אחד. כשצריך להשאיר חלק (המשתמש בכתובת, המילה Bearer) — קבוצות בסוגריים, ו־fn בונה מחדש.

💡 רמז 2

שורות .env: /^(\s*[A-Z0-9_]*(?:SECRET|PASSWORD|TOKEN|API_KEY)[A-Z0-9_]*\s*=\s*)(\S.*)$/gm — הדגל m הופך את ^ ו־$ לתחילת וסוף שורה.

✅ פתרון
// P30-E28
// מנקה טקסט לפני שמדביקים אותו לצ'אט של AI (או ל־issue, או ל־Slack): סודות לא יוצאים מהמחשב שלך
const RULES: { type: string; re: RegExp; keep?: (m: string, ...g: string[]) => string }[] = [
  // בלוק מפתח פרטי — מהשורה BEGIN ועד END
  { type: "private-key", re: /-----BEGIN [A-Z ]*PRIVATE KEY-----[\s\S]*?-----END [A-Z ]*PRIVATE KEY-----/g },
  // כתובת חיבור עם סיסמה: משאירים את המשתמש והשרת, מסתירים רק את הסיסמה
  { type: "url-password", re: /\b([a-z][a-z0-9+.-]*:\/\/[^\s:@/]+:)([^\s@/]+)(@)/gi, keep: (_m, a, _p, c) => `${a}[REDACTED:url-password]${c}` },
  { type: "jwt", re: /\beyJ[A-Za-z0-9_-]{5,}\.[A-Za-z0-9_-]{5,}\.[A-Za-z0-9_-]{5,}/g },
  { type: "api-key", re: /\bsk-[A-Za-z0-9_-]{16,}/g },
  { type: "aws-key", re: /\bAKIA[0-9A-Z]{16}\b/g },
  { type: "github-token", re: /\bgh[pousr]_[A-Za-z0-9]{30,}/g },
  { type: "bearer", re: /(\bBearer\s+)([A-Za-z0-9._~+/-]{8,}=*)/g, keep: (_m, a) => `${a}[REDACTED:bearer]` },
  // שורות של .env: KEY=value, כשהשם נשמע כמו סוד
  { type: "env-secret", re: /^(\s*[A-Z0-9_]*(?:SECRET|PASSWORD|PASSWD|TOKEN|API_KEY|PRIVATE_KEY)[A-Z0-9_]*\s*=\s*)(\S.*)$/gm, keep: (_m, a) => `${a}[REDACTED:env-secret]` },
];

export function redactForPrompt(text: string): { text: string; found: string[] } {
  const found = new Set<string>();
  let out = text;
  for (const { type, re, keep } of RULES) {
    out = out.replace(re, (...args) => {
      // ערך שכבר הוסתר (מכלל קודם) — לא מסתירים שוב
      if (String(args[0]).includes("[REDACTED:")) return args[0];
      found.add(type);
      return keep ? keep(...(args as [string, ...string[]])) : `[REDACTED:${type}]`;
    });
  }
  return { text: out, found: [...found].sort() };
}

הכלי הזה הוא רשת ביטחון, לא היתר: מה שלא מתאים לאף כלל (סיסמה בתוך JSON, נתוני לקוחות) עובר כמו שהוא. הכלל הראשון נשאר "לא מדביקים מה שלא צריך".

⚠️ טעויות נפוצות
  • להסתיר את כל הכתובת במקום רק את הסיסמה — ה־AI לא יכול לעזור עם "connection refused" אם הוא לא רואה host ופורט.
  • regex של JWT בלי אורך מינימלי — כל מילה שמתחילה ב־eyJ ויש בה שתי נקודות נמחקת.

P30-E29 · סקירה של diff שה־AI הציע 🤖 רמה 3

💡 רמז 1

שורת @@ -10,6 +10,5 @@ אומרת: החלק מתחיל בשורה 10 בקובץ הישן ובשורה 10 בחדש. משם: שורה שמתחילה ב־+ מקדמת רק את המונה החדש, - רק את הישן, ורווח — את שניהם.

💡 רמז 2

שורות +++ ו־--- מתחילות גם הן ב־+ ו־−. לבדוק אותן לפני שבודקים שורות שנוספו או נמחקו.

✅ פתרון
// P30-E29
// עובר על diff (הפורמט של git diff) שעוזר AI הציע, ומסמן שינויים שחייבים עין אנושית
export interface Finding {
  file: string;
  line: number; // בשורה שנוספה — מספר השורה בקובץ החדש; בשורה שנמחקה — בקובץ הישן
  rule: "test-disabled" | "assertion-removed" | "new-dependency" | "dangerous-eval" | "type-escape";
}

export function reviewDiff(diff: string): Finding[] {
  const findings: Finding[] = [];
  let file = "";
  let oldLine = 0;
  let newLine = 0;
  for (const raw of diff.split("\n")) {
    if (raw.startsWith("diff --git")) continue;
    if (raw.startsWith("+++ ")) {
      file = raw.slice(4).replace(/^b\//, "");
      continue;
    }
    if (raw.startsWith("--- ")) continue;
    const hunk = /^@@ -(\d+)(?:,\d+)? \+(\d+)(?:,\d+)? @@/.exec(raw);
    if (hunk) {
      oldLine = +hunk[1];
      newLine = +hunk[2];
      continue;
    }
    if (!file) continue;
    const add = (rule: Finding["rule"], line: number) => findings.push({ file, line, rule });
    if (raw.startsWith("+")) {
      const code = raw.slice(1);
      // AI "מתקן" בדיקה שנכשלת בכך שהוא מכבה אותה
      if (/\b(?:test|it|describe)\.(?:skip|only)\s*\(/.test(code)) add("test-disabled", newLine);
      if (/\beval\s*\(|\bnew Function\s*\(/.test(code)) add("dangerous-eval", newLine);
      if (/@ts-ignore|@ts-expect-error|\bas any\b/.test(code)) add("type-escape", newLine);
      // חבילה חדשה: לבדוק שהיא קיימת, מי מתחזק אותה, ושבאמת צריך אותה
      if (/(^|\/)package\.json$/.test(file) && /^\s*"(?!version")[^"]+"\s*:\s*"[\^~>=<]*\d/.test(code)) add("new-dependency", newLine);
      newLine++;
    } else if (raw.startsWith("-")) {
      // בדיקה שאיבדה assertion — "עוברת" כי היא כבר לא בודקת כלום
      if (/\bassert\b|\bexpect\s*\(/.test(raw.slice(1))) add("assertion-removed", oldLine);
      oldLine++;
    } else {
      oldLine++;
      newLine++;
    }
  }
  return findings;
}
⚠️ טעויות נפוצות
  • לספור את שורות ה־+++/--- כשורות שנוספו/נמחקו.
  • לקדם את שני המונים בכל שורה — מספרי השורות "זזים" אחרי המחיקה הראשונה.
  • /skip/ בלי ההקשר — items.skip או משתנה בשם skipped מסומנים בטעות.

זה לא מחליף קריאה. אבל אלה בדיוק השינויים שקל לפספס ב־diff של 300 שורות: "הבדיקות עוברות" — כי ה־AI כיבה את זו שנכשלה.

💡 רמז 1

💡 רמז 2

בפרק 21 (OWASP API Top 10): הבעיה הראשונה ברשימה היא בדיוק מה שחסר כאן ב־WHERE.

✅ פתרון
// P30-E30
// GET /tasks?status=&sort=&order= — אחרי תיקון של הקוד שה־AI הציע
interface Db {
  query(text: string, params: unknown[]): Promise<{ rows: unknown[] }>;
}

// עמודות שמותר למיין לפיהן. שם עמודה לא יכול להיות פרמטר ($2) — לכן רשימה סגורה
const SORTS: Record<string, string> = { created: "created_at", due: "due_date", title: "title" };

export async function listTasks(
  db: Db,
  userId: number,
  { status, sort = "created", order = "asc" }: { status?: string; sort?: string; order?: string } = {}
) {
  const column = SORTS[sort];
  if (!column) throw new RangeError(`אי אפשר למיין לפי ${sort}`);
  if (order !== "asc" && order !== "desc") throw new RangeError("order: asc או desc");
  // תקלה 1: בלי user_id — כל משתמש ראה את המשימות של כולם
  const params: unknown[] = [userId];
  let sql = "SELECT id, title, status, due_date FROM tasks WHERE user_id = $1";
  if (status !== undefined) {
    params.push(status); // תקלה 2: ערך מהמשתמש — רק כפרמטר, אף פעם לא בתוך הטקסט
    sql += ` AND status = $${params.length}`;
  }
  sql += ` ORDER BY ${column} ${order.toUpperCase()}`; // תקלה 3: רק מהרשימה הסגורה
  const { rows } = await db.query(sql, params);
  return rows;
}
  1. אין WHERE user_id — כל משתמש מחובר רואה את המשימות של כולם (Broken Object Level Authorization, מקום 1 ב־OWASP API Top 10).
  2. status בתוך הטקסט — SQL injection. x' OR '1'='1 מחזיר הכול, ומשהו גרוע יותר מוחק. ערך מהמשתמש — רק כפרמטר ($2).
  3. sort ו־order בלי רשימה סגורה — columns[sort] ?? sort נראה כמו הגנה, אבל כל ערך לא מוכר עובר כמו שהוא. שם עמודה וכיוון מיון לא יכולים להיות פרמטרים, ולכן רק ערכים מרשימה שאתה כתבת.
⚠️ טעויות נפוצות
  • "לנקות" את status עם replace של גרשיים — תמיד יש עוד דרך. פרמטרים, לא ניקוי.
  • לתקן את ה־injection ולשכוח את ה־user_id: הקוד "מאובטח" ועדיין מחזיר משימות של אחרים. אוטומציה לא תתפוס את זה; רק מי שיודע מה ה־endpoint צריך לעשות.
