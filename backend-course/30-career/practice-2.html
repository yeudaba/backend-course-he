type Stats = { p50Before: number; p95Before: number; p50After: number; p95After: number; speedup: number; bullet: string | null };
export function latencyClaim(before: number[], after: number[], what: string): Stats
  • percentile בשיטת nearest-rank: ממיינים, ולוקחים את האיבר במקום ceil(p/100 × n) (מ־1). מעגלים למספר שלם.

  • מערך ריק, או ‏p95After שהוא 0 ← RangeError.
