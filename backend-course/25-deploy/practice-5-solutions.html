P25-E83 · ‏seed.ts ← seed.mts 🧠 הבנה רמה 2

💡 רמז 1

פרויקט ה־Nest הוא CommonJS (פרק 20) — בלי "type": "module".

💡 רמז 2

מה בודק טיפוסים, ומה רק מריץ?

✅ פתרון
  1. בלי "type": "module" ב־package.json, קובץ .ts הוא CommonJS — וב־CommonJS אין await ברמה העליונה.
  2. .mts = "המודול הזה הוא ESM", בלי קשר ל־package.json — כמו .mjs ב־JavaScript. TypeScript בודק אותו כ־ESM, ו־Node מריץ אותו כ־ESM.
  3. זה הופך את כל הפרויקט ל־ESM — ה־build של Nest, ה־imports בלי סיומת, Jest. שינוי גדול בשביל קובץ אחד.
  4. כי Node (עם strip-types) פשוט הריץ אותו — ולא בדק טיפוסים. רק npm run typecheck על כל הפרויקט ראה את השגיאה — ואף אחד לא הריץ אותו עד ה־CI.
⚠️ טעויות נפוצות
  • "לתקן" עם main().then(...) בלי await — עובד, אבל מסתיר שהבעיה היא סוג המודול, ותחזור בקובץ הבא.

P25-E84 · ‏app.listen(PORT, "0.0.0.0") 🧠 הבנה רמה 2

💡 רמז 1

‏127.0.0.1 — "רק מתוך אותה מכונה". ומה זה "אותה מכונה" בתוך קונטיינר?

💡 רמז 2

ה־proxy של Render פונה לקונטיינר דרך הרשת שלו, לא דרך localhost של הקונטיינר.

✅ פתרון
  1. 127.0.0.1 — רק חיבורים מתוך אותה מכונה (loopback). ‏0.0.0.0 — מכל הממשקים: גם מהרשת.
  2. הקונטיינר הוא "מכונה" משלו. ה־proxy (או ה־port mapping של Docker) מתחבר מבחוץ לממשק הרשת של הקונטיינר — והשרת לא מאזין שם. ‏Render: ‏"No open ports detected", וה־deploy נכשל.
  3. הדפדפן ו־curl רצים על אותה מכונה, אז loopback מספיק.
  4. כן, קצת: כל מי שברשת יכול לפנות לפורט 3000 אצלך. לכן בפיתוח בלי Docker — עדיף 127.0.0.1 (או ש־firewall של ה־Mac שואל). בקונטיינר זה בטוח — רק מה שפותחים עם -p נגיש מבחוץ.
⚠️ טעויות נפוצות
  • "עובד אצלי" עם localhost — וב־Docker השרת עולה, ה־HEALTHCHECK (שרץ בפנים) עובר, ומבחוץ אין תשובה.

P25-E85 · למה Auto-Deploy כבוי? 🧠 הבנה רמה 1

💡 רמז 1

Auto-Deploy מתחיל לבנות ברגע ה־push — במקביל ל־CI, בלי לדעת עליו.

💡 רמז 2

מי מריץ את ה־migrations?

✅ פתרון
  1. Render בונה ומעלה את הקוד החדש, בזמן שה־CI עוד מריץ בדיקות — וה־migration עוד לא רץ. הקוד החדש מחפש עמודה שלא קיימת ← 500, עד שה־CI מגיע לשלב ה־migrations (ואם הבדיקות נכשלות — לעולם לא).
  2. הקוד עולה לאוויר למרות הבדיקה האדומה. ה־CI הופך ל"דיווח", לא לשער.
  3. בערך יחד — ובדיקת העשן של ה־CI עלולה לפגוש גרסה שעוד מתחלפת. ועם hook — Render בונה פעם שנייה. הסדר לא מוגדר, וזו כל הבעיה.

עם hook: בדיקות ← migrations ← deploy ← עשן. סדר אחד, ושער אחד.

⚠️ טעויות נפוצות
  • להשאיר את שניהם דולקים "ליתר ביטחון" — שני deploys לכל push, ואחד מהם לפני ה־migrations.

// P25-E89: הפרמטרים של smoke.mjs
function positiveInt(env: Record<string, string | undefined>, name: string, fallback: number): number {
  const v = env[name];
  if (v === undefined || v === "") return fallback;
  if (!/^\d+$/.test(v) || Number(v) <= 0) throw new Error(`${name} לא תקין`);
  return Number(v);
}

export function parseSmokeEnv(env: Record<string, string | undefined>) {
  const raw = env.BASE_URL;
  if (!raw) throw new Error("חסר BASE_URL");
  let u: URL;
  try {
    u = new URL(raw);
  } catch {
    throw new Error("BASE_URL לא תקין");
  }
  if (u.protocol !== "http:" && u.protocol !== "https:") throw new Error("BASE_URL לא תקין");
  const baseUrl = raw.replace(/\/+$/, "");
  const commit = env.EXPECT_COMMIT || null;
  if (commit !== null && !/^[0-9a-f]{7,40}$/.test(commit)) throw new Error("EXPECT_COMMIT לא תקין");
  const timeoutSeconds = positiveInt(env, "TIMEOUT_SECONDS", 900);
  const intervalSeconds = positiveInt(env, "INTERVAL_SECONDS", 10);
  if (intervalSeconds > timeoutSeconds) throw new Error("INTERVAL_SECONDS גדול מ־TIMEOUT_SECONDS");
  return { baseUrl, expectCommit: commit, timeoutSeconds, intervalSeconds };
}
⚠️ טעויות נפוצות
  • Number(env.TIMEOUT_SECONDS) || 900 — ‏"abc" הופך בשקט ל־900, ו־"-5" עובר.

// P25-E90: ההודעות של בדיקת העשן
const short = (sha: string) => sha.slice(0, 7);

export function waitingLine(n: number, v: { status: number; commit?: string; error?: string }, expect: string | null): string {
  let line = `#${n}: /version ← ${v.status || v.error}`;
  if (v.commit) {
    line += `, commit ${short(v.commit)}`;
    if (expect) line += ` (מחכים ל־${short(expect)})`;
  }
  return line;
}

export function notReadyLine(n: number, status: number): string {
  return `#${n}: הגרסה החדשה רצה, אבל /ready ← ${status}`;
}

export function successLine(base: string, commit: string, seconds: number): string {
  return `✔ ${base} — commit ${short(commit)}, ready (${Math.round(seconds)}s)`;
}

export function failLine(expect: string | null, timeoutSeconds: number): string {
  return `✖ הגרסה${expect ? ` ${short(expect)}` : ""} לא עלתה תוך ${timeoutSeconds} שניות`;
}

שורה טובה ב־CI עונה על "מה קרה" בלי לפתוח קוד: איזה ניסיון, מה השרת ענה, איזו גרסה רצה, ולמה מחכים.

export function readmeProblems(md: string, { user, repo, liveUrl }: { user: string; repo: string; liveUrl: string }): string[] {
  const lines = md.split("\n");
  const problems: string[] = [];
  const badge = `![CI](https://github.com/${user}/${repo}/actions/workflows/ci.yml/badge.svg)`;
  if (!lines.slice(0, 5).some((l) => l.includes(badge))) problems.push("badge");
  if (!md.includes(`${liveUrl.replace(/\/+$/, "")}/docs`)) problems.push("live-link");
  if (!lines.some((l) => l.includes("NestJS") && l.includes("PostgreSQL") && l.includes("Docker"))) problems.push("architecture");
  if (/YOUR-(USER|SERVICE)/.test(md)) problems.push("placeholder");
  return problems;
}

// P25-E93: הכתובת של ה־API בדף
export function apiBase(html: string, pageUrl: string): string {
  const page = new URL(pageUrl);
  if (page.hostname === "localhost" || page.hostname === "127.0.0.1") return "http://localhost:3000";
  let api: string | undefined;
  for (const [tag] of html.matchAll(/<meta\b[^>]*>/gi)) {
    if (/\bname="api-url"/i.test(tag)) api = /\bcontent="([^"]*)"/i.exec(tag)?.[1];
  }
  if (!api) throw new Error("חסר api-url");
  if (api.includes("YOUR-SERVICE")) throw new Error("api-url לא הוגדר");
  if (page.protocol === "https:" && api.startsWith("http:")) throw new Error("mixed content");
  return api.replace(/\/+$/, "");
}
⚠️ טעויות נפוצות
  • regex שמניח ש־name לפני content — ואז עורך HTML שמסדר תכונות שובר את הדף בפריסה.

// P25-E94: הכתובת של Neon — לפני שמדביקים
export function neonUrlProblems(url: string): string[] {
  let u: URL;
  try {
    u = new URL(url);
  } catch {
    return ["not-a-url"];
  }
  const problems: string[] = [];
  if (u.protocol !== "postgres:" && u.protocol !== "postgresql:") problems.push("protocol");
  if (!u.password) problems.push("password");
  if (!u.hostname.endsWith(".neon.tech")) problems.push("host");
  if (!u.pathname.slice(1)) problems.push("database");
  if (u.searchParams.get("sslmode") !== "require") problems.push("sslmode");
  return problems;
}

export function shellQuote(value: string): string {
  return "'" + value.replaceAll("'", "'\\''") + "'";
}

// P25-E95: JWT_SECRET לפרודקשן
import { randomBytes } from "node:crypto";

export function generateSecret(bytes = 48): string {
  if (bytes < 32) throw new RangeError("לפחות 32 בתים");
  return randomBytes(bytes).toString("base64url");
}

export function secretProblems(secret: string): string[] {
  const problems: string[] = [];
  if (secret.length < 32) problems.push("too-short");
  if (secret !== secret.trim()) problems.push("whitespace");
  if (/change-?me|secret|password|example|dev-only/i.test(secret)) problems.push("placeholder");
  if (new Set(secret).size < 10) problems.push("low-variety");
  return problems;
}

48 בתים אקראיים = 64 תווי base64url, ‏384 ביטים — הרבה מעבר למה ש־HS256 צריך. ו־whitespace — כי העתקה מהטרמינל לממשק של Render מוסיפה לפעמים ירידת שורה, והטוקנים "לא מתאמתים" בין שני שרתים שאחד מהם עם הרווח.

⚠️ טעויות נפוצות
  • Math.random().toString(36) — לא קריפטוגרפי, וקצר.

  const p: string[] = [];
  const db = env.DATABASE_URL;
  if (!db) p.push("DATABASE_URL: חסר");
  else if (!/[?&]sslmode=require(&|$)/.test(db)) p.push("DATABASE_URL: חסר sslmode=require");
  if (!env.JWT_SECRET) p.push("JWT_SECRET: חסר");
  else if (env.JWT_SECRET.length < 32) p.push("JWT_SECRET: קצר מ־32 תווים");
  const cors = env.CORS_ORIGINS;
  if (!cors) p.push("CORS_ORIGINS: חסר");
  else if (cors.split(",").map((s) => s.trim()).some((o) => !/^https:\/\/[^/*\s]+$/.test(o))) p.push("CORS_ORIGINS: רק https://host, בלי נתיב");
  if (env.TRUST_PROXY !== "1") p.push("TRUST_PROXY: ב־Render יש proxy אחד — 1");
  if ("PORT" in env) p.push("PORT: Render קובע לבד");
  if ("NODE_ENV" in env && env.NODE_ENV !== "production") p.push("NODE_ENV: רק production (או לא להגדיר)");
  if ("RENDER_GIT_COMMIT" in env) p.push("RENDER_GIT_COMMIT: Render קובע לבד");
  return p;
}

P25-E24 · אתגר: linter ל־workflows 🤖 💻 כתיבת קוד 🤖 רמה 5

💡 רמז 1

💡 רמז 2

✅ פתרון
// P25-E24
import { load as parseYaml } from "js-yaml";

export interface Finding {
  job: string | null;
  rule: "yaml" | "permissions" | "timeout" | "unpinned-action" | "npm-install" | "secret-in-run" | "unknown-needs" | "deploy-without-needs" | "pr-secrets";
}

interface Step { uses?: string; run?: string; env?: Record<string, string> }
interface Job { needs?: string | string[]; "timeout-minutes"?: number; steps?: Step[]; env?: Record<string, string>; if?: string }

export function lintWorkflow(text: string): Finding[] {
  let wf: { on?: unknown; permissions?: unknown; jobs?: Record<string, Job>; env?: Record<string, string> } & Record<string, unknown>;
  try {
    wf = parseYaml(text) as typeof wf;
  } catch {
    return [{ job: null, rule: "yaml" }];
  }
  const out: Finding[] = [];
  const add = (job: string | null, rule: Finding["rule"]) => out.push({ job, rule });
  if (wf.permissions === undefined) add(null, "permissions");
  const on = wf.on ?? wf["true"];
  const onPr = Array.isArray(on) ? on.includes("pull_request") : typeof on === "object" && on !== null ? "pull_request" in on : on === "pull_request";
  const jobs = wf.jobs ?? {};
  for (const [name, job] of Object.entries(jobs)) {
    if (job["timeout-minutes"] === undefined) add(name, "timeout");
    const steps = job.steps ?? [];
    if (steps.some((s) => s.uses && (!s.uses.includes("@") || /@(main|master)$/.test(s.uses)))) add(name, "unpinned-action");
    if (steps.some((s) => /\bnpm (install|i)\b/.test(s.run ?? ""))) add(name, "npm-install");
    // ${{ secrets.X }} בתוך הסקריפט עצמו: הערך נכנס לטקסט של הסקריפט (וגם injection). דרך env — כמשתנה
    if (steps.some((s) => /\$\{\{\s*secrets\./.test(s.run ?? ""))) add(name, "secret-in-run");
    const needs = [job.needs ?? []].flat();
    if (needs.some((n) => !(n in jobs))) add(name, "unknown-needs");
    if (/deploy/i.test(name) && needs.length === 0) add(name, "deploy-without-needs");
    // job שרץ גם על pull request ומשתמש בסודות, בלי if שמגביל ל־push
    const usesSecrets = /secrets\./.test(JSON.stringify(job));
    if (onPr && usesSecrets && !/push|refs\/heads/.test(job.if ?? "")) add(name, "pr-secrets");
  }
  return out;
}

הכלי האמיתי: actionlint — מריצים אותו ב־CI על ה־workflows עצמם. ולמה secret-in-run? ${{ }} מוחלף בטקסט לפני שה־shell רץ: הסוד הופך לחלק מהסקריפט (ואם הערך מגיע ממשתמש — script injection). דרך env הוא משתנה רגיל.

⚠️ טעויות נפוצות

// P25-E101: למה ה־deploy נכשל? לקרוא את הלוג
const PREFIXES = ["הגדרות שרת לא תקינות — ", "הגדרות לא תקינות ב־.env — "];

function candidates(line: string): string[] {
  try {
    const o = JSON.parse(line) as { msg?: unknown; err?: { message?: unknown } };
    return [o?.msg, o?.err?.message].filter((x): x is string => typeof x === "string");
  } catch {
    return [line];
  }
}

export function configProblemsFromLogs(text: string) {
  const out: { name: string; message: string }[] = [];
  const seen = new Set<string>();
  for (const line of text.split("\n")) {
    for (const c of candidates(line)) {
      for (const prefix of PREFIXES) {
        const i = c.indexOf(prefix);
        if (i === -1) continue;
        for (const part of c.slice(i + prefix.length).split("; ")) {
          const j = part.indexOf(": ");
          if (j === -1) continue;
          const name = part.slice(0, j).trim();
          const message = part.slice(j + 2).trim();
          if (seen.has(`${name}\n${message}`)) continue;
          seen.add(`${name}\n${message}`);
          out.push({ name, message });
        }
      }
    }
  }
  return out;
}
