💡 רמז 1

indexOf("=") — רק הראשון מפריד.

💡 רמז 2

מירכאות: אם הערך מתחיל ונגמר באותה מירכאה — slice(1, -1).

✅ פתרון
// P13-E10
export function parseDotenv(text) {
  const result = {};
  const lines = text.split("\n");
  for (let i = 0; i < lines.length; i++) {
    const line = lines[i].trim();
    if (line === "" || line.startsWith("#")) continue;
    const eq = line.indexOf("=");
    if (eq === -1) throw new Error(`שורה ${i + 1} לא תקינה`);
    const key = line.slice(0, eq).trim();
    let value = line.slice(eq + 1).trim();
    const q = value[0];
    if ((q === '"' || q === "'") && value.length >= 2 && value.at(-1) === q) {
      value = value.slice(1, -1);
    }
    result[key] = value;
  }
  return result;
}

console.log(parseDotenv('# db\nDATABASE_URL = postgres://a:b@h/db?x=1\nAPP_NAME="Dana Shop"\n\nEMPTY='));
{
  DATABASE_URL: 'postgres://a:b@h/db?x=1',
  APP_NAME: 'Dana Shop',
  EMPTY: ''
}
⚠️ טעויות נפוצות
  • line.split("=") — ערך עם = נחתך.

// P13-E15
const LEVELS = ["debug", "info", "warn", "error"];

export function loadConfig(env) {
  const problems = [];
  if (!env.DATABASE_URL) problems.push("DATABASE_URL חסר");
  if (!env.JWT_SECRET) problems.push("JWT_SECRET חסר");
  else if (env.JWT_SECRET.length < 32) problems.push("JWT_SECRET קצר מדי");
  const port = env.PORT === undefined ? 3000 : Number(env.PORT);
  if (!Number.isInteger(port) || port < 1 || port > 65535) problems.push("PORT לא תקין");
  const logLevel = env.LOG_LEVEL ?? "info";
  if (!LEVELS.includes(logLevel)) problems.push("LOG_LEVEL לא תקין");
  if (problems.length > 0) {
    throw new Error(`הגדרות לא תקינות: ${problems.join("; ")}`);
  }
  return Object.freeze({ port, databaseUrl: env.DATABASE_URL, jwtSecret: env.JWT_SECRET, logLevel });
}

export function redact(config) {
  return Object.fromEntries(
    Object.entries(config).map(([k, v]) => [k, /secret|password|url/i.test(k) ? "***" : v])
  );
}

try {
  loadConfig({ JWT_SECRET: "short" });
} catch (err) {
  console.log(err.message);
}
console.log(redact(loadConfig({ DATABASE_URL: "postgres://u:p@h/db", JWT_SECRET: "x".repeat(32) })));

P13-E16 · הגדרות לפי סביבה 🤖 💻 כתיבת קוד 🤖 רמה 3

💡 רמז 1

אובייקט ברירות מחדל לפי סביבה, ואז env.X ?? defaults.x.

💡 רמז 2

ב־production: אין ברירות מחדל לשדות החובה — בדיוק כדי שלא "יעבוד בטעות" מול מסד לא נכון.

✅ פתרון
// P13-E16
const DEFAULTS = {
  development: { databaseUrl: "postgres://localhost:5432/app_dev", logLevel: "debug", cors: "*" },
  test: { databaseUrl: "postgres://localhost:5432/app_test", logLevel: "error", cors: "*" },
  production: { logLevel: "info" },
};

export function configFor(env) {
  const name = env.NODE_ENV ?? "development";
  const defaults = DEFAULTS[name];
  if (!defaults) {
    throw new Error(`NODE_ENV לא מוכר: ${name}`);
  }
  if (name === "production") {
    for (const key of ["DATABASE_URL", "CORS_ORIGIN"]) {
      if (!env[key]) throw new Error(`${key} חובה ב־production`);
    }
  }
  return {
    env: name,
    databaseUrl: env.DATABASE_URL ?? defaults.databaseUrl,
    logLevel: env.LOG_LEVEL ?? defaults.logLevel,
    cors: env.CORS_ORIGIN ?? defaults.cors,
  };
}

console.log(configFor({}));
console.log(configFor({ NODE_ENV: "production", DATABASE_URL: "postgres://prod/db", CORS_ORIGIN: "https://shop.example.com" }));
{
  env: 'development',
  databaseUrl: 'postgres://localhost:5432/app_dev',
  logLevel: 'debug',
  cors: '*'
}
{
  env: 'production',
  databaseUrl: 'postgres://prod/db',
  logLevel: 'info',
  cors: 'https://shop.example.com'
}
⚠️ טעויות נפוצות
  • ברירת מחדל ל־DATABASE_URL גם בייצור — שרת שפריסה שלו "שכחה" משתנה יעלה מול localhost ויראה תקין.

// P13-E22: מתוקן
export function buildConfig(env, log = console.log) {
  for (const name of ["DATABASE_URL", "JWT_SECRET"]) {
    if (!env[name]) {
      throw new Error(`חסר ${name}`);
    }
  }
  const config = { port: Number(env.PORT ?? 3000), databaseUrl: env.DATABASE_URL, jwtSecret: env.JWT_SECRET };
  log(`config: ${JSON.stringify({ ...config, databaseUrl: "***", jwtSecret: "***" })}`);
  return config;
}
