P13-E06 · בדיקת ^ 🤖 💻 כתיבת קוד 🤖 רמה 2

💡 רמז 2

השוואת גרסאות: קודם major, אחר כך minor, אחר כך patch.

✅ פתרון
// P13-E06
export function parseVersion(v) {
  const parts = String(v).split(".");
  if (parts.length !== 3 || !parts.every((p) => /^\d+$/.test(p))) {
    throw new Error(`גרסה לא תקינה: ${v}`);
  }
  return parts.map(Number);
}

function compare(a, b) {
  for (let i = 0; i < 3; i++) {
    if (a[i] !== b[i]) return a[i] - b[i];
  }
  return 0;
}

export function satisfiesCaret(version, range) {
  const v = parseVersion(version);
  const min = parseVersion(range.replace(/^\^/, ""));
  if (compare(v, min) < 0) return false;
  if (min[0] === 0) return v[0] === 0 && v[1] === min[1];
  return v[0] === min[0];
}

console.log(satisfiesCaret("1.9.0", "^1.2.3"), satisfiesCaret("2.0.0", "^1.2.3"), satisfiesCaret("0.3.0", "^0.2.3"));
true false false
⚠️ טעויות נפוצות
  • השוואת מחרוזות ("1.10.0" < "1.9.0" מילונית!) — חייבים מספרים.

// P13-E07
import fs from "node:fs/promises";
import path from "node:path";

export function depsReport(pkg) {
  const prod = Object.keys(pkg.dependencies ?? {}).sort();
  const dev = Object.keys(pkg.devDependencies ?? {}).sort();
  return { prod, dev, total: prod.length + dev.length };
}

export function scriptLines(pkg) {
  return Object.entries(pkg.scripts ?? {}).map(([name, cmd]) => {
    const run = ["start", "test"].includes(name) ? `npm ${name}` : `npm run ${name}`;
    return `${run}  →  ${cmd}`;
  });
}

export async function readPackage(dir) {
  try {
    return JSON.parse(await fs.readFile(path.join(dir, "package.json"), "utf8"));
  } catch (err) {
    if (err.code === "ENOENT") throw new Error("לא נמצא package.json");
    throw err;
  }
}

P13-E13 · מיון ובחירת גרסאות 🤖 💻 כתיבת קוד 🤖 רמה 3

💡 רמז 1

parse פעם אחת, ואז השוואה לפי major, minor, patch.

💡 רמז 2

~: אותו major ו־minor, ≥ המינימום.

✅ פתרון
// P13-E13
const parse = (v) => v.split(".").map(Number);

export function compareVersions(a, b) {
  const x = parse(a);
  const y = parse(b);
  for (let i = 0; i < 3; i++) {
    if (x[i] !== y[i]) return x[i] - y[i];
  }
  return 0;
}

export function satisfies(version, range) {
  const op = range[0];
  if (op !== "^" && op !== "~") return compareVersions(version, range) === 0;
  const min = range.slice(1);
  if (compareVersions(version, min) < 0) return false;
  const [vMaj, vMin] = parse(version);
  const [mMaj, mMin] = parse(min);
  return op === "^" ? vMaj === mMaj : vMaj === mMaj && vMin === mMin;
}

export function maxSatisfying(versions, range) {
  const ok = versions.filter((v) => satisfies(v, range)).sort(compareVersions);
  return ok.at(-1) ?? null;
}

const available = ["4.17.1", "4.18.2", "4.21.0", "5.0.0", "5.1.0"];
console.log(maxSatisfying(available, "^4.18.0"), maxSatisfying(available, "~4.18.0"), maxSatisfying(available, "^6.0.0"));
4.21.0 4.18.2 null
⚠️ טעויות נפוצות
  • versions.sort() בלי comparator — מיון מילוני: "4.9.0" אחרי "4.10.0".

export function isValidPackageName(name) {
  if (typeof name !== "string" || name.length < 1 || name.length > 214) return false;

// P13-E34: איזה סוג טווח?
export function rangeKind(range) {
  const m = String(range).trim().match(/^([~^]?)\d+\.\d+\.\d+$/);
  if (!m) return "other";
  if (m[1] === "^") return "caret";
  if (m[1] === "~") return "tilde";
  return "exact";
}

export function countKinds(deps) {
  const counts = { caret: 0, tilde: 0, exact: 0, other: 0 };
  for (const range of Object.values(deps)) counts[rangeKind(range)]++;
  return counts;
}

💡 רמז 2

מיון מפתחות: Object.fromEntries(Object.entries(obj).sort(([a], [b]) => a.localeCompare(b))). מחיקה מהחלק השני — על עותק: const rest = { ...pkg[other] }; delete rest[name];

✅ פתרון
// P13-E35: להוסיף תלות
const sortKeys = (obj) => Object.fromEntries(Object.entries(obj).sort(([a], [b]) => (a < b ? -1 : a > b ? 1 : 0)));

export function addDependency(pkg, name, version, { dev = false } = {}) {
  const target = dev ? "devDependencies" : "dependencies";
  const other = dev ? "dependencies" : "devDependencies";
  const result = { ...pkg, [target]: sortKeys({ ...pkg[target], [name]: `^${version}` }) };
  if (pkg[other] && name in pkg[other]) {
    const rest = { ...pkg[other] };
    delete rest[name];
    if (Object.keys(rest).length === 0) delete result[other];
    else result[other] = rest;
  }
  return result;
}

P13-E36 · מה ישן? (npm outdated) 🤖 💻 כתיבת קוד 🤖 רמה 3

💡 רמז 1

פונקציית עזר: const parts = (v) => v.replace(/^[~^]/, "").split(".").map(Number); ‏— ‏"4.1.12" ← [4, 1, 12]. משווים מספרים, לא מחרוזות!

💡 רמז 2

לולאה על שלושת החלקים: החלק הראשון שבו l[i] !== c[i] קובע. אם l[i] < c[i] — הנוכחית חדשה יותר (לא outdated).

✅ פתרון
// P13-E36: מה ישן?
const parts = (v) => v.replace(/^[~^]/, "").split(".").map(Number);
const TYPES = ["major", "minor", "patch"];

export function outdated(deps, latest) {
  const result = [];
  for (const [name, range] of Object.entries(deps)) {
    if (!(name in latest)) continue;
    const c = parts(range);
    const l = parts(latest[name]);
    for (let i = 0; i < 3; i++) {
      if (l[i] === c[i]) continue;
      if (l[i] > c[i]) {
        result.push({ name, current: c.join("."), latest: latest[name], type: TYPES[i] });
      }
      break;
    }
  }
  return result.sort((a, b) => (a.name < b.name ? -1 : 1));
}

P13-E37 · מה יש ב־lockfile 🤖 💻 כתיבת קוד 🤖 רמה 3

💡 רמז 1

✅ פתרון
// P13-E37: מה יש ב־lockfile
const NM = "node_modules/";

export function lockSummary(lock) {
  let total = 0;
  let dev = 0;
  const sets = new Map();
  for (const [key, info] of Object.entries(lock.packages ?? {})) {
    if (key === "") continue;
    total++;
    if (info.dev === true) dev++;
    const name = key.slice(key.lastIndexOf(NM) + NM.length);
    if (!sets.has(name)) sets.set(name, new Set());
    sets.get(name).add(info.version);
  }
  const versions = new Map();
  for (const [name, set] of sets) versions.set(name, [...set].sort());
  const duplicates = [...versions].filter(([, v]) => v.length > 1).map(([n]) => n).sort();
  return { total, prod: total - dev, dev, versions, duplicates };
}

    "": { name: "app" },
    "node_modules/express": { version: "5.1.0" },
    "node_modules/debug": { version: "4.4.1" },
    "node_modules/send/node_modules/debug": { version: "2.6.9" },
    "node_modules/@types/node": { version: "24.3.0", dev: true },
  },
};

P13-E27 · העלאת גרסה (semver) 🤖 💻 כתיבת קוד 🤖 רמה 3

💡 רמז 1

קודם בדיקה: /^\d+\.\d+\.\d+$/.test(version) (הנקודה עם \, אחרת היא "כל תו"). אחר כך: const [major, minor, patch] = version.split(".").map(Number).

💡 רמז 2

בכל case אפשר להחזיר ישר: case "minor": return ${major}.${minor + 1}.0;. כש־case מסתיים ב־return, לא צריך break. ה־default זורק.

✅ פתרון
// P13-E27: העלאת גרסה (semver) עם switch
export function bump(version, kind) {
  if (!/^\d+\.\d+\.\d+$/.test(version)) {
    throw new Error(`גרסה לא תקינה: ${version}`);
  }
  const [major, minor, patch] = version.split(".").map(Number);
  switch (kind) {
    case "major":
      return `${major + 1}.0.0`;
    case "minor":
      return `${major}.${minor + 1}.0`;
    case "patch":
      return `${major}.${minor}.${patch + 1}`;
    default:
      throw new Error(`סוג לא מוכר: ${kind}`);
  }
}

map(Number) חשוב: בלי זה minor + 1 על המחרוזת "9" נותן "91" ולא 10. ‏0.9.9 ← 0.10.0 מראה שגרסאות הן מספרים ולא טקסט (את זה ראית כבר ב־P13-E13, במיון גרסאות).

  • /^\d+.\d+.\d+$/ בלי \ לפני הנקודות — "1x4x2" נחשבת תקינה.
  • case "minor": בלי return ובלי break — ממשיכים ל־case הבא.

P13-E42 · תלויות לפי scope 🤖 💻 כתיבת קוד 🤖 רמה 2

💡 רמז 1

💡 רמז 2

✅ פתרון
// P13-E42: תלויות לפי scope
const NONE = "(ללא)";

export function byScope(pkg) {
  const names = Object.keys({ ...pkg.dependencies, ...pkg.devDependencies });
  const groups = new Map();
  for (const name of names) {
    const scope = name.startsWith("@") ? name.slice(0, name.indexOf("/")) : NONE;
    if (!groups.has(scope)) groups.set(scope, []);
    groups.get(scope).push(name);
  }
  const result = new Map();
  const keys = [...groups.keys()].filter((k) => k !== NONE).sort();
  if (groups.has(NONE)) keys.unshift(NONE);
  for (const k of keys) result.set(k, groups.get(k).sort());
  return result;
}
