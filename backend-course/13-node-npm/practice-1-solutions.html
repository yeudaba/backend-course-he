P13-E06 · בדיקת ^ 🤖 💻 כתיבת קוד 🤖 רמה 2

💡 רמז 1

parseVersion: split("."), 3 חלקים, כל אחד /^\d+$/.

💡 רמז 2

השוואת גרסאות: קודם major, אחר כך minor, אחר כך patch.

✅ פתרון
// P13-E06
export function parseVersion(v) {
  const parts = String(v).split(".");
  if (parts.length !== 3 || !parts.every((p) => /^\d+$/.test(p))) {
    throw new Error(`גרסה לא תקינה: ${v}`);
  }
  return parts.map(Number);
}

function compare(a, b) {
  for (let i = 0; i < 3; i++) {
    if (a[i] !== b[i]) return a[i] - b[i];
  }
  return 0;
}

export function satisfiesCaret(version, range) {
  const v = parseVersion(version);
  const min = parseVersion(range.replace(/^\^/, ""));
  if (compare(v, min) < 0) return false;
  if (min[0] === 0) return v[0] === 0 && v[1] === min[1];
  return v[0] === min[0];
}

console.log(satisfiesCaret("1.9.0", "^1.2.3"), satisfiesCaret("2.0.0", "^1.2.3"), satisfiesCaret("0.3.0", "^0.2.3"));
true false false
⚠️ טעויות נפוצות
  • השוואת מחרוזות ("1.10.0" < "1.9.0" מילונית!) — חייבים מספרים.

✅ פתרון
// P13-E07
import fs from "node:fs/promises";
import path from "node:path";

export function depsReport(pkg) {
  const prod = Object.keys(pkg.dependencies ?? {}).sort();
  const dev = Object.keys(pkg.devDependencies ?? {}).sort();
  return { prod, dev, total: prod.length + dev.length };
}

export function scriptLines(pkg) {
  return Object.entries(pkg.scripts ?? {}).map(([name, cmd]) => {
    const run = ["start", "test"].includes(name) ? `npm ${name}` : `npm run ${name}`;
    return `${run}  →  ${cmd}`;
  });
}

export async function readPackage(dir) {
  try {
    return JSON.parse(await fs.readFile(path.join(dir, "package.json"), "utf8"));
  } catch (err) {
    if (err.code === "ENOENT") throw new Error("לא נמצא package.json");
    throw err;
  }
}

const pkg = { scripts: { start: "node app.js", dev: "node --watch app.js" }, dependencies: { zod: "^3", express: "^5" } };
console.log(depsReport(pkg), scriptLines(pkg));
{ prod: [ 'express', 'zod' ], dev: [], total: 2 } [ 'npm start  →  node app.js', 'npm run dev  →  node --watch app.js' ]

שים לב ל־readPackage: מתרגמים רק את השגיאה שאנחנו מבינים (ENOENT); JSON פגום — נזרק כמו שהוא.

⚠️ טעויות נפוצות
  • catch { throw new Error("לא נמצא") } לכל שגיאה — package.json פגום ידווח כ"לא נמצא", ויבזבז זמן.

P13-E13 · מיון ובחירת גרסאות 🤖 💻 כתיבת קוד 🤖 רמה 3

💡 רמז 1

parse פעם אחת, ואז השוואה לפי major, minor, patch.

💡 רמז 2

~: אותו major ו־minor, ≥ המינימום.

✅ פתרון
// P13-E13
const parse = (v) => v.split(".").map(Number);

export function compareVersions(a, b) {
  const x = parse(a);
  const y = parse(b);
  for (let i = 0; i < 3; i++) {
    if (x[i] !== y[i]) return x[i] - y[i];
  }
  return 0;
}

export function satisfies(version, range) {
  const op = range[0];
  if (op !== "^" && op !== "~") return compareVersions(version, range) === 0;
  const min = range.slice(1);
  if (compareVersions(version, min) < 0) return false;
  const [vMaj, vMin] = parse(version);
  const [mMaj, mMin] = parse(min);
  return op === "^" ? vMaj === mMaj : vMaj === mMaj && vMin === mMin;
}

export function maxSatisfying(versions, range) {
  const ok = versions.filter((v) => satisfies(v, range)).sort(compareVersions);
  return ok.at(-1) ?? null;
}

const available = ["4.17.1", "4.18.2", "4.21.0", "5.0.0", "5.1.0"];
console.log(maxSatisfying(available, "^4.18.0"), maxSatisfying(available, "~4.18.0"), maxSatisfying(available, "^6.0.0"));
4.21.0 4.18.2 null
⚠️ טעויות נפוצות
  • versions.sort() בלי comparator — מיון מילוני: "4.9.0" אחרי "4.10.0".

P13-E27 · העלאת גרסה (semver) 🤖 💻 כתיבת קוד 🤖 רמה 3

💡 רמז 1

קודם בדיקה: /^\d+\.\d+\.\d+$/.test(version) (הנקודה עם \, אחרת היא "כל תו"). אחר כך: const [major, minor, patch] = version.split(".").map(Number).

💡 רמז 2

בכל case אפשר להחזיר ישר: case "minor": return ${major}.${minor + 1}.0;. כש־case מסתיים ב־return, לא צריך break. ה־default זורק.

✅ פתרון
// P13-E27: העלאת גרסה (semver) עם switch
export function bump(version, kind) {
  if (!/^\d+\.\d+\.\d+$/.test(version)) {
    throw new Error(`גרסה לא תקינה: ${version}`);
  }
  const [major, minor, patch] = version.split(".").map(Number);
  switch (kind) {
    case "major":
      return `${major + 1}.0.0`;
    case "minor":
      return `${major}.${minor + 1}.0`;
    case "patch":
      return `${major}.${minor}.${patch + 1}`;
    default:
      throw new Error(`סוג לא מוכר: ${kind}`);
  }
}

console.log(bump("1.4.2", "patch"), bump("1.4.2", "minor"), bump("1.4.2", "major"));
console.log(bump("0.9.9", "minor"));
try {
  bump("1.4", "patch");
} catch (err) {
  console.log(err.message);
}
1.4.3 1.5.0 2.0.0
0.10.0
גרסה לא תקינה: 1.4

map(Number) חשוב: בלי זה minor + 1 על המחרוזת "9" נותן "91" ולא 10. ‏0.9.9 ← 0.10.0 מראה שגרסאות הן מספרים ולא טקסט (את זה ראית כבר ב־P13-E13, במיון גרסאות).

ה־switch מתאים כאן כי משווים ערך אחד לכמה מחרוזות קבועות, ולכל אחת יש פעולה אחרת. ‏return בתוך case חוסך את ה־break. ה־default הוא המקום הטבעי לשגיאה.

⚠️ טעויות נפוצות
  • לשכוח לאפס: minor על 1.4.2 נותן 1.5.2 במקום 1.5.0.
  • /^\d+.\d+.\d+$/ בלי \ לפני הנקודות — "1x4x2" נחשבת תקינה.
  • case "minor": בלי return ובלי break — ממשיכים ל־case הבא.
