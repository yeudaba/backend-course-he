// P13-E11
import path from "node:path";

export function safeJoin(baseDir, userPath) {
  const base = path.resolve(baseDir);
  const full = path.resolve(base, userPath);
  if (full !== base && !full.startsWith(base + path.sep)) {
    throw new Error("נתיב לא מורשה");
  }
  return full;
}

// P13-E12
import fs from "node:fs/promises";
import path from "node:path";

export async function dirStats(dir) {
  const entries = await fs.readdir(dir, { withFileTypes: true });
  const files = entries.filter((e) => e.isFile()).map((e) => e.name);
  const sizes = await Promise.all(files.map(async (name) => (await fs.stat(path.join(dir, name))).size));
  const byExtension = {};
  for (const name of files) {
    const ext = path.extname(name).slice(1).toLowerCase() || "(none)";
    byExtension[ext] = (byExtension[ext] ?? 0) + 1;
  }
  return { files: files.length, totalBytes: sizes.reduce((s, n) => s + n, 0), byExtension };
}

// P13-E17
import fs from "node:fs/promises";
import path from "node:path";
import { randomUUID } from "node:crypto";

const ALLOWED = [".png", ".jpg", ".jpeg", ".pdf"];

export async function storeUpload(baseDir, originalName, content) {
  const ext = path.extname(originalName).toLowerCase();
  if (!ALLOWED.includes(ext)) {
    throw new Error("סוג קובץ לא נתמך");
  }
  const storedName = `${randomUUID()}${ext}`;
  await fs.mkdir(baseDir, { recursive: true });
  await fs.writeFile(path.join(baseDir, storedName), content);
  return { storedName, originalName: path.basename(originalName), size: Buffer.byteLength(content) };
}

✅ פתרון
// P13-E18
import fs from "node:fs";
import readline from "node:readline";

export async function countStatuses(file) {
  const result = { "2xx": 0, "3xx": 0, "4xx": 0, "5xx": 0, invalid: 0, slowest: null };
  const rl = readline.createInterface({ input: fs.createReadStream(file), crlfDelay: Infinity });
  for await (const raw of rl) {
    const line = raw.trim();
    if (line === "") continue;
    const parts = line.split(/\s+/);
    const status = Number(parts[2]);
    const ms = Number(parts[3]);
    const group = `${Math.floor(status / 100)}xx`;
    if (parts.length !== 4 || !Number.isInteger(status) || Number.isNaN(ms) || !(group in result) || group === "invalid") {
      result.invalid++;
      continue;
    }
    result[group]++;
    if (result.slowest === null || ms > result.slowest.ms) {
      result.slowest = { path: parts[1], ms };
    }
  }
  return result;
}

💡 רמז 2

✅ פתרון
// P13-E19
import fs from "node:fs";
import fsp from "node:fs/promises";
import path from "node:path";
import { createHash } from "node:crypto";

export async function fileHash(file) {
  const hash = createHash("sha256");
  for await (const chunk of fs.createReadStream(file)) {
    hash.update(chunk);
  }
  return hash.digest("hex");
}

export async function findDuplicates(dir) {
  const entries = await fsp.readdir(dir, { withFileTypes: true });
  const names = entries.filter((e) => e.isFile()).map((e) => e.name);
  const hashes = await Promise.all(names.map((n) => fileHash(path.join(dir, n))));
  const groups = new Map();
  names.forEach((name, i) => {
    const list = groups.get(hashes[i]) ?? [];
    list.push(name);
    groups.set(hashes[i], list);
  });
  return [...groups.values()]
    .filter((g) => g.length > 1)
    .map((g) => g.sort())
    .sort((a, b) => a[0].localeCompare(b[0]));
}

  await fsp.writeFile(path.join(tmp, name), text);
}

import fs from "node:fs";
import readline from "node:readline";

export async function countMatchingLines(file, word) {

P13-E26 · פירוק שורת לוג עם regex 🤖 💻 כתיבת קוד 🤖 רמה 3

💡 רמז 1

בונים את ה־regex חלק אחר חלק, משמאל לימין: \S+ ("רצף בלי רווחים") ל־ip ולשני המקפים, \[([^\]]+)\] לזמן ("כל מה שאינו ]"), ואז "([A-Z]+) (\S+) HTTP\/[\d.]+". מסיימים ב־(\d{3}) (\d+|-). ‏^ בהתחלה ו־$ בסוף — כדי שכל השורה תתאים, ולא רק חלק ממנה.

💡 רמז 2

קבוצה בשם: (?<ip>\S+). אחרי const m = line.trim().match(LINE) הערכים נמצאים ב־m.groups, ואפשר לפרק אותם: const { ip, time, ... } = m.groups. אם m הוא null — אין התאמה.

✅ פתרון
// P13-E26: פירוק שורת לוג עם regex
const LINE = /^(?<ip>\S+) \S+ \S+ \[(?<time>[^\]]+)\] "(?<method>[A-Z]+) (?<path>\S+) HTTP\/[\d.]+" (?<status>\d{3}) (?<bytes>\d+|-)$/;

export function parseAccessLine(line) {
  const m = line.trim().match(LINE);
  if (!m) {
    return null;
  }
  const { ip, time, method, path, status, bytes } = m.groups;
  return {
    ip,
    time,
    method,
    path,
    status: Number(status),
    bytes: bytes === "-" ? 0 : Number(bytes),
  };
}

console.log(parseAccessLine("משהו אחר לגמרי"));

ה־regex נקרא משמאל לימין, בדיוק כמו השורה: ip, שני שדות שלא מעניינים אותנו (\S+ \S+), זמן בסוגריים, בקשה במירכאות, סטטוס ובתים. ‏[^\]]+ פירושו "תו אחד או יותר שאינם ]" — כך הזמן נעצר בדיוק בסוגר, גם אם יש בו רווח.

קבוצות בשם (?<name>) הופכות את התוצאה לאובייקט קריא. בלעדיהן הייתי כותב m[1], m[5], ובכל שינוי ב־regex המספרים זזים.

מה שיוצא מ־regex הוא תמיד מחרוזת. לכן Number(status), ולכן גם (\d+|-): הבתים הם מספר או מקף, ובקוד הופכים מקף ל־0.

⚠️ טעויות נפוצות
  • בלי ^ ו־$ — שורה עם זבל בסוף עדיין "מתאימה", והבודק מצפה ל־null.
  • .* במקום [^\]]+ — ‏.* "חמדן" ובולע עד ה־] האחרון בשורה.
  • לשכוח \ לפני [, ] או / — בתוך regex אלה תווים מיוחדים.
  • להחזיר status: "200" — מחרוזת ולא מספר.

P13-E28 · מיזוג קובצי הגדרות במקביל 🤖 💻 כתיבת קוד 🤖 רמה 3

💡 רמז 1

פונקציית עזר readJson(file): ‏await fs.readFile(file, "utf8"), ואז JSON.parse בתוך try/catch שזורק את השגיאה החדשה. אחר כך: await Promise.all(files.map(readJson)).

שים לב ש־try עוטף רק את JSON.parse, ולא את readFile. כך שגיאת "קובץ לא קיים" לא הופכת ל"JSON לא תקין".

💡 רמז 2

מיזוג: configs.reduce((merged, config) => ({ ...merged, ...config }), {}). ב־spread, מה שמופיע מאוחר יותר גובר.

✅ פתרון
// P13-E28: מיזוג קובצי הגדרות במקביל
import fs from "node:fs/promises";
import os from "node:os";
import path from "node:path";

async function readJson(file) {
  const text = await fs.readFile(file, "utf8");
  try {
    return JSON.parse(text);
  } catch {
    throw new Error(`${path.basename(file)}: JSON לא תקין`);
  }
}

export async function loadConfigs(files) {
  const configs = await Promise.all(files.map(readJson));
  return configs.reduce((merged, config) => ({ ...merged, ...config }), {});
}

const dir = await fs.mkdtemp(path.join(os.tmpdir(), "cfg-"));
await fs.writeFile(path.join(dir, "default.json"), '{ "port": 3000, "logLevel": "info" }');
await fs.writeFile(path.join(dir, "local.json"), '{ "logLevel": "debug" }');
await fs.writeFile(path.join(dir, "broken.json"), "{ port: 1 }");

console.log(await loadConfigs([path.join(dir, "default.json"), path.join(dir, "local.json")]));
try {
  await loadConfigs([path.join(dir, "default.json"), path.join(dir, "broken.json")]);
} catch (err) {
  console.log(err.message);
}

Promise.all שומר על הסדר של המערך, גם אם קובץ אחד נקרא מהר מהאחר. לכן ה־reduce ממזג בדיוק לפי הסדר שביקשו, והקובץ האחרון גובר.

ה־try הצר הוא העיקר כאן: הוא תופס רק את השגיאה שאנחנו יודעים להסביר (JSON שבור), ומוסיף לה את שם הקובץ. שגיאה אחרת, כמו קובץ חסר, עוברת כמו שהיא. ככה מי שקורא את ההודעה יודע מה באמת קרה.

⚠️ טעויות נפוצות
  • for ... of עם await בתוכו — עובד, אבל קורא קובץ אחרי קובץ. הבודק מחפש Promise.all.
  • { ...config, ...merged } — הסדר הפוך, והקובץ הראשון גובר.
  • try סביב הכול עם catch שזורק "JSON לא תקין" — קובץ חסר מקבל הודעה שקרית.
  • להחזיר את err.message המקורי של JSON.parse — הוא לא אומר באיזה קובץ הבעיה.

P13-E24 · אתגר: פותר תלויות (mini-npm) 🤖 💻 כתיבת קוד 🤖 רמה 5

💡 רמז 1

תור של [name, range] להוספה. לכל אחד: אם כבר נבחרה גרסה — בדוק שהיא מקיימת את הטווח החדש (אחרת התנגשות). אם לא — בחר max ותוסיף את התלויות שלה לתור.

💡 רמז 2

אל תשכח: אותה חבילה יכולה להופיע כמה פעמים בתור — זה תקין.

✅ פתרון
// P13-E24: פותר תלויות שטוח
const parse = (v) => v.split(".").map(Number);
function compare(a, b) {
  const x = parse(a);
  const y = parse(b);
  for (let i = 0; i < 3; i++) if (x[i] !== y[i]) return x[i] - y[i];
  return 0;
}
function satisfies(version, range) {
  if (!range.startsWith("^")) return compare(version, range) === 0;
  const min = range.slice(1);
  return compare(version, min) >= 0 && parse(version)[0] === parse(min)[0];
}

export function resolveDeps(rootDeps, registry) {
  const chosen = {};
  const queue = Object.entries(rootDeps);
  while (queue.length > 0) {
    const [name, range] = queue.shift();
    if (chosen[name]) {
      if (!satisfies(chosen[name], range)) {
        throw new Error(`התנגשות: ${name}`);
      }
      continue;
    }
    const versions = Object.keys(registry[name] ?? {}).filter((v) => satisfies(v, range)).sort(compare);
    const best = versions.at(-1);
    if (!best) {
      throw new Error(`אין גרסה של ${name} שמתאימה ל־${range}`);
    }
    chosen[name] = best;
    queue.push(...Object.entries(registry[name][best].deps ?? {}));
  }
  return Object.fromEntries(Object.entries(chosen).sort(([a], [b]) => a.localeCompare(b)));
}

const registry = {
  express: { "5.0.0": { deps: { router: "^2.0.0", debug: "^4.0.0" } }, "5.1.0": { deps: { router: "^2.1.0", debug: "^4.3.0" } } },
  router: { "2.0.0": {}, "2.2.0": { deps: { debug: "^4.0.0" } } },
  debug: { "4.3.4": {}, "4.4.0": {} },
};
console.log(resolveDeps({ express: "^5.0.0" }, registry));
{ debug: '4.4.0', express: '5.1.0', router: '2.2.0' }

npm אמיתי לא "שטוח" לגמרי: כשיש התנגשות, הוא מתקין שתי גרסאות (אחת בתוך node_modules של החבילה שצריכה אותה). זו הסיבה ש־node_modules כל כך גדול. וזו גם הסיבה שה־lockfile קיים — הפתרון הזה מסובך, ושומרים את התוצאה.

⚠️ טעויות נפוצות
  • לבחור גרסה בלי לעבור על התלויות שלה — עץ חסר.

// P13-E25: log rotation
import fs from "node:fs/promises";
import path from "node:path";

async function exists(file) {
  try {
    await fs.access(file);
    return true;
  } catch {
    return false;
  }
}

export async function rotate(dir, name, maxFiles) {
  const actions = [];
  const file = (i) => path.join(dir, i === 0 ? name : `${name}.${i}`);
  const label = (i) => (i === 0 ? name : `${name}.${i}`);

  if (await exists(file(maxFiles))) {
    await fs.rm(file(maxFiles));
    actions.push(`delete ${label(maxFiles)}`);
  }
  for (let i = maxFiles - 1; i >= 0; i--) {
    if (await exists(file(i))) {
      await fs.rename(file(i), file(i + 1));
      actions.push(`${label(i)} -> ${label(i + 1)}`);
    }
  }
  await fs.writeFile(file(0), "");
  actions.push(`create ${name}`);
  return actions;
}

const tmp = await fs.mkdtemp(path.join((await import("node:os")).tmpdir(), "rot-"));
for (const f of ["app.log", "app.log.1", "app.log.3"]) await fs.writeFile(path.join(tmp, f), f);
console.log(await rotate(tmp, "app.log", 3));
