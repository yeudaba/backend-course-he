💡 רמז 2

// P26.5-E14: consumer group — commit רק אחרי שהטיפול הצליח
import { Consumer, MessagesStreamModes, MessagesStreamFallbackModes, stringDeserializers } from "@platformatic/kafka";

export interface ConsumeOptions {
  brokers: string[];
  groupId: string;
  topic: string;
}

export async function consumeAndCommit(
  { brokers, groupId, topic }: ConsumeOptions,
  handler: (event: any) => Promise<void>,
  { max }: { max: number },
): Promise<number> {
  const consumer = new Consumer({ groupId, clientId: `${groupId}-worker`, bootstrapBrokers: brokers, deserializers: stringDeserializers });
  const stream = await consumer.consume({
    topics: [topic],
    mode: MessagesStreamModes.COMMITTED, // ממשיכים מאיפה שה־group עצר
    fallbackMode: MessagesStreamFallbackModes.EARLIEST, // group חדש ← מההתחלה
    autocommit: false,
  });
  let processed = 0;
  try {
    for await (const m of stream) {
      await handler(JSON.parse(m.value)); // נכשל ← יוצאים בלי commit; ההודעה תגיע שוב
      await m.commit();
      if (++processed >= max) break;
    }
  } finally {
    await stream.close().catch(() => {});
    await consumer.close();
  }
  return processed;
}

commit אחרי כל הודעה הוא הכי בטוח, ולא הכי מהיר: בנפח גבוה עושים commit כל N הודעות או כל שנייה (autocommit עם מרווח) — ומקבלים שאחרי קריסה כמה הודעות יטופלו שוב. שוב: consumer אידמפוטנטי.

✅ פתרון
// P26.5-E19: consumer lag — כמה ה־consumers מאחור, והאם הפער גדל
export interface LagReport {
  perPartition: Record<number, number>;
  total: number;
  worstPartition: number | null;
}

// end: ה־offset הבא שייכתב בכל partition (high watermark); committed: הבא שה־group יקרא (‎-1 = אין commit)
export function consumerLag(end: Record<number, number>, committed: Record<number, number>, earliest: Record<number, number> = {}): LagReport {
  const perPartition: Record<number, number> = {};
  let total = 0;
  let worstPartition: number | null = null;
  for (const [p, hw] of Object.entries(end)) {
    const part = Number(p);
    const c = committed[part];
    const from = c === undefined || c < 0 ? (earliest[part] ?? 0) : c; // בלי commit — כל מה שעוד שמור ב־topic
    const lag = Math.max(0, hw - from);
    perPartition[part] = lag;
    total += lag;
    if (worstPartition === null || lag > perPartition[worstPartition]) worstPartition = part;
  }
  return { perPartition, total, worstPartition };
}

// מתריעים כשה־lag עולה ברציפות — לא על קפיצה אחת
export function isFallingBehind(samples: number[], { window = 3 } = {}): boolean {
  if (samples.length < window + 1) return false;
  const last = samples.slice(-(window + 1));
  return last.every((v, i) => i === 0 || v > last[i - 1]);
}

lag גבוה אחרי deploy זה טבעי (consumer עלה ומדביק פער). lag שעולה כל הזמן — הצרכנים איטיים מהקצב: מוסיפים consumers (עד מספר ה־partitions), או מגלים שה־handler איטי (שאילתה בלי אינדקס, קריאת API בכל הודעה).

  save: (orderId: number, status: string) => Promise<void>,
  max: number,

P26.5-E26 · להחזיר סדר להודעות 🤖 💻 כתיבת קוד 🤖 רמה 3

💡 רמז 1

שני שדות מספיקים: #next (הבא בתור) ו־#waiting = new Map<number, T>(). ב־push, שלושת המקרים לפי הסדר: seq < #next || #waiting.has(seq) ← כפילות. seq > #next ← לשמור ב־Map. אחרת ← למסור.

💡 רמז 2

אחרי שמסרת את seq: ‏this.#next++, ואז while (this.#waiting.has(this.#next)): מוציאים מה־Map, מוסרים, ומקדמים. ‏gaps: ‏Math.max(...this.#waiting.keys()), ולולאה מ־#next עד אליו.

✅ פתרון
// P26.5-E26 — להחזיר סדר להודעות
export type PushResult = "delivered" | "buffered" | "duplicate";

export class ReorderBuffer<T> {
  #next: number;
  #waiting = new Map<number, T>();
  #deliver: (seq: number, message: T) => void;

  constructor(deliver: (seq: number, message: T) => void, startSeq = 1) {
    this.#deliver = deliver;
    this.#next = startSeq;
  }

  push(seq: number, message: T): PushResult {
    if (!Number.isInteger(seq)) throw new TypeError(`seq חייב להיות מספר שלם: ${seq}`);
    if (seq < this.#next || this.#waiting.has(seq)) return "duplicate";
    if (seq > this.#next) {
      this.#waiting.set(seq, message);
      return "buffered";
    }
    this.#deliver(seq, message);
    this.#next++;
    while (this.#waiting.has(this.#next)) {
      const waiting = this.#waiting.get(this.#next) as T;
      this.#waiting.delete(this.#next);
      this.#deliver(this.#next, waiting);
      this.#next++;
    }
    return "delivered";
  }

  get nextSeq(): number {
    return this.#next;
  }

  get pending(): number {
    return this.#waiting.size;
  }

  gaps(): number[] {
    if (this.#waiting.size === 0) return [];
    const max = Math.max(...this.#waiting.keys());
    const missing: number[] = [];
    for (let seq = this.#next; seq < max; seq++) if (!this.#waiting.has(seq)) missing.push(seq);
    return missing;
  }
}

  console.log(`push ${seq} → ${result}`, result === "buffered" ? `(חסר: ${buffer.gaps().join(", ")})` : "");

מה חסר כאן, ובמערכת אמיתית חייב להיות? גבול. אם 4 לא מגיעה לעולם, ה־Map גדל בלי סוף, וההזמנה תקועה. פתרונות: זמן המתנה מקסימלי ואז התראה, או מסירה עם "דילוג" מתועד. ובנוסף, ‏#next חי בזיכרון: אחרי restart הוא חוזר ל־1. לכן שומרים אותו במסד, באותה טרנזקציה של הטיפול (כמו processed_events בשיעור 5), ומעבירים ל־startSeq.

הרבה פעמים אפשר לחסוך את כל זה: אם כל ההודעות של הזמנה הולכות לאותו partition (‏key = מזהה ההזמנה), ‏Kafka כבר שומר את הסדר. ‏ReorderBuffer נחוץ כשאין לך שליטה על התשתית, או כשההודעות מגיעות מכמה מקורות.

⚠️ טעויות נפוצות
  • למסור רק את seq ולא לבדוק מי מחכה אחריו — 3 נשארת ב־Map לנצח.
  • לבדוק כפילות רק מול #next — הודעה שמחכה ומגיעה שוב דורסת את הראשונה.
  • לשמור את כל ההודעות במערך ולמיין בכל push — עובד, אבל O(n log n) לכל הודעה, ואין דרך פשוטה לזהות כפילות.
  • Math.max(...[]) כשאין ממתינות — ‏-Infinity. בדוק קודם שה־Map לא ריק.

consumer במקום i מקבל base + (i < extra ? 1 : 0), החל מ־partition שאחרי הקודם.

✅ פתרון
// P26.5-E82: חלוקת partitions בין consumers
export function assignPartitions(partitions: number, consumers: string[]): Record<string, number[]> {
  if (new Set(consumers).size !== consumers.length) throw new Error("שמות כפולים");
  const sorted = [...consumers].sort();
  const n = sorted.length;
  const out: Record<string, number[]> = {};
  if (n === 0) return out;
  const base = Math.floor(partitions / n);
  const extra = partitions % n;
  let next = 0;
  sorted.forEach((c, i) => {
    const count = base + (i < extra ? 1 : 0);
    out[c] = Array.from({ length: count }, (_, k) => next + k);
    next += count;
  });
  return out;
}

מכאן רואים שתי אמיתות מהשיעור: consumers מעבר למספר ה־partitions מקבלים [], וכל שינוי ברשימה (מישהו הצטרף או נפל) מזיז partitions בין כולם. (Kafka היום מעדיף שיטות "דביקות" — cooperative sticky — שמזיזות כמה שפחות.)

⚠️ טעויות נפוצות
  • בלי מיון — אותה קבוצה מקבלת חלוקה אחרת בכל פעם, לפי סדר ההצטרפות.

💡 רמז 1

💡 רמז 2

✅ פתרון
// P26.5-E83: להחזיר group אחורה (replay)
import type { Admin } from "@platformatic/kafka";

export async function rewindGroup(admin: Admin, groupId: string, topic: string, offsets: Record<number, number>) {
  const entries = Object.entries(offsets).map(([p, o]) => ({ partition: Number(p), offset: o }));
  for (const e of entries) {
    if (!Number.isInteger(e.partition) || e.partition < 0) throw new RangeError(`partition לא תקין: ${e.partition}`);
    if (!Number.isInteger(e.offset) || e.offset < 0) throw new RangeError(`offset לא תקין: ${e.offset}`);
  }
  if (entries.length === 0) return;
  await admin.alterConsumerGroupOffsets({
    groupId,
    topics: [{ name: topic, partitionOffsets: entries.map((e) => ({ partition: e.partition, offset: BigInt(e.offset) })) }],
  });
}

שני תנאים לפני replay אמיתי: ה־group לא רץ באותו רגע (Kafka לא ייתן לשנות offsets של group פעיל), וה־consumer אידמפוטנטי — כי כל מה שבין ה־offset החדש לישן יטופל שוב. ‏(בשורת הפקודה: kafka-consumer-groups --reset-offsets.)

⚠️ טעויות נפוצות
  • לקרוא ל־admin לכל partition בנפרד — partition לא תקין באמצע משאיר group חצי מוזז.

💡 רמז 2

✅ פתרון
// P26.5-E84: dedupe עם חלון — לזכור רק את האחרונים
export class RecentIds {
  #ids = new Set<string>();
  #max: number;
  constructor(max: number) {
    this.#max = max;
  }

  seen(id: string) {
    if (this.#ids.has(id)) return true;
    this.#ids.add(id);
    if (this.#ids.size > this.#max) this.#ids.delete(this.#ids.values().next().value as string);
    return false;
  }

  get size() {
    return this.#ids.size;
  }
}

זיכרון קבוע, ו־O(1) לכל הודעה. המחיר: כפילות שמגיעה אחרי יותר מ־max הודעות (replay של יום שלם!) לא תיתפס. לכן זו אופטימיזציה מעל ההגנה האמיתית (UNIQUE במסד), לא במקומה.

⚠️ טעויות נפוצות
  • מערך עם includes — O(n) לכל הודעה; עם 100,000 זה מורגש.

P26.5-E88 · ‏lag לכל partition ב־SQL 🤖 💻 כתיבת קוד 🤖 רמה 3

💡 רמז 1

קודם — רשימת (group, topic): SELECT DISTINCT group_name, topic FROM committed_offsets.

💡 רמז 2

אחר כך JOIN end_offsets USING (topic), ו־LEFT JOIN committed_offsets על group, ‏topic ו־partition.

✅ פתרון
-- P26.5-E88: lag לכל partition ב־SQL
WITH groups AS (SELECT DISTINCT group_name, topic FROM committed_offsets)
SELECT g.group_name, g.topic, e.partition,
       e.end_offset - COALESCE(c.committed, 0) AS lag
FROM groups g
JOIN end_offsets e ON e.topic = g.topic
LEFT JOIN committed_offsets c
  ON c.group_name = g.group_name AND c.topic = g.topic AND c.partition = e.partition
ORDER BY lag DESC, g.group_name, g.topic, e.partition;

ה־LEFT JOIN הוא העיקר: partition שה־group עוד לא נגע בו (למשל אחרי שהוסיפו partitions) הוא בדיוק המקום שבו lag מתחבא — ו־JOIN רגיל היה מעלים אותו מהדוח.

⚠️ טעויות נפוצות
  • להתחיל מ־committed_offsets בלבד — partitions בלי commit לא מופיעים.

✅ פתרון
// P26.5-E89: לפענח הודעה לפי event-type, עם zod
import { z } from "zod";

const id = z.number().int().positive();
const schemas: Record<string, z.ZodType<{ orderId: number }>> = {
  "order.placed": z.object({ orderId: id, email: z.string().includes("@"), totalCents: id }),
  "order.paid": z.object({ orderId: id, amountCents: id }),
  "order.shipped": z.object({ orderId: id, carrier: z.string().min(1) }),
};

type Result = { ok: true; type: string; event: unknown } | { ok: false; reason: string };

export function decodeMessage(m: { key: string | null; value: string; headers: Record<string, string> }): Result {
  const type = m.headers["event-type"];
  if (!type) return { ok: false, reason: "missing-type" };
  const schema = schemas[type];
  if (!schema) return { ok: false, reason: "unknown-type" };
  let data: unknown;
  try {
    data = JSON.parse(m.value);
  } catch {
    return { ok: false, reason: "bad-json" };
  }
  const r = schema.safeParse(data);
  if (!r.success) return { ok: false, reason: "invalid" };
  if (m.key !== String(r.data.orderId)) return { ok: false, reason: "wrong-key" };
  return { ok: true, type, event: data };
}

‏z.object (לא strict) בכוונה: producer שמוסיף שדה לא שובר consumers ישנים (שיעור 5). וכל ok: false הוא שגיאה קבועה — ההודעה הולכת ל־DLQ (topic של dead letters), לא ל־retry.

⚠️ טעויות נפוצות
  • להחזיר את r.data — zod מוריד את השדות הלא מוכרים, ו־consumer שמעביר את האירוע הלאה מאבד מידע.

P26.5-E90 · סדר בתוך key, מקביליות בין keys 🤖 💻 כתיבת קוד 🤖 רמה 3

💡 רמז 1

‏Map<key, Promise<void>> — "הזנב" של כל key. הודעה חדשה: tail.then(() => handler(m)).

💡 רמז 2

הזנב לא צריך להיכשל: next.catch(() => {}). ניקוי: אם בסוף הזנב עדיין הוא אותו Promise — delete.

✅ פתרון
// P26.5-E90: סדר בתוך key, מקביליות בין keys
export function perKeySerial<M extends { key: string }>(handler: (m: M) => Promise<void>) {
  const tails = new Map<string, Promise<void>>();
  const all = new Set<Promise<void>>();
  return {
    push(m: M) {
      const prev = tails.get(m.key) ?? Promise.resolve();
      const run = prev.then(() => handler(m));
      const tail = run.catch(() => {});
      tails.set(m.key, tail);
      all.add(tail);
      tail.then(() => {
        all.delete(tail);
        if (tails.get(m.key) === tail) tails.delete(m.key);
      });
      return run;
    },
    async idle() {
      while (all.size) await Promise.all([...all]);
    },
  };
}

ככה consumer אחד מטפל באלף הזמנות במקביל — וכל הזמנה בסדר שלה. הסכנה: commit. offset 100 יכול להסתיים לפני 98 (של key אחר) — ו־commit של 101 "מדלג" על 98 שעוד לא נגמרה. לכן ב־Kafka עושים commit רק עד ה־offset הרציף שהסתיים (או מחכים ל־idle() לפני commit).

⚠️ טעויות נפוצות
  • prev.then(() => handler(m)) בלי catch בזנב — כישלון אחד גורם לכל ההודעות הבאות של אותו key להידחות בלי לרוץ.
