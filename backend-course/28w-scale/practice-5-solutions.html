P28.5-E24 · אתגר: API אנליטיקה תחת עומס 🤖 💻 כתיבת קוד 🤖 רמה 5

💡 רמז 1

DuckDB בתוך התהליך: בעלייה — CREATE TABLE orders AS SELECT *, <day בישראל> AS day FROM read_parquet($1). מחשבים את היום פעם אחת, לא בכל שאילתה.

💡 רמז 2

פרמטרים ($1, $2) — לא שרשור של ה־query string ל־SQL. LRU פשוט: Map שומר סדר הכנסה; מחיקה והכנסה מחדש = "הכי חדש".

✅ פתרון
// P28.5-E24: API אנליטיקה מעל Parquet — DuckDB בתוך התהליך, cache, ו־p95 נמוך תחת עומס
import http from "node:http";
import { DuckDBInstance } from "@duckdb/node-api";

const DAY = /^\d{4}-\d{2}-\d{2}$/;
const isDay = (s: string | null): s is string => !!s && DAY.test(s) && !Number.isNaN(Date.parse(`${s}T00:00:00Z`)) && new Date(`${s}T00:00:00Z`).toISOString().startsWith(s);
const daysBetween = (a: string, b: string) => (Date.parse(`${b}T00:00:00Z`) - Date.parse(`${a}T00:00:00Z`)) / 86_400_000;

class TtlLru<V> {
  private map = new Map<string, { value: V; expires: number }>();
  private max: number;
  private ttlMs: number;
  private now: () => number;
  constructor(max: number, ttlMs: number, now: () => number) {
    this.max = max;
    this.ttlMs = ttlMs;
    this.now = now;
  }
  get(k: string): V | undefined {
    const e = this.map.get(k);
    if (!e) return undefined;
    if (e.expires <= this.now()) {
      this.map.delete(k);
      return undefined;
    }
    this.map.delete(k); // LRU: הכי חדש בסוף
    this.map.set(k, e);
    return e.value;
  }
  set(k: string, value: V) {
    this.map.delete(k);
    this.map.set(k, { value, expires: this.now() + this.ttlMs });
    if (this.map.size > this.max) this.map.delete(this.map.keys().next().value!);
  }
}

export async function createAnalyticsServer({ parquetPath, cacheTtlMs = 60_000, cacheSize = 100, now = Date.now }: { parquetPath: string; cacheTtlMs?: number; cacheSize?: number; now?: () => number }): Promise<http.Server> {
  const db = await DuckDBInstance.create(":memory:");
  const conn = await db.connect();
  // טוענים פעם אחת לזיכרון, עם עמודת יום מחושבת (שעון ישראל) — כל שאילתה אחר כך מהירה
  await conn.run(
    `CREATE TABLE orders AS SELECT *, strftime(timezone('Asia/Jerusalem', created_at), '%Y-%m-%d') AS day FROM read_parquet($1)`,
    [parquetPath],
  );
  const cache = new TtlLru<string>(cacheSize, cacheTtlMs, now);

  async function revenue(from: string, to: string) {
    const r = await conn.runAndReadAll(
      `SELECT day, count(*)::INTEGER AS orders, round(sum(quantity * price_cents) / 100.0, 2)::DOUBLE AS revenue
       FROM orders WHERE day BETWEEN $1 AND $2 GROUP BY day ORDER BY day`,
      [from, to],
    );
    const days = r.getRowObjects().map((x) => ({ day: String(x.day), orders: Number(x.orders), revenue: Number(x.revenue) }));
    const total = { orders: days.reduce((s, d) => s + d.orders, 0), revenue: Math.round(days.reduce((s, d) => s + d.revenue * 100, 0)) / 100 };
    return { from, to, days, total };
  }
  async function topProducts(day: string, limit: number) {
    const r = await conn.runAndReadAll(
      `SELECT product, sum(quantity)::INTEGER AS units, round(sum(quantity * price_cents) / 100.0, 2)::DOUBLE AS revenue
       FROM orders WHERE day = $1 GROUP BY product ORDER BY revenue DESC, product LIMIT $2`,
      [day, limit],
    );
    return r.getRowObjects().map((x) => ({ product: String(x.product), units: Number(x.units), revenue: Number(x.revenue) }));
  }

  const send = (res: http.ServerResponse, status: number, body: unknown, cacheState?: string) => {
    res.statusCode = status;
    res.setHeader("content-type", "application/json");
    if (cacheState) res.setHeader("x-cache", cacheState);
    res.end(typeof body === "string" ? body : JSON.stringify(body));
  };

  const server = http.createServer(async (req, res) => {
    const url = new URL(req.url ?? "/", "http://x");
    try {
      if (url.pathname === "/health") return send(res, 200, { ok: true });
      let key: string;
      let compute: () => Promise<unknown>;
      if (url.pathname === "/revenue") {
        const from = url.searchParams.get("from");
        const to = url.searchParams.get("to");
        if (!isDay(from) || !isDay(to)) return send(res, 400, { error: "from ו־to בפורמט YYYY-MM-DD" });
        if (daysBetween(from, to) < 0) return send(res, 400, { error: "from אחרי to" });
        if (daysBetween(from, to) > 92) return send(res, 400, { error: "עד 92 ימים" });
        key = `revenue:${from}:${to}`;
        compute = () => revenue(from, to);
      } else if (url.pathname === "/top-products") {
        const day = url.searchParams.get("day");
        const limit = Number(url.searchParams.get("limit") ?? 5);
        if (!isDay(day)) return send(res, 400, { error: "day בפורמט YYYY-MM-DD" });
        if (!Number.isInteger(limit) || limit < 1 || limit > 20) return send(res, 400, { error: "limit בין 1 ל־20" });
        key = `top:${day}:${limit}`;
        compute = () => topProducts(day, limit);
      } else return send(res, 404, { error: "not found" });

      const hit = cache.get(key);
      if (hit !== undefined) return send(res, 200, hit, "HIT");
      const body = JSON.stringify(await compute());
      cache.set(key, body);
      send(res, 200, body, "MISS");
    } catch {
      send(res, 500, { error: "internal" });
    }
  });
  server.on("close", () => {
    conn.closeSync();
    db.closeSync();
  });
  return server;
}

הארכיטקטורה הזו — Parquet (על S3) + DuckDB בתוך שירות — נפוצה יותר ויותר לדשבורדים פנימיים: בלי cluster של מחסן נתונים, ושאילתות של מילי־שניות על מיליוני שורות. כשהנתונים גדלים לטרה־בייטים — אותן שאילתות עוברות ל־BigQuery/Snowflake כמעט בלי שינוי.

P28.5-E25 · אתגר: data lake מצטבר 🤖 💻 כתיבת קוד 🤖 רמה 5

💡 רמז 1

watermark על (updated_at, id) ולא רק updated_at: עשרים שורות עם אותו updated_at בעמודים של 3 — WHERE (updated_at, id) > ($1, $2) ORDER BY updated_at, id. ו־updated_at::text שומר מיקרו־שניות, שה־Date של JavaScript מאבד.

כתיבה: טבלה זמנית ב־DuckDB, conn.createAppender, ו־COPY batch TO 'file' (FORMAT parquet).

💡 רמז 2

הגרסה האחרונה: QUALIFY row_number() OVER (PARTITION BY id ORDER BY updated_at DESC) = 1.

✅ פתרון
// P28.5-E25: סנכרון מצטבר מ־PostgreSQL ל־data lake (קובצי Parquet), ודוח שמתעלם מגרסאות ישנות
import fs from "node:fs/promises";
import path from "node:path";
import type { Pool } from "pg";
import type { DuckDBConnection } from "@duckdb/node-api";

interface State {
  seq: number;
  updatedAt: string | null; // ה־watermark: (updated_at, id) של השורה האחרונה שיוצאה
  id: number;
}

const q = (s: string) => `'${s.replaceAll("'", "''")}'`;

async function readState(dir: string): Promise<State> {
  try {
    return JSON.parse(await fs.readFile(path.join(dir, "_state.json"), "utf8"));
  } catch {
    return { seq: 0, updatedAt: null, id: 0 };
  }
}

export async function syncOrders({ pool, conn, lakeDir, pageSize = 1000 }: { pool: Pool; conn: DuckDBConnection; lakeDir: string; pageSize?: number }): Promise<{ exported: number; file: string | null }> {
  await fs.mkdir(lakeDir, { recursive: true });
  const state = await readState(lakeDir);
  const rows: any[] = [];
  let cursor = { updatedAt: state.updatedAt, id: state.id };
  for (;;) {
    // keyset על (updated_at, id): שורות עם אותו updated_at לא הולכות לאיבוד בין עמודים
    const { rows: page } = await pool.query(
      `SELECT id, customer_id, total_cents, status, created_at, updated_at, updated_at::text AS updated_at_text
       FROM orders
       WHERE $1::timestamptz IS NULL OR (updated_at, id) > ($1::timestamptz, $2::int)
       ORDER BY updated_at, id LIMIT $3`,
      [cursor.updatedAt, cursor.id, pageSize],
    );
    rows.push(...page);
    if (page.length < pageSize) break;
    const last = page.at(-1);
    cursor = { updatedAt: last.updated_at_text, id: last.id };
  }
  if (rows.length === 0) return { exported: 0, file: null }; // אין חדש ← אין קובץ ריק

  await conn.run(`CREATE OR REPLACE TEMP TABLE batch (id INTEGER, customer_id INTEGER, total_cents INTEGER, status VARCHAR, created_at TIMESTAMPTZ, updated_at TIMESTAMPTZ)`);
  const app = await conn.createAppender("batch");
  for (const r of rows) {
    app.appendInteger(r.id);
    app.appendInteger(r.customer_id);
    app.appendInteger(r.total_cents);
    app.appendVarchar(r.status);
    app.appendVarchar(r.created_at.toISOString());
    app.appendVarchar(r.updated_at_text);
    app.endRow();
  }
  app.closeSync();
  const seq = state.seq + 1;
  const file = path.join(lakeDir, `part-${String(seq).padStart(5, "0")}.parquet`);
  await conn.run(`COPY batch TO ${q(file)} (FORMAT parquet)`);
  // ה־state נכתב רק אחרי שהקובץ קיים; ובכתיבה אטומית (קובץ זמני ← rename)
  const last = rows.at(-1);
  const tmp = path.join(lakeDir, "_state.json.tmp");
  await fs.writeFile(tmp, JSON.stringify({ seq, updatedAt: last.updated_at_text, id: last.id }));
  await fs.rename(tmp, path.join(lakeDir, "_state.json"));
  return { exported: rows.length, file };
}

export async function lakeReport(conn: DuckDBConnection, lakeDir: string) {
  const r = await conn.runAndReadAll(`
    WITH latest AS (
      SELECT * FROM read_parquet(${q(path.join(lakeDir, "part-*.parquet"))})
      QUALIFY row_number() OVER (PARTITION BY id ORDER BY updated_at DESC) = 1 -- הגרסה האחרונה של כל הזמנה
    )
    SELECT strftime(timezone('Asia/Jerusalem', created_at), '%Y-%m-%d') AS day,
           count(*)::INTEGER AS orders, (sum(total_cents) / 100.0)::DOUBLE AS revenue
    FROM latest WHERE status <> 'cancelled'
    GROUP BY day ORDER BY day`);
  return r.getRowObjects().map((x) => ({ day: String(x.day), orders: Number(x.orders), revenue: Number(x.revenue) }));
}
