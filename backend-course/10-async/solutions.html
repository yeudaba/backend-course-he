// P10-E06
function delay(ms, value) {
  return new Promise((resolve) => setTimeout(() => resolve(value), ms));
}

async function countdown(from, ms, log = console.log) {
  for (let n = from; n >= 1; n--) {
    log(n);
    await delay(ms);
  }
  log("🚀");
}

P10-E07 · callback בסגנון Node 🤖 רמה 2

💡 רמז 1

בתוך ה־callback של readFile: אם יש err — return callback(err).

💡 רמז 2

JSON.parse בתוך try/catch. היזהר לא לקרוא ל־callback בתוך ה־try (אם הוא עצמו יזרוק — ייקרא שוב מה־catch).

✅ פתרון
// P10-E07: callback בסגנון error-first
const fs = require("node:fs");

function readJson(file, callback) {
  fs.readFile(file, "utf8", (err, text) => {
    if (err) {
      callback(err);
      return;
    }
    let data;
    try {
      data = JSON.parse(text);
    } catch (parseError) {
      callback(parseError);
      return;
    }
    callback(null, data);
  });
}

readJson("/no/such.json", (err) => console.log(err.code));

module.exports = { readJson };
ENOENT
⚠️ טעויות נפוצות
  • try { callback(null, JSON.parse(text)) } catch (e) { callback(e) } — אם ה־callback של מי שקרא זורק, הוא יקרא פעמיים.

// P10-E08
function legacyGetPrice(code, callback) {
  const prices = { "KB-101": 199, "MS-201": 89 };
  setTimeout(() => {
    if (code in prices) {
      callback(null, prices[code]);
    } else {
      callback(new Error(`אין מחיר ל־${code}`));
    }
  }, 5);
}

function getPrice(code) {
  return new Promise((resolve, reject) => {
    legacyGetPrice(code, (err, price) => {
      if (err) {
        reject(err);
      } else {
        resolve(price);
      }
    });
  });
}

module.exports = { legacyGetPrice, getPrice };

(ל־Node יש עוזר מובנה: require("node:util").promisify(legacyGetPrice) — עושה בדיוק את זה לכל פונקציה error-first.)

P10-E09 · שרשרת then 🤖 רמה 2

const users = [{ id: 1, name: "דנה" }, { id: 2, name: "עומר" }];

// P10-E10
function slowSquareRoot(n, ms = 10) {
  return new Promise((resolve, reject) => {
    setTimeout(() => {
      if (typeof n !== "number" || Number.isNaN(n)) {
        reject(new Error("לא מספר"));
        return;
      }
      if (n < 0) {
        reject(new Error("אין שורש למספר שלילי"));
        return;
      }
      resolve(Math.sqrt(n));
    }, ms);
  });
}

⚠️ טעויות נפוצות

P10-E14 · שורות בכל הקבצים 🤖 רמה 3

💡 רמז 1

💡 רמז 2

✅ פתרון
// P10-E14
const fs = require("node:fs/promises");
const path = require("node:path");

function countLines(text) {
  if (text === "") {
    return 0;
  }
  const trimmed = text.endsWith("\n") ? text.slice(0, -1) : text;
  return trimmed.split("\n").length;
}

async function lineCounts(dir) {
  const names = (await fs.readdir(dir)).filter((n) => n.endsWith(".txt"));
  const pairs = await Promise.all(
    names.map(async (name) => [name, countLines(await fs.readFile(path.join(dir, name), "utf8"))])
  );
  return Object.fromEntries(pairs);
}

module.exports = { lineCounts, countLines };
⚠️ טעויות נפוצות
  • fs.readFile(name) בלי path.join(dir, ...) — מחפש בתיקייה הנוכחית.

P10-E15 · timeout 🤖 רמה 3

💡 רמז 1

Promise.race בין ה־promise לבין promise של טיימר שעושה reject.

💡 רמז 2

שמור את ה־id של setTimeout, ו־.finally(() => clearTimeout(id)) על ה־race.

✅ פתרון
// P10-E15
function withTimeout(promise, ms) {
  let timer;
  const timeout = new Promise((_, reject) => {
    timer = setTimeout(() => {
      const err = new Error(`timeout after ${ms}ms`);
      err.code = "TIMEOUT";
      reject(err);
    }, ms);
  });
  return Promise.race([promise, timeout]).finally(() => clearTimeout(timer));
}

const slow = new Promise((r) => setTimeout(() => r("done"), 200));
withTimeout(slow, 20).catch((e) => console.log(e.code, e.message));

module.exports = { withTimeout };
TIMEOUT timeout after 20ms

(שים לב: ה־Promise האיטי ממשיך לרוץ ברקע — timeout לא "מבטל" אותו. ביטול אמיתי דורש AbortController, שנפגוש עם fetch בפרק 14.)

⚠️ טעויות נפוצות
  • בלי clearTimeout — התוכנית מחכה לטיימר של 30 שניות גם אחרי שהכול נגמר.

P10-E16 · retry עם המתנה 🤖 רמה 3

// P10-E16
const sleep = (ms) => new Promise((r) => setTimeout(r, ms));

async function retryAsync(fn, { attempts = 3, baseDelay = 10, log = () => {} } = {}) {
  for (let i = 1; i <= attempts; i++) {
    try {
      return await fn(i);
    } catch (err) {
      if (i === attempts) {
        throw err;
      }
      log(`attempt ${i} failed: ${err.message}`);
      await sleep(baseDelay * 2 ** (i - 1));
    }
  }
}

retryAsync(async (n) => {
  if (n < 2) throw new Error("busy");
  return "ok";
}, { log: console.log }).then(console.log);

module.exports = { retryAsync };

ok
⚠️ טעויות נפוצות
  • return fn(i) בלי await — השגיאה בורחת מה־try, ואין ניסיון חוזר.

// P10-E17
async function loadDashboard(sources) {
  const names = Object.keys(sources);
  const results = await Promise.allSettled(names.map((name) => sources[name]()));
  const data = {};
  const errors = {};
  results.forEach((result, i) => {
    if (result.status === "fulfilled") {
      data[names[i]] = result.value;
    } else {
      errors[names[i]] = result.reason?.message ?? String(result.reason);
    }
  });
  return { data, errors };
}

loadDashboard({
  user: async () => ({ name: "דנה" }),
  weather: async () => {
    throw new Error("השירות לא זמין");
  },
  cart: async () => 3,
}).then(console.log);

  errors: { weather: 'השירות לא זמין' }

P10-E20 · callback שנקרא פעמיים 🤖 רמה 4

💡 רמז 1

חסר return אחרי טיפול בשגיאה; ה־parse לא בתוך try.

💡 רמז 2

מבנה נכון: אם err ← cb(err) ו־return; try parse; catch ← cb(e) ו־return; cb(null, data).

✅ פתרון
// P10-E20: מתוקן
const fs = require("node:fs");

function loadSettings(file, cb) {
  fs.readFile(file, "utf8", (err, text) => {
    if (err) {
      cb(err);
      return;
    }
    let settings;
    try {
      settings = JSON.parse(text);
    } catch (parseErr) {
      cb(parseErr);
      return;
    }
    cb(null, { theme: "light", ...settings });
  });
}

P10-E24 · אתגר: מגבלת מקביליות 🤖 רמה 5

💡 רמז 1

"עובדים": צור limit פונקציות async שכל אחת לוקחת את האינדקס הבא מתוך מונה משותף בלולאה, עד שנגמרים הפריטים.

💡 רמז 2

let next = 0; const worker = async () => { while (next < items.length) { const i = next++; results[i] = await fn(items[i], i); } }; await Promise.all(Array.from({ length: limit }, worker)); — זה בטוח כי JavaScript לא קוטעת קוד סינכרוני (event loop!).

✅ פתרון
// P10-E24: מקביליות מוגבלת
async function mapLimit(items, limit, fn) {
  const results = new Array(items.length);
  let next = 0;
  async function worker() {
    while (next < items.length) {
      const i = next++;
      results[i] = await fn(items[i], i);
    }
  }
  const workers = Array.from({ length: Math.min(limit, items.length) }, () => worker());
  await Promise.all(workers);
  return results;
}

let active = 0;
let maxActive = 0;
mapLimit([1, 2, 3, 4, 5, 6], 2, async (n) => {
  active++;
  maxActive = Math.max(maxActive, active);
  await new Promise((r) => setTimeout(r, 10));
  active--;
  return n * n;
}).then((r) => console.log(r, "max in parallel:", maxActive));

next++ בטוח בלי נעילה כי שום worker לא קוטע אחר באמצע שורה — רק ב־await. זה אחד המקומות שבהם הבנת ה־event loop נותנת יתרון אמיתי.

⚠️ טעויות נפוצות
  • חלוקה ל"מנות" של limit ו־Promise.all על כל מנה — עובד, אבל איטי: כל מנה מחכה לאיטי שבה.

P10-E25 · אתגר: cache אסינכרוני 🤖 רמה 5

💡 רמז 1

Map של key ← { promise, expires }. שומרים את ה־Promise עצמו (לא את התוצאה) — כך בקשות מקבילות משתפות אותו.

💡 רמז 2

על כישלון: promise.catch(() => cache.delete(key)). את expires קובעים ברגע השמירה.

✅ פתרון
// P10-E25: cache עם שיתוף בקשות בטיסה
function memoizeAsync(fn, ttlMs) {
  const cache = new Map();
  return function (key) {
    const hit = cache.get(key);
    if (hit && hit.expires > Date.now()) {
      return hit.promise;
    }
    const promise = fn(key);
    cache.set(key, { promise, expires: Date.now() + ttlMs });
    promise.catch(() => cache.delete(key));
    return promise;
  };
}

let calls = 0;
const slowDouble = async (n) => {
  calls++;
  await new Promise((r) => setTimeout(r, 10));
  return n * 2;
};
const cached = memoizeAsync(slowDouble, 1000);
Promise.all([cached(5), cached(5), cached(5)]).then((r) => console.log(r, "calls:", calls));

module.exports = { memoizeAsync };
[ 10, 10, 10 ] calls: 1

זה בדיוק "cache stampede protection" — כשאלף משתמשים מבקשים את אותו דבר באותה שנייה, השרת עושה עבודה אחת. נחזור לזה עם Redis בפרק 26.

⚠️ טעויות נפוצות
  • לשמור את התוצאה רק אחרי ש־fn סיימה — בקשות מקבילות כולן מפספסות את ה־cache וקוראות ל־fn.
