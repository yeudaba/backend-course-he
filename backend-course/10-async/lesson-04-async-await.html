שיעור 4 — async/await, מקביליות וכשלים

🎯 בשיעור הזה

1. async ו־await

const sleep = (ms) => new Promise((resolve) => setTimeout(resolve, ms));

async function getUser(id) {
  await sleep(20);
  return { id, name: id === 1 ? "דנה" : "עומר" };
}

async function main() {
  console.log("מביא משתמש...");
  const user = await getUser(1);
  console.log("קיבלתי:", user.name);
}

main();
console.log("main התחילה, ממשיכים");

main התחילה, ממשיכים
קיבלתי: דנה
מילהמה עושה
async functionהפונקציה תמיד מחזירה Promise. return 5 ← Promise שמצליח עם 5
await promise"עצור את הפונקציה הזו עד שה־Promise מסתיים, ותן לי את הערך". מותר רק בתוך async (או ברמה העליונה של קובץ .mjs)

שים לב: await עוצר רק את main, לא את כל התוכנית — לכן "main התחילה, ממשיכים" הודפס לפני "קיבלתי". המלצר ממשיך לשולחנות אחרים.

2. שגיאות: try/catch חוזר לעבוד

import fs from "node:fs/promises";

async function loadConfig(file) {
  try {
    const text = await fs.readFile(file, "utf8");
    return JSON.parse(text);
  } catch (err) {
    console.log(`לא הצלחתי לטעון ${file}: ${err.code ?? err.message}`);
    return { port: 3000 };
  }
}

console.log(await loadConfig("/no/such/config.json"));
לא הצלחתי לטעון /no/such/config.json: ENOENT
{ port: 3000 }

עם await, Promise שנכשל זורק — כמו שגיאה רגילה. ו־try/catch תופס אותה. השווה לפירמידת ה־callbacks: אותה לוגיקה, בלי קינון.

3. ⚡ רצף מול מקביליות

⚡ הטעות הנפוצה ביותר עם await 
const sleep = (ms, v) => new Promise((r) => setTimeout(() => r(v), ms));
const fetchUser = () => sleep(100, "user");
const fetchOrders = () => sleep(100, "orders");
const fetchStock = () => sleep(100, "stock");

let t = Date.now();
const a = await fetchUser();
const b = await fetchOrders();
const c = await fetchStock();
console.log("ברצף:", Math.round((Date.now() - t) / 100) * 100, "ms");

t = Date.now();
const [x, y, z] = await Promise.all([fetchUser(), fetchOrders(), fetchStock()]);
console.log("במקביל:", Math.round((Date.now() - t) / 100) * 100, "ms", x, y, z);
ברצף: 300 ms
במקביל: 100 ms user orders stock

שלוש פעולות בלתי תלויות עם await אחרי await מחכות אחת לשנייה בלי סיבה. Promise.all מתחיל את כולן יחד ומחכה לכולן. כלל: אם פעולה ב' צריכה את התוצאה של א' — רצף. אם לא — במקביל.

כלימחזיראם אחת נכשלת
Promise.all([...])מערך תוצאות, באותו סדרנכשל מיד עם השגיאה הראשונה ("הכול או כלום")
Promise.allSettled([...])מערך של { status: "fulfilled", value } / { status: "rejected", reason }לא נכשל — מדווח על כל אחת
Promise.race([...])התוצאה של הראשונה שמסתיימת (הצלחה או כישלון)תלוי מי ראשונה

4. 👣 גישור: כשהתוצאה מאחרת או נכשלת

בשרת אמיתי, כל קריאה לשירות אחר יכולה: (א) להצליח; (ב) להיכשל מיד; (ג) לא לענות בכלל. המקרה השלישי הוא המסוכן — בלי הגבלה, הבקשה של המשתמש תחכה לנצח. שני כלים:

Timeout עם race

const sleep = (ms, v) => new Promise((r) => setTimeout(() => r(v), ms));

function withTimeout(promise, ms) {
  const timeout = new Promise((_, reject) =>
    setTimeout(() => reject(new Error(`timeout אחרי ${ms}ms`)), ms)
  );
  return Promise.race([promise, timeout]);
}

try {
  console.log(await withTimeout(sleep(20, "מהיר"), 100));
  console.log(await withTimeout(sleep(500, "איטי"), 100));
} catch (err) {
  console.log("נכשל:", err.message);
}
מהיר
נכשל: timeout אחרי 100ms

Retry עם המתנה שגדלה

const sleep = (ms) => new Promise((r) => setTimeout(r, ms));

async function retry(fn, attempts, baseDelay = 10) {
  for (let i = 1; i <= attempts; i++) {
    try {
      return await fn(i);
    } catch (err) {
      if (i === attempts) throw err;
      const delay = baseDelay * 2 ** (i - 1);
      console.log(`ניסיון ${i} נכשל, מחכה ${delay}ms`);
      await sleep(delay);
    }
  }
}

const flaky = async (attempt) => {
  if (attempt < 3) throw new Error("השירות לא זמין");
  return "הצלחה";
};
console.log(await retry(flaky, 5));
ניסיון 1 נכשל, מחכה 10ms
ניסיון 2 נכשל, מחכה 20ms
הצלחה

המתנה שמכפילה את עצמה ("exponential backoff") נותנת לשירות שקרס זמן להתאושש, במקום להפציץ אותו. שים לב ל־return await בתוך try: בלי ה־await, השגיאה הייתה "בורחת" מה־try.

⚠️ Promise שאף אחד לא תפס

async function שנזרקה בה שגיאה, ונקראה בלי await/catch — Node קורס (unhandled rejection). בשרת זה אומר שרת שנפל. כל Promise צריך מישהו שמחכה לו ותופס.

💼 בעולם העבודה

שאלות ראיון: "מה ההבדל בין Promise.all ל־allSettled?", "מה יודפס קודם?" (event loop), "מה הבעיה בקוד הזה?" — ושם כמעט תמיד יש await בתוך לולאה על דברים בלתי תלויים, או forEach עם async (שלא מחכה לכלום — תרגיל E22).
