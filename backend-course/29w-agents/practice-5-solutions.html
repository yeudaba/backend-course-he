P29.5-E12 · tracing 🤖 💻 כתיבת קוד 🤖 רמה 2

💡 רמז 1

Proxy על ה־client שמחליף רק את messages; ו־Object.create(client.messages) עם create משלך.

💡 רמז 2

העלות: (input × מחיר + output × מחיר + cache_read × מחיר) / 1,000,000. וזמן — performance.now() לפני ואחרי ה־await.

✅ פתרון
// P29.5-E12: tracing לסוכן — כל קריאה למודל וכל כלי: זמן, טוקנים ועלות
import type Anthropic from "@anthropic-ai/sdk";

export interface Price {
  input: number; // $ למיליון טוקנים
  output: number;
  cacheRead: number;
}
export const PRICES: Record<string, Price> = {
  "claude-opus-5-5": { input: 4, output: 20, cacheRead: 0.2 },
  "claude-sonnet-5-5": { input: 2, output: 10, cacheRead: 0.2 },
  "claude-haiku-5-5": { input: 0.1, output: 0.5, cacheRead: 0.01 },
};

export interface ModelSpan {
  kind: "model";
  step: number;
  model: string;
  stopReason: string | null;
  inputTokens: number;
  outputTokens: number;
  cacheReadTokens: number;
  costUsd: number;
  ms: number;
}
export interface ToolSpan {
  kind: "tool";
  name: string;
  ms: number;
  isError: boolean;
}

export function createTracer(prices: Record<string, Price> = PRICES) {
  const spans: (ModelSpan | ToolSpan)[] = [];
  let step = 0;
  const round = (x: number) => Math.round(x * 1e6) / 1e6;
  return {
    spans,
    // עוטף את client.messages.create בלי לשנות את הקוד שקורא לו
    wrapClient(client: Anthropic): Anthropic {
      const create = client.messages.create.bind(client.messages);
      const messages = Object.create(client.messages);
      messages.create = async (params: any, opts?: any) => {
        const t0 = performance.now();
        const res = (await create(params, opts)) as Anthropic.Message;
        const p = prices[params.model];
        if (!p) throw new Error(`אין מחיר למודל ${params.model}`);
        const u = res.usage;
        const cacheRead = u.cache_read_input_tokens ?? 0;
        spans.push({
          kind: "model",
          step: ++step,
          model: params.model,
          stopReason: res.stop_reason,
          inputTokens: u.input_tokens,
          outputTokens: u.output_tokens,
          cacheReadTokens: cacheRead,
          costUsd: round((u.input_tokens * p.input + u.output_tokens * p.output + cacheRead * p.cacheRead) / 1e6),
          ms: Math.round(performance.now() - t0),
        });
        return res;
      };
      return new Proxy(client, { get: (t, k) => (k === "messages" ? messages : Reflect.get(t, k)) });
    },
    wrapTool<I>(name: string, run: (input: I) => Promise<string>) {
      return async (input: I) => {
        const t0 = performance.now();
        try {
          const out = await run(input);
          spans.push({ kind: "tool", name, ms: Math.round(performance.now() - t0), isError: false });
          return out;
        } catch (err) {
          spans.push({ kind: "tool", name, ms: Math.round(performance.now() - t0), isError: true });
          throw err;
        }
      };
    },
    summary() {
      const m = spans.filter((s): s is ModelSpan => s.kind === "model");
      const t = spans.filter((s): s is ToolSpan => s.kind === "tool");
      return {
        steps: m.length,
        toolCalls: t.length,
        toolErrors: t.filter((x) => x.isError).length,
        inputTokens: m.reduce((s, x) => s + x.inputTokens, 0),
        outputTokens: m.reduce((s, x) => s + x.outputTokens, 0),
        costUsd: round(m.reduce((s, x) => s + x.costUsd, 0)),
      };
    },
  };
}

בפרודקשן שולחים את זה ל־OpenTelemetry (ה־SDK של Anthropic יודע להוציא spans בעצמו) — וכל שיחה של סוכן נראית כמו בקשה מבוזרת בפרק 24: עץ של צעדים, עם זמן ועלות לכל אחד (E05).

⚠️ טעויות נפוצות
  • לעטוף את client.messages.create ישירות (להחליף את הפונקציה באובייקט המקורי). זה משנה את ה־client לכל מי שמשתמש בו.
  • לשכוח את cache_read_input_tokens. העלות יוצאת נמוכה מהאמת בשיחות ארוכות.
  • ב־wrapTool לתפוס את השגיאה ולא לזרוק הלאה. ה־tracing לא אמור לשנות את ההתנהגות של הכלי.

✅ פתרון
// P29.5-E13: eval על המסלול — לא רק "מה ענה", אלא "איך הגיע לשם"
export interface Trajectory {
  toolCalls: { name: string; input: unknown }[];
  finalText: string;
  steps: number;
}

export interface Expectations {
  mustCall?: string[];
  mustNotCall?: string[];
  order?: [string, string][]; // [a, b]: הקריאה הראשונה ל־a לפני הקריאה הראשונה ל־b
  maxSteps?: number;
  maxToolCalls?: number;
  finalIncludes?: string[];
  finalExcludes?: string[];
}

export function evaluateTrajectory(t: Trajectory, e: Expectations): { pass: boolean; failures: string[] } {
  const failures: string[] = [];
  const names = t.toolCalls.map((c) => c.name);
  for (const n of e.mustCall ?? []) if (!names.includes(n)) failures.push(`לא קרא ל־${n}`);
  for (const n of e.mustNotCall ?? []) if (names.includes(n)) failures.push(`קרא ל־${n}, שאסור`);
  for (const [a, b] of e.order ?? []) {
    const ia = names.indexOf(a);
    const ib = names.indexOf(b);
    if (ib !== -1 && (ia === -1 || ia > ib)) failures.push(`קרא ל־${b} לפני ${a}`);
  }
  if (e.maxSteps !== undefined && t.steps > e.maxSteps) failures.push(`${t.steps} צעדים (מותר ${e.maxSteps})`);
  if (e.maxToolCalls !== undefined && names.length > e.maxToolCalls) failures.push(`${names.length} קריאות לכלים (מותר ${e.maxToolCalls})`);
  const text = t.finalText.toLowerCase();
  for (const s of e.finalIncludes ?? []) if (!text.includes(s.toLowerCase())) failures.push(`התשובה לא כוללת "${s}"`);
  for (const s of e.finalExcludes ?? []) if (text.includes(s.toLowerCase())) failures.push(`התשובה כוללת "${s}"`);
  return { pass: failures.length === 0, failures };
}

תשובה נכונה במסלול שגוי היא באג שמחכה לקרות: "הוחזרו 89.90 ₪" — אבל בלי get_order לפני, כלומר הסכום נוחש. evals של מסלול תופסים את זה לפני הלקוחות.

⚠️ טעויות נפוצות
  • ב־order להשוות את הקריאה האחרונה במקום הראשונה, או לא להיכשל כש־b קרה בלי a.
  • finalIncludes רגיש לרישיות. "Refund" ו־"refund" צריכים להיחשב אותו דבר.
  • להחזיר רק את הכישלון הראשון. התרגיל מבקש הודעה לכל בעיה, כדי לראות את כל מה שנשבר בריצה אחת.

💡 רמז 1

💡 רמז 2

האירועים יוצאים בסדר שהם קורים: yield של model אחרי כל תשובה, tool_start לפני כל כלי ו־tool_end אחריו. ושגיאה נתפסת והופכת ל־yield של error.

✅ פתרון
// P29.5-E19: אירועי התקדמות — כדי שהמשתמש יראה מה הסוכן עושה (SSE, WebSocket)
import type Anthropic from "@anthropic-ai/sdk";

export type AgentEvent =
  | { type: "model"; step: number; stopReason: string | null }
  | { type: "tool_start"; id: string; name: string }
  | { type: "tool_end"; id: string; name: string; isError: boolean }
  | { type: "done"; text: string }
  | { type: "error"; message: string };

export async function* agentEvents(
  client: Anthropic,
  { system, tools, prompt, maxSteps = 10 }: { system: string; tools: Record<string, { definition: Anthropic.Tool; run: (input: any) => Promise<string> }>; prompt: string; maxSteps?: number },
): AsyncGenerator<AgentEvent> {
  const messages: Anthropic.MessageParam[] = [{ role: "user", content: prompt }];
  try {
    for (let step = 1; step <= maxSteps; step++) {
      const res = await client.messages.create({ model: "claude-opus-5-5", max_tokens: 16000, system, tools: Object.values(tools).map((t) => t.definition), messages });
      yield { type: "model", step, stopReason: res.stop_reason };
      messages.push({ role: "assistant", content: res.content });
      if (res.stop_reason !== "tool_use") {
        yield { type: "done", text: res.content.flatMap((b) => (b.type === "text" ? [b.text] : [])).join("") };
        return;
      }
      const results: Anthropic.ToolResultBlockParam[] = [];
      for (const c of res.content.filter((b): b is Anthropic.ToolUseBlock => b.type === "tool_use")) {
        yield { type: "tool_start", id: c.id, name: c.name };
        let r: Anthropic.ToolResultBlockParam;
        try {
          const t = tools[c.name];
          if (!t) throw new Error(`אין כלי בשם ${c.name}`);
          r = { type: "tool_result", tool_use_id: c.id, content: await t.run(c.input) };
        } catch (err) {
          r = { type: "tool_result", tool_use_id: c.id, content: `שגיאה: ${(err as Error).message}`, is_error: true };
        }
        results.push(r);
        yield { type: "tool_end", id: c.id, name: c.name, isError: !!r.is_error };
      }
      messages.push({ role: "user", content: results });
    }
    yield { type: "error", message: "עבר את תקרת הצעדים" };
  } catch (err) {
    yield { type: "error", message: (err as Error).message };
  }
}

⚠️ טעויות נפוצות

import type Anthropic from "@anthropic-ai/sdk";
import { evaluateTrajectory, type Expectations, type Trajectory } from "./P29.5-E13.ts";

export interface Scenario {
  name: string;
  prompt: string;
  expect: Expectations;
}

export interface RunOutput extends Trajectory {
  costUsd: number;
}

export async function runEvalSuite({
  scenarios,
  makeClient,
  agent,
  trials = 3,
  concurrency = 4,
}: {
  scenarios: Scenario[];
  makeClient: (scenario: Scenario, trial: number) => Anthropic;
  agent: (client: Anthropic, prompt: string) => Promise<RunOutput>;
  trials?: number;
  concurrency?: number;
}) {
  const jobs = scenarios.flatMap((s) => Array.from({ length: trials }, (_, t) => ({ s, t })));
  const outcomes = new Map<string, { pass: boolean; failures: string[]; costUsd: number }[]>();
  let next = 0;
  async function worker() {
    while (next < jobs.length) {
      const { s, t } = jobs[next++];
      let r: { pass: boolean; failures: string[]; costUsd: number };
      try {
        const out = await agent(makeClient(s, t), s.prompt);
        r = { ...evaluateTrajectory(out, s.expect), costUsd: out.costUsd };
      } catch (err) {
        r = { pass: false, failures: [`הריצה נכשלה: ${(err as Error).message}`], costUsd: 0 }; // קריסה היא כישלון — לא מדלגים
      }
      outcomes.set(s.name, [...(outcomes.get(s.name) ?? []), r]);
    }
  }
  await Promise.all(Array.from({ length: Math.min(concurrency, jobs.length) }, worker));

  const results = scenarios.map((s) => {
    const rs = outcomes.get(s.name)!;
    const passes = rs.filter((r) => r.pass).length;
    return {
      name: s.name,
      trials: rs.length,
      passes,
      passRate: passes / rs.length,
      failures: [...new Set(rs.flatMap((r) => r.failures))],
      avgCostUsd: Math.round((rs.reduce((x, r) => x + r.costUsd, 0) / rs.length) * 1e6) / 1e6,
    };
  });
  const totalPasses = results.reduce((x, r) => x + r.passes, 0);
  return {
    results,
    passRate: totalPasses / jobs.length,
    flaky: results.filter((r) => r.passRate > 0 && r.passRate < 1).map((r) => r.name), // עובר לפעמים — הכי מסוכן
    failing: results.filter((r) => r.passRate === 0).map((r) => r.name),
  };
}

type Span = { conversationId: string; kind: "model" | "tool"; costUsd?: number; ms: number; isError?: boolean };
type Row = { conversationId: string; steps: number; tools: number; toolErrors: number; costUsd: number; ms: number };

export function costPerConversation(spans: Span[]): Row[] {
  const by = new Map<string, Row>();
  for (const s of spans) {
    const r = by.get(s.conversationId) ?? { conversationId: s.conversationId, steps: 0, tools: 0, toolErrors: 0, costUsd: 0, ms: 0 };
    if (s.kind === "model") r.steps++;
    else {
      r.tools++;
      if (s.isError) r.toolErrors++;
    }
    r.costUsd += s.costUsd ?? 0;
    r.ms += s.ms;
    by.set(s.conversationId, r);
  }
  return [...by.values()]
    .map((r) => ({ ...r, costUsd: Math.round(r.costUsd * 1e6) / 1e6 }))
    .sort((a, b) => b.costUsd - a.costUsd || (a.conversationId < b.conversationId ? -1 : a.conversationId > b.conversationId ? 1 : 0));
}

// P29.5-E92: למצוא בעיות ב־trace בקוד
type Step = { step: number; stopReason: string; cacheRead: number; tools: { name: string; input: unknown; ms: number; isError: boolean }[] };

function stableKey(v: unknown): string {
  if (Array.isArray(v)) return `[${v.map(stableKey).join(",")}]`;
  if (v && typeof v === "object") {
    const o = v as Record<string, unknown>;
    return `{${Object.keys(o).sort().map((k) => `${JSON.stringify(k)}:${stableKey(o[k])}`).join(",")}}`;
  }
  return JSON.stringify(v);
}

export function findTraceIssues(steps: Step[], opts: { slowMs: number; repeat: number }): string[] {
  const counts = new Map<string, number>();
  const repeat: string[] = [];
  const slow: string[] = [];
  const noCache: string[] = [];
  const maxTokens: string[] = [];
  steps.forEach((s, i) => {
    for (const t of s.tools) {
      const key = `${t.name}\u0000${stableKey(t.input)}`;
      const n = (counts.get(key) ?? 0) + 1;
      counts.set(key, n);
      if (n === opts.repeat) repeat.push(`repeat:${t.name}`);
      if (t.ms > opts.slowMs) slow.push(`slow:${t.name}@${s.step}`);
    }
    if (i > 0 && s.cacheRead === 0) noCache.push(`no_cache@${s.step}`);
    if (s.stopReason === "max_tokens") maxTokens.push(`max_tokens@${s.step}`);
  });
  return [...repeat, ...slow, ...noCache, ...maxTokens];
}

// P29.5-E94: pass@k ו־pass^k
function check(n: number, c: number, k: number) {
  if (![n, c, k].every(Number.isInteger) || c < 0 || c > n || k < 1 || k > n) throw new RangeError(`n=${n}, c=${c}, k=${k} לא תקינים`);
}

// C(a, k) / C(n, k)
function ratio(a: number, n: number, k: number): number {
  if (a < k) return 0;
  let r = 1;
  for (let i = 0; i < k; i++) r *= (a - i) / (n - i);
  return r;
}

const r4 = (x: number) => Math.round(x * 1e4) / 1e4;

export function passAtK(n: number, c: number, k: number): number {
  check(n, c, k);
  return r4(1 - ratio(n - c, n, k));
}

export function passHatK(n: number, c: number, k: number): number {
  check(n, c, k);
  return r4(ratio(c, n, k));
}

// P29.5-E28 — trace כעץ טקסט
export type Span = { id: string; parentId: string | null; name: string; start: number; end: number };

export function renderTrace(spans: Span[]): string {
  const children = new Map<string | null, Span[]>();
  const ids = new Set(spans.map((s) => s.id));
  for (const span of spans) {
    // הורה שלא נמצא ב־trace (נחתך, או עוד לא הגיע) — מציגים את ה־span כשורש
    const parent = span.parentId !== null && ids.has(span.parentId) ? span.parentId : null;
    const list = children.get(parent) ?? [];
    list.push(span);
    children.set(parent, list);
  }
  const lines: string[] = [];
  const walk = (parent: string | null, depth: number) => {
    const list = [...(children.get(parent) ?? [])].sort((a, b) => a.start - b.start || a.name.localeCompare(b.name));
    for (const span of list) {
      lines.push(`${"  ".repeat(depth)}${span.name}`.padEnd(32) + `${span.end - span.start}ms`.padStart(8));
      walk(span.id, depth + 1);
    }
  };
  walk(null, 0);
  return lines.join("\n");
}

// P29.5-E98: spans בפורמט של OpenTelemetry
import { createHash } from "node:crypto";

type In = { id: string; parentId: string | null; name: string; start: number; end: number; attrs?: Record<string, string | number | boolean> };
type Value = { stringValue?: string; intValue?: string; doubleValue?: number; boolValue?: boolean };
type Out = { traceId: string; spanId: string; parentSpanId?: string; name: string; startTimeUnixNano: string; endTimeUnixNano: string; attributes: { key: string; value: Value }[] };

const spanId = (id: string) => createHash("sha256").update(id).digest("hex").slice(0, 16);
const nanos = (ms: number) => (BigInt(ms) * 1_000_000n).toString();

function value(v: string | number | boolean): Value {
  if (typeof v === "string") return { stringValue: v };
  if (typeof v === "boolean") return { boolValue: v };
  return Number.isInteger(v) ? { intValue: String(v) } : { doubleValue: v };
}

export function toOtelSpans(traceId: string, spans: In[]): Out[] {
  if (!/^[0-9a-f]{32}$/.test(traceId)) throw new Error(`traceId לא תקין: ${traceId}`);
  const ids = new Set(spans.map((s) => s.id));
  return spans.map((s) => {
    if (s.parentId !== null && !ids.has(s.parentId)) throw new Error(`ההורה ${s.parentId} של ${s.id} לא ברשימה`);
    const out: Out = {
      traceId,
      spanId: spanId(s.id),
      name: s.name,
      startTimeUnixNano: nanos(s.start),
      endTimeUnixNano: nanos(s.end),
      attributes: Object.keys(s.attrs ?? {})
        .sort()
        .map((key) => ({ key, value: value(s.attrs![key]) })),
    };
    if (s.parentId !== null) out.parentSpanId = spanId(s.parentId);
    return out;
  });
}

בעבודה — ה־SDK של OpenTelemetry בונה את זה בשבילך. כאן ראית מה יש בפנים: מזהים, זמנים בננו־שניות, ו־attributes עם טיפוס.

⚠️ טעויות נפוצות
  • ‏String(ms * 1e6) — ‏1763290000000 * 1e6 גדול מ־MAX_SAFE_INTEGER, והספרות האחרונות שגויות (או בכתיב מדעי).

// P29.5-E99: מגמה של evals במסד
type Db = { query(text: string, params?: unknown[]): Promise<{ rows: any[] }> };
type Result = { scenario: string; trials: number; passes: number };

export async function saveEvalRun(db: Db, run: { gitSha: string; results: Result[] }): Promise<number> {
  for (const r of run.results) {
    if (!Number.isInteger(r.trials) || r.trials < 1 || !Number.isInteger(r.passes) || r.passes < 0 || r.passes > r.trials) {
      throw new RangeError(`תוצאה לא תקינה ב־${r.scenario}: ${r.passes}/${r.trials}`);
    }
  }
  await db.query("BEGIN");
  try {
    const { rows } = await db.query("INSERT INTO eval_runs (git_sha) VALUES ($1) RETURNING id", [run.gitSha]);
    const id: number = rows[0].id;
    for (const r of run.results) {
      await db.query("INSERT INTO eval_results (run_id, scenario, trials, passes) VALUES ($1, $2, $3, $4)", [id, r.scenario, r.trials, r.passes]);
    }
    await db.query("COMMIT");
    return id;
  } catch (err) {
    await db.query("ROLLBACK");
    throw err;
  }
}

export async function passRateTrend(db: Db, scenario: string, lastN: number) {
  const { rows } = await db.query(
    `SELECT r.id, r.git_sha, e.trials, e.passes
       FROM eval_runs r JOIN eval_results e ON e.run_id = r.id
      WHERE e.scenario = $1
      ORDER BY r.id DESC
      LIMIT $2`,
    [scenario, lastN],
  );
  return rows.reverse().map((r) => ({ runId: r.id, gitSha: r.git_sha, passRate: Math.round((r.passes / r.trials) * 100) / 100 }));
}

  • לסנן לפי תרחיש אחרי ה־LIMIT — הרצות שלא כללו את התרחיש תופסות מקום.
