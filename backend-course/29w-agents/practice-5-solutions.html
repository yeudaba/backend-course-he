✅ פתרון
// P29.5-E12: tracing לסוכן — כל קריאה למודל וכל כלי: זמן, טוקנים ועלות
import type Anthropic from "@anthropic-ai/sdk";

export interface Price {
  input: number; // $ למיליון טוקנים
  output: number;
  cacheRead: number;
}
export const PRICES: Record<string, Price> = {
  "claude-opus-5-5": { input: 4, output: 20, cacheRead: 0.2 },
  "claude-sonnet-5-5": { input: 2, output: 10, cacheRead: 0.2 },
  "claude-haiku-5-5": { input: 0.1, output: 0.5, cacheRead: 0.01 },
};

export interface ModelSpan {
  kind: "model";
  step: number;
  model: string;
  stopReason: string | null;
  inputTokens: number;
  outputTokens: number;
  cacheReadTokens: number;
  costUsd: number;
  ms: number;
}
export interface ToolSpan {
  kind: "tool";
  name: string;
  ms: number;
  isError: boolean;
}

export function createTracer(prices: Record<string, Price> = PRICES) {
  const spans: (ModelSpan | ToolSpan)[] = [];
  let step = 0;
  const round = (x: number) => Math.round(x * 1e6) / 1e6;
  return {
    spans,
    // עוטף את client.messages.create בלי לשנות את הקוד שקורא לו
    wrapClient(client: Anthropic): Anthropic {
      const create = client.messages.create.bind(client.messages);
      const messages = Object.create(client.messages);
      messages.create = async (params: any, opts?: any) => {
        const t0 = performance.now();
        const res = (await create(params, opts)) as Anthropic.Message;
        const p = prices[params.model];
        if (!p) throw new Error(`אין מחיר למודל ${params.model}`);
        const u = res.usage;
        const cacheRead = u.cache_read_input_tokens ?? 0;
        spans.push({
          kind: "model",
          step: ++step,
          model: params.model,
          stopReason: res.stop_reason,
          inputTokens: u.input_tokens,
          outputTokens: u.output_tokens,
          cacheReadTokens: cacheRead,
          costUsd: round((u.input_tokens * p.input + u.output_tokens * p.output + cacheRead * p.cacheRead) / 1e6),
          ms: Math.round(performance.now() - t0),
        });
        return res;
      };
      return new Proxy(client, { get: (t, k) => (k === "messages" ? messages : Reflect.get(t, k)) });
    },
    wrapTool<I>(name: string, run: (input: I) => Promise<string>) {
      return async (input: I) => {
        const t0 = performance.now();
        try {
          const out = await run(input);
          spans.push({ kind: "tool", name, ms: Math.round(performance.now() - t0), isError: false });
          return out;
        } catch (err) {
          spans.push({ kind: "tool", name, ms: Math.round(performance.now() - t0), isError: true });
          throw err;
        }
      };
    },
    summary() {
      const m = spans.filter((s): s is ModelSpan => s.kind === "model");
      const t = spans.filter((s): s is ToolSpan => s.kind === "tool");
      return {
        steps: m.length,
        toolCalls: t.length,
        toolErrors: t.filter((x) => x.isError).length,
        inputTokens: m.reduce((s, x) => s + x.inputTokens, 0),
        outputTokens: m.reduce((s, x) => s + x.outputTokens, 0),
        costUsd: round(m.reduce((s, x) => s + x.costUsd, 0)),
      };
    },
  };
}

✅ פתרון
// P29.5-E13: eval על המסלול — לא רק "מה ענה", אלא "איך הגיע לשם"
export interface Trajectory {
  toolCalls: { name: string; input: unknown }[];
  finalText: string;
  steps: number;
}

export interface Expectations {
  mustCall?: string[];
  mustNotCall?: string[];
  order?: [string, string][]; // [a, b]: הקריאה הראשונה ל־a לפני הקריאה הראשונה ל־b
  maxSteps?: number;
  maxToolCalls?: number;
  finalIncludes?: string[];
  finalExcludes?: string[];
}

export function evaluateTrajectory(t: Trajectory, e: Expectations): { pass: boolean; failures: string[] } {
  const failures: string[] = [];
  const names = t.toolCalls.map((c) => c.name);
  for (const n of e.mustCall ?? []) if (!names.includes(n)) failures.push(`לא קרא ל־${n}`);
  for (const n of e.mustNotCall ?? []) if (names.includes(n)) failures.push(`קרא ל־${n}, שאסור`);
  for (const [a, b] of e.order ?? []) {
    const ia = names.indexOf(a);
    const ib = names.indexOf(b);
    if (ib !== -1 && (ia === -1 || ia > ib)) failures.push(`קרא ל־${b} לפני ${a}`);
  }
  if (e.maxSteps !== undefined && t.steps > e.maxSteps) failures.push(`${t.steps} צעדים (מותר ${e.maxSteps})`);
  if (e.maxToolCalls !== undefined && names.length > e.maxToolCalls) failures.push(`${names.length} קריאות לכלים (מותר ${e.maxToolCalls})`);
  const text = t.finalText.toLowerCase();
  for (const s of e.finalIncludes ?? []) if (!text.includes(s.toLowerCase())) failures.push(`התשובה לא כוללת "${s}"`);
  for (const s of e.finalExcludes ?? []) if (text.includes(s.toLowerCase())) failures.push(`התשובה כוללת "${s}"`);
  return { pass: failures.length === 0, failures };
}

// P29.5-E19: אירועי התקדמות — כדי שהמשתמש יראה מה הסוכן עושה (SSE, WebSocket)
import type Anthropic from "@anthropic-ai/sdk";

export type AgentEvent =
  | { type: "model"; step: number; stopReason: string | null }
  | { type: "tool_start"; id: string; name: string }
  | { type: "tool_end"; id: string; name: string; isError: boolean }
  | { type: "done"; text: string }
  | { type: "error"; message: string };

export async function* agentEvents(
  client: Anthropic,
  { system, tools, prompt, maxSteps = 10 }: { system: string; tools: Record<string, { definition: Anthropic.Tool; run: (input: any) => Promise<string> }>; prompt: string; maxSteps?: number },
): AsyncGenerator<AgentEvent> {
  const messages: Anthropic.MessageParam[] = [{ role: "user", content: prompt }];
  try {
    for (let step = 1; step <= maxSteps; step++) {
      const res = await client.messages.create({ model: "claude-opus-5-5", max_tokens: 16000, system, tools: Object.values(tools).map((t) => t.definition), messages });
      yield { type: "model", step, stopReason: res.stop_reason };
      messages.push({ role: "assistant", content: res.content });
      if (res.stop_reason !== "tool_use") {
        yield { type: "done", text: res.content.flatMap((b) => (b.type === "text" ? [b.text] : [])).join("") };
        return;
      }
      const results: Anthropic.ToolResultBlockParam[] = [];
      for (const c of res.content.filter((b): b is Anthropic.ToolUseBlock => b.type === "tool_use")) {
        yield { type: "tool_start", id: c.id, name: c.name };
        let r: Anthropic.ToolResultBlockParam;
        try {
          const t = tools[c.name];
          if (!t) throw new Error(`אין כלי בשם ${c.name}`);
          r = { type: "tool_result", tool_use_id: c.id, content: await t.run(c.input) };
        } catch (err) {
          r = { type: "tool_result", tool_use_id: c.id, content: `שגיאה: ${(err as Error).message}`, is_error: true };
        }
        results.push(r);
        yield { type: "tool_end", id: c.id, name: c.name, isError: !!r.is_error };
      }
      messages.push({ role: "user", content: results });
    }
    yield { type: "error", message: "עבר את תקרת הצעדים" };
  } catch (err) {
    yield { type: "error", message: (err as Error).message };
  }
}

✅ פתרון
// P29.5-E25: חבילת evals לסוכן — כמה תרחישים, כמה ניסיונות לכל אחד, ודוח
import type Anthropic from "@anthropic-ai/sdk";
import { evaluateTrajectory, type Expectations, type Trajectory } from "./P29.5-E13.ts";

export interface Scenario {
  name: string;
  prompt: string;
  expect: Expectations;
}

export interface RunOutput extends Trajectory {
  costUsd: number;
}

export async function runEvalSuite({
  scenarios,
  makeClient,
  agent,
  trials = 3,
  concurrency = 4,
}: {
  scenarios: Scenario[];
  makeClient: (scenario: Scenario, trial: number) => Anthropic;
  agent: (client: Anthropic, prompt: string) => Promise<RunOutput>;
  trials?: number;
  concurrency?: number;
}) {
  const jobs = scenarios.flatMap((s) => Array.from({ length: trials }, (_, t) => ({ s, t })));
  const outcomes = new Map<string, { pass: boolean; failures: string[]; costUsd: number }[]>();
  let next = 0;
  async function worker() {
    while (next < jobs.length) {
      const { s, t } = jobs[next++];
      let r: { pass: boolean; failures: string[]; costUsd: number };
      try {
        const out = await agent(makeClient(s, t), s.prompt);
        r = { ...evaluateTrajectory(out, s.expect), costUsd: out.costUsd };
      } catch (err) {
        r = { pass: false, failures: [`הריצה נכשלה: ${(err as Error).message}`], costUsd: 0 }; // קריסה היא כישלון — לא מדלגים
      }
      outcomes.set(s.name, [...(outcomes.get(s.name) ?? []), r]);
    }
  }
  await Promise.all(Array.from({ length: Math.min(concurrency, jobs.length) }, worker));

  const results = scenarios.map((s) => {
    const rs = outcomes.get(s.name)!;
    const passes = rs.filter((r) => r.pass).length;
    return {
      name: s.name,
      trials: rs.length,
      passes,
      passRate: passes / rs.length,
      failures: [...new Set(rs.flatMap((r) => r.failures))],
      avgCostUsd: Math.round((rs.reduce((x, r) => x + r.costUsd, 0) / rs.length) * 1e6) / 1e6,
    };
  });
  const totalPasses = results.reduce((x, r) => x + r.passes, 0);
  return {
    results,
    passRate: totalPasses / jobs.length,
    flaky: results.filter((r) => r.passRate > 0 && r.passRate < 1).map((r) => r.name), // עובר לפעמים — הכי מסוכן
    failing: results.filter((r) => r.passRate === 0).map((r) => r.name),
  };
}

P29.5-E28 · trace כעץ טקסט 🤖 💻 כתיבת קוד 🤖 רמה 2

💡 רמז 1

קודם קיבוץ: Map<string | null, Span[]> מהורה לילדים. ‏null הוא מפתח חוקי ב־Map, והוא מייצג "שורש". הורה שלא קיים: const ids = new Set(spans.map((s) => s.id)), ו־ids.has(parentId) ? parentId : null.

💡 רמז 2

אחר כך פונקציה פנימית walk(parent, depth): לוקחת את הילדים, ממיינת עותק ([...list].sort(...)), ולכל אחד מוסיפה שורה וקוראת ל־walk(span.id, depth + 1). השורה: ${" ".repeat(depth)}${span.name}.padEnd(32) + ${span.end - span.start}ms.padStart(8).

✅ פתרון
// P29.5-E28 — trace כעץ טקסט
export type Span = { id: string; parentId: string | null; name: string; start: number; end: number };

export function renderTrace(spans: Span[]): string {
  const children = new Map<string | null, Span[]>();
  const ids = new Set(spans.map((s) => s.id));
  for (const span of spans) {
    // הורה שלא נמצא ב־trace (נחתך, או עוד לא הגיע) — מציגים את ה־span כשורש
    const parent = span.parentId !== null && ids.has(span.parentId) ? span.parentId : null;
    const list = children.get(parent) ?? [];
    list.push(span);
    children.set(parent, list);
  }
  const lines: string[] = [];
  const walk = (parent: string | null, depth: number) => {
    const list = [...(children.get(parent) ?? [])].sort((a, b) => a.start - b.start || a.name.localeCompare(b.name));
    for (const span of list) {
      lines.push(`${"  ".repeat(depth)}${span.name}`.padEnd(32) + `${span.end - span.start}ms`.padStart(8));
      walk(span.id, depth + 1);
    }
  };
  walk(null, 0);
  return lines.join("\n");
}

⚠️ טעויות נפוצות
  • לולאה בתוך לולאה (לכל span — לחפש את הילדים ב־filter) — עובד, אבל O(n²). קיבוץ ב־Map פעם אחת הוא O(n).
  • להניח שההורה מופיע לפני הילדים ברשימה — spans נשלחים כשהם נגמרים, ולכן ילדים מגיעים בדרך כלל לפני ההורה.
  • לדלג על span עם הורה חסר — הוא נעלם מהתצוגה.
  • להזיח רק את השם אחרי padEnd — העמודה של הזמנים זזה בכל רמה.
