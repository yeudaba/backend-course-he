// P29.5-E10: אישור אנושי לפני פעולה — הסוכן עוצר, שומר מצב, וממשיך אחרי החלטה
import type Anthropic from "@anthropic-ai/sdk";

export interface ApprovalConfig {
  system: string;
  tools: Record<string, { definition: Anthropic.Tool; run: (input: any) => Promise<string> }>;
  needsApproval: (name: string, input: any) => boolean;
  maxSteps?: number;
  model?: string;
}

export interface RunState {
  messages: Anthropic.MessageParam[];
  waiting: { id: string; name: string; input: unknown }[]; // הקריאות של התור האחרון, שמחכות להחלטה
  steps: number;
}

export type AdvanceResult =
  | { status: "needs_approval"; state: RunState; pending: { id: string; name: string; input: unknown }[] }
  | { status: "done"; state: RunState; text: string }
  | { status: "max_steps"; state: RunState };

export const startState = (prompt: string): RunState => ({ messages: [{ role: "user", content: prompt }], waiting: [], steps: 0 });

async function execute(cfg: ApprovalConfig, calls: RunState["waiting"], decisions: Record<string, "approve" | "deny">): Promise<Anthropic.ToolResultBlockParam[]> {
  return Promise.all(
    calls.map(async (c): Promise<Anthropic.ToolResultBlockParam> => {
      const tool = cfg.tools[c.name];
      if (!tool) return { type: "tool_result", tool_use_id: c.id, content: `אין כלי בשם ${c.name}`, is_error: true };
      if (cfg.needsApproval(c.name, c.input) && decisions[c.id] !== "approve") {
        return { type: "tool_result", tool_use_id: c.id, content: "המשתמש לא אישר את הפעולה. אל תנסה שוב — שאל אותו מה לעשות.", is_error: true };
      }
      try {
        return { type: "tool_result", tool_use_id: c.id, content: await tool.run(c.input) };
      } catch (err) {
        return { type: "tool_result", tool_use_id: c.id, content: `שגיאה: ${(err as Error).message}`, is_error: true };
      }
    }),
  );
}

export async function advance(client: Anthropic, cfg: ApprovalConfig, state: RunState, decisions: Record<string, "approve" | "deny"> = {}): Promise<AdvanceResult> {
  const s: RunState = structuredClone(state); // לא משנים את המצב שקיבלנו — הוא אולי שמור במסד
  const maxSteps = cfg.maxSteps ?? 10;
  if (s.waiting.length) {
    const needing = s.waiting.filter((c) => cfg.needsApproval(c.name, c.input));
    const missing = needing.filter((c) => !(c.id in decisions));
    if (missing.length) return { status: "needs_approval", state: s, pending: missing };
    s.messages.push({ role: "user", content: await execute(cfg, s.waiting, decisions) });
    s.waiting = [];
  }
  while (s.steps < maxSteps) {
    s.steps++;
    const res = await client.messages.create({
      model: cfg.model ?? "claude-opus-5-5",
      max_tokens: 16000,
      system: cfg.system,
      tools: Object.values(cfg.tools).map((t) => t.definition),
      messages: s.messages,
    });
    s.messages.push({ role: "assistant", content: res.content });
    if (res.stop_reason !== "tool_use") {
      return { status: "done", state: s, text: res.content.flatMap((b) => (b.type === "text" ? [b.text] : [])).join("") };
    }
    const calls = res.content.filter((b): b is Anthropic.ToolUseBlock => b.type === "tool_use").map((b) => ({ id: b.id, name: b.name, input: b.input }));
    const needing = calls.filter((c) => cfg.needsApproval(c.name, c.input));
    if (needing.length) {
      s.waiting = calls; // עוצרים לפני שמריצים משהו מהתור הזה — גם את הכלים שלא צריכים אישור
      return { status: "needs_approval", state: s, pending: needing };
    }
    s.messages.push({ role: "user", content: await execute(cfg, calls, {}) });
  }
  return { status: "max_steps", state: s };
}

// P29.5-E14: ריצה עמידה — מצב נשמר אחרי כל צעד; אחרי קריסה ממשיכים, בלי לחזור על מה שכבר קרה
import type Anthropic from "@anthropic-ai/sdk";

export interface KV {
  get(key: string): Promise<string | undefined>;
  set(key: string, value: string): Promise<void>;
}

interface Saved {
  messages: Anthropic.MessageParam[];
  results: Record<string, Anthropic.ToolResultBlockParam>; // tool_use_id ← תוצאה שכבר חושבה
  steps: number;
  done?: string;
}

export async function runDurable(
  client: Anthropic,
  { runId, store, system, tools, prompt, maxSteps = 10, model = "claude-opus-5-5" }: {
    runId: string;
    store: KV;
    system: string;
    tools: Record<string, { definition: Anthropic.Tool; run: (input: any) => Promise<string> }>;
    prompt: string;
    maxSteps?: number;
    model?: string;
  },
): Promise<string> {
  const key = `agent-run:${runId}`;
  const raw = await store.get(key);
  const s: Saved = raw ? JSON.parse(raw) : { messages: [{ role: "user", content: prompt }], results: {}, steps: 0 };
  const save = () => store.set(key, JSON.stringify(s));
  if (s.done !== undefined) return s.done;

  for (;;) {
    const last = s.messages.at(-1)!;
    if (last.role === "assistant") {
      // התור האחרון של המודל ביקש כלים — מריצים רק את מה שעוד לא רץ, ושומרים כל תוצאה מיד
      const calls = (last.content as Anthropic.ContentBlockParam[]).filter((b): b is Anthropic.ToolUseBlockParam => b.type === "tool_use");
      for (const c of calls) {
        if (s.results[c.id]) continue;
        const tool = tools[c.name];
        let result: Anthropic.ToolResultBlockParam;
        try {
          if (!tool) throw new Error(`אין כלי בשם ${c.name}`);
          result = { type: "tool_result", tool_use_id: c.id, content: await tool.run(c.input) };
        } catch (err) {
          result = { type: "tool_result", tool_use_id: c.id, content: `שגיאה: ${(err as Error).message}`, is_error: true };
        }
        s.results[c.id] = result;
        await save();
      }
      s.messages.push({ role: "user", content: calls.map((c) => s.results[c.id]) });
      await save();
      continue;
    }
    if (s.steps >= maxSteps) throw new Error(`הריצה ${runId} עברה ${maxSteps} צעדים`);
    const res = await client.messages.create({ model, max_tokens: 16000, system, tools: Object.values(tools).map((t) => t.definition), messages: s.messages });
    s.steps++;
    if (res.stop_reason === "tool_use") {
      s.messages.push({ role: "assistant", content: res.content });
      await save();
      continue;
    }
    s.messages.push({ role: "assistant", content: res.content });
    s.done = res.content.flatMap((b) => (b.type === "text" ? [b.text] : [])).join("");
    await save();
    return s.done;
  }
}

const luhn = (digits: string) => {
  let sum = 0;
  for (let i = 0; i < digits.length; i++) {
    let d = Number(digits[digits.length - 1 - i]);
    if (i % 2 === 1) d = d * 2 > 9 ? d * 2 - 9 : d * 2;
    sum += d;
  }
  return sum % 10 === 0;
};
const israeliId = (id: string) => {
  const s = id.padStart(9, "0");
  let sum = 0;
  for (let i = 0; i < 9; i++) {
    const d = Number(s[i]) * ((i % 2) + 1);
    sum += d > 9 ? d - 9 : d;
  }
  return sum % 10 === 0;
};

export function redactPII(text: string): string {
  return text
    .replace(/\b(?:\d[ -]?){13,19}\b/g, (m) => (luhn(m.replace(/\D/g, "")) ? "[כרטיס]" : m))
    .replace(/(?<!\d)\d{9}(?!\d)/g, (m) => (israeliId(m) ? "[ת״ז]" : m))
    .replace(/(?<!\d)0(?:5\d|[2-489])-?\d{7}(?!\d)/g, "[טלפון]")
    .replace(/[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.[A-Za-z]{2,}/g, "[אימייל]");
}

export function checkRefund(input: { orderId: number; amountCents: number }, ctx: { store: Store; customerId: number; maxRefundCents: number }): { ok: true } | { ok: false; reason: string } {
  const o = ctx.store.orders.get(input.orderId);
  if (!o || o.customerId !== ctx.customerId) return { ok: false, reason: "ההזמנה לא נמצאה" };
  if (!Number.isInteger(input.amountCents) || input.amountCents <= 0) return { ok: false, reason: "סכום לא תקין" };
  if (o.status !== "paid" && o.status !== "shipped") return { ok: false, reason: `אי אפשר להחזיר הזמנה בסטטוס ${o.status}` };
  const already = ctx.store.refunds.filter((r) => r.orderId === o.id).reduce((s, r) => s + r.amountCents, 0);
  if (already + input.amountCents > o.totalCents) return { ok: false, reason: "הסכום גדול ממה שנשאר להחזיר" };
  if (input.amountCents > ctx.maxRefundCents) return { ok: false, reason: "מעל הסכום שסוכן רשאי להחזיר — דרוש נציג" };
  return { ok: true };
}

export function wrapUntrusted(source: string, text: string): string {
  // תיקון 3: תוכן שמנסה לסגור את התגית בעצמו — לא יוצא ממנה

}

export function createTaintPolicy({ untrustedTools, sensitiveTools }: { untrustedTools: string[]; sensitiveTools: string[] }) {
  let tainted = false; // תיקון 1: אחרי שתוכן לא אמין נכנס להקשר — הוא שם עד סוף השיחה
  return {
    afterTool(name: string) {
      if (untrustedTools.includes(name)) tainted = true;
    },
    // תיקון 2: הכלים הרגישים צריכים אישור — לא הכלים שמביאים תוכן
    requiresApproval: (name: string) => tainted && sensitiveTools.includes(name),
    get tainted() {
      return tainted;
    },
  };
}

        return JSON.stringify({ id: o.id, status: o.status, total: o.totalCents / 100, items: o.items });

Amount in agorot.", input_schema: { type: "object", properties: { orderId: { type: "integer" }, amountCents: { type: "integer" } }, required: ["orderId", "amountCents"], additionalProperties: false } },

      if (pending.length) {
        s.waiting = calls;
        return { status: "needs_approval", state: s, pending };
      }
      s.messages.push({ role: "user", content: await runTools(calls, {}) });
    }
    throw new Error("יותר מדי צעדים");
  }

  return {
    start: (message: string) => loop({ messages: [{ role: "user", content: message }], waiting: [] }),
    async resume(state: State, decisions: Record<string, "approve" | "deny">) {
      const s = structuredClone(state);
      s.messages.push({ role: "user", content: await runTools(s.waiting, decisions) });
      s.waiting = [];
      return loop(s);
    },
  };
}
