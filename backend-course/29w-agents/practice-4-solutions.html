P29.5-E11 · תוצאות כלי שלא מציפות 🤖 💻 כתיבת קוד 🤖 רמה 2

💡 רמז 1

ההודעה מיועדת למודל: "בקש offset/limit מצומצם יותר".

💡 רמז 2

ב־limitToolOutput חשב קודם כמה תווים לוקחים מכל צד, ורק אז את ההפרש. ב־paginate: מתי offset + limit מגיע לסוף?

✅ פתרון
// P29.5-E11: תוצאות כלי שלא מציפות את ה־context — קיצור, ועימוד
export function limitToolOutput(text: string, { maxChars = 4000 } = {}): string {
  if (text.length <= maxChars) return text;
  const head = Math.floor(maxChars * 0.7);
  const tail = maxChars - head;
  const omitted = text.length - head - tail;
  return `${text.slice(0, head)}\n…[הושמטו ${omitted} תווים מתוך ${text.length}. בקש טווח מצומצם יותר (offset/limit) כדי לראות את השאר]…\n${text.slice(text.length - tail)}`;
}

export function paginate<T>(items: T[], { offset = 0, limit = 20, maxLimit = 50 } = {}) {
  if (!Number.isInteger(offset) || offset < 0) throw new Error("offset לא תקין");
  if (!Number.isInteger(limit) || limit < 1) throw new Error("limit לא תקין");
  const n = Math.min(limit, maxLimit);
  const page = items.slice(offset, offset + n);
  const next = offset + page.length;
  return { items: page, total: items.length, nextOffset: next < items.length ? next : null };
}

למה לא "לקצר את ההיסטוריה" אחר כך — למחוק תוצאות ישנות מהודעות קודמות? כי במודלים הנוכחיים, עריכה של תורות קודמים פוסלת את בלוקי ה־thinking שאחריהם — ובחשבונות שנפתחו מסוף אוגוסט 2026, הבקשה נכשלת עם 400. ההיסטוריה היא append-only: מקטינים לפני שמוסיפים, ולשיחות ארוכות באמת — context editing או compaction של ה־API עצמו (שיעור 4).

⚠️ טעויות נפוצות
  • לחתוך רק את ההתחלה. הסוף של פלט (שורת סיכום, שגיאה) הוא לעתים קרובות החלק החשוב.
  • לחתוך בלי הודעה למודל. הוא לא יודע שחסר מידע, ומסיק מסקנות מחלק מהנתונים.
  • nextOffset שלא חוזר null בסוף, או שחוזר null כשנשארו פריטים. המודל יבקש עמוד ריק בלולאה, או יפספס את האחרונים.

💡 רמז 2

אל תעביר לעובד את messages של המוביל. בנה לו מערך חדש עם הודעה אחת, וחתוך את מה שהוא מחזיר לפני שהוא נכנס ל־tool_result.

✅ פתרון
// P29.5-E15: מוביל ועובדים — כל עובד מתחיל בהקשר נקי, ומחזיר רק סיכום
import type Anthropic from "@anthropic-ai/sdk";

export interface Worker {
  system: string;
  model?: string;
}

const MAX_REPORT = 2000;
const textOf = (m: Anthropic.Message) => m.content.flatMap((b) => (b.type === "text" ? [b.text] : [])).join("");

async function runWorker(client: Anthropic, w: Worker, instructions: string): Promise<string> {
  // הקשר חדש לגמרי: רק ההוראות — לא ההיסטוריה של המוביל
  const res = await client.messages.create({ model: w.model ?? "claude-opus-5-5", max_tokens: 8000, system: w.system, messages: [{ role: "user", content: instructions }] });
  const t = textOf(res);
  return t.length > MAX_REPORT ? t.slice(0, MAX_REPORT) + "…[קוצר]" : t;
}

export async function runLead(client: Anthropic, { system, task, workers, maxSteps = 8 }: { system: string; task: string; workers: Record<string, Worker>; maxSteps?: number }): Promise<string> {
  const delegate: Anthropic.Tool = {
    name: "delegate",
    description: `Give a self-contained sub-task to a worker. The worker sees only these instructions. Workers: ${Object.keys(workers).join(", ")}.`,
    input_schema: {
      type: "object",
      properties: { worker: { type: "string", enum: Object.keys(workers) }, instructions: { type: "string" } },
      required: ["worker", "instructions"],
      additionalProperties: false,
    },
  };
  const messages: Anthropic.MessageParam[] = [{ role: "user", content: task }];
  for (let i = 0; i < maxSteps; i++) {
    const res = await client.messages.create({ model: "claude-opus-5-5", max_tokens: 16000, system, tools: [delegate], messages });
    messages.push({ role: "assistant", content: res.content });
    if (res.stop_reason !== "tool_use") return textOf(res);
    const calls = res.content.filter((b): b is Anthropic.ToolUseBlock => b.type === "tool_use");
    const results = await Promise.all( // כמה עובדים במקביל
      calls.map(async (c): Promise<Anthropic.ToolResultBlockParam> => {
        const { worker, instructions } = c.input as { worker: string; instructions: string };
        const w = workers[worker];
        if (!w) return { type: "tool_result", tool_use_id: c.id, content: `אין עובד בשם ${worker}`, is_error: true };
        try {
          return { type: "tool_result", tool_use_id: c.id, content: await runWorker(client, w, instructions) };
        } catch (err) {
          return { type: "tool_result", tool_use_id: c.id, content: `העובד נכשל: ${(err as Error).message}`, is_error: true };
        }
      }),
    );
    messages.push({ role: "user", content: results });
  }
  throw new Error("המוביל עבר את תקרת הצעדים");
}

למה בכלל? כי סוכן אחד שקורא 200 מסמכים ממלא את ההקשר שלו בקריאה, ושוכח מה ביקשו ממנו. עובדים קוראים בהקשר משלהם, ומחזירים תקציר — המוביל רואה 200 תקצירים קצרים. המחיר: יותר טוקנים בסך הכול, ותיאום. ל־Managed Agents של Anthropic יש מנגנון כזה מובנה (multiagent), שמריץ את העובדים בשבילך.

⚠️ טעויות נפוצות
  • להעביר לעובד את ההיסטוריה של המוביל. זה יקר, מבלבל, ומבטל את כל היתרון של הקשר נקי.
  • להריץ האצלות באותו תור אחת אחרי השנייה. התרגיל מבקש מקביליות, ושלושה עובדים של 10 שניות הופכים ל־30.
  • להחזיר למוביל את כל התשובה של העובד. 20,000 תווים מכל עובד ממלאים את ה־context של המוביל.

💡 רמז 2

אחרי שהתשובה חזרה, בדוק את stop_reason לפני שאתה מחזיר אותה. ושים לב לא לדרוס betas שכבר הגיעו בפרמטרים.

✅ פתרון
// P29.5-E17: קריאה עם מדיניות — fallback בצד השרת, ושגיאות ברורות על refusal ועל תשובה חתוכה
import Anthropic from "@anthropic-ai/sdk";

export class RefusalError extends Error {
  category: string | null;
  constructor(category: string | null) {
    super(`המודל סירב (${category ?? "ללא קטגוריה"})`);
    this.category = category;
  }
}
export class TruncatedError extends Error {
  constructor() {
    super("התשובה נחתכה ב־max_tokens");
  }
}

export async function createWithPolicy(client: Anthropic, params: Record<string, any>): Promise<Anthropic.Message> {
  const res = (await client.beta.messages.create({
    ...params,
    betas: [...(params.betas ?? []), "server-side-fallback-2026-07-01"],
    fallbacks: "default", // אם המודל מסרב, השרת מנתב למודל אחר לפי קטגוריית הסירוב
  } as any)) as any;
  if (res.stop_reason === "refusal") throw new RefusalError(res.stop_details?.category ?? null);
  if (res.stop_reason === "max_tokens") throw new TruncatedError();
  return res as Anthropic.Message;
}

בסוכן, refusal באמצע הלולאה הוא סטטוס, לא תשובה: אין content לקרוא. ו־max_tokens באמצע tool_use — קלט חתוך לכלי. שניהם — בודקים לפני שממשיכים (E06).

⚠️ טעויות נפוצות
  • לחפש refusal ב־catch. סירוב מגיע כ־HTTP 200 עם stop_reason: "refusal", לא כשגיאה.
  • betas: ["server-side-fallback-2026-07-01"] שמחליף את ה־betas של ה־caller. צריך לצרף לרשימה הקיימת.
  • להחזיר תשובה עם stop_reason: "max_tokens" כרגילה. היא חתוכה, ועלולה להיות JSON שבור או משפט באמצע.

export function createLimiter(limits: Record<string, number>, defaultLimit = 4) {
  const active = new Map<string, number>();
  const queues = new Map<string, (() => void)[]>();
  const limitOf = (name: string) => limits[name] ?? defaultLimit;

  async function acquire(name: string) {
    if ((active.get(name) ?? 0) < limitOf(name)) {
      active.set(name, (active.get(name) ?? 0) + 1);
      return;
    }
    await new Promise<void>((resolve) => {
      const q = queues.get(name) ?? [];
      q.push(resolve);
      queues.set(name, q);
    });
  }
  function release(name: string) {
    const next = queues.get(name)?.shift();
    if (next) next(); // המקום עובר ישר לממתין הבא — active לא משתנה
    else active.set(name, (active.get(name) ?? 1) - 1);
  }
  return {
    async run<T>(name: string, fn: () => Promise<T>): Promise<T> {
      await acquire(name);
      try {
        return await fn();
      } finally {
        release(name);
      }
    },
    active: (name: string) => active.get(name) ?? 0,
  };
}
⚠️ טעויות נפוצות
  • לשחרר את המקום רק בהצלחה. שגיאה אחת "גונבת" מקום לתמיד, ואחרי כמה שגיאות הכלי נתקע.
  • ב־release להוריד את המונה ואז להעיר מחכה שמעלה אותו. בין השניים קריאה חדשה יכולה "לקפוץ" לפני המחכה, והסדר כבר לא FIFO.
  • מונה אחד לכל הכלים. אז מיילים מחכים לשאילתות למסד, ולהפך.

export function contextCost(addedPerStep: number[], opts: { systemTokens: number; usdPerMTokIn: number; usdPerMTokCacheRead: number }) {
  let sent = opts.systemTokens;
  let total = 0;
  let withCache = 0;
  addedPerStep.forEach((added, i) => {
    const previous = sent;
    sent += added;
    total += sent;
    withCache += i === 0 ? sent * opts.usdPerMTokIn : previous * opts.usdPerMTokCacheRead + added * opts.usdPerMTokIn;
  });
  return {
    totalInputTokens: total,
    usdNoCache: r6((total * opts.usdPerMTokIn) / 1e6),
    usdWithCache: r6(withCache / 1e6),
  };
}

  if (!last) throw new Error("אין הודעות במצב");

type Store = { get(id: string): Promise<string | null>; set(id: string, value: string): Promise<void> };

export function idempotentTool(run: (input: unknown) => Promise<string>, store: Store) {
  const inflight = new Map<string, Promise<string>>();
  return (toolUseId: string, input: unknown): Promise<string> => {
    let p = inflight.get(toolUseId);
    if (!p) {
      p = (async () => {
        const saved = await store.get(toolUseId);
        if (saved !== null) return saved;
        const out = await run(input);
        await store.set(toolUseId, out);
        return out;
      })().finally(() => inflight.delete(toolUseId));
      inflight.set(toolUseId, p);
    }
    return p;
  };
}

import type { Redis } from "ioredis";

type Status = "running" | "waiting" | "done";
const RUNNING = "agent:runs:running";

export function createRunStore(redis: Redis, opts: { ttlSeconds: number }) {
  const key = (runId: string) => `agent:run:${runId}`;
  return {
    async save(runId: string, state: unknown, status: Status) {
      const m = redis.multi().set(key(runId), JSON.stringify(state), "EX", opts.ttlSeconds);
      if (status === "running") m.sadd(RUNNING, runId);
      else m.srem(RUNNING, runId);
      await m.exec();
    },
    async load(runId: string) {
      const v = await redis.get(key(runId));
      return v === null ? null : (JSON.parse(v) as unknown);
    },
    async running() {
      return (await redis.smembers(RUNNING)).sort();
    },
    async remove(runId: string) {
      await redis.multi().del(key(runId)).srem(RUNNING, runId).exec();
    },
  };
}

// P29.5-E84: עובדים במקביל, עם מגבלה ו־timeout
type Task = { worker: string; run: (signal: AbortSignal) => Promise<string> };
type Out = { worker: string; ok: boolean; text: string };

async function runOne(task: Task, timeoutMs: number): Promise<Out> {
  const ac = new AbortController();
  let timer: ReturnType<typeof setTimeout> | undefined;
  const timeout = new Promise<Out>((resolve) => {
    timer = setTimeout(() => {
      ac.abort();
      resolve({ worker: task.worker, ok: false, text: "timeout" });
    }, timeoutMs);
  });
  const work = task.run(ac.signal).then(
    (text): Out => ({ worker: task.worker, ok: true, text: text.slice(0, 2000) }),
    (err): Out => ({ worker: task.worker, ok: false, text: `error: ${(err as Error).message}` }),
  );
  try {
    return await Promise.race([work, timeout]);
  } finally {
    clearTimeout(timer);
  }
}

export async function delegateAll(tasks: Task[], opts: { concurrency: number; timeoutMs: number }): Promise<Out[]> {
  const results: Out[] = new Array(tasks.length);
  let next = 0;
  const lane = async () => {
    while (next < tasks.length) {
      const i = next++;
      results[i] = await runOne(tasks[i], opts.timeoutMs);
    }
  };
  await Promise.all(Array.from({ length: Math.min(opts.concurrency, tasks.length) }, lane));
  return results;
}
