// P29.5-E11: תוצאות כלי שלא מציפות את ה־context — קיצור, ועימוד
export function limitToolOutput(text: string, { maxChars = 4000 } = {}): string {
  if (text.length <= maxChars) return text;
  const head = Math.floor(maxChars * 0.7);
  const tail = maxChars - head;
  const omitted = text.length - head - tail;
  return `${text.slice(0, head)}\n…[הושמטו ${omitted} תווים מתוך ${text.length}. בקש טווח מצומצם יותר (offset/limit) כדי לראות את השאר]…\n${text.slice(text.length - tail)}`;
}

export function paginate<T>(items: T[], { offset = 0, limit = 20, maxLimit = 50 } = {}) {
  if (!Number.isInteger(offset) || offset < 0) throw new Error("offset לא תקין");
  if (!Number.isInteger(limit) || limit < 1) throw new Error("limit לא תקין");
  const n = Math.min(limit, maxLimit);
  const page = items.slice(offset, offset + n);
  const next = offset + page.length;
  return { items: page, total: items.length, nextOffset: next < items.length ? next : null };
}

// P29.5-E15: מוביל ועובדים — כל עובד מתחיל בהקשר נקי, ומחזיר רק סיכום
import type Anthropic from "@anthropic-ai/sdk";

export interface Worker {
  system: string;
  model?: string;
}

const MAX_REPORT = 2000;
const textOf = (m: Anthropic.Message) => m.content.flatMap((b) => (b.type === "text" ? [b.text] : [])).join("");

async function runWorker(client: Anthropic, w: Worker, instructions: string): Promise<string> {
  // הקשר חדש לגמרי: רק ההוראות — לא ההיסטוריה של המוביל
  const res = await client.messages.create({ model: w.model ?? "claude-opus-5-5", max_tokens: 8000, system: w.system, messages: [{ role: "user", content: instructions }] });
  const t = textOf(res);
  return t.length > MAX_REPORT ? t.slice(0, MAX_REPORT) + "…[קוצר]" : t;
}

export async function runLead(client: Anthropic, { system, task, workers, maxSteps = 8 }: { system: string; task: string; workers: Record<string, Worker>; maxSteps?: number }): Promise<string> {
  const delegate: Anthropic.Tool = {
    name: "delegate",
    description: `Give a self-contained sub-task to a worker. The worker sees only these instructions. Workers: ${Object.keys(workers).join(", ")}.`,
    input_schema: {
      type: "object",
      properties: { worker: { type: "string", enum: Object.keys(workers) }, instructions: { type: "string" } },
      required: ["worker", "instructions"],
      additionalProperties: false,
    },
  };
  const messages: Anthropic.MessageParam[] = [{ role: "user", content: task }];
  for (let i = 0; i < maxSteps; i++) {
    const res = await client.messages.create({ model: "claude-opus-5-5", max_tokens: 16000, system, tools: [delegate], messages });
    messages.push({ role: "assistant", content: res.content });
    if (res.stop_reason !== "tool_use") return textOf(res);
    const calls = res.content.filter((b): b is Anthropic.ToolUseBlock => b.type === "tool_use");
    const results = await Promise.all( // כמה עובדים במקביל
      calls.map(async (c): Promise<Anthropic.ToolResultBlockParam> => {
        const { worker, instructions } = c.input as { worker: string; instructions: string };
        const w = workers[worker];
        if (!w) return { type: "tool_result", tool_use_id: c.id, content: `אין עובד בשם ${worker}`, is_error: true };
        try {
          return { type: "tool_result", tool_use_id: c.id, content: await runWorker(client, w, instructions) };
        } catch (err) {
          return { type: "tool_result", tool_use_id: c.id, content: `העובד נכשל: ${(err as Error).message}`, is_error: true };
        }
      }),
    );
    messages.push({ role: "user", content: results });
  }
  throw new Error("המוביל עבר את תקרת הצעדים");
}

import Anthropic from "@anthropic-ai/sdk";

export class RefusalError extends Error {
  category: string | null;
  constructor(category: string | null) {
    super(`המודל סירב (${category ?? "ללא קטגוריה"})`);
    this.category = category;
  }
}
export class TruncatedError extends Error {
  constructor() {
    super("התשובה נחתכה ב־max_tokens");
  }
}

export async function createWithPolicy(client: Anthropic, params: Record<string, any>): Promise<Anthropic.Message> {
  const res = (await client.beta.messages.create({
    ...params,
    betas: [...(params.betas ?? []), "server-side-fallback-2026-07-01"],
    fallbacks: "default", // אם המודל מסרב, השרת מנתב למודל אחר לפי קטגוריית הסירוב
  } as any)) as any;
  if (res.stop_reason === "refusal") throw new RefusalError(res.stop_details?.category ?? null);
  if (res.stop_reason === "max_tokens") throw new TruncatedError();
  return res as Anthropic.Message;
}

// P29.5-E18: הגבלת מקביליות לכל כלי — המודל ביקש 8 שאילתות במקביל; המסד מחזיק 2
export function createLimiter(limits: Record<string, number>, defaultLimit = 4) {
  const active = new Map<string, number>();
  const queues = new Map<string, (() => void)[]>();
  const limitOf = (name: string) => limits[name] ?? defaultLimit;

  async function acquire(name: string) {
    if ((active.get(name) ?? 0) < limitOf(name)) {
      active.set(name, (active.get(name) ?? 0) + 1);
      return;
    }
    await new Promise<void>((resolve) => {
      const q = queues.get(name) ?? [];
      q.push(resolve);
      queues.set(name, q);
    });
  }
  function release(name: string) {
    const next = queues.get(name)?.shift();
    if (next) next(); // המקום עובר ישר לממתין הבא — active לא משתנה
    else active.set(name, (active.get(name) ?? 1) - 1);
  }
  return {
    async run<T>(name: string, fn: () => Promise<T>): Promise<T> {
      await acquire(name);
      try {
        return await fn();
      } finally {
        release(name);
      }
    },
    active: (name: string) => active.get(name) ?? 0,
  };
}
