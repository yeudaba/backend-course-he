P29.5-E06 · לולאת סוכן לפרודקשן 🤖 💻 כתיבת קוד 🤖 רמה 2

💡 רמז 2

✅ פתרון
// P29.5-E06: לולאת סוכן שמחזיקה מעמד — כלים במקביל, שגיאות ככלי, ותקרת צעדים
import type Anthropic from "@anthropic-ai/sdk";

export interface AgentTool {
  definition: Anthropic.Tool;
  run: (input: any) => Promise<string>;
}

export interface AgentResult {
  status: "done" | "max_steps" | "refused" | "truncated";
  text: string;
  steps: number;
  usage: { inputTokens: number; outputTokens: number };
}

export async function runAgent(
  client: Anthropic,
  { system, tools, prompt, maxSteps = 10, model = "claude-opus-5-5" }: { system: string; tools: Record<string, AgentTool>; prompt: string; maxSteps?: number; model?: string },
): Promise<AgentResult> {
  const messages: Anthropic.MessageParam[] = [{ role: "user", content: prompt }];
  const usage = { inputTokens: 0, outputTokens: 0 };
  const textOf = (m: Anthropic.Message) => m.content.flatMap((b) => (b.type === "text" ? [b.text] : [])).join("");

  for (let step = 1; step <= maxSteps; step++) {
    const res = await client.messages.create({
      model,
      max_tokens: 16000,
      system,
      tools: Object.values(tools).map((t) => t.definition),
      messages,
    });
    usage.inputTokens += res.usage.input_tokens;
    usage.outputTokens += res.usage.output_tokens;

    if (res.stop_reason === "refusal") return { status: "refused", text: "", steps: step, usage };
    if (res.stop_reason === "max_tokens") return { status: "truncated", text: textOf(res), steps: step, usage };
    messages.push({ role: "assistant", content: res.content }); // כל התוכן — כולל tool_use ו־thinking — בלי לשנות
    if (res.stop_reason !== "tool_use") return { status: "done", text: textOf(res), steps: step, usage };

    const calls = res.content.filter((b): b is Anthropic.ToolUseBlock => b.type === "tool_use");
    // כלים במקביל; כל תוצאה — באותו סדר, וכולן בהודעת user אחת
    const results = await Promise.all(
      calls.map(async (c): Promise<Anthropic.ToolResultBlockParam> => {
        const tool = tools[c.name];
        if (!tool) return { type: "tool_result", tool_use_id: c.id, content: `אין כלי בשם ${c.name}`, is_error: true };
        try {
          return { type: "tool_result", tool_use_id: c.id, content: await tool.run(c.input) };
        } catch (err) {
          // שגיאה של כלי היא מידע למודל (יכול לתקן ולנסות שוב), לא קריסה של הסוכן
          return { type: "tool_result", tool_use_id: c.id, content: `שגיאה: ${(err as Error).message}`, is_error: true };
        }
      }),
    );
    messages.push({ role: "user", content: results });
  }
  return { status: "max_steps", text: "", steps: maxSteps, usage };
}

למה כל התוצאות בהודעה אחת? כי זו הצורה שבה המודל ביקש אותן. תוצאות בהודעות נפרדות "מלמדות" את המודל, לאורך השיחה, להפסיק לבקש כלים במקביל — והסוכן נהיה איטי.

// P29.5-E07: כלי עם סכמה אחת — להגדרה (JSON Schema strict) ולאימות הקלט בפועל
import { z } from "zod";
import type Anthropic from "@anthropic-ai/sdk";

export interface DefinedTool<T> {
  definition: Anthropic.Tool;
  execute: (input: unknown) => Promise<{ content: string; is_error?: true }>;
}

export function defineTool<T extends z.ZodObject<any>>({ name, description, schema, run }: { name: string; description: string; schema: T; run: (input: z.infer<T>) => Promise<string> }): DefinedTool<z.infer<T>> {
  if (!/^[a-zA-Z0-9_-]{1,64}$/.test(name)) throw new Error(`שם כלי לא תקין: ${name}`);
  const json = z.toJSONSchema(schema.strict()) as Record<string, unknown>; // strict ← additionalProperties: false
  delete json.$schema;
  return {
    definition: { name, description, input_schema: json as Anthropic.Tool.InputSchema, strict: true },
    async execute(input) {
      const parsed = schema.strict().safeParse(input);
      if (!parsed.success) {
        const msg = parsed.error.issues.map((i) => `${i.path.join(".") || "(root)"}: ${i.message}`).join("; ");
        return { content: `קלט לא תקין — ${msg}`, is_error: true };
      }
      try {
        return { content: await run(parsed.data) };
      } catch (err) {
        return { content: `שגיאה: ${(err as Error).message}`, is_error: true };
      }
    },
  };
}

strict: true אומר ל־API להבטיח שהקלט של המודל תואם לסכמה. אז למה לבדוק שוב? כי הבדיקה בקוד שלך היא שכבת ההגנה — מול באג, מול שינוי בהגדרה, ומול מי שקורא לכלי לא דרך המודל. ובקריאות עם streaming של קלט כלים, ה־API לא מאמת בכלל.

P29.5-E26 · שומר לולאה לסוכן 🤖 💻 כתיבת קוד 🤖 רמה 3

💡 רמז 1

stableKey רקורסיבית, עם שלושה מקרים: מערך ← `[${value.map(stableKey).join(",")}]`. אובייקט (לא null) ← המפתחות ממוינים, וכל אחד `${JSON.stringify(k)}:${stableKey(v)}`. כל השאר ← JSON.stringify(value).

💡 רמז 2

שדות: #steps, ‏#cost, ‏#calls = new Map<string, number>(). הטיפוס של התוצאה הוא union: { ok: true } | { ok: false; reason: "steps" | "repeat" | "cost"; message: string }. בעיגול: Math.round(x * 10_000) / 10_000. ‏0.1 + 0.1 + 0.1 + 0.1 ב־JavaScript הוא 0.4000000000000001.

✅ פתרון
// P29.5-E26 — שומר לולאה לסוכן: צעדים, חזרות ועלות
export function stableKey(value: unknown): string {
  if (Array.isArray(value)) return `[${value.map(stableKey).join(",")}]`;
  if (value !== null && typeof value === "object") {
    const entries = Object.keys(value)
      .sort()
      .map((k) => `${JSON.stringify(k)}:${stableKey((value as Record<string, unknown>)[k])}`);
    return `{${entries.join(",")}}`;
  }
  return JSON.stringify(value) ?? "null";
}

type Limits = { maxSteps: number; maxRepeats: number; maxCostUsd: number };
export type Verdict = { ok: true } | { ok: false; reason: "steps" | "repeat" | "cost"; message: string };

export class LoopGuard {
  #limits: Limits;
  #steps = 0;
  #cost = 0;
  #calls = new Map<string, number>();

  constructor(limits: Limits) {
    this.#limits = limits;
  }

  check(tool: string, input: unknown, costUsd: number): Verdict {
    this.#steps++;
    this.#cost += costUsd;
    const key = `${tool}:${stableKey(input)}`;
    const times = (this.#calls.get(key) ?? 0) + 1;
    this.#calls.set(key, times);

    if (this.#steps > this.#limits.maxSteps) return { ok: false, reason: "steps", message: `יותר מ־${this.#limits.maxSteps} צעדים` };
    if (this.#cost > this.#limits.maxCostUsd) return { ok: false, reason: "cost", message: `העלות עברה $${this.#limits.maxCostUsd}` };
    if (times > this.#limits.maxRepeats) return { ok: false, reason: "repeat", message: `${tool} נקרא ${times} פעמים עם אותו קלט` };
    return { ok: true };
  }

  get steps(): number {
    return this.#steps;
  }

  get costUsd(): number {
    return Math.round(this.#cost * 10_000) / 10_000;
  }
}

// P29.5-E33: מ־stop_reason לפעולה הבאה
import type Anthropic from "@anthropic-ai/sdk";

type Next =
  | { action: "run_tools"; calls: { id: string; name: string; input: unknown }[] }
  | { action: "done"; text: string }
  | { action: "continue" }
  | { action: "truncated" }
  | { action: "refused"; category: string | null };

export function nextAction(msg: Anthropic.Message): Next {
  switch (msg.stop_reason) {
    case "tool_use":
      return {
        action: "run_tools",
        calls: msg.content
          .filter((b): b is Anthropic.ToolUseBlock => b.type === "tool_use")
          .map((b) => ({ id: b.id, name: b.name, input: b.input })),
      };
    case "end_turn":
    case "stop_sequence": {
      const text = msg.content
        .filter((b): b is Anthropic.TextBlock => b.type === "text")
        .map((b) => b.text)
        .join("")
        .trim();
      return { action: "done", text };
    }
    case "pause_turn":
      return { action: "continue" };
    case "max_tokens":
      return { action: "truncated" };
    case "refusal": {
      const details = (msg as { stop_details?: { category?: string | null } | null }).stop_details;

    }
    default:
      throw new Error(`stop_reason לא מוכר: ${msg.stop_reason}`);
  }
}

// P29.5-E34: כלים במקביל, עם timeout לכל כלי
import type Anthropic from "@anthropic-ai/sdk";

type Tool = { run(input: unknown): Promise<string> };
type Call = { id: string; name: string; input: unknown };

async function withTimeout<T>(p: Promise<T>, ms: number): Promise<T> {
  let timer: ReturnType<typeof setTimeout> | undefined;
  const timeout = new Promise<never>((_, reject) => {
    timer = setTimeout(() => reject(new Error(`timeout after ${ms}ms`)), ms);
  });
  try {
    return await Promise.race([p, timeout]);
  } finally {
    clearTimeout(timer);
  }
}

export function runToolCalls(calls: Call[], tools: Record<string, Tool>, opts: { timeoutMs: number }) {
  return Promise.all(
    calls.map(async (c): Promise<Anthropic.ToolResultBlockParam> => {
      if (!Object.hasOwn(tools, c.name)) return { type: "tool_result", tool_use_id: c.id, content: `unknown tool: ${c.name}`, is_error: true };
      try {
        return { type: "tool_result", tool_use_id: c.id, content: await withTimeout(tools[c.name].run(c.input), opts.timeoutMs) };
      } catch (err) {
        const msg = (err as Error).message;
        return { type: "tool_result", tool_use_id: c.id, content: msg.startsWith("timeout after") ? msg : `שגיאה: ${msg}`, is_error: true };
      }
    }),
  );
}

הכלי האיטי ממשיך לרוץ ברקע — Promise.race לא עוצר אותו. כדי לעצור באמת, מעבירים לו AbortSignal (E42).

⚠️ טעויות נפוצות

  • ‏tools[c.name] בלי hasOwn — המודל ביקש toString, והקוד מנסה להריץ פונקציה של Object.prototype.

export function chooseApproach(task: Task): { approach: Approach; reasons: string[] } {
  if (task.steps === "one") return { approach: "single_call", reasons: [] };

// P29.5-E36: ההיסטוריה רק גדלה
function same(a: unknown, b: unknown): boolean {
  if (a === b) return true;
  if (a === null || b === null || typeof a !== "object" || typeof b !== "object") return false;
  if (Array.isArray(a) !== Array.isArray(b)) return false;
  if (Array.isArray(a) && Array.isArray(b)) return a.length === b.length && a.every((x, i) => same(x, b[i]));
  const ka = Object.keys(a);
  const kb = Object.keys(b);
  return ka.length === kb.length && ka.every((k) => Object.hasOwn(b, k) && same((a as Record<string, unknown>)[k], (b as Record<string, unknown>)[k]));
}

export function firstChangedMessage(prev: unknown[], next: unknown[]): number | null {
  for (let i = 0; i < prev.length; i++) {
    if (i >= next.length || !same(prev[i], next[i])) return i;
  }
  return null;
}

type Db = { query(text: string, params?: unknown[]): Promise<{ rows: any[] }> };
type Msg = { role: string; content: unknown };

export async function appendMessages(db: Db, conversationId: string, expectedCount: number, messages: Msg[]): Promise<number> {
  await db.query("BEGIN");
  try {
    const { rows } = await db.query("SELECT count(*)::int AS n FROM agent_messages WHERE conversation_id = $1", [conversationId]);
    if (rows[0].n !== expectedCount) throw Object.assign(new Error(`conflict: יש ${rows[0].n} הודעות, ציפינו ל־${expectedCount}`), { code: "CONFLICT" });
    for (let i = 0; i < messages.length; i++) {
      await db.query("INSERT INTO agent_messages (conversation_id, seq, role, content) VALUES ($1, $2, $3, $4::jsonb)", [
        conversationId,
        expectedCount + i,
        messages[i].role,
        JSON.stringify(messages[i].content),
      ]);
    }
    await db.query("COMMIT");
    return expectedCount + messages.length;
  } catch (err) {

    if ((err as { code?: string }).code === "23505") throw new Error("conflict: מישהו אחר הוסיף הודעות לשיחה בינתיים");

  }
}

}

// P29.5-E41: לכל היותר שתי ריצות ללקוח, ב־Redis
import type { Redis } from "ioredis";

export function createRunSlots(redis: Redis, opts: { maxPerUser: number; ttlSeconds: number }) {
  const key = (userId: string) => `agent:runs:${userId}`;
  return {
    async acquire(userId: string) {
      const n = await redis.incr(key(userId));
      if (n > opts.maxPerUser) {
        await redis.decr(key(userId));
        return false;
      }
      await redis.expire(key(userId), opts.ttlSeconds);
      return true;
    },
    async release(userId: string) {
      if ((await redis.decr(key(userId))) < 0) await redis.del(key(userId));
    },
    async active(userId: string) {
      return Number((await redis.get(key(userId))) ?? 0);
    },
  };
}

// P29.5-E42: POST /agent עם deadline
import express from "express";
import { z } from "zod";

type Run = (prompt: string, signal: AbortSignal) => Promise<{ status: "done"; text: string } | { status: "max_steps" }>;
const Body = z.object({ prompt: z.string().trim().min(1).max(4000) });

export function agentApp(run: Run, opts: { timeoutMs: number }) {
  const app = express();
  app.use(express.json());
  app.post("/agent", async (req, res) => {
    const parsed = Body.safeParse(req.body);
    if (!parsed.success) return res.status(400).json({ error: "prompt: 1–4000 תווים" });

    const send = (status: number, body: unknown) => {
      if (!res.headersSent) res.status(status).json(body);
    };
    const ac = new AbortController();
    const timer = setTimeout(() => {
      ac.abort();
      send(504, { error: "timeout" });
    }, opts.timeoutMs);
    res.on("close", () => {
      if (!res.writableEnded) ac.abort();
    });

    try {
      const r = await run(parsed.data.prompt, ac.signal);
      if (r.status === "done") send(200, { status: "done", text: r.text });
      else send(200, { status: "handed_off", text: "לא הצלחתי להשלים — העברתי לנציג" });
    } catch {
      send(500, { error: "agent_failed" });
    } finally {
      clearTimeout(timer);
    }
  });
  return app;
}

‏run מעביר את ה־signal הלאה — ל־client.messages.create(params, { signal }) ולכלים — כך שהביטול עוצר באמת, ולא רק את התשובה.
