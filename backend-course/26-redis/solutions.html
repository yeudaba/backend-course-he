P26-E31 · sessions ב־Redis 🤖 ✍️ מאפס רמה 3

💡 רמז 1

await redis.multi().hset(key, { ...data, userId: String(userId) }).expire(key, ttlSec).sadd(...).exec(). ‏HGETALL על מפתח שלא קיים מחזיר {}, לא null. ‏const { userId, ...data } = hash מפריד את userId משאר השדות.

💡 רמז 2

תוקף לרשימה: EXPIRE key ttl NX קובע תוקף רק אם אין לה עדיין, ו־EXPIRE key ttl GT רק מאריך. שניהם ברצף. ב־ioredis: .expire(key, ttl, "NX").expire(key, ttl, "GT"). ב־destroyAllForUser: ‏SMEMBERS, ואז redis.del(...ids.map(k)). ‏DEL מחזיר כמה מפתחות היו קיימים ונמחקו. זה בדיוק המספר.

✅ פתרון
// P26-E31 — sessions ב־Redis
import { randomUUID } from "node:crypto";
import type { Redis } from "ioredis";

export type Session = { userId: number; data: Record<string, string> };

const sessionKey = (id: string) => `session:${id}`;
const userKey = (userId: number) => `user:${userId}:sessions`;

export async function createSession(redis: Redis, userId: number, data: Record<string, string> = {}, ttlSec = 1800): Promise<string> {
  const id = randomUUID();
  await redis
    .multi()
    .hset(sessionKey(id), { ...data, userId: String(userId) })
    .expire(sessionKey(id), ttlSec)
    .sadd(userKey(userId), id)
    // הרשימה חיה כמו ה־session הארוך ביותר שבה: NX — אם אין לה עדיין תוקף, GT — רק להאריך
    .expire(userKey(userId), ttlSec, "NX")
    .expire(userKey(userId), ttlSec, "GT")
    .exec();
  return id;
}

export async function getSession(redis: Redis, id: string, ttlSec = 1800): Promise<Session | null> {
  const hash = await redis.hgetall(sessionKey(id));
  if (Object.keys(hash).length === 0) return null;
  const { userId, ...data } = hash;
  await redis.multi().expire(sessionKey(id), ttlSec).expire(userKey(Number(userId)), ttlSec, "GT").exec();
  return { userId: Number(userId), data };
}

export async function destroySession(redis: Redis, id: string): Promise<boolean> {
  const userId = await redis.hget(sessionKey(id), "userId");
  if (userId === null) return false;
  await redis.multi().del(sessionKey(id)).srem(userKey(Number(userId)), id).exec();
  return true;
}

export async function destroyAllForUser(redis: Redis, userId: number): Promise<number> {
  const ids = await redis.smembers(userKey(userId));
  const deleted = ids.length > 0 ? await redis.del(...ids.map(sessionKey)) : 0;
  await redis.del(userKey(userId));
  return deleted;
}

שורה 2: ה־session של הטלפון חי 60 שניות, והרשימה של המשתמש 1,800. בלי NX/‏GT, ה־session הקצר היה קובע לרשימה 60 שניות. אחרי דקה היא הייתה נמחקת, ו־"התנתק מכל המכשירים" לא היה מוצא את הלפטופ.

למה הרשימה בכלל? בלעדיה, כדי למצוא את כל ה־sessions של משתמש צריך לסרוק את כל המפתחות (SCAN session:*) ולקרוא כל אחד. זה איטי, ובשרת גדול זה מיליוני מפתחות. האינדקס ההפוך (משתמש ← מזהים) הוא אותו רעיון כמו אינדקס במסד. המחיר: לשמור על שניהם מעודכנים, ולכן MULTI.

הרשימה יכולה להכיל מזהים של sessions שכבר פגו. זה לא מזיק: ‏DEL עליהם לא עושה כלום, והרשימה עצמה פגה בסוף. בגלל זה destroyAllForUser סופרת לפי מה ש־DEL החזיר, ולא לפי אורך הרשימה.

⚠️ טעויות נפוצות
  • Math.random().toString(36) כמזהה — אפשר לנחש. מזהה session הוא סיסמה זמנית.
  • HSET ואז EXPIRE בלי MULTI — אם השרת נופל ביניהם, נשאר session בלי תוקף, לנצח.
  • if (!hash) — ‏{} הוא truthy, ו־session שלא קיים חוזר כ־{ userId: NaN }.
  • לקבוע לרשימה EXPIRE רגיל בכל יצירה — session קצר מקצר את הרשימה של כולם.

P26-E32 · לקרוא את INFO של Redis 🤖 🔁 חזרה רמה 2

💡 רמז 1

text.split(/\r?\n/) מפריד גם \r\n וגם \n. בלולאה: משתנה section שמצביע על האובייקט של המקטע הנוכחי. שורה שמתחילה ב־# יוצרת אובייקט חדש ומכניסה אותו ל־info. שורה רגילה נכנסת ל־section.

💡 רמז 2

Object.fromEntries(value.split(",").map((pair) => { const [k, n] = pair.split("="); return [k, Number(n)]; })). ב־hitRatio: ‏Number(info.stats?.keyspace_hits ?? 0), ואז Math.round(ratio * 1000) / 1000.

✅ פתרון
// P26-E32 — לקרוא את INFO של Redis
export type Info = Record<string, Record<string, string>>;

export function parseInfo(text: string): Info {
  const info: Info = {};
  let section: Record<string, string> | undefined;
  for (const raw of text.split(/\r?\n/)) {
    const line = raw.trim();
    if (line === "") continue;
    if (line.startsWith("#")) {
      section = {};
      info[line.slice(1).trim().toLowerCase()] = section;
      continue;
    }
    const colon = line.indexOf(":");
    if (colon === -1 || section === undefined) continue;
    section[line.slice(0, colon)] = line.slice(colon + 1);
  }
  return info;
}

export function parseKeyspace(info: Info): Record<string, Record<string, number>> {
  const result: Record<string, Record<string, number>> = {};
  for (const [db, value] of Object.entries(info.keyspace ?? {})) {
    result[db] = Object.fromEntries(
      value.split(",").map((pair) => {
        const [key, n] = pair.split("=");
        return [key, Number(n)];
      }),
    );
  }
  return result;
}

export function hitRatio(info: Info): number | null {
  const hits = Number(info.stats?.keyspace_hits ?? 0);
  const misses = Number(info.stats?.keyspace_misses ?? 0);
  if (hits + misses === 0) return null;
  return Math.round((hits / (hits + misses)) * 1000) / 1000;
}

P26-E33 · TypedEmitter: אירועים עם טיפוסים 🤖 🔁 חזרה רמה 3

💡 רמז 1

החתימה שעושה את כל העבודה: on<K extends keyof Events>(event: K, listener: (payload: Events[K]) => void). ‏K הוא שם אירוע מסוים, ו־Events[K] הוא ה־payload שלו. אותו דבר ב־emit. האחסון: #listeners = new Map<keyof Events, Set<(payload: any) => void>>().

💡 רמז 2

once: עוטפים את המאזין בפונקציה שקודם קוראת ל־this.off(event, wrapper) ורק אז ל־listener, ורושמים את העטיפה. ב־emit עוברים על עותק: [...set]. מחיקה מ־Set תוך כדי מעבר עליו עלולה לדלג על איבר.

✅ פתרון
// P26-E33 — TypedEmitter: אירועים עם טיפוסים
type Listener<T> = (payload: T) => void;

export class TypedEmitter<Events extends Record<string, unknown>> {
  #listeners = new Map<keyof Events, Set<Listener<any>>>();

  on<K extends keyof Events>(event: K, listener: Listener<Events[K]>): () => void {
    let set = this.#listeners.get(event);
    if (!set) {
      set = new Set();
      this.#listeners.set(event, set);
    }
    set.add(listener);
    return () => this.off(event, listener);
  }

  off<K extends keyof Events>(event: K, listener: Listener<Events[K]>): void {
    const set = this.#listeners.get(event);
    if (!set) return;
    set.delete(listener);
    if (set.size === 0) this.#listeners.delete(event);
  }

  once<K extends keyof Events>(event: K, listener: Listener<Events[K]>): () => void {
    const wrapper: Listener<Events[K]> = (payload) => {
      this.off(event, wrapper);
      listener(payload);
    };
    return this.on(event, wrapper);
  }

  emit<K extends keyof Events>(event: K, payload: Events[K]): number {
    const set = this.#listeners.get(event);
    if (!set) return 0;
    const listeners = [...set];
    for (const listener of listeners) listener(payload);
    return listeners.length;
  }

  listenerCount(event: keyof Events): number {
    return this.#listeners.get(event)?.size ?? 0;
  }
}

שורה 3, ‏1 0: ה־once נקרא באירוע הראשון ונעלם, ולכן השני כבר לא מגיע לאף אחד. בשורה האחרונה stop() ביטל את המאזין היחיד של order.paid.

K extends keyof Events ו־Events[K] הם מה שהופך את ה־emitter מ"כל מחרוזת" ל־API עם חוזה. העורך משלים שמות אירועים, ושינוי במבנה של order.paid מראה שגיאה בכל מקום ששולח אותו. בפנים יש any אחד: ‏Map לא יכול לזכור שלכל מפתח יש טיפוס אחר. ה־API החיצוני מגן על הכול, כמו ב־Repository<T> של פרק 12.

‏on שמחזיר פונקציית ביטול נוח במיוחד בקוד שנרשם לזמן מוגבל, כמו חיבור SSE (שיעור 6): נרשמים כשהלקוח מתחבר, וקוראים ל־stop() ב־req.on("close"). בלי זה, כל חיבור שנסגר משאיר מאזין, והזיכרון גדל.

⚠️ טעויות נפוצות
  • מערך במקום Set — אותו מאזין נרשם פעמיים ונקרא פעמיים.
  • לעבור על ה־Set עצמו כש־once מוחק באמצע — מאזין שנרשם אחריו עלול לא להיקרא.
  • emit(event: string, payload: unknown) — עובד, אבל בלי שום בדיקה של TypeScript, כמו EventEmitter.
  • ב־once, לקרוא ל־listener לפני off — אם המאזין עצמו קורא ל־emit לאותו אירוע, הוא נקרא שוב.

P26-E01 · Redis או PostgreSQL? רמה 1

💡 רמז 1

שאלה אחת לכל שורה: מה קורה אם Redis מאבד את המידע (restart בלי persistence, או מחיקה)?

💡 רמז 2

Redis שומר בזיכרון. הוא יכול לשמור לדיסק — אבל ברוב ההגדרות לא מתייחסים אליו כמקור אמת לכסף.

✅ פתרון
  1. PostgreSQL. כסף, טרנזקציות, constraints, גיבויים. אסור לאבד.
  2. שניהם: מקור האמת ב־PostgreSQL, עותק ב־Redis (cache). נמחק? טוענים שוב.
  3. Redis. נתון קצר־חיים שמשתנה בכל בקשה, ושכל השרתים צריכים לראות (פרק 24 עשה את זה ב־Map — לשרת אחד). נאבד? הכי גרוע — מישהו קיבל עוד כמה ניסיונות.
  4. תלוי במוצר. בפרויקט — PostgreSQL (המשתמש מצפה לראות אותה גם מחר, ממכשיר אחר). אתרים עם המון אורחים — לפעמים ב־Redis עם TTL, עגלה של אורח.
  5. Redis — sorted set לכל דקה (E24). סטטיסטיקה "חמה", מקורבת, שמתה לבד.
  6. תלוי: בפרויקט — PostgreSQL, כי הוא חלק מאותה טרנזקציה ומאותן החלטות. ב־Redis עם TTL זה עובד ומהיר יותר, אבל אז "ההזמנה נשמרה" ו"המפתח נשמר" כבר לא אטומיים יחד.
  7. Redis (BullMQ). העבודה עצמה — המייל — יכולה להיווצר מחדש מהמסד; לכן גם אובדן התור לא שובר נכונות, רק מעכב.
⚠️ טעויות נפוצות
  • "Redis מהיר, אז הכול שם". מהירות לא שווה הרבה אם נתון שאסור לאבד נעלם ב־restart.

P26-E03 · מלכודות של cache רמה 1

💡 רמז 1

"שני הדברים הקשים במדעי המחשב: ביטול cache, ונתינת שמות."

💡 רמז 2

מה נמצא במפתח של ה־cache? וכל מה שמשפיע על התשובה — נמצא שם?

✅ פתרון
  1. ביטול (invalidation). TTL הוא רשת ביטחון, לא תוכנית. בכל שינוי — מבטלים: מוחקים את המפתח, או מעלים "גרסה" לקבוצה (E14, ובפרויקט). ו־TTL קצר לנתונים שמשתנים.
  2. stampede. כולם מגלים יחד שאין ערך, וכולם טוענים. פתרונות: רק אחד טוען (מנעול / single-flight — E13), TTL עם "רעש" אקראי כדי שלא כל המפתחות יפוגו יחד, או רענון ברקע לפני שפג.
  3. penetration. שומרים גם את "אין תוצאה" (רשימה ריקה, ל־TTL קצר) — E20. ובנוסף: rate limit.
  4. מפתח בלי המשתמש. כל מה שמשפיע על התשובה — כולל מי שואל — חייב להיות במפתח. או: לא לשמור ב־cache משותף נתונים פרטיים בכלל. זו דליפת מידע, לא באג קטן.
  5. cache הפך לתלות. cache הוא אופטימיזציה: אם Redis לא עונה — הולכים למסד (איטי, אבל עובד). בפרויקט: X-Cache: OFF, והכול ממשיך.
⚠️ טעויות נפוצות
  • "נשים TTL של יום, יהיה מהר" — ואז באג "המחיר לא מתעדכן" שאף אחד לא מצליח לשחזר.

import type { Redis } from "ioredis";

// מטמון JSON פשוט מעל Redis: מחרוזות בלבד שם — לכן JSON.stringify / parse
export function createJsonStore(redis: Redis) {
  return {
    async set(key: string, value: unknown, ttlSeconds?: number) {
      if (value === undefined) throw new Error("אין ערך לשמור");

      if (ttlSeconds) await redis.set(key, json, "EX", ttlSeconds);
      else await redis.set(key, json);
    },

      const raw = await redis.get(key);
      return raw === null ? null : (JSON.parse(raw) as T);
    },
    async del(key: string) {
      return (await redis.del(key)) === 1;
    },

      const t = await redis.ttl(key);
      return t === -2 ? null : t === -1 ? Infinity : t;
    },
  };
}

P26-E07 · cache-aside 🤖 רמה 2

💡 רמז 1

💡 רמז 2

✅ פתרון
// P26-E07
import type { Redis } from "ioredis";

// cache-aside: קודם Redis; אין ← טוענים מהמקור ושומרים עם TTL
export function createCacheAside(redis: Redis) {
  const stats = { hits: 0, misses: 0 };
  async function get<T>(key: string, ttlSeconds: number, load: () => Promise<T>): Promise<T> {
    const raw = await redis.get(key);
    if (raw !== null) {
      stats.hits++;
      return JSON.parse(raw) as T;
    }
    stats.misses++;
    const value = await load(); // אם load זורק — לא שומרים כלום (לא מחזיקים שגיאות ב־cache)
    if (value !== undefined && value !== null) await redis.set(key, JSON.stringify(value), "EX", ttlSeconds);
    return value;
  }
  return { get, stats, invalidate: (key: string) => redis.del(key) };
}

P26-E09 · הגבלת קצב משותפת 🤖 רמה 2

💡 רמז 2

✅ פתרון
// P26-E09
import type { Redis } from "ioredis";

// fixed window ב־Redis: מונה לכל (לקוח, חלון). משותף לכל השרתים — בניגוד ל־Map בזיכרון (פרק 24)
export async function hit(redis: Redis, clientId: string, { limit, windowSeconds, now = Date.now() }: { limit: number; windowSeconds: number; now?: number }) {
  const window = Math.floor(now / 1000 / windowSeconds);
  const key = `rl:${clientId}:${window}`;
  // INCR אטומי; EXPIRE רק כשהמפתח נוצר עכשיו (NX) — המונה מת לבד בסוף החלון
  const [[, count]] = (await redis.multi().incr(key).expire(key, windowSeconds, "NX").exec()) as [[null, number], [null, number]];
  const resetSeconds = (window + 1) * windowSeconds - Math.floor(now / 1000);
  return { allowed: count <= limit, remaining: Math.max(0, limit - count), resetSeconds };
}

// טבלת מובילים: sorted set — Redis שומר ממוין לפי score, וכל פעולה ב־O(log n)
export function leaderboard(redis: Redis, name: string) {
  const key = `lb:${name}`;
  return {
    add: (player: string, points: number) => redis.zincrby(key, points, player).then(Number),
    async top(n: number) {
      const flat = await redis.zrevrange(key, 0, n - 1, "WITHSCORES");

      return r === null ? null : r + 1;

      const r = await redis.zrevrank(key, player);
      if (r === null) return [];
      const start = Math.max(0, r - radius);
      const flat = await redis.zrevrange(key, start, r + radius, "WITHSCORES");
      return Array.from({ length: flat.length / 2 }, (_, i) => ({ player: flat[2 * i], score: Number(flat[2 * i + 1]), rank: start + i + 1 }));
    },
  };

P26-E12 · מנעול מבוזר 🤖 רמה 2

💡 רמז 1

💡 רמז 2

שחרור: "אם הערך הוא ה־token שלי — מחק" חייב להיות אטומי. Lua: redis.eval(script, 1, key, token).

✅ פתרון
// P26-E12
import { randomUUID } from "node:crypto";
import type { Redis } from "ioredis";

// מחיקה רק אם הערך הוא ה־token שלי — בפקודה אחת (Lua רץ ב־Redis באופן אטומי)
const RELEASE = `if redis.call("GET", KEYS[1]) == ARGV[1] then return redis.call("DEL", KEYS[1]) else return 0 end`;

export async function acquire(redis: Redis, name: string, ttlMs: number): Promise<string | null> {
  const token = randomUUID();
  // SET NX PX: "רק אם לא קיים" ועם תוקף — באותה פקודה. אם התהליך מת, המנעול משתחרר לבד
  const ok = await redis.set(`lock:${name}`, token, "PX", ttlMs, "NX");
  return ok === "OK" ? token : null;
}

export async function release(redis: Redis, name: string, token: string): Promise<boolean> {
  return (await redis.eval(RELEASE, 1, `lock:${name}`, token)) === 1;
}

export async function withLock<T>(redis: Redis, name: string, ttlMs: number, fn: () => Promise<T>): Promise<T | null> {
  const token = await acquire(redis, name, ttlMs);
  if (!token) return null; // מישהו אחר מחזיק — לא מחכים (לפי הצורך: לנסות שוב אחר כך)
  try {
    return await fn();
  } finally {
    await release(redis, name, token);
  }
}

P26-E13 · cache שעומד בעומס 🤖 רמה 3

💡 רמז 1

Map<string, Promise> — ומוחקים ב־finally.

💡 רמז 2

המנעול — כמו ב־E12, ומשחררים רק אם הוא שלך.

✅ פתרון
// P26-E13
import { randomUUID } from "node:crypto";
import type { Redis } from "ioredis";

const RELEASE = `if redis.call("GET", KEYS[1]) == ARGV[1] then return redis.call("DEL", KEYS[1]) else return 0 end`;
const sleep = (ms: number) => new Promise((r) => setTimeout(r, ms));

// מניעת stampede: מפתח פופולרי פג תוקף, ו־500 בקשות טוענות אותו מהמסד בבת אחת.
// בתוך תהליך — "single flight" (Promise משותף). בין שרתים — מנעול: רק אחד טוען, האחרים מחכים לערך
export function createStampedeCache(redis: Redis, { lockMs = 5000, waitMs = 3000, pollMs = 25 } = {}) {
  const inFlight = new Map<string, Promise<unknown>>();

  async function load<T>(key: string, ttlSeconds: number, loader: () => Promise<T>): Promise<T> {
    const cached = await redis.get(key);
    if (cached !== null) return JSON.parse(cached) as T;
    const token = randomUUID();
    if ((await redis.set(`${key}:lock`, token, "PX", lockMs, "NX")) === "OK") {
      try {
        const value = await loader();
        await redis.set(key, JSON.stringify(value), "EX", ttlSeconds);
        return value;
      } finally {
        await redis.eval(RELEASE, 1, `${key}:lock`, token);
      }
    }
    // מישהו אחר טוען — מחכים שהערך יופיע. לא הופיע בזמן? טוענים בעצמנו (עדיף איטי מתקוע)
    for (const end = Date.now() + waitMs; Date.now() < end; ) {
      await sleep(pollMs);
      const v = await redis.get(key);
      if (v !== null) return JSON.parse(v) as T;
    }
    return loader();
  }

  return {
    get<T>(key: string, ttlSeconds: number, loader: () => Promise<T>): Promise<T> {
      const existing = inFlight.get(key);
      if (existing) return existing as Promise<T>;
      const p = load(key, ttlSeconds, loader).finally(() => inFlight.delete(key));
      inFlight.set(key, p);
      return p;
    },
  };
}
⚠️ טעויות נפוצות
  • לחכות למנעול בלי גבול — הטוען נתקע, וכל השרתים נתקעים איתו.

P26-E14 · ביטול עם גרסה 🤖 רמה 3

💡 רמז 1

שתי קריאות ל־Redis ב־get: הגרסה, ואז הערך.

💡 רמז 2

המפתחות של הגרסה הישנה לא נמחקים — יש להם TTL.

✅ פתרון
// P26-E14
import type { Redis } from "ioredis";

// cache עם "גרסה" לכל קבוצה: invalidate = INCR אחד, במקום לחפש ולמחוק את כל הווריאציות
export function createVersionedCache(redis: Redis) {
  const version = async (group: string) => (await redis.get(`v:${group}`)) ?? "0";
  return {
    async get<T>(group: string, key: string, ttlSeconds: number, load: () => Promise<T>): Promise<{ value: T; hit: boolean }> {
      const full = `c:${group}:${await version(group)}:${key}`;
      const raw = await redis.get(full);
      if (raw !== null) return { value: JSON.parse(raw) as T, hit: true };
      const value = await load();
      await redis.set(full, JSON.stringify(value), "EX", ttlSeconds);
      return { value, hit: false };
    },
    // המפתחות הישנים לא נמחקים — הם פשוט לא נקראים יותר, ומתים ב־TTL
    invalidate: (group: string) => redis.incr(`v:${group}`),
  };
}

P26-E15 · חלון נע ב־sorted set 🤖 רמה 3

💡 רמז 1

💡 רמז 2

✅ פתרון
// P26-E15
import { randomUUID } from "node:crypto";
import type { Redis } from "ioredis";

// sliding window ב־sorted set: כל בקשה = איבר עם score = זמן. סופרים רק את ה־windowMs האחרונות
export async function slidingHit(redis: Redis, clientId: string, { limit, windowMs, now = Date.now() }: { limit: number; windowMs: number; now?: number }) {
  const key = `sw:${clientId}`;
  const results = await redis
    .multi()
    .zremrangebyscore(key, "-inf", now - windowMs) // מנקים את מה שיצא מהחלון
    .zcard(key)
    .zrange(key, 0, 0, "WITHSCORES")
    .exec();
  const count = Number(results![1][1]);
  const oldest = results![2][1] as string[];
  if (count >= limit) {
    // נדחה — לא נספר (פרק 24). Retry-After: עד שהוותיק יוצא מהחלון
    const retryAfterMs = Number(oldest[1]) + windowMs - now;
    return { allowed: false, remaining: 0, retryAfterSeconds: Math.max(1, Math.ceil(retryAfterMs / 1000)) };
  }
  await redis.multi().zadd(key, now, `${now}-${randomUUID()}`).pexpire(key, windowMs).exec();
  return { allowed: true, remaining: limit - count - 1, retryAfterSeconds: 0 };
}

// P26-E18

// "צרכן idempotent": תורים מבטיחים "לפחות פעם אחת" — לא "בדיוק פעם אחת". retry, worker שנפל אחרי העבודה ולפני ack…
// לכן כל הודעה מסומנת כ"טופלה". SET NX: רק הראשון מצליח לסמן. נכשל באמצע? מוחקים את הסימון — שינסו שוב
export async function processOnce<T>(redis: Redis, messageId: string, ttlSeconds: number, handler: () => Promise<T>): Promise<{ processed: boolean; result?: T }> {
  const key = `processed:${messageId}`;
  if ((await redis.set(key, "1", "EX", ttlSeconds, "NX")) !== "OK") return { processed: false };
  try {
    return { processed: true, result: await handler() };
  } catch (err) {
    await redis.del(key);
    throw err;
  }
}

// cache בזיכרון של התהליך (L1, הכי מהיר) — אבל עם כמה שרתים, כל אחד מחזיק עותק משלו.
// כששרת אחד משנה, הוא מפרסם ב־pub/sub, וכל השרתים מוחקים את העותק שלהם
export async function createL1Cache(redis: Redis, subscriber: Redis, channel = "cache-invalidate") {
  const local = new Map<string, unknown>();
  await subscriber.subscribe(channel);
  subscriber.on("message", (ch: string, key: string) => {
    if (ch === channel) local.delete(key);
  });
  return {
    async get<T>(key: string, load: () => Promise<T>): Promise<T> {
      if (local.has(key)) return local.get(key) as T;
      const value = await load();
      local.set(key, value);
      return value;
    },
    // מקומית מיד, ולכל השאר דרך Redis
    async invalidate(key: string) {
      local.delete(key);
      await redis.publish(channel, key);
    },
    size: () => local.size,
  };
}
⚠️ טעויות נפוצות

export async function trending(redis: Redis, { windowMinutes, n, now = Date.now() }: { windowMinutes: number; n: number; now?: number }) {
  const m = minuteOf(now);
  const buckets = Array.from({ length: windowMinutes }, (_, i) => `trend:${m - i}`);
  const tmp = `trend:tmp:${m}:${windowMinutes}:${Math.random().toString(36).slice(2)}`;
  const results = await redis
    .multi()
    .zunionstore(tmp, buckets.length, ...buckets)
    .zrevrange(tmp, 0, n - 1, "WITHSCORES")
    .del(tmp)
    .exec();
  const flat = results![1][1] as string[];
  return Array.from({ length: flat.length / 2 }, (_, i) => ({ productId: Number(flat[2 * i]), views: Number(flat[2 * i + 1]) }));
}

// token bucket אטומי: הקריאה, החישוב והכתיבה — בסקריפט Lua אחד. Redis מריץ סקריפט בבת אחת,
// בלי שפקודה אחרת תיכנס באמצע: אין מרוץ בין שרתים, ואין round-trip נוסף
const TAKE = `
local capacity = tonumber(ARGV[1])
local rate = tonumber(ARGV[2])      -- אסימונים לשנייה
local now = tonumber(ARGV[3])       -- מילישניות
local cost = tonumber(ARGV[4])
local state = redis.call("HMGET", KEYS[1], "tokens", "ts")
local tokens = tonumber(state[1]) or capacity
local ts = tonumber(state[2]) or now
tokens = math.min(capacity, tokens + math.max(0, now - ts) / 1000 * rate)
local allowed = 0
if tokens >= cost then
  tokens = tokens - cost
  allowed = 1
end
redis.call("HSET", KEYS[1], "tokens", tostring(tokens), "ts", tostring(now))
redis.call("PEXPIRE", KEYS[1], math.ceil(capacity / rate * 1000) + 1000)
local retry = 0
if allowed == 0 then retry = math.ceil((cost - tokens) / rate) end
return { allowed, math.floor(tokens), retry }
`;

export async function take(redis: Redis, key: string, { capacity, refillPerSecond, now = Date.now(), cost = 1 }: { capacity: number; refillPerSecond: number; now?: number; cost?: number }) {
  const [allowed, remaining, retryAfterSeconds] = (await redis.eval(TAKE, 1, `tb:${key}`, capacity, refillPerSecond, now, cost)) as number[];
  return { allowed: allowed === 1, remaining, retryAfterSeconds };
}

P26-E26 · polling, SSE או WebSocket? רמה 1

💡 רמז 1

שתי שאלות: מי מדבר (רק השרת, או גם הלקוח)? וכמה מהר זה צריך להגיע?

💡 רמז 2

SSE — שרת ← לקוח, HTTP רגיל, עם התחברות מחדש ו־Last-Event-ID מובנים. WebSocket — שני הכיוונים, הודעות קטנות ותכופות.

✅ פתרון
  1. SSE — רק השרת מעדכן, ומעט. אפשר גם polling כל 30 שניות, אבל SSE מגיע מיד ולא מעמיס בקשות ריקות. ההתחברות מחדש וה־Last-Event-ID חינם.
  2. WebSocket — שני הצדדים כותבים, הרבה הודעות קטנות. (SSE + POST לכל הודעה עובד גם — פשוט פחות טבעי.)
  3. שום דבר — טעינה רגילה של הדף. אין "זמן אמת" במשהו שמשתנה פעם ביום.
  4. SSE — שרת ← מסכים. ואם יש כמה שרתים — Redis pub/sub מאחור (E30).
  5. WebSocket — שני כיוונים, תדירות גבוהה, וחשוב שלא יהיה overhead של HTTP לכל הודעה.
⚠️ טעויות נפוצות
  • WebSocket לכל דבר "כי הוא זמן אמת" — יותר קוד (התחברות מחדש, heartbeat, אימות) בשביל משהו ש־SSE נותן מובנה.

P26-E27 · הודעת SSE 🤖 רמה 2

💡 רמז 1

כל שדה הוא "name: value\n", וההודעה מסתיימת ב־"\n" נוסף.

💡 רמז 2

text.split(/\r\n|\r|\n/), ולכל שורה — data: ${line}\n.

✅ פתרון
// P26-E27
// הודעה אחת בפורמט של Server-Sent Events: שורות "שדה: ערך", ושורה ריקה בסוף
export function formatEvent({ data, event, id, retry }: { data: unknown; event?: string; id?: string | number; retry?: number }): string {
  let out = "";
  if (id !== undefined) out += `id: ${id}\n`;
  if (event !== undefined) out += `event: ${event}\n`;
  if (retry !== undefined) out += `retry: ${retry}\n`;
  const text = typeof data === "string" ? data : JSON.stringify(data);
  // ירידת שורה בתוך data הייתה מסיימת את השדה — לכן כל שורה מקבלת data: משלה, והדפדפן מחבר אותן עם \n
  for (const line of text.split(/\r\n|\r|\n/)) out += `data: ${line}\n`;
  return out + "\n"; // שורה ריקה = סוף ההודעה
}
⚠️ טעויות נפוצות
  • if (id) במקום id !== undefined — id: 0 נעלם.
  • לשלוח data עם ירידת שורה כמו שהיא — הדפדפן חושב שהשורה השנייה היא שדה לא מוכר, ומשמיט אותה.

P26-E28 · רכזת SSE עם Last-Event-ID 🤖 רמה 3

💡 רמז 1

res.writeHead(200, {...}) ו־res.flushHeaders() — כדי שהלקוח יקבל את ה־headers מיד, לפני האירוע הראשון. Set<Response> של המחוברים.

💡 רמז 2

req.on("close", () => clients.delete(res)). ו־Number(req.get("last-event-id")) — NaN כשאין header.

✅ פתרון
// P26-E28
// "רכזת" של SSE: כל לקוח מחובר מקבל כל אירוע, עם id עולה — ומי שהתנתק וחזר מקבל את מה שפספס
import type { Request, Response } from "express";

const format = (id: number, event: string, data: unknown) => `id: ${id}\nevent: ${event}\ndata: ${JSON.stringify(data)}\n\n`;

export function eventHub({ bufferSize = 100 }: { bufferSize?: number } = {}) {
  const clients = new Set<Response>();
  const buffer: { id: number; text: string }[] = []; // האירועים האחרונים — בשביל Last-Event-ID
  let lastId = 0;

  function handler(req: Request, res: Response) {
    res.writeHead(200, {
      "Content-Type": "text/event-stream",
      "Cache-Control": "no-cache",
      Connection: "keep-alive",
      "X-Accel-Buffering": "no", // שלא יחזיק את ההודעות ב־proxy (nginx)
    });
    res.flushHeaders();
    // הדפדפן שולח Last-Event-ID לבד כשהוא מתחבר מחדש
    const since = Number(req.get("last-event-id"));
    if (Number.isInteger(since)) for (const e of buffer) if (e.id > since) res.write(e.text);
    clients.add(res);
    req.on("close", () => clients.delete(res)); // אחרת: דליפה, וכתיבה לחיבור סגור
  }

  function publish(event: string, data: unknown) {
    const text = format(++lastId, event, data);
    buffer.push({ id: lastId, text });
    if (buffer.length > bufferSize) buffer.shift();
    for (const res of clients) res.write(text);
    return lastId;
  }

  return { handler, publish, size: () => clients.size };
}
⚠️ טעויות נפוצות
  • לשכוח את req.on("close") — כל רענון של הדף משאיר חיבור "מת" ברשימה. אחרי יום: אלפי res.write לחיבורים סגורים, וזיכרון שרק עולה.
  • לשמור את כל האירועים לנצח בשביל Last-Event-ID — דליפה מסוג אחר. מאגר מוגבל, ומי שפספס יותר מזה — טוען את הדף מחדש.

P26-E29 · חדרי צ'אט ב־WebSocket 🤖 רמה 3

💡 רמז 1

new WebSocketServer({ noServer: true }), ו־server.on("upgrade", (req, socket, head) => …). שם בודקים את ה־token, ואם תקין — wss.handleUpgrade(req, socket, head, (ws) => …).

לבדוק בעצמך: ב־Node 24 יש WebSocket גלובלי (כמו בדפדפן) — new WebSocket("ws://localhost:3000/chat?token=…").

💡 רמז 2

דחייה לפני ה־upgrade: socket.end("HTTP/1.1 401 Unauthorized\r\nConnection: close\r\n\r\n"). חדרים: Map<string, Set<WebSocket>>, ו־ws.on("close") מוציא מהחדר.

✅ פתרון
// P26-E29
// חדרי צ'אט ב־WebSocket: אימות בזמן ה־upgrade, הצטרפות לחדר, ושידור לחברי החדר בלבד
import type { Server } from "node:http";
import { WebSocketServer, WebSocket } from "ws";

export function createChatServer(server: Server, { verifyToken }: { verifyToken: (token: string) => string | null }) {
  const wss = new WebSocketServer({ noServer: true });
  const rooms = new Map<string, Set<WebSocket>>();
  const send = (ws: WebSocket, msg: object) => ws.readyState === WebSocket.OPEN && ws.send(JSON.stringify(msg));

  // האימות קורה לפני שהחיבור נפתח: בלי token תקין — 401 ו־socket סגור. אחרי upgrade כבר אין status codes
  server.on("upgrade", (req, socket, head) => {
    const url = new URL(req.url ?? "/", "http://x");
    const user = url.pathname === "/chat" ? verifyToken(url.searchParams.get("token") ?? "") : null;
    if (!user) {
      socket.end("HTTP/1.1 401 Unauthorized\r\nConnection: close\r\n\r\n");
      return;
    }
    wss.handleUpgrade(req, socket, head, (ws) => {
      let room: string | null = null;
      const leave = () => { if (room) rooms.get(room)?.delete(ws); room = null; };
      ws.on("message", (raw) => {
        let msg: { type?: string; room?: unknown; text?: unknown };
        try { msg = JSON.parse(String(raw)); } catch { return send(ws, { type: "error", error: "invalid json" }); }
        if (msg.type === "join" && typeof msg.room === "string" && msg.room) {
          leave();
          room = msg.room;
          if (!rooms.has(room)) rooms.set(room, new Set());
          rooms.get(room)!.add(ws);
          return send(ws, { type: "joined", room });
        }
        if (msg.type === "say" && typeof msg.text === "string") {
          if (!room) return send(ws, { type: "error", error: "join a room first" });
          // השולח (user) נקבע בשרת לפי ה־token — לא לפי מה שהלקוח כותב בהודעה
          for (const peer of rooms.get(room) ?? []) send(peer, { type: "message", room, from: user, text: msg.text });
          return;
        }
        send(ws, { type: "error", error: "unknown message" });
      });
      ws.on("close", leave);
    });
  });

  return { close: () => new Promise<void>((resolve) => { for (const ws of wss.clients) ws.terminate(); wss.close(() => resolve()); }) };
}
⚠️ טעויות נפוצות
  • לאמת אחרי שהחיבור נפתח ("שלח token בהודעה הראשונה") — בינתיים כל אחד מחזיק חיבור פתוח בשרת. ואם שוכחים לבדוק בהודעה אחת — פרצה.
  • from: msg.from — כל אחד יכול להתחזות לכל אחד.
  • לזרוק על JSON שבור — חריגה ב־handler של message מפילה את כל התהליך.

P26-E30 · fan-out בין שרתים עם Redis 🤖 רמה 3

💡 רמז 1

await sub.subscribe(channel) פעם אחת, ו־sub.on("message", (ch, message) => …) שמפיץ ל־Set של מאזינים.

💡 רמז 2

"פעם אחת": אל תקרא למאזינים המקומיים ישירות ב־publish — ההודעה חוזרת מ־Redis גם לשרת ששלח.

✅ פתרון
// P26-E30
// כמה שרתים, לקוחות מחוברים לכל אחד: אירוע שקרה בשרת אחד צריך להגיע ללקוחות של כולם — דרך Redis pub/sub
import type { Redis } from "ioredis";

export async function createRelay({ pub, sub, channel }: { pub: Redis; sub: Redis; channel: string }) {
  const listeners = new Set<(data: unknown) => void>();
  // מנוי אחד לשרת, לא אחד לכל לקוח. החיבור sub נפרד: חיבור במצב SUBSCRIBE לא מריץ פקודות אחרות
  await sub.subscribe(channel);
  const onMessage = (ch: string, message: string) => {
    if (ch !== channel) return; // אותו חיבור יכול להיות מנוי לערוצים אחרים
    const data = JSON.parse(message);
    for (const l of listeners) l(data);
  };
  sub.on("message", onMessage);

  return {
    // לא שולחים ישירות למאזינים המקומיים: ההודעה תחזור מ־Redis גם לשרת הזה — אחרת כל אחד יקבל פעמיים
    publish: async (data: unknown) => { await pub.publish(channel, JSON.stringify(data)); },
    subscribe(listener: (data: unknown) => void) {
      listeners.add(listener);
      return () => listeners.delete(listener);
    },
    async close() {
      sub.off("message", onMessage);
      await sub.unsubscribe(channel);
    },
  };
}
⚠️ טעויות נפוצות
  • publish שגם שולח מקומית וגם ל־Redis — הלקוחות של השרת השולח מקבלים כל הודעה פעמיים.
  • subscribe ב־Redis לכל לקוח SSE — אלף לקוחות = אלף מנויים, ואלף קריאות ל־JSON.parse לכל הודעה.
  • pub/sub לא שומר כלום: שרת שהיה למטה ברגע ה־publish — פספס. לכן ל־SSE יש Last-Event-ID, ולדברים שאסור לפספס — תור (BullMQ, שיעור 4) או Redis Streams.
