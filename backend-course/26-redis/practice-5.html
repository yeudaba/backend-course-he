P26-E93 · ‏getOrSet: ‏HIT, ‏MISS או OFF 🤖

export async function getOrSet<T>(redis: Redis | null, group: string, key: string, ttl: number, load: () => Promise<T>): Promise<{ value: T; status: "HIT" | "MISS" | "OFF" }>
export async function invalidate(redis: Redis | null, group: string): Promise<void>
  • הגרסה ב־v:<group> (אין ← "0"). המפתח: cache:<group>:<version>:<key>.

  • invalidate ← ‏INCR לגרסה. ‏null או שגיאה ← לא זורק.
