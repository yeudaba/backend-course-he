// P06-E09: split ו־join
function parseNumbers(text) {
  if (text.trim() === "") {
    return [];
  }
  const result = [];
  for (const part of text.split(",")) {
    result.push(Number(part.trim()));
  }
  return result;
}

function humanList(items) {
  if (items.length === 0) {
    return "";
  }
  if (items.length === 1) {
    return items[0];
  }
  return `${items.slice(0, -1).join(", ")} ו־${items[items.length - 1]}`;
}

P06-E13 · ספירה וייחודיים 🤖 רמה 3

💡 רמז 1

unique: מערך תוצאה, ולכל איבר — if (!result.includes(x)) result.push(x).

💡 רמז 2

mostFrequent: עבור על unique(arr), ולכל ערך חשב countOccurrences — מקסימום.

✅ פתרון
// P06-E13: ספירה וערכים ייחודיים
function countOccurrences(arr, value) {
  let count = 0;
  for (const x of arr) {
    if (x === value) {
      count++;
    }
  }
  return count;
}

function unique(arr) {
  const result = [];
  for (const x of arr) {
    if (!result.includes(x)) {
      result.push(x);
    }
  }
  return result;
}

function mostFrequent(arr) {
  let best;
  let bestCount = 0;
  for (const value of unique(arr)) {
    const count = countOccurrences(arr, value);
    if (count > bestCount) {
      best = value;
      bestCount = count;
    }
  }
  return best;
}

console.log(unique([3, 1, 3, 2, 1]), mostFrequent(["a", "b", "b", "a", "b"]));

module.exports = { countOccurrences, unique, mostFrequent };
[ 3, 1, 2 ] b

💡 mostFrequent כאן היא לולאה בתוך לולאה (O(n²)) — ראה פרק 4, שיעור 4. בפרק 7 נכתוב אותה עם Map במעבר אחד. ול־unique יש קיצור: [...new Set(arr)] — גם הוא בפרק 7.

⚠️ טעויות נפוצות
  • >= במקום > במקסימום — בתיקו ינצח האחרון.

// P06-E14: ניקוי תגיות
function cleanTags(input, max = 5) {
  return input
    .split(",")
    .map((tag) => tag.trim().toLowerCase())
    .filter((tag) => tag !== "")
    .filter((tag, i, all) => all.indexOf(tag) === i)

// P06-E16: טיפול במילים עם מערכים
function words(text) {
  return text.trim().split(" ").filter((w) => w !== "");
}

function toSlug(title) {
  return words(title.toLowerCase()).join("-");
}

function titleCase(text) {
  return words(text)

}

function wordCount(text) {
  return words(text).length;
}

P06-E17 · סטטיסטיקה עם reduce 🤖 רמה 3

// P06-E17: min, max, average עם reduce
function stats(nums) {
  if (nums.length === 0) {
    return null;
  }
  const min = nums.reduce((m, n) => (n < m ? n : m), nums[0]);
  const max = nums.reduce((m, n) => (n > m ? n : m), nums[0]);
  const total = nums.reduce((s, n) => s + n, 0);
  const average = Math.round((total / nums.length) * 100) / 100;
  return [min, max, average];
}

// P06-E18: topN וחציון
function topN(nums, n) {
  return [...nums].sort((a, b) => b - a).slice(0, n);
}

function median(nums) {
  if (nums.length === 0) {
    return null;
  }
  const sorted = [...nums].sort((a, b) => a - b);
  const mid = Math.floor(sorted.length / 2);
  if (sorted.length % 2 === 1) {
    return sorted[mid];
  }
  return (sorted[mid - 1] + sorted[mid]) / 2;
}

P06-E19 · עמודים 🤖 רמה 3

💡 רמז 1

chunk: לולאה עם i += size, ובכל סיבוב arr.slice(i, i + size).

💡 רמז 2

✅ פתרון
// P06-E19: חלוקה לעמודים
function chunk(arr, size) {
  const result = [];
  for (let i = 0; i < arr.length; i += size) {
    result.push(arr.slice(i, i + size));
  }
  return result;
}

function getPage(arr, page, pageSize) {
  if (page < 1) {
    return [];
  }
  return arr.slice((page - 1) * pageSize, page * pageSize);
}

function totalPages(count, pageSize) {
  return Math.max(1, Math.ceil(count / pageSize));
}

console.log(chunk([1, 2, 3, 4, 5], 2), getPage(["a", "b", "c", "d", "e"], 2, 2), totalPages(0, 10));

module.exports = { chunk, getPage, totalPages };

מערך של מערכים — chunk מחזיר אחד כזה. ב־getPage, עמוד אחרי הסוף "עובד לבד": slice מחוץ לגבולות מחזיר [].

⚠️ טעויות נפוצות
  • getPage עם page * pageSize כהתחלה — עמוד 1 יתחיל מהעמוד השני.

💡 רמז 2

אפשר להחליף את כל הלולאה ב־for...of — וכך באג הגבולות נעלם מעצמו.

✅ פתרון
// P06-E20: סיכום מכירות — מתוקן
function summarize(sales) {
  if (sales.length === 0) {
    return "אין מכירות";
  }
  let total = 0;
  let best = sales[0];
  for (const s of sales) {
    total += s;
    if (s > best) {
      best = s;
    }
  }
  return `${sales.length} מכירות, סה״כ ${total}, הגדולה ביותר ${best}`;
}

console.log(summarize([100, 250, 40]));
console.log(summarize([-20, -5]));

module.exports = { summarize };
3 מכירות, סה״כ 390, הגדולה ביותר 250
2 מכירות, סה״כ -25, הגדולה ביותר -5

(1) i < sales.length - 1 דילג על האחרון. (2) best = 0 — עם החזרות בלבד התוצאה 0. (3) רשימה ריקה החזירה "0 מכירות, הגדולה ביותר 0" — הבודק דורש הודעה מפורשת.

⚠️ טעויות נפוצות
  • לתקן רק מה שרואים בדוגמה הראשונה. הבודק בודק גם קלטים "רעים".

P06-E21 · map ו־filter שבורים 🤖 רמה 4

💡 רמז 1

אחת מחזירה מערך של undefined, אחת לא מסננת כלום, ואחת מחזירה undefined בכלל.

💡 רמז 2

{ } בלי return; callback של filter שמחזיר ערך במקום בוליאני; forEach במקום map.

✅ פתרון
// P06-E21: callbacks — מתוקן
function doublePrices(prices) {
  return prices.map((p) => p * 2);
}

function activeUsers(statuses) {
  return statuses.filter((s) => s === "active");
}

function shout(words) {
  return words.map((w) => w.toUpperCase());
}

console.log(doublePrices([5, 10]), activeUsers(["active", "blocked", "active"]), shout(["hi"]));

module.exports = { doublePrices, activeUsers, shout };
[ 10, 20 ] [ 'active', 'active' ] [ 'HI' ]

(1) map((p) => { p * 2 }) ← [undefined, undefined]. (2) filter((s) => s) — כל מחרוזת לא ריקה truthy, אז כולן נשארו. (3) forEach לא מחזיר כלום.

⚠️ טעויות נפוצות
  • לתקן את (3) עם forEach ו־push למערך — עובד, אבל זה בדיוק מה ש־map עושה.

💡 רמז 2

✅ פתרון
// P06-E24: דוח מכירות שבועי
const sum = (nums) => nums.reduce((s, n) => s + n, 0);
const money = (n) => `₪${n.toLocaleString("en-US")}`;

function bestDay(days, revenue) {
  const index = revenue.indexOf(Math.max(...revenue));
  return days[index];
}

function averageOrderValue(orders, revenue) {
  const totalOrders = sum(orders);
  if (totalOrders === 0) {
    return 0;
  }
  return Math.round((sum(revenue) / totalOrders) * 100) / 100;
}

function daysAbove(days, revenue, threshold) {
  return days.filter((_, i) => revenue[i] > threshold);
}

function reportLines(days, orders, revenue) {
  const lines = days.map((d, i) => `${d}: ${orders[i]} הזמנות, ${money(revenue[i])}`);
  lines.push(`סה״כ: ${sum(orders)} הזמנות, ${money(sum(revenue))}`);
  return lines;
}

const orders = [12, 8, 10, 9, 15, 16, 0];
const revenue = [1840, 1100, 1390, 1200, 1950, 1850, 0];
console.log(bestDay(days, revenue), averageOrderValue(orders, revenue));
console.log(reportLines(days, orders, revenue).join("\n"));

module.exports = { bestDay, averageOrderValue, daysAbove, reportLines };

P06-E25 · אתגר: שאלות ראיון — two sum ומיזוג 🤖 רמה 5

💡 רמז 1

twoSum: לולאה מקוננת, כש־j מתחיל מ־i + 1.

💡 רמז 2

mergeSorted: let i = 0, j = 0; ו־while (i < a.length && j < b.length); אחרי הלולאה — מוסיפים את מה שנשאר מאחד מהם.

✅ פתרון
// P06-E25: twoSum ו־mergeSorted
function twoSum(nums, target) {
  for (let i = 0; i < nums.length; i++) {
    for (let j = i + 1; j < nums.length; j++) {
      if (nums[i] + nums[j] === target) {
        return [i, j];
      }
    }
  }
  return null;
}

function mergeSorted(a, b) {
  const result = [];
  let i = 0;
  let j = 0;
  while (i < a.length && j < b.length) {
    if (a[i] <= b[j]) {
      result.push(a[i]);
      i++;
    } else {
      result.push(b[j]);
      j++;
    }
  }
  return [...result, ...a.slice(i), ...b.slice(j)];
}

console.log(twoSum([2, 7, 11, 15], 9), twoSum([1, 2], 10));
console.log(mergeSorted([1, 4, 9], [2, 3, 10, 12]).join(", "));

module.exports = { twoSum, mergeSorted };
[ 0, 1 ] null
1, 2, 3, 4, 9, 10, 12

צעדים: twoSum — עד כחצי מיליון השוואות (לולאה מקוננת, O(n²)). mergeSorted — לכל היותר 2,000 (כל איבר נלקח פעם אחת, O(n)). בפרק 7 נפתור את twoSum במעבר אחד עם Map — זה מה שהמראיין מחכה לו אחרי הפתרון הראשון.

⚠️ טעויות נפוצות
  • twoSum עם j שמתחיל מ־0 — עלול להחזיר [i, i] (אותו איבר פעמיים).
