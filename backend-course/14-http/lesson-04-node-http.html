1. השרת הכי קטן

playground/hello-server.mjs 
import http from "node:http";

const server = http.createServer((req, res) => {
  console.log(req.method, req.url);
  res.writeHead(200, { "Content-Type": "application/json; charset=utf-8" });
  res.end(JSON.stringify({ message: "שלום מהשרת", path: req.url }));
});

server.listen(3000, () => console.log("http://localhost:3000"));
חלקמה זה
http.createServer(handler)יוצר שרת. ה־handler נקרא על כל בקשה — callback (פרק 5!)
reqהבקשה: req.method, req.url, req.headers, והגוף (stream)
res.writeHead(status, headers)שורת ה־status ו־headers
res.end(body)שולח את הגוף ומסיים. בלי end — הלקוח מחכה לנצח
server.listen(3000)מתחיל להאזין בפורט. התוכנית לא מסתיימת — Ctrl+C

הרץ, ופתח http://localhost:3000/abc בדפדפן. כל רענון — שורה חדשה ב־Terminal (ולפעמים גם בקשה ל־/favicon.ico — הדפדפן מבקש אייקון).

2. Routing ידני וגוף JSON

פתח את 14-http/files/demo-server.mjs ב־VS Code וקרא אותו — הוא קצר. שלושה רעיונות מרכזיים בו:

const url = new URL(req.url, `http://${req.headers.host}`);
const parts = url.pathname.split("/").filter(Boolean);

req.url הוא רק "/products/7?x=1". new URL מפרק אותו; parts הוא ["products", "7"] — ומשם if־ים לפי method ו־path.

async function readJson(req) {
  let raw = "";
  for await (const chunk of req) raw += chunk;
  if (raw === "") return undefined;
  return JSON.parse(raw);
}

הגוף מגיע כ־stream (פרק 13) — בחתיכות. אוספים, ואז JSON.parse (שעלול לזרוק → 400).

} catch (err) {
  if (err instanceof SyntaxError) return send(res, 400, { error: "JSON לא תקין" });
  console.error(err);
  send(res, 500, { error: "שגיאה פנימית" });

try/catch אחד סביב הכול — שום בקשה לא מפילה את השרת, וללקוח לא נחשפים פרטים (פרק 8).

💡 למה בכלל Express?

תסתכל כמה קוד השרת הזה דורש לחמישה routes: פירוק URL, קריאת גוף, if־ים, writeHead. Express (פרק 15) עושה את כל זה בשורה לכל route. אבל עכשיו אתה יודע מה קורה מתחת — וזה יעזור בכל דיבוג.

3. fetch — Node כלקוח

שרתים מדברים גם עם שרתים אחרים (תשלומים, מיילים, מזג אוויר). fetch מובנה ב־Node:

playground/client.mjs 
const res = await fetch("http://localhost:4000/products/1");
console.log(res.status, res.ok, res.headers.get("content-type"));
const product = await res.json();
console.log(product.name);

const created = await fetch("http://localhost:4000/products", {
  method: "POST",
  headers: { "Content-Type": "application/json" },
  body: JSON.stringify({ name: "רמקול", price: 199 }),
});
console.log(created.status, await created.json());
% node playground/client.mjs
200 true application/json; charset=utf-8
מקלדת
201 { id: 3, name: 'רמקול', price: 199 }

(ה־id תלוי בכמה מוצרים כבר יצרת מאז שהשרת עלה — הנתונים בזיכרון, ונמחקים בכל הפעלה מחדש.)

⚠️ fetch לא זורק על 404 או 500

fetch נכשל (זורק) רק אם אין חיבור בכלל. תשובת 404 או 500 היא "הצלחה" מבחינתו! חייבים לבדוק res.ok (true ל־2xx) בעצמך. ו־timeout? אין כברירת מחדל — שרת שלא עונה ישאיר אותך מחכה. הפתרון: signal: AbortSignal.timeout(3000).

async function getJson(url, ms = 3000) {
  const res = await fetch(url, { signal: AbortSignal.timeout(ms) });
  if (!res.ok) {
    throw new Error(`HTTP ${res.status} מ־${url}`);
  }
  return res.json();
}
