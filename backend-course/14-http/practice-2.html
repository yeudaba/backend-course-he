let currentUser = null;

function handle(req) {
  if (req.path === "/login") {

  if (req.path === "/me") return `אתה ${currentUser}`;
}

P14-E08 · פירוק headers 🤖

💻 כתיבת קוד 🤖רמה 2⏱ 15 דק׳ידע נדרש: שיעור 2

בקובץ 14-http/starter/P14-E08.mjs השלם parseHeaders(lines) — מקבלת מערך שורות "Name: value" ומחזירה אובייקט:

  • שמות באותיות קטנות (כמו ב־Node).

P14-E13 · לפרק בקשה גולמית 🤖

💻 כתיבת קוד 🤖רמה 3⏱ 20 דק׳ידע נדרש: שיעור 2

בקובץ 14-http/starter/P14-E13.mjs השלם parseRequest(raw), שמקבלת בקשת HTTP כטקסט (שורות מופרדות ב־\r\n או \n) ומחזירה:

  • method, path (בלי query), query (אובייקט), headers (שמות באותיות קטנות).
  • body: אחרי השורה הריקה הראשונה. ריק ← undefined. אם content-type מכיל json ← אובייקט (JSON.parse), אחרת — המחרוזת.
  • שורה ראשונה שאין בה בדיוק 3 חלקים ← Error("שורת בקשה לא תקינה").

P14-E14 · לבנות תשובה גולמית 🤖

💻 כתיבת קוד 🤖רמה 3⏱ 20 דק׳ידע נדרש: שיעור 2

בקובץ 14-http/starter/P14-E14.mjs השלם buildResponse(status, body, headers = {}) שמחזירה את התשובה כטקסט עם \r\n:

  • שורה ראשונה: HTTP/1.1 201 Created — לפי המפה STATUS_TEXT שבקובץ. status שלא במפה ← Error(status לא מוכר: ${status}).
  • אחריה ה־headers שקיבלת, לפי הסדר.
  • אם יש גוף: מחרוזת ← כמו שהיא; אחרת ← JSON.stringify ו־Content-Type: application/json; charset=utf-8 (אם לא נשלח content-type). ובכל מקרה — Content-Length בבתים.
  • שורה ריקה, ואז הגוף.
  • 204 עם גוף ← Error("204 לא יכול להכיל גוף").

P14-E26 · פירוק header של Cookie 🤖

💻 כתיבת קוד 🤖🔥 חימוםרמה 2⏱ 10 דק׳ידע נדרש: שיעור 2

הדפדפן שולח את כל העוגיות (cookies) של האתר ב־header אחד:

Cookie: theme=dark; lang=he; cart=%D7%A1%D7%9C

כתוב וייצא parseCookies(header) שהופכת את הערך של ה־header (מה שאחרי Cookie: ) לאובייקט:

  • הזוגות מופרדים ב־;. רווחים מסביב לשם ולערך — מסירים.
  • הערכים מקודדים כמו ב־URL. מפענחים עם decodeURIComponent (‏%D7%A1%D7%9C ← "סל").
  • בערך עצמו יכול להופיע = (‏token=a=b=c ← { token: "a=b=c" }). רק ה־= הראשון מפריד.
  • חלק בלי =, או עם שם ריק — מדלגים עליו.
  • undefined או מחרוזת ריקה (אין header) ← {}.

P14-E49 · פירוק שורת status 🤖

💻 כתיבת קוד 🤖🔥 חימוםרמה 2⏱ 10 דק׳ידע נדרש: שיעור 2

P14-E50 · לקרוא Content-Type 🤖

💻 כתיבת קוד 🤖רמה 2⏱ 15 דק׳ידע נדרש: שיעור 2

ה־header ‏Content-Type יכול להגיע בהרבה צורות: application/json; charset=utf-8, ‏Application/JSON, ‏text/html;charset="UTF-8". כתוב וייצא שתי פונקציות:

  • parseContentType(value) ← { type, charset }. ‏type — החלק שלפני ה־; הראשון, באותיות קטנות ובלי רווחים. ‏charset — הערך של הפרמטר charset (שם הפרמטר בלי תלות באותיות), באותיות קטנות ובלי מירכאות, או null. פרמטרים אחרים — מתעלמים. ערך חסר או ריק ← null.
  • isJson(value) ← true אם ה־type הוא application/json, או מסתיים ב־+json (למשל application/problem+json).

P14-E51 · לבחור פורמט לפי Accept 🤖

💻 כתיבת קוד 🤖רמה 3⏱ 20 דק׳ידע נדרש: שיעור 2

הלקוח שולח Accept עם רשימת פורמטים, ולכל אחד אפשר לתת "ציון" q בין 0 ל־1 (ברירת מחדל 1; ‏q=0 פירושו "לא מקבל"):

Accept: text/html, application/json;q=0.9, */*;q=0.1

כתוב וייצא chooseFormat(accept, available). ‏available — מערך הפורמטים שהשרת יודע להחזיר, לפי סדר העדפה של השרת.

  • לכל פורמט ב־available מוצאים את ה־q שלו: התאמה מדויקת, או text/* (כל סוג שמתחיל ב־text/), או */*. אם כמה רשומות מתאימות — לוקחים את הספציפית ביותר (מדויקת ← ‏text/* ← ‏*/*).
  • מחזירים את הפורמט עם ה־q הגבוה ביותר (מעל 0). שוויון ← הראשון ב־available. אין אף אחד ← null.
  • accept חסר או ריק ← available[0]. אותיות ורווחים לא משנים.

P14-E53 · הלקוח קורא שגיאה

const responses = [
  '{"error":{"code":"NOT_FOUND","message":"לא נמצא"}}',
  '{"error":"לא נמצא"}',
  '{"message":"Internal Server Error"}',
];
for (const text of responses) {
  const data = JSON.parse(text);
  try {
    console.log(data.error.message ?? "(אין הודעה)");
  } catch (err) {
    console.log("קרס:", err.message);
  }
}
for (const text of responses) console.log(JSON.parse(text).error?.message ?? "(אין הודעה)");

P14-E54 · כסף ב־JSON

🧠 הבנהרמה 2⏱ 10 דק׳ידע נדרש: פרק 2, פרק 6, שיעור 2

מה יודפס?

const items = [
  { name: "מחברת", price: 19.9, qty: 3 },
  { name: "עט", price: 0.1, qty: 1 },
  { name: "מחק", price: 0.2, qty: 2 },

P14-E55 · משגיאה לתשובה 🤖

💻 כתיבת קוד 🤖רמה 3⏱ 20 דק׳ידע נדרש: פרק 8, פרק 12, שיעור 2

כתוב וייצא שלוש מחלקות שגיאה ופונקציה אחת שהופכת כל שגיאה לתשובת HTTP במבנה הקבוע מהשיעור:

  • class NotFoundError extends Error, ‏class ConflictError extends Error — רק הודעה.
  • class ValidationError extends Error — ‏constructor(message, details); ‏details הוא מערך (למשל ["price: חייב להיות מספר"]).
  • toApiError(err) ← { status, body }:
    • ValidationError ← 400, ‏{ error: { code: "VALIDATION", message, details } }
    • NotFoundError ← 404, ‏code: "NOT_FOUND"; ‏ConflictError ← 409, ‏code: "CONFLICT" (בלי details).
    • SyntaxError (JSON שבור) ← 400, ‏{ code: "INVALID_JSON", message: "JSON לא תקין" }.
    • כל שגיאה אחרת ← 500, ‏{ code: "INTERNAL", message: "שגיאה פנימית" } — בלי ההודעה המקורית.

% node 14-http/starter/P14-E55.mjs
% npm run check P14-E55

🔗 איך זה מתחבר: מחלקות שגיאה עם extends Error ו־instanceof מפרקים 8 ו־12 (כמו toHttpResponse בפרק 8). השיעור: מבנה שגיאה קבוע, ו־4xx מול 5xx — הודעת 500 לא חושפת פרטים פנימיים.

P14-E56 · מאובייקט פנימי ל־JSON של API 🤖

💻 כתיבת קוד 🤖רמה 3⏱ 20 דק׳ידע נדרש: פרק 5, פרק 7, שיעור 2

במסד הנתונים (פרק 16) השמות הם snake_case (created_at), אבל ה־API מחזיר camelCase. כתוב וייצא toApiJson(value) שהופכת ערך — לעומק — למבנה שמוכן ל־JSON לפי המוסכמות:

  • אובייקט רגיל ← אובייקט חדש שבו כל מפתח הוא camelCase: user_id ← userId, ‏created_at ← createdAt (מפתח שכבר camelCase נשאר). מפתחות שהערך שלהם undefined — מושמטים.
  • מערך ← מערך חדש, כל איבר עובר המרה.
  • Date ← מחרוזת ISO.
