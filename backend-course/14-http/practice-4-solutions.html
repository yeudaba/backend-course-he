💡 רמז 2

import http from "node:http";

// P14-E11
import http from "node:http";

function send(res, status, body, headers = {}) {
  res.writeHead(status, { "Content-Type": "application/json; charset=utf-8", ...headers });
  res.end(JSON.stringify(body));
}

const routes = {
  "/": () => ({ message: "שלום" }),
  "/time": () => ({ now: new Date().toISOString() }),
};

export function createHelloServer() {
  return http.createServer((req, res) => {
    const { pathname } = new URL(req.url, "http://localhost");
    const route = routes[pathname];
    if (!route) return send(res, 404, { error: "לא נמצא" });
    if (req.method !== "GET") return send(res, 405, { error: "Method Not Allowed" }, { Allow: "GET" });
    send(res, 200, route());
  });
}

מבנה "טבלת routes" (אובייקט של path ← פונקציה) קורא הרבה יותר טוב משרשרת if־ים, וזה בדיוק מה ש־Express עושה בשבילך.

⚠️ טעויות נפוצות

  • לשכוח return אחרי send ← שתי תשובות לאותה בקשה, ושגיאה ERR_HTTP_HEADERS_SENT.

P14-E16 · router עם פרמטרים 🤖 💻 כתיבת קוד 🤖 רמה 3

💡 רמז 2

חלק שמתחיל ב־: מתאים לכל ערך, ונכנס ל־params.

✅ פתרון
// P14-E16
const segments = (p) => p.split("/").filter(Boolean);

function matchParts(pattern, parts) {
  if (pattern.length !== parts.length) return null;
  const params = {};
  for (let i = 0; i < pattern.length; i++) {
    if (pattern[i].startsWith(":")) params[pattern[i].slice(1)] = decodeURIComponent(parts[i]);
    else if (pattern[i] !== parts[i]) return null;
  }
  return params;
}

export function createRouter() {
  const routes = [];
  return {
    add(method, pattern, handler) {
      routes.push({ method, pattern: segments(pattern), handler });
    },
    match(method, path) {
      const parts = segments(path);
      const allow = [];
      for (const route of routes) {
        const params = matchParts(route.pattern, parts);
        if (!params) continue;
        if (route.method === method) return { status: 200, handler: route.handler, params };
        allow.push(route.method);
      }
      return allow.length > 0 ? { status: 405, allow } : { status: 404 };
    },
  };
}

const router = createRouter();
router.add("GET", "/products/:id", "getOne");
router.add("DELETE", "/products/:id", "remove");
const m = router.match("GET", "/products/7");
console.log(m.status, m.handler, m.params);
console.log(router.match("PUT", "/products/7"), router.match("GET", "/nope"));
200 getOne { id: '7' }
{ status: 405, allow: [ 'GET', 'DELETE' ] } { status: 404 }

זה לב ה־routing של Express (פרק 15) — כולל req.params.

⚠️ טעויות נפוצות
  • להחזיר 404 ברגע שה־method לא מתאים — מפספס את ה־405 ואת ה־route הבא באותו path.

P14-E17 · לקרוא גוף JSON בבטחה 🤖 💻 כתיבת קוד 🤖 רמה 3

💡 רמז 1

💡 רמז 2

Buffer.concat(chunks).toString("utf8")

✅ פתרון
// P14-E17
import { Readable } from "node:stream";

export class HttpError extends Error {
  constructor(status, message) {
    super(message);
    this.name = "HttpError";
    this.status = status;
  }
}

export async function readJsonBody(req, { limit = 1_000_000 } = {}) {
  if (!req.headers["content-type"]?.includes("application/json")) {
    throw new HttpError(415, "Content-Type חייב להיות application/json");
  }
  const chunks = [];
  let size = 0;
  for await (const chunk of req) {
    const buf = Buffer.from(chunk);
    size += buf.length;
    if (size > limit) throw new HttpError(413, "הגוף גדול מדי");
    chunks.push(buf);
  }
  const text = Buffer.concat(chunks).toString("utf8");
  if (text.trim() === "") throw new HttpError(400, "גוף ריק");
  try {
    return JSON.parse(text);
  } catch {
    throw new HttpError(400, "JSON לא תקין");
  }
}

// אות עברית (2 בתים) שנחתכת בין שתי חתיכות:
const bytes = Buffer.from('{"name":"שלום"}');
const req = Readable.from([bytes.subarray(0, 10), bytes.subarray(10)]);
req.headers = { "content-type": "application/json" };
console.log(await readJsonBody(req));
{ name: 'שלום' }

עם raw += chunk (כמו בשרת ההדגמה) הדוגמה הזו הייתה מקבלת תו פגום. בשרת ההדגמה זה "עובד" כי הבקשות קטנות ומגיעות בחתיכה אחת — באג שמופיע רק בעומס או עם קבצים גדולים.

⚠️ טעויות נפוצות
  • לבדוק את הגודל רק בסוף — תוקף שולח 2GB והשרת אוסף הכול לזיכרון לפני שמסרב.

💡 רמז 2

פונקציה אחת validateNote(body, { partial }) משרתת גם POST וגם PATCH.

✅ פתרון
// P14-E18
import http from "node:http";

function send(res, status, body, headers = {}) {
  if (status === 204) {
    res.writeHead(204, headers);
    return res.end();
  }
  res.writeHead(status, { "Content-Type": "application/json; charset=utf-8", ...headers });
  res.end(JSON.stringify(body));
}

async function readJson(req) {
  const chunks = [];
  for await (const chunk of req) chunks.push(Buffer.from(chunk));
  return JSON.parse(Buffer.concat(chunks).toString("utf8") || "null");
}

// מחזירה הודעת שגיאה, או null אם תקין
function validateNote(body, { partial }) {
  if (typeof body !== "object" || body === null || Array.isArray(body)) return "הגוף חייב להיות אובייקט";
  if (partial && body.text === undefined && body.done === undefined) return "צריך text או done";
  if ((!partial || body.text !== undefined) && (typeof body.text !== "string" || body.text.trim().length < 1 || body.text.trim().length > 200)) {
    return "text: מחרוזת של 1–200 תווים";
  }
  if (body.done !== undefined && typeof body.done !== "boolean") return "done: true או false";
  return null;
}

export function createNotesServer() {
  const notes = [];
  let nextId = 1;

  return http.createServer(async (req, res) => {
    const parts = new URL(req.url, "http://localhost").pathname.split("/").filter(Boolean);
    try {
      if (parts[0] !== "notes" || parts.length > 2) return send(res, 404, { error: "לא נמצא" });

      if (parts.length === 1) {
        if (req.method === "GET") return send(res, 200, notes);
        if (req.method === "POST") {
          const body = await readJson(req);
          const error = validateNote(body, { partial: false });
          if (error) return send(res, 400, { error });
          const note = { id: nextId++, text: body.text.trim(), done: false };
          notes.push(note);
          return send(res, 201, note, { Location: `/notes/${note.id}` });
        }
        return send(res, 405, { error: "Method Not Allowed" }, { Allow: "GET, POST" });
      }

      const note = notes.find((n) => n.id === Number(parts[1]));
      if (!["GET", "PATCH", "DELETE"].includes(req.method)) {
        return send(res, 405, { error: "Method Not Allowed" }, { Allow: "GET, PATCH, DELETE" });
      }
      if (!note) return send(res, 404, { error: "הפתק לא נמצא" });
      if (req.method === "GET") return send(res, 200, note);
      if (req.method === "DELETE") {
        notes.splice(notes.indexOf(note), 1);
        return send(res, 204);
      }
      const body = await readJson(req);
      const error = validateNote(body, { partial: true });
      if (error) return send(res, 400, { error });
      if (body.text !== undefined) note.text = body.text.trim();
      if (body.done !== undefined) note.done = body.done;
      return send(res, 200, note);
    } catch (err) {
      if (err instanceof SyntaxError) return send(res, 400, { error: "JSON לא תקין" });
      console.error(err);
      send(res, 500, { error: "שגיאה פנימית" });
    }
  });
}

שים לב כמה מהקוד הוא "צנרת" (פירוק כתובת, קריאת גוף, status, headers) לעומת לוגיקה. בפרק 15 נכתוב את אותו API ב־Express — ונראה כמה ממנו נעלם.

⚠️ טעויות נפוצות
  • להשוות n.id === parts[1] — מספר מול מחרוזת, תמיד false.
  • לבדוק if (!body.text) — מחרוזת של רווחים עוברת, ו־{ "text": 5 } נשמר כמספר.

P14-E19 · ניסיון חוזר חכם 🤖 💻 כתיבת קוד 🤖 רמה 3

💡 רמז 1

לולאת for (let attempt = 0; attempt <= retries; attempt++).

💡 רמז 2

await new Promise((r) => setTimeout(r, delayMs * 2 ** attempt))

✅ פתרון
// P14-E19
const RETRYABLE = new Set([408, 429, 500, 502, 503, 504]);
const sleep = (ms) => new Promise((r) => setTimeout(r, ms));

export async function fetchWithRetry(url, init = {}, { retries = 2, delayMs = 200 } = {}) {
  let lastError;
  let lastResponse;
  for (let attempt = 0; attempt <= retries; attempt++) {
    if (attempt > 0) await sleep(delayMs * 2 ** (attempt - 1));
    try {
      lastResponse = await fetch(url, init);
      lastError = undefined;
      if (!RETRYABLE.has(lastResponse.status)) return lastResponse;
    } catch (err) {
      lastError = err;
      lastResponse = undefined;
    }
  }
  if (lastError) throw lastError;
  return lastResponse;
}

שני דברים חשובים בעולם האמיתי: (1) ניסיון חוזר מתאים רק לבקשות idempotent — POST של הזמנה צריך "מפתח idempotency" (פרויקט הדגל); (2) מערכות גדולות מוסיפות "jitter" (רעש אקראי) להמתנה, כדי שאלף לקוחות לא ינסו שוב באותה מילישנייה.

⚠️ טעויות נפוצות
  • לנסות שוב על כל status שאינו ok — 400 יישלח 3 פעמים ויקבל 400 שלוש פעמים.
  • ניסיון חוזר בלי המתנה — מכה בשרת שכבר מתקשה.

P14-E22 · השרת שקורס מבקשה אחת 🤖 💻 כתיבת קוד 🤖 רמה 4

💡 רמז 1

handler שהוא async וזורק ← unhandled rejection ← Node מפיל את התהליך. ושים לב איך נראה req.url כשהכתובת בעברית.

💡 רמז 2

(counts[word] ?? 0) + 1, ו־decodeURIComponent.

✅ פתרון
// P14-E22: מתוקן
import http from "node:http";

function send(res, status, body) {
  res.writeHead(status, { "Content-Type": "application/json; charset=utf-8" });
  res.end(JSON.stringify(body));
}

export function createCounterServer() {
  const counts = {};
  return http.createServer(async (req, res) => {
    try {
      if (req.method === "POST" && req.url === "/count") {
        let raw = "";
        for await (const chunk of req) raw += chunk;
        const { word } = JSON.parse(raw);
        if (typeof word !== "string" || word.trim() === "") return send(res, 400, { error: "word חובה" });
        counts[word] = (counts[word] ?? 0) + 1;
        return send(res, 200, { word, count: counts[word] });
      }
      if (req.method === "GET" && req.url.startsWith("/count/")) {
        const word = decodeURIComponent(req.url.slice("/count/".length));
        return send(res, 200, { word, count: counts[word] ?? 0 });
      }
      send(res, 404, { error: "לא נמצא" });
    } catch (err) {
      if (err instanceof SyntaxError) return send(res, 400, { error: "JSON לא תקין" });
      console.error(err);
      send(res, 500, { error: "שגיאה פנימית" });
    }
  });
}

באג 1 הוא שבירת שרת על ידי כל משתמש אנונימי — Denial of Service בבקשה אחת. (גם JSON.parse("null") היה מפיל את ה־destructuring — ה־catch תופס גם את זה, כ־500. ניסיון לשפר: לבדוק שהגוף הוא אובייקט ולהחזיר 400.)

⚠️ טעויות נפוצות
  • לעטוף רק את JSON.parse — שגיאה אחרת בהמשך עדיין מפילה את השרת. try/catch אחד סביב כל ה־handler.

P14-E27 · סטטיסטיקת בקשות כמחלקה 🤖 💻 כתיבת קוד 🤖 רמה 3

💡 רמז 1

#routes = new Map(). הערך לכל route: { count, totalMs, errors }. ב־record: ‏const s = this.#routes.get(route) ?? { count: 0, totalMs: 0, errors: 0 }, מעדכנים, ו־set.

💡 רמז 2

[...this.#routes] נותן מערך של [route, s]. ‏map לאובייקט התוצאה, ואז sort((a, b) => b.count - a.count || a.route.localeCompare(b.route)). כשההפרש הוא 0 (תיקו), ה־|| עובר לקריטריון השני.

✅ פתרון
// P14-E27: סטטיסטיקת בקשות כמחלקה
export class RequestStats {
  #routes = new Map();

  record(method, path, status, ms) {
    const route = `${method} ${path}`;
    const s = this.#routes.get(route) ?? { count: 0, totalMs: 0, errors: 0 };
    s.count++;
    s.totalMs += ms;
    if (status >= 500) {
      s.errors++;
    }
    this.#routes.set(route, s);
  }

  get total() {
    let sum = 0;
    for (const s of this.#routes.values()) {
      sum += s.count;
    }
    return sum;
  }

  summary() {
    return [...this.#routes]
      .map(([route, s]) => ({ route, count: s.count, avgMs: Math.round(s.totalMs / s.count), errors: s.errors }))
      .sort((a, b) => b.count - a.count || a.route.localeCompare(b.route));
  }
}

const stats = new RequestStats();
stats.record("GET", "/users", 200, 12);
stats.record("GET", "/users", 200, 20);
stats.record("POST", "/users", 201, 35);
stats.record("GET", "/orders", 500, 101);
stats.record("GET", "/users", 503, 3);
console.log(stats.total);
console.log(stats.summary());

שומרים totalMs ולא את הממוצע. ממוצע אי אפשר לעדכן בלי לדעת על כמה בקשות הוא חושב. סכום ומונה מעדכנים בפעולה אחת, ומחשבים את הממוצע רק כשמבקשים summary().

b.count - a.count || ...: ההפרש 0 הוא falsy, ולכן רק בתיקו נבדק הקריטריון השני. זה הדפוס שראית בפרק 6 למיון לפי כמה שדות.

כך עובדים כלי ניטור כמו Prometheus (פרק 24): לכל route יש מונים מצטברים, ומהם מחשבים ממוצעים ואחוזי שגיאות.

⚠️ טעויות נפוצות
  • לשמור avgMs ולעדכן אותו ב־(avg + ms) / 2 — זה לא ממוצע. בשלוש בקשות 12, 20, 3 יוצא 9.5 במקום 11.67.
  • לספור status >= 400 כשגיאה — 404 היא תשובה תקינה של השרת. כאן סופרים תקלות של השרת (5xx).
  • routes = new Map() בלי # — הבודק מגלה שדה ציבורי.
  • total() כמתודה במקום get total() — ‏stats.total מחזיר פונקציה ולא מספר.

P14-E28 · לחכות שהשרת יעלה 🤖 💻 כתיבת קוד 🤖 רמה 3

💡 רמז 1

השהיה: const sleep = (ms) => new Promise((resolve) => setTimeout(resolve, ms)); (פרק 10). הלולאה: let attempt = 0; while (attempt < tries) { attempt++; ... }, ואחרי הלולאה throw.

💡 רמז 2

בתוך הלולאה: try { const res = await fetch(url); if (res.ok) return attempt; } catch { }, ואחריו if (attempt < tries) await sleep(delayMs);. ה־catch הריק בכוונה: שגיאת רשת היא בדיוק המצב שבו ממשיכים לחכות.

✅ פתרון
// P14-E28: לחכות שהשרת יעלה
import http from "node:http";

const sleep = (ms) => new Promise((resolve) => setTimeout(resolve, ms));

export async function waitUntilReady(url, { tries = 10, delayMs = 100 } = {}) {
  let attempt = 0;
  while (attempt < tries) {
    attempt++;
    try {
      const res = await fetch(url);
      if (res.ok) {
        return attempt;
      }
    } catch {
      // השרת עוד לא מקשיב — ננסה שוב
    }
    if (attempt < tries) {
      await sleep(delayMs);
    }
  }
  throw new Error(`השרת לא מוכן אחרי ${tries} ניסיונות`);
}

// שרת שעונה 503 בשתי הבקשות הראשונות, ואז 200
let calls = 0;
const server = http.createServer((req, res) => {
  calls++;
  res.writeHead(calls <= 2 ? 503 : 200).end();
});
server.listen(0, async () => {
  const url = `http://localhost:${server.address().port}/health`;
  console.log("מוכן בניסיון", await waitUntilReady(url, { delayMs: 20 }));
  try {
    await waitUntilReady("http://localhost:1/health", { tries: 3, delayMs: 20 });
  } catch (err) {
    console.log(err.message);
  }
  server.close();
});
מוכן בניסיון 3
השרת לא מוכן אחרי 3 ניסיונות

כמה פעמים הלולאה תרוץ? לא יודעים מראש — היא נעצרת כשהשרת עונה. זה המקרה הקלאסי ל־while (כמו fetchAll ב־P10-E28). ‏return מתוך הלולאה יוצא גם מהפונקציה, ולכן ה־throw אחרי הלולאה קורה רק אם כל הניסיונות נכשלו.

ההבדל מ־P14-E19 (ניסיון חוזר חכם): שם מנסים שוב פעולה שנכשלה. כאן מחכים שמשהו יהיה מוכן. גם 503 וגם "החיבור נדחה" פירושם "עוד לא".

בפרק 23 תראה את אותו רעיון ב־healthcheck של Docker Compose.

⚠️ טעויות נפוצות
  • בלי try/catch סביב fetch — כשהשרת עוד לא מקשיב, fetch זורק (ECONNREFUSED), והפונקציה נכשלת כבר בניסיון הראשון.
  • while (true) בלי תנאי עצירה — אם השרת לא עולה אף פעם, הסקריפט תקוע לנצח.
  • לחכות גם אחרי הניסיון האחרון — עוד delayMs מבוזבז לפני השגיאה.
  • setTimeout(() => {}, delayMs) בלי Promise ו־await — לא מחכה בכלל.

P14-E24 · אתגר: Express משלך 🤖 💻 כתיבת קוד 🤖 רמה 5

💡 רמז 1

שלב 1: שרת שמכין req/res (path, query, body, status, json). שלב 2: router מ־E16. שלב 3: שרשרת middleware.

💡 רמז 2

middleware: let i = 0; const next = async () => { const fn = stack[i++]; if (fn) await fn(req, res, next); else await runRoute(); }

✅ פתרון
// P14-E24
import fs from "node:fs";
import http from "node:http";

const segments = (p) => p.split("/").filter(Boolean);

function matchParts(pattern, parts) {
  if (pattern.length !== parts.length) return null;
  const params = {};
  for (let i = 0; i < pattern.length; i++) {
    if (pattern[i].startsWith(":")) params[pattern[i].slice(1)] = decodeURIComponent(parts[i]);
    else if (pattern[i] !== parts[i]) return null;
  }
  return params;
}

async function readBody(req) {
  if (!req.headers["content-type"]?.includes("application/json")) return undefined;
  const chunks = [];
  for await (const chunk of req) chunks.push(Buffer.from(chunk));
  const text = Buffer.concat(chunks).toString("utf8");
  if (text === "") return undefined;
  try {
    return JSON.parse(text);
  } catch {
    throw Object.assign(new Error("JSON לא תקין"), { status: 400 });
  }
}

function enhance(res) {
  res.status = (code) => {
    res.statusCode = code;
    return res;
  };
  res.json = (body) => {
    if (res.statusCode === 204) return res.end();
    res.setHeader("Content-Type", "application/json; charset=utf-8");
    res.end(JSON.stringify(body));
  };
}

export function createApp() {
  const middleware = [];
  const routes = [];

  async function route(req, res) {
    const parts = segments(req.path);
    const allow = [];
    for (const r of routes) {
      const params = matchParts(r.pattern, parts);
      if (!params) continue;
      if (r.method !== req.method) {
        allow.push(r.method);
        continue;
      }
      req.params = params;
      return r.handler(req, res);
    }
    if (allow.length > 0) {
      res.setHeader("Allow", allow.join(", "));
      return res.status(405).json({ error: "Method Not Allowed" });
    }
    res.status(404).json({ error: "לא נמצא" });
  }

  async function handle(req, res) {
    enhance(res);
    try {
      const url = new URL(req.url, "http://localhost");
      req.path = url.pathname;
      req.query = Object.fromEntries(url.searchParams);
      req.params = {};
      req.body = await readBody(req);
      let i = 0;
      const next = async () => {
        const fn = middleware[i++];
        if (fn) await fn(req, res, next);
        else await route(req, res);
      };
      await next();
    } catch (err) {
      if (res.headersSent) return res.end();
      if (err.status) return res.status(err.status).json({ error: err.message });
      console.error(err);
      res.status(500).json({ error: "שגיאה פנימית" });
    }
  }

  const app = {
    use(fn) {
      middleware.push(fn);
    },
    listen(port, cb) {
      return http.createServer(handle).listen(port, cb);
    },
  };
  for (const method of ["get", "post", "put", "patch", "delete"]) {
    app[method] = (pattern, handler) => routes.push({ method: method.toUpperCase(), pattern: segments(pattern), handler });
  }
  return app;
}

// הדגמה — רק כשמריצים את הקובץ ישירות (node P14-E24.mjs), לא כשהבדיקות טוענות אותו
if (fs.realpathSync(process.argv[1]) === import.meta.filename) {
  const app = createApp();
  app.get("/products/:id", (req, res) => res.json({ id: req.params.id, sort: req.query.sort }));
  const server = app.listen(0, async () => {
    const res = await fetch(`http://localhost:${server.address().port}/products/7?sort=price`);
    console.log(res.status, await res.json());
    server.close();
  });
}

  • לא לעשות await ל־handler — שגיאה ב־handler async לא נתפסת, והשרת קורס (כמו ב־E22).

P14-E25 · אתגר: הגבלת קצב (rate limit) 🤖 💻 כתיבת קוד 🤖 רמה 5

💡 רמז 1

Map מלקוח ל־{ start, count }.

💡 רמז 2

חלון נגמר כש־now() - start >= windowMs. Retry-After: Math.ceil((start + windowMs - now()) / 1000).

✅ פתרון
// P14-E25
import http from "node:http";

export function createLimitedServer({ limit, windowMs, now = Date.now }) {
  const windows = new Map();

  return http.createServer((req, res) => {
    const client = req.headers["x-client-id"] ?? "anonymous";
    const t = now();
    let w = windows.get(client);
    if (!w || t - w.start >= windowMs) {
      w = { start: t, count: 0 };
      windows.set(client, w);
    }
    w.count++;

    const headers = { "Content-Type": "application/json; charset=utf-8" };
    if (w.count > limit) {
      headers["Retry-After"] = String(Math.ceil((w.start + windowMs - t) / 1000));
      headers["X-RateLimit-Remaining"] = "0";
      res.writeHead(429, headers);
      return res.end(JSON.stringify({ error: "יותר מדי בקשות" }));
    }
    headers["X-RateLimit-Remaining"] = String(limit - w.count);
    res.writeHead(200, headers);
    res.end(JSON.stringify({ ok: true }));
  });
}

שתי חולשות לחשוב עליהן: (1) ה־Map גדל לנצח — לקוחות ישנים לא נמחקים (פתרון: ניקוי תקופתי, או Redis עם תפוגה — פרק 26); (2) עם שני עותקים של השרת, לכל אחד מונה משלו, והלקוח מקבל פי 2. גם את זה פותר Redis. ומזהה שהלקוח שולח בעצמו אפשר לזייף — בפועל מגבילים לפי IP או לפי משתמש מחובר.

⚠️ טעויות נפוצות
  • Retry-After במילישניות — התקן מגדיר שניות.
