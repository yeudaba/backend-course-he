2. ההזמנה

src/orders/orders.service.ts 
  // עגלה ← הזמנה, בטרנזקציה אחת: או שהכול קורה (מלאי ירד, הזמנה נוצרה, עגלה התרוקנה) — או כלום
  createFromCart(userId: number) {
    return this.prisma.$transaction(async (tx) => {
      // לפי productId: שתי הזמנות עם אותם מוצרים נועלות אותם באותו סדר — בלי deadlock
      const cart = await tx.cartItem.findMany({ where: { userId, product: { active: true } }, orderBy: { productId: "asc" } });
      if (cart.length === 0) throw new HttpError(422, "CART_EMPTY", "העגלה ריקה");

      const shortages: { productId: number; requested: number; available: number }[] = [];
      for (const item of cart) {
        // UPDATE מותנה ואטומי: "הורד, רק אם יש מספיק". שני לקוחות על היחידה האחרונה — PostgreSQL נועל את השורה,
        // השני מחכה, ואחרי ה־COMMIT של הראשון התנאי כבר לא מתקיים: count = 0. בלי SELECT-ואז-UPDATE
        const { count } = await tx.product.updateMany({
          where: { id: item.productId, active: true, stock: { gte: item.quantity } },
          data: { stock: { decrement: item.quantity } },
        });
        if (count === 0) {
          const p = await tx.product.findUnique({ where: { id: item.productId }, select: { stock: true } });
          shortages.push({ productId: item.productId, requested: item.quantity, available: p?.stock ?? 0 });
        }
      }
      // זריקה בתוך $transaction = ROLLBACK: גם מה שכבר ירד מהמלאי — חוזר
      if (shortages.length > 0) throw new HttpError(409, "OUT_OF_STOCK", "חלק מהמוצרים אזלו", shortages);

      // המחיר והשם — עכשיו, אחרי שהשורות נעולות: זה המחיר שהלקוח משלם, גם אם ישתנה מחר
      const products = await tx.product.findMany({ where: { id: { in: cart.map((c) => c.productId) } }, select: { id: true, name: true, priceCents: true } });
      const byId = new Map(products.map((p) => [p.id, p]));
      const items = cart.map((c) => ({ productId: c.productId, name: byId.get(c.productId)!.name, priceCents: byId.get(c.productId)!.priceCents, quantity: c.quantity }));
      const totalCents = items.reduce((sum, i) => sum + i.priceCents * i.quantity, 0);

      const order = await tx.order.create({ data: { userId, totalCents, createdAt: this.clock.now(), items: { create: items } }, select: orderView });
      await tx.cartItem.deleteMany({ where: { userId } });
      return order;
    });
  }
  • ה־findMany של המחירים בא אחרי העדכונים — השורות כבר נעולות על ידי הטרנזקציה, ולכן המחיר שנקרא הוא בדיוק המחיר שבו נמכר.
  • ה־details של OUT_OF_STOCK אוספים את כל החוסרים, לא רק את הראשון — הלקוח מתקן את העגלה פעם אחת.

3. מכונת המצבים

src/orders/order-status.ts 
export type OrderStatus = "pending_payment" | "paid" | "shipped" | "cancelled";
export type OrderAction = "pay" | "ship" | "cancel";

// מכונת המצבים של הזמנה — במקום אחד, בלי מסד, ונבדקת ב־unit tests
const TRANSITIONS: Record<OrderAction, Partial<Record<OrderStatus, OrderStatus>>> = {
  pay: { pending_payment: "paid" },
  ship: { paid: "shipped" },
  cancel: { pending_payment: "cancelled", paid: "cancelled" }, // נשלחה — כבר אי אפשר לבטל
};

export function nextStatus(current: OrderStatus, action: OrderAction): OrderStatus | null {
  return TRANSITIONS[action][current] ?? null;
}

export function fromStatuses(action: OrderAction): OrderStatus[] {
  return Object.keys(TRANSITIONS[action]) as OrderStatus[];
}

וב־service, כל מעבר הוא updateMany עם status: { in: fromStatuses(action) } — הטבלה והמסד מסכימים, ומקביליות מטופלת במסד.

4. החלטות שכדאי לשים לב אליהן

  • תשלום מחוץ לטרנזקציה. קריאה לספק תשלום (רשת, שניות) בתוך טרנזקציה מחזיקה נעילות כל הזמן הזה. לכן: חיוב ← ואז טרנזקציה קצרה עם מעבר מותנה. ואם ההזמנה בוטלה בינתיים (count = 0) — refund מיד.
  • Idempotency שומר גם 4xx. OUT_OF_STOCK עם אותו מפתח יחזור כמו שהוא — "אותה בקשה, אותה תשובה". רוצים לנסות שוב אחרי שהמלאי חזר? מפתח חדש. 5xx — לא נשמר.
  • הטביעה של תשלום — 4 ספרות אחרונות, לא כל הכרטיס (גם לא ב־hash: יש רק 1016 מספרים, ו־hash שלהם אפשר לנחש).
  • /health ו־/ready לא בתיעוד (@ApiExcludeController) — הם לפלטפורמה, לא ללקוחות של ה־API.
  • הבדיקות של הפרויקט מחליפות רק את PrismaService ואת CONFIG — כל השאר אמיתי, כולל ה־guard, ה־pipes וה־filter.
