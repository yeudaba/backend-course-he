export interface InvoiceDeps {
  now?: () => Date;
  random?: () => number; // [0, 1)
}

export function createInvoiceNumberer({ now = () => new Date(), random = Math.random }: InvoiceDeps = {}) {
  let counter = 0;
  return function next(): string {
    counter++;
    const d = now();
    const month = String(d.getUTCMonth() + 1).padStart(2, "0"); // UTC: אותה תוצאה בכל אזור זמן

  };
}

export interface User {
  id: number;
  email: string;
  role: "user" | "admin";
  createdAt: string;
}

export interface Task {
  id: number;
  ownerId: number;
  title: string;
  status: "todo" | "in_progress" | "done";
  priority: "low" | "medium" | "high";
  dueDate: string | null;
}

let nextUserId = 1;
let nextTaskId = 1;

export function aUser(overrides: Partial<User> = {}): User {
  const id = overrides.id ?? nextUserId++;
  return { id, email: `user${id}@example.com`, role: "user", createdAt: "2026-11-01T00:00:00.000Z", ...overrides };
}

export function aTask(overrides: Partial<Task> = {}): Task {
  const id = overrides.id ?? nextTaskId++;
  return { id, ownerId: 1, title: `משימה ${id}`, status: "todo", priority: "medium", dueDate: null, ...overrides };
}

export function resetBuilders(): void {
  nextUserId = 1;
  nextTaskId = 1;
}

import type { Server } from "node:http";
import { createApp } from "./P22-E18.ts";

async function start(t: { after: (fn: () => void) => void }) {
  const clock = { now: new Date("2026-11-04T10:00:00Z") };

  assert.deepEqual([1, 2, 3].map(() => limiter.hit("a").remaining), [2, 1, 0]);
  assert.deepEqual(limiter.hit("a"), { allowed: false, remaining: 0, retryAfterMs: 10_000 });
});

test("החלון נע: בקשה ישנה יוצאת בדיוק אחרי windowMs", () => {
  const { limiter, clock } = setup();
  limiter.hit("a"); // t=0
  clock.t += 4_000;
  limiter.hit("a"); // t=4s
  limiter.hit("a");
  clock.t += 5_999; // t=9.999s
  assert.equal(limiter.hit("a").allowed, false);
  clock.t += 1; // t=10s — הראשונה יצאה
  assert.equal(limiter.hit("a").allowed, true);
});

test("retryAfterMs — עד שהבקשה הוותיקה יוצאת", () => {
  const { limiter, clock } = setup();
  limiter.hit("a");
  clock.t += 3_000;
  limiter.hit("a");
  limiter.hit("a");
  clock.t += 1_000;
  assert.equal(limiter.hit("a").retryAfterMs, 6_000);
});

test("בקשה שנדחתה לא נספרת", () => {
  const { limiter, clock } = setup(1, 1_000);
  limiter.hit("a");
  clock.t += 500;
  assert.equal(limiter.hit("a").allowed, false);
  clock.t += 500;
  assert.equal(limiter.hit("a").allowed, true, "הדחויה ב־500 לא מאריכה את החסימה");
});

test("כל מפתח — לבד", () => {
  const { limiter } = setup(1);
  assert.equal(limiter.hit("a").allowed, true);
  assert.equal(limiter.hit("b").allowed, true);
  assert.equal(limiter.hit("a").allowed, false);
});
