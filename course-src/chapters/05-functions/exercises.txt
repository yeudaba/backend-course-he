```js run save=solutions/P05-E12.js
// P05-E12: פונקציה שמקבלת פונקציה
function applyTwice(fn, value) {
  return fn(fn(value));
}

function applyN(fn, value, times) {
  let result = value;
  for (let i = 0; i < times; i++) {
    result = fn(result);
  }
  return result;
}

console.log(applyTwice((x) => x + 3, 10));
console.log(applyN((s) => s + "!", "היי", 3));

module.exports = { applyTwice, applyN };
```

```js run save=solutions/P05-E13.js
// P05-E13: בדיקת אימייל עם guard clauses
function validateEmail(email) {
  const clean = email.trim();
  if (clean === "") {
    return "חובה להזין אימייל";
  }
  if (clean.includes(" ")) {
    return "אימייל לא יכול להכיל רווחים";
  }
  const at = clean.indexOf("@");
  if (at <= 0 || at === clean.length - 1) {
    return "חסר @ במקום הנכון";
  }
  if (!clean.slice(at + 1).includes(".")) {
    return "חסר דומיין תקין";
  }
  return null;
}

console.log(validateEmail("  dana@mail.com "));
console.log(validateEmail("dana@mail"));

module.exports = { validateEmail };
```

```out
null
חסר דומיין תקין
```

```js run save=solutions/P05-E16.js
// P05-E16: ספירת תווים
function countMatching(text, predicate) {
  let count = 0;
  for (const ch of text) {
    if (predicate(ch)) {
      count++;
    }
  }
  return count;
}

function countChar(text, ch) {
  return countMatching(text, (c) => c === ch);
}

function countVowels(text) {
  return countMatching(text.toLowerCase(), (ch) => "aeiou".includes(ch));
}
```

```js run save=solutions/P05-E17.js
// P05-E17: slug וקיצור טקסט
function toSlug(title) {
  const clean = title.trim().toLowerCase();
  let slug = "";
  for (const ch of clean) {
    if (ch === " ") {
      if (slug[slug.length - 1] !== "-") {
        slug += "-";
      }
    } else {
      slug += ch;
    }
  }
  return slug;
}

function truncate(text, maxLength) {
  if (text.length <= maxLength) {
    return text;
  }
  return text.slice(0, maxLength - 3) + "...";
}
```

```js run save=solutions/P05-E18.js
// P05-E18: ציון חוזק סיסמה
function hasMatching(text, predicate) {
  for (const ch of text) {
    if (predicate(ch)) {
      return true;
    }
  }
  return false;
}

const hasDigit = (text) => hasMatching(text, (ch) => "0123456789".includes(ch));
const hasUpper = (text) => hasMatching(text, (ch) => ch !== ch.toLowerCase());
const hasSpecial = (text) => hasMatching(text, (ch) => "!@#$%".includes(ch));

function passwordScore(password) {
  let score = 0;
  if (password.length >= 8) score++;
  if (hasDigit(password)) score++;
  if (hasUpper(password)) score++;
  if (hasSpecial(password)) score++;
  return score;
}

function passwordLabel(password) {
  const score = passwordScore(password);
  if (score <= 1) {
    return "חלשה";
  }
  if (score <= 3) {
    return "בינונית";
  }
  return "חזקה";
}

console.log(passwordScore("abc"), passwordLabel("Secret123"), passwordLabel("Secret123!"));

module.exports = { hasDigit, hasUpper, hasSpecial, passwordScore, passwordLabel };
```

```js run save=solutions/P05-E19.js
const AMOUNT_WIDTH = 8;

function formatLine(label, amount, width = 28) {
  return label.toUpperCase().padEnd(width - AMOUNT_WIDTH, ".") + amount.toFixed(2).padStart(AMOUNT_WIDTH);
}

function calcTotals(subtotal, couponPercent, vatRate) {
  const afterDiscount = subtotal * (1 - couponPercent / 100);
  const total = afterDiscount * (1 + vatRate);
  return Math.round(total * 100) / 100;
}
```

```js run save=solutions/P05-E22.js
// P05-E22: העברת פונקציות כארגומנט — מתוקן
function transformAll(from, to, fn) {
  let result = "";
  for (let i = from; i <= to; i++) {
    result += fn(i) + " ";
  }
  return result.trim();
}

const double = (n) => n * 2;
const label = (n) => `#${n}`;

const doubles = transformAll(1, 3, double);
const labels = transformAll(1, 3, label);
const squares = transformAll(1, 3, (n) => n * n);

console.log(squares);

module.exports = { transformAll, doubles, labels, squares };
```

```js run save=solutions/P05-E23.js
// P05-E23: קבוצת גיל — guard clauses בסדר הנכון
function ageGroup(input, minAge = 0) {
  if (input.trim() === "") {
    return "שגיאה: חובה להזין גיל";
  }
  const age = Number(input);
  if (Number.isNaN(age)) {
    return "שגיאה: גיל חייב להיות מספר";
  }
  if (age < minAge || age > 120) {
    return "שגיאה: גיל לא הגיוני";
  }
  if (age < 18) {
    return "קבוצה: קטין";
  }
  if (age < 65) {
    return "קבוצה: מבוגר";
  }
  return "קבוצה: גמלאי";
}

module.exports = { ageGroup };
```

```js run save=solutions/P05-E24.js
function getUnitPrice(code) {
  switch (code.trim().toUpperCase()) {
    case "KB-101":
      return 199;
    case "KB-102":
      return 249;
    case "MS-201":
      return 89;
    default:
      return null;
  }
}

function validateQuantity(input, max = 5) {
  const quantity = Number(input);
  if (input.trim() === "" || Number.isNaN(quantity)) {
    return "כמות חייבת להיות מספר";
  }
  if (!Number.isInteger(quantity) || quantity < 1 || quantity > max) {
    return `כמות חייבת להיות מספר שלם בין 1 ל־${max}`;
  }
  return null;
}

function couponDiscount(coupon, subtotal) {
  const code = coupon.trim().toUpperCase();
  if (code === "") {
    return 0;
  }
  if (code === "VIP20") {
    return subtotal > 500 ? subtotal * 0.2 : null;
  }
  if (code === "WELCOME") {
    return Math.min(30, subtotal);
  }
  return null;
}

function checkout(code, quantityInput, coupon, isLoggedIn) {
  if (!isLoggedIn) {
    return "שגיאה: יש להתחבר כדי להזמין";
  }
  const unitPrice = getUnitPrice(code);
  if (unitPrice === null) {
    return "שגיאה: מוצר לא קיים";
  }
  const qtyError = validateQuantity(quantityInput);
  if (qtyError !== null) {
    return `שגיאה: ${qtyError}`;
  }
  const subtotal = unitPrice * Number(quantityInput);
  const discount = couponDiscount(coupon, subtotal);
  if (discount === null) {
    return "שגיאה: הקופון לא קיים או לא תקף להזמנה הזו";
  }
  return `לתשלום: ${(subtotal - discount).toFixed(2)}`;
}

console.log(checkout("KB-101", "2", "VIP20", true));
```

```js run save=solutions/P05-E25.js
// P05-E25: עיצוב מספר עם פסיקים
function addCommas(digits) {
  let result = "";
  let count = 0;
  for (let i = digits.length - 1; i >= 0; i--) {
    if (count > 0 && count % 3 === 0) {
      result = "," + result;
    }
    result = digits[i] + result;
    count++;
  }
  return result;
}

function formatNumber(n, decimals = 2) {
  const sign = n < 0 ? "-" : "";
  const fixed = Math.abs(n).toFixed(decimals);
  const dot = fixed.indexOf(".");
  const intPart = dot === -1 ? fixed : fixed.slice(0, dot);
  const decPart = dot === -1 ? "" : fixed.slice(dot);
  return sign + addCommas(intPart) + decPart;
}

console.log(formatNumber(1234567.891));
console.log(formatNumber(-1234.5, 1));
console.log(formatNumber(1000000, 0));

module.exports = { formatNumber, addCommas };
```

```js run save=solutions/P05-E28.js
function isPrime(n) {
  if (!Number.isInteger(n) || n < 2) return false;
  for (let d = 2; d * d <= n; d++) {
    if (n % d === 0) return false;

}

}
```
