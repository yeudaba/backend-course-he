// P26-E06
import type { Redis } from "ioredis";

// מטמון JSON פשוט מעל Redis: מחרוזות בלבד שם — לכן JSON.stringify / parse
export function createJsonStore(redis: Redis) {
  return {
    async set(key: string, value: unknown, ttlSeconds?: number) {
      if (value === undefined) throw new Error("אין ערך לשמור");
      const json = JSON.stringify(value);
      if (ttlSeconds) await redis.set(key, json, "EX", ttlSeconds);
      else await redis.set(key, json);
    },
    async get<T>(key: string): Promise<T | null> {
      const raw = await redis.get(key);
      return raw === null ? null : (JSON.parse(raw) as T);
    },
    async del(key: string) {
      return (await redis.del(key)) === 1;
    },
    // כמה שניות נשארו: null — אין מפתח; Infinity — בלי תוקף
    async ttl(key: string) {
      const t = await redis.ttl(key);
      return t === -2 ? null : t === -1 ? Infinity : t;
    },
  };
}

@hint1
<p>"cache-aside" = הקוד שלך מנהל את ה־cache: בודק, טוען, שומר. Redis לא יודע כלום על המסד.</p>
@hint2
<p>למה לא לשמור <code>null</code>? "המוצר לא קיים" יישאר ל־TTL שלם גם אחרי שייווצר. (רשימה ריקה <code>[]</code> — כן שומרים: זו תשובה אמיתית, ראה E20.)</p>
@solution
```ts save=solutions/P26-E07.ts
// P26-E07
import type { Redis } from "ioredis";

// cache-aside: קודם Redis; אין ← טוענים מהמקור ושומרים עם TTL
export function createCacheAside(redis: Redis) {
  const stats = { hits: 0, misses: 0 };
  async function get<T>(key: string, ttlSeconds: number, load: () => Promise<T>): Promise<T> {
    const raw = await redis.get(key);
    if (raw !== null) {
      stats.hits++;
      return JSON.parse(raw) as T;
    }
    stats.misses++;
    const value = await load(); // אם load זורק — לא שומרים כלום (לא מחזיקים שגיאות ב־cache)
    if (value !== undefined && value !== null) await redis.set(key, JSON.stringify(value), "EX", ttlSeconds);
    return value;
  }
  return { get, stats, invalidate: (key: string) => redis.del(key) };
}
```

// fixed window ב־Redis: מונה לכל (לקוח, חלון). משותף לכל השרתים — בניגוד ל־Map בזיכרון (פרק 24)
export async function hit(redis: Redis, clientId: string, { limit, windowSeconds, now = Date.now() }: { limit: number; windowSeconds: number; now?: number }) {
  const window = Math.floor(now / 1000 / windowSeconds);
  const key = `rl:${clientId}:${window}`;
  // INCR אטומי; EXPIRE רק כשהמפתח נוצר עכשיו (NX) — המונה מת לבד בסוף החלון

import type { Redis } from "ioredis";

const RELEASE = `if redis.call("GET", KEYS[1]) == ARGV[1] then return redis.call("DEL", KEYS[1]) else return 0 end`;

export async function acquire(redis: Redis, name: string, ttlMs: number): Promise<string | null> {
  const token = randomUUID();
  // SET NX PX: "רק אם לא קיים" ועם תוקף — באותה פקודה. אם התהליך מת, המנעול משתחרר לבד

  return ok === "OK" ? token : null;

  return (await redis.eval(RELEASE, 1, `lock:${name}`, token)) === 1;
}

  if (!token) return null; // מישהו אחר מחזיק — לא מחכים (לפי הצורך: לנסות שוב אחר כך)
  try {
    return await fn();
  } finally {
    await release(redis, name, token);
  }

// P26-E13
import { randomUUID } from "node:crypto";
import type { Redis } from "ioredis";

const RELEASE = `if redis.call("GET", KEYS[1]) == ARGV[1] then return redis.call("DEL", KEYS[1]) else return 0 end`;
const sleep = (ms: number) => new Promise((r) => setTimeout(r, ms));

// מניעת stampede: מפתח פופולרי פג תוקף, ו־500 בקשות טוענות אותו מהמסד בבת אחת.
// בתוך תהליך — "single flight" (Promise משותף). בין שרתים — מנעול: רק אחד טוען, האחרים מחכים לערך
export function createStampedeCache(redis: Redis, { lockMs = 5000, waitMs = 3000, pollMs = 25 } = {}) {
  const inFlight = new Map<string, Promise<unknown>>();

  async function load<T>(key: string, ttlSeconds: number, loader: () => Promise<T>): Promise<T> {
    const cached = await redis.get(key);
    if (cached !== null) return JSON.parse(cached) as T;
    const token = randomUUID();
    if ((await redis.set(`${key}:lock`, token, "PX", lockMs, "NX")) === "OK") {
      try {
        const value = await loader();
        await redis.set(key, JSON.stringify(value), "EX", ttlSeconds);
        return value;
      } finally {
        await redis.eval(RELEASE, 1, `${key}:lock`, token);
      }
    }
    // מישהו אחר טוען — מחכים שהערך יופיע. לא הופיע בזמן? טוענים בעצמנו (עדיף איטי מתקוע)
    for (const end = Date.now() + waitMs; Date.now() < end; ) {
      await sleep(pollMs);
      const v = await redis.get(key);
      if (v !== null) return JSON.parse(v) as T;
    }
    return loader();
  }

  return {
    get<T>(key: string, ttlSeconds: number, loader: () => Promise<T>): Promise<T> {
      const existing = inFlight.get(key);
      if (existing) return existing as Promise<T>;
      const p = load(key, ttlSeconds, loader).finally(() => inFlight.delete(key));
      inFlight.set(key, p);
      return p;
    },
  };
}

// sliding window ב־sorted set: כל בקשה = איבר עם score = זמן. סופרים רק את ה־windowMs האחרונות
export async function slidingHit(redis: Redis, clientId: string, { limit, windowMs, now = Date.now() }: { limit: number; windowMs: number; now?: number }) {
  const key = `sw:${clientId}`;
  const results = await redis
    .multi()
    .zremrangebyscore(key, "-inf", now - windowMs) // מנקים את מה שיצא מהחלון
    .zcard(key)
    .zrange(key, 0, 0, "WITHSCORES")

  const oldest = results![2][1] as string[];
  if (count >= limit) {

  return { allowed: true, remaining: limit - count - 1, retryAfterSeconds: 0 };
}

import { Queue, Worker, type ConnectionOptions } from "bullmq";

// תזכורת מתוזמנת: עבודה עם delay. jobId לפי ההזמנה — כדי שאפשר יהיה לבטל אותה כשהלקוח שילם
export function createReminders(connection: ConnectionOptions, prefix: string) {
  const queue = new Queue<{ orderId: number }>("reminders", { connection, prefix });
  return {
    queue,
    schedule: (orderId: number, delayMs: number) => queue.add("remind", { orderId }, { jobId: `remind-${orderId}`, delay: delayMs }),
    // שולם לפני הזמן — מבטלים. מחזיר true אם הייתה עבודה ממתינה שבוטלה
    async cancel(orderId: number) {
      const job = await queue.getJob(`remind-${orderId}`);
      if (!job || !(await job.isDelayed())) return false;
      await job.remove();
      return true;
    },
    close: () => queue.close(),
  };
}

export function startReminderWorker(connection: ConnectionOptions, prefix: string, remind: (orderId: number) => Promise<void>) {
  return new Worker<{ orderId: number }>("reminders", (job) => remind(job.data.orderId), { connection, prefix });
}

import type { Redis } from "ioredis";

// "צרכן idempotent": תורים מבטיחים "לפחות פעם אחת" — לא "בדיוק פעם אחת". retry, worker שנפל אחרי העבודה ולפני ack…
// לכן כל הודעה מסומנת כ"טופלה". SET NX: רק הראשון מצליח לסמן. נכשל באמצע? מוחקים את הסימון — שינסו שוב
export async function processOnce<T>(redis: Redis, messageId: string, ttlSeconds: number, handler: () => Promise<T>): Promise<{ processed: boolean; result?: T }> {
  const key = `processed:${messageId}`;
  if ((await redis.set(key, "1", "EX", ttlSeconds, "NX")) !== "OK") return { processed: false };
  try {
    return { processed: true, result: await handler() };
  } catch (err) {
    await redis.del(key);
    throw err;
  }
}

// cache בזיכרון של התהליך (L1, הכי מהיר) — אבל עם כמה שרתים, כל אחד מחזיק עותק משלו.
// כששרת אחד משנה, הוא מפרסם ב־pub/sub, וכל השרתים מוחקים את העותק שלהם
export async function createL1Cache(redis: Redis, subscriber: Redis, channel = "cache-invalidate") {
  const local = new Map<string, unknown>();
  await subscriber.subscribe(channel);
  subscriber.on("message", (ch: string, key: string) => {
    if (ch === channel) local.delete(key);
  });
  return {
    async get<T>(key: string, load: () => Promise<T>): Promise<T> {
      if (local.has(key)) return local.get(key) as T;
      const value = await load();
      local.set(key, value);
      return value;
    },
    // מקומית מיד, ולכל השאר דרך Redis
    async invalidate(key: string) {
      local.delete(key);
      await redis.publish(channel, key);
    },
    size: () => local.size,
  };
}

// P26-E24
import type { Redis } from "ioredis";

const minuteOf = (now: number) => Math.floor(now / 60_000);

// "חם עכשיו": sorted set לכל דקה. צפייה ← ZINCRBY בדלי של הדקה הנוכחית (שמת לבד אחרי שעתיים)
export async function recordView(redis: Redis, productId: number, now = Date.now()) {
  const key = `trend:${minuteOf(now)}`;
  await redis.multi().zincrby(key, 1, String(productId)).expire(key, 7200).exec();
}

// N הדקות האחרונות: ZUNIONSTORE מחבר את הדליים (סכום ה־scores) למפתח זמני, ומשם — ה־top
export async function trending(redis: Redis, { windowMinutes, n, now = Date.now() }: { windowMinutes: number; n: number; now?: number }) {
  const m = minuteOf(now);
  const buckets = Array.from({ length: windowMinutes }, (_, i) => `trend:${m - i}`);
  const tmp = `trend:tmp:${m}:${windowMinutes}:${Math.random().toString(36).slice(2)}`;
  const results = await redis
    .multi()
    .zunionstore(tmp, buckets.length, ...buckets)
    .zrevrange(tmp, 0, n - 1, "WITHSCORES")
    .del(tmp)
    .exec();
  const flat = results![1][1] as string[];
  return Array.from({ length: flat.length / 2 }, (_, i) => ({ productId: Number(flat[2 * i]), views: Number(flat[2 * i + 1]) }));
}

// token bucket אטומי: הקריאה, החישוב והכתיבה — בסקריפט Lua אחד. Redis מריץ סקריפט בבת אחת,
// בלי שפקודה אחרת תיכנס באמצע: אין מרוץ בין שרתים, ואין round-trip נוסף
const TAKE = `
local capacity = tonumber(ARGV[1])
local rate = tonumber(ARGV[2])      -- אסימונים לשנייה
local now = tonumber(ARGV[3])       -- מילישניות
local cost = tonumber(ARGV[4])
local state = redis.call("HMGET", KEYS[1], "tokens", "ts")
local tokens = tonumber(state[1]) or capacity
local ts = tonumber(state[2]) or now
tokens = math.min(capacity, tokens + math.max(0, now - ts) / 1000 * rate)
local allowed = 0
if tokens >= cost then
  tokens = tokens - cost
  allowed = 1
end
redis.call("HSET", KEYS[1], "tokens", tostring(tokens), "ts", tostring(now))
redis.call("PEXPIRE", KEYS[1], math.ceil(capacity / rate * 1000) + 1000)
local retry = 0
if allowed == 0 then retry = math.ceil((cost - tokens) / rate) end
return { allowed, math.floor(tokens), retry }
`;

export async function take(redis: Redis, key: string, { capacity, refillPerSecond, now = Date.now(), cost = 1 }: { capacity: number; refillPerSecond: number; now?: number; cost?: number }) {
  const [allowed, remaining, retryAfterSeconds] = (await redis.eval(TAKE, 1, `tb:${key}`, capacity, refillPerSecond, now, cost)) as number[];
  return { allowed: allowed === 1, remaining, retryAfterSeconds };
}

// P26-E28
// "רכזת" של SSE: כל לקוח מחובר מקבל כל אירוע, עם id עולה — ומי שהתנתק וחזר מקבל את מה שפספס
import type { Request, Response } from "express";

const format = (id: number, event: string, data: unknown) => `id: ${id}\nevent: ${event}\ndata: ${JSON.stringify(data)}\n\n`;

export function eventHub({ bufferSize = 100 }: { bufferSize?: number } = {}) {
  const clients = new Set<Response>();
  const buffer: { id: number; text: string }[] = []; // האירועים האחרונים — בשביל Last-Event-ID
  let lastId = 0;

  function handler(req: Request, res: Response) {
    res.writeHead(200, {
      "Content-Type": "text/event-stream",
      "Cache-Control": "no-cache",
      Connection: "keep-alive",
      "X-Accel-Buffering": "no", // שלא יחזיק את ההודעות ב־proxy (nginx)
    });
    res.flushHeaders();
    // הדפדפן שולח Last-Event-ID לבד כשהוא מתחבר מחדש
    const since = Number(req.get("last-event-id"));
    if (Number.isInteger(since)) for (const e of buffer) if (e.id > since) res.write(e.text);
    clients.add(res);
    req.on("close", () => clients.delete(res)); // אחרת: דליפה, וכתיבה לחיבור סגור
  }

  function publish(event: string, data: unknown) {
    const text = format(++lastId, event, data);
    buffer.push({ id: lastId, text });
    if (buffer.length > bufferSize) buffer.shift();
    for (const res of clients) res.write(text);
    return lastId;
  }

  return { handler, publish, size: () => clients.size };
}
```

// P26-E29
// חדרי צ'אט ב־WebSocket: אימות בזמן ה־upgrade, הצטרפות לחדר, ושידור לחברי החדר בלבד
import type { Server } from "node:http";
import { WebSocketServer, WebSocket } from "ws";

export function createChatServer(server: Server, { verifyToken }: { verifyToken: (token: string) => string | null }) {
  const wss = new WebSocketServer({ noServer: true });
  const rooms = new Map<string, Set<WebSocket>>();
  const send = (ws: WebSocket, msg: object) => ws.readyState === WebSocket.OPEN && ws.send(JSON.stringify(msg));

  // האימות קורה לפני שהחיבור נפתח: בלי token תקין — 401 ו־socket סגור. אחרי upgrade כבר אין status codes
  server.on("upgrade", (req, socket, head) => {
    const url = new URL(req.url ?? "/", "http://x");
    const user = url.pathname === "/chat" ? verifyToken(url.searchParams.get("token") ?? "") : null;
    if (!user) {
      socket.end("HTTP/1.1 401 Unauthorized\r\nConnection: close\r\n\r\n");
      return;
    }
    wss.handleUpgrade(req, socket, head, (ws) => {
      let room: string | null = null;
      const leave = () => { if (room) rooms.get(room)?.delete(ws); room = null; };
      ws.on("message", (raw) => {
        let msg: { type?: string; room?: unknown; text?: unknown };
        try { msg = JSON.parse(String(raw)); } catch { return send(ws, { type: "error", error: "invalid json" }); }
        if (msg.type === "join" && typeof msg.room === "string" && msg.room) {
          leave();
          room = msg.room;
          if (!rooms.has(room)) rooms.set(room, new Set());
          rooms.get(room)!.add(ws);
          return send(ws, { type: "joined", room });
        }
        if (msg.type === "say" && typeof msg.text === "string") {
          if (!room) return send(ws, { type: "error", error: "join a room first" });
          // השולח (user) נקבע בשרת לפי ה־token — לא לפי מה שהלקוח כותב בהודעה
          for (const peer of rooms.get(room) ?? []) send(peer, { type: "message", room, from: user, text: msg.text });
          return;
        }
        send(ws, { type: "error", error: "unknown message" });
      });
      ws.on("close", leave);
    });
  });

  return { close: () => new Promise<void>((resolve) => { for (const ws of wss.clients) ws.terminate(); wss.close(() => resolve()); }) };
}
```

// כמה שרתים, לקוחות מחוברים לכל אחד: אירוע שקרה בשרת אחד צריך להגיע ללקוחות של כולם — דרך Redis pub/sub
import type { Redis } from "ioredis";

export async function createRelay({ pub, sub, channel }: { pub: Redis; sub: Redis; channel: string }) {
  const listeners = new Set<(data: unknown) => void>();
  // מנוי אחד לשרת, לא אחד לכל לקוח. החיבור sub נפרד: חיבור במצב SUBSCRIBE לא מריץ פקודות אחרות
  await sub.subscribe(channel);
  const onMessage = (ch: string, message: string) => {
    if (ch !== channel) return; // אותו חיבור יכול להיות מנוי לערוצים אחרים
    const data = JSON.parse(message);
    for (const l of listeners) l(data);
  };
  sub.on("message", onMessage);

  return {
    // לא שולחים ישירות למאזינים המקומיים: ההודעה תחזור מ־Redis גם לשרת הזה — אחרת כל אחד יקבל פעמיים
    publish: async (data: unknown) => { await pub.publish(channel, JSON.stringify(data)); },
    subscribe(listener: (data: unknown) => void) {
      listeners.add(listener);
      return () => listeners.delete(listener);
    },
    async close() {
      sub.off("message", onMessage);
      await sub.unsubscribe(channel);
    },
  };
}

import { randomUUID } from "node:crypto";
import type { Redis } from "ioredis";

export type Session = { userId: number; data: Record<string, string> };

const sessionKey = (id: string) => `session:${id}`;
const userKey = (userId: number) => `user:${userId}:sessions`;

export async function createSession(redis: Redis, userId: number, data: Record<string, string> = {}, ttlSec = 1800): Promise<string> {
  const id = randomUUID();
  await redis
    .multi()
    .hset(sessionKey(id), { ...data, userId: String(userId) })
    .expire(sessionKey(id), ttlSec)
    .sadd(userKey(userId), id)
    // הרשימה חיה כמו ה־session הארוך ביותר שבה: NX — אם אין לה עדיין תוקף, GT — רק להאריך
    .expire(userKey(userId), ttlSec, "NX")
    .expire(userKey(userId), ttlSec, "GT")
    .exec();
  return id;
}

export async function getSession(redis: Redis, id: string, ttlSec = 1800): Promise<Session | null> {
  const hash = await redis.hgetall(sessionKey(id));
  if (Object.keys(hash).length === 0) return null;
  const { userId, ...data } = hash;
  await redis.multi().expire(sessionKey(id), ttlSec).expire(userKey(Number(userId)), ttlSec, "GT").exec();
  return { userId: Number(userId), data };
}

export async function destroySession(redis: Redis, id: string): Promise<boolean> {
  const userId = await redis.hget(sessionKey(id), "userId");
  if (userId === null) return false;
  await redis.multi().del(sessionKey(id)).srem(userKey(Number(userId)), id).exec();
  return true;
}

export async function destroyAllForUser(redis: Redis, userId: number): Promise<number> {
  const ids = await redis.smembers(userKey(userId));
  const deleted = ids.length > 0 ? await redis.del(...ids.map(sessionKey)) : 0;
  await redis.del(userKey(userId));
  return deleted;
}

// P26-E32 — לקרוא את INFO של Redis
export type Info = Record<string, Record<string, string>>;

export function parseInfo(text: string): Info {
  const info: Info = {};
  let section: Record<string, string> | undefined;
  for (const raw of text.split(/\r?\n/)) {
    const line = raw.trim();
    if (line === "") continue;
    if (line.startsWith("#")) {
      section = {};
      info[line.slice(1).trim().toLowerCase()] = section;
      continue;
    }
    const colon = line.indexOf(":");
    if (colon === -1 || section === undefined) continue;
    section[line.slice(0, colon)] = line.slice(colon + 1);
  }
  return info;
}

export function parseKeyspace(info: Info): Record<string, Record<string, number>> {
  const result: Record<string, Record<string, number>> = {};
  for (const [db, value] of Object.entries(info.keyspace ?? {})) {
    result[db] = Object.fromEntries(
      value.split(",").map((pair) => {
        const [key, n] = pair.split("=");
        return [key, Number(n)];
      }),
    );
  }
  return result;
}

export function hitRatio(info: Info): number | null {
  const hits = Number(info.stats?.keyspace_hits ?? 0);
  const misses = Number(info.stats?.keyspace_misses ?? 0);
  if (hits + misses === 0) return null;
  return Math.round((hits / (hits + misses)) * 1000) / 1000;
}

// P26-E33 — TypedEmitter: אירועים עם טיפוסים
type Listener<T> = (payload: T) => void;

export class TypedEmitter<Events extends Record<string, unknown>> {
  #listeners = new Map<keyof Events, Set<Listener<any>>>();

  on<K extends keyof Events>(event: K, listener: Listener<Events[K]>): () => void {
    let set = this.#listeners.get(event);
    if (!set) {
      set = new Set();
      this.#listeners.set(event, set);
    }
    set.add(listener);
    return () => this.off(event, listener);
  }

  off<K extends keyof Events>(event: K, listener: Listener<Events[K]>): void {
    const set = this.#listeners.get(event);
    if (!set) return;
    set.delete(listener);
    if (set.size === 0) this.#listeners.delete(event);
  }

  once<K extends keyof Events>(event: K, listener: Listener<Events[K]>): () => void {
    const wrapper: Listener<Events[K]> = (payload) => {
      this.off(event, wrapper);
      listener(payload);
    };
    return this.on(event, wrapper);
  }

  emit<K extends keyof Events>(event: K, payload: Events[K]): number {
    const set = this.#listeners.get(event);
    if (!set) return 0;
    const listeners = [...set];
    for (const listener of listeners) listener(payload);
    return listeners.length;
  }

  listenerCount(event: keyof Events): number {
    return this.#listeners.get(event)?.size ?? 0;
  }
}

type ShopEvents = {
  "order.paid": { orderId: number; totalCents: number };
  "stock.low": { productId: number; left: number };
};
