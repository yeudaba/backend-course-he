```ts save=solutions/P26.5-E06.ts
// P26.5-E06: topic exchange — מי מקבל איזה אירוע
import type { Channel } from "amqplib";

export interface Topology {
  exchange: string;
  emailQueue: string;
  analyticsQueue: string;
}

export interface OrderEvent {
  id: string;
  type: "created" | "paid" | "shipped" | "cancelled";
  orderId: number;
  country: string;
}

export async function setupTopology(ch: Channel, t: Topology): Promise<void> {
  await ch.assertExchange(t.exchange, "topic", { durable: true });
  await ch.assertQueue(t.emailQueue, { durable: true });
  await ch.assertQueue(t.analyticsQueue, { durable: true });
  // routing key: order.<type>.<country>  —  * = מילה אחת, # = אפס מילים או יותר
  await ch.bindQueue(t.emailQueue, t.exchange, "order.created.*");
  await ch.bindQueue(t.emailQueue, t.exchange, "order.shipped.*");
  await ch.bindQueue(t.analyticsQueue, t.exchange, "order.#");
}

export const routingKey = (e: OrderEvent) => `order.${e.type}.${e.country.toLowerCase()}`;

export function publishOrderEvent(ch: Channel, exchange: string, e: OrderEvent): boolean {
  return ch.publish(exchange, routingKey(e), Buffer.from(JSON.stringify(e)), {
    persistent: true, // נשמר בדיסק — שורד restart של RabbitMQ (בתור durable)
    contentType: "application/json",
    messageId: e.id,
  });
}
```

```ts save=solutions/P26.5-E07.ts
// P26.5-E07: worker עם prefetch, ack אחרי הצלחה, ו־dead letter לכישלונות
import type { Channel, ConsumeMessage } from "amqplib";

export interface WorkQueue {
  queue: string;
  deadLetterExchange: string;
  deadLetterQueue: string;
}

export async function setupWorkQueue(ch: Channel, w: WorkQueue): Promise<void> {
  await ch.assertExchange(w.deadLetterExchange, "fanout", { durable: true });
  await ch.assertQueue(w.deadLetterQueue, { durable: true });
  await ch.bindQueue(w.deadLetterQueue, w.deadLetterExchange, "");
  await ch.assertQueue(w.queue, { durable: true, deadLetterExchange: w.deadLetterExchange });
}

export async function startWorker(
  ch: Channel,
  queue: string,
  handler: (job: unknown) => Promise<void>,
  { prefetch = 5 } = {},
): Promise<string> {
  await ch.prefetch(prefetch); // לכל היותר prefetch הודעות שלא אושרו — בזמן נתון
  const { consumerTag } = await ch.consume(queue, async (msg: ConsumeMessage | null) => {
    if (!msg) return; // התור נמחק / ה־consumer בוטל
    try {
      const job = JSON.parse(msg.content.toString());
      await handler(job);
      ch.ack(msg); // רק אחרי שהעבודה הצליחה
    } catch {
      ch.nack(msg, false, false); // requeue=false ← ל־dead letter exchange, לא חזרה לראש התור
    }
  });
  return consumerTag;
}
```

```ts save=solutions/P26.5-E09.ts
// P26.5-E09: שליחה לתור FIFO — סדר לכל הזמנה, בלי כפילויות, ב־batches
import { SQSClient, SendMessageBatchCommand } from "@aws-sdk/client-sqs";

export interface OrderEvent {
  eventId: string;
  orderId: number;
  type: string;
  data?: unknown;
}

export async function sendOrderEvents(sqs: SQSClient, queueUrl: string, events: OrderEvent[]): Promise<number> {
  let sent = 0;
  for (let i = 0; i < events.length; i += 10) {
    const chunk = events.slice(i, i + 10); // SQS: עד 10 הודעות ב־batch
    const res = await sqs.send(
      new SendMessageBatchCommand({
        QueueUrl: queueUrl,
        Entries: chunk.map((e, j) => ({
          Id: String(j), // ייחודי בתוך ה־batch בלבד
          MessageBody: JSON.stringify(e),
          MessageGroupId: String(e.orderId), // סדר נשמר בתוך קבוצה
          MessageDeduplicationId: e.eventId, // אותו id תוך 5 דקות ← נשלח פעם אחת
          MessageAttributes: { type: { DataType: "String", StringValue: e.type } },
        })),
      }),
    );
    if (res.Failed?.length) {
      const ids = res.Failed.map((f) => chunk[Number(f.Id)].eventId);
      throw new Error(`שליחה נכשלה עבור: ${ids.join(", ")} (${res.Failed[0].Code})`);
    }
    sent += res.Successful?.length ?? 0;
```

```ts save=solutions/P26.5-E10.ts
import { SQSClient, ReceiveMessageCommand, DeleteMessageBatchCommand } from "@aws-sdk/client-sqs";

export async function pollOnce(
  sqs: SQSClient,
  queueUrl: string,
  handler: (body: unknown) => Promise<void>,
  { waitSeconds = 20 } = {},
): Promise<{ handled: number; failed: number }> {
  const res = await sqs.send(
    new ReceiveMessageCommand({
      QueueUrl: queueUrl,
      MaxNumberOfMessages: 10,
      WaitTimeSeconds: waitSeconds, // long polling: מחכה עד שיש הודעה, במקום לחזור ריק מיד
    }),
  );
  const done: { Id: string; ReceiptHandle: string }[] = [];
  let failed = 0;
  for (const [i, m] of (res.Messages ?? []).entries()) {
    try {
      await handler(JSON.parse(m.Body!));
      done.push({ Id: String(i), ReceiptHandle: m.ReceiptHandle! });
    } catch {
      failed++; // לא מוחקים: אחרי ה־visibility timeout ההודעה תחזור (ובסוף — ל־DLQ)
    }
  }
  if (done.length) {
    const del = await sqs.send(new DeleteMessageBatchCommand({ QueueUrl: queueUrl, Entries: done }));
    if (del.Failed?.length) throw new Error(`מחיקה נכשלה: ${del.Failed[0].Message}`);
  }
  return { handled: done.length, failed };
}
```

```ts save=solutions/P26.5-E13.ts
// P26.5-E13: projection — מצב ההזמנות מתוך זרם אירועים (עם כפילויות ואי־סדר)
export interface OrderEvent {
  orderId: number;
  seq: number; // מספר רץ לכל הזמנה: 1, 2, 3…
  type: "order.placed" | "order.paid" | "order.shipped" | "order.cancelled";
}

export interface OrderView {
  status: "placed" | "paid" | "shipped" | "cancelled";
  seq: number;
}

const STATUS = { "order.placed": "placed", "order.paid": "paid", "order.shipped": "shipped", "order.cancelled": "cancelled" } as const;

export function applyEvent(state: Map<number, OrderView>, e: OrderEvent): boolean {
  const cur = state.get(e.orderId);
  if (cur && e.seq <= cur.seq) return false; // כפילות, או אירוע ישן שהגיע מאוחר
  if (cur?.status === "cancelled") return false; // סופי
  state.set(e.orderId, { status: STATUS[e.type], seq: e.seq });
  return true;
}

export function project(events: OrderEvent[]) {
  const state = new Map<number, OrderView>();
  let applied = 0;
  for (const e of events) if (applyEvent(state, e)) applied++;
  return { state, applied, ignored: events.length - applied };
}
```

```ts save=solutions/P26.5-E14.ts
import { Consumer, MessagesStreamModes, MessagesStreamFallbackModes, stringDeserializers } from "@platformatic/kafka";

export interface ConsumeOptions {
  brokers: string[];
  groupId: string;
  topic: string;
}

export async function consumeAndCommit(
  { brokers, groupId, topic }: ConsumeOptions,
  handler: (event: any) => Promise<void>,
  { max }: { max: number },
): Promise<number> {
  const consumer = new Consumer({ groupId, clientId: `${groupId}-worker`, bootstrapBrokers: brokers, deserializers: stringDeserializers });
  const stream = await consumer.consume({
    topics: [topic],
    mode: MessagesStreamModes.COMMITTED, // ממשיכים מאיפה שה־group עצר
    fallbackMode: MessagesStreamFallbackModes.EARLIEST, // group חדש ← מההתחלה
    autocommit: false,
  });
  let processed = 0;
  try {
    for await (const m of stream) {
      await handler(JSON.parse(m.value)); // נכשל ← יוצאים בלי commit; ההודעה תגיע שוב
      await m.commit();
      if (++processed >= max) break;
    }
  } finally {
    await stream.close().catch(() => {});
    await consumer.close();
  }
  return processed;
}
```

```ts save=solutions/P26.5-E16.ts
// P26.5-E16: מתי לנסות שוב, ומתי לוותר — exponential backoff עם jitter
export type RetryDecision = { action: "retry"; delayMs: number } | { action: "dead-letter"; reason: string };

export interface RetryOptions {
  baseMs?: number;
  maxMs?: number;
  maxAttempts?: number;
  random?: () => number;
}

export function isPermanent(err: any): boolean {
  if (err?.permanent === true) return true;
  const s = err?.status;
  return typeof s === "number" && s >= 400 && s < 500 && s !== 408 && s !== 429; // 4xx — לא ישתנה בניסיון חוזר
}

// attempt = כמה ניסיונות כבר נכשלו (1 אחרי הכישלון הראשון)
export function nextRetry(attempt: number, err: unknown, { baseMs = 500, maxMs = 30_000, maxAttempts = 5, random = Math.random }: RetryOptions = {}): RetryDecision {
  if (isPermanent(err)) return { action: "dead-letter", reason: "permanent" };
  if (attempt >= maxAttempts) return { action: "dead-letter", reason: "max-attempts" };
  const cap = Math.min(maxMs, baseMs * 2 ** (attempt - 1));
  let delayMs = Math.floor(random() * cap); // "full jitter": אקראי בין 0 לתקרה — כדי שאלף consumers לא יחזרו באותה שנייה
  const retryAfter = (err as any)?.retryAfterMs;
  if (typeof retryAfter === "number" && retryAfter > delayMs) delayMs = Math.min(retryAfter, maxMs);
  return { action: "retry", delayMs };
}
```
<p>למה jitter? שירות התשלומים נפל לדקה, ואלף consumers נכשלו יחד. בלי jitter — כולם חוזרים יחד אחרי 500ms, ושוב אחרי 1000ms: גלים שמפילים את השירות בדיוק כשהוא מנסה לקום ("thundering herd"). עם jitter — הניסיונות מתפזרים.</p>

```ts save=solutions/P26.5-E17.ts
// P26.5-E17: transactional outbox — ההזמנה והאירוע נשמרים יחד; relay שולח אחר כך
import type { Pool } from "pg";

export interface OutboxRow {
  id: number;
  topic: string;
  key: string;
  payload: any;
}

export async function placeOrder(pool: Pool, order: { email: string; totalCents: number }): Promise<number> {
  const client = await pool.connect();
  try {
    await client.query("BEGIN");
    const { rows } = await client.query("INSERT INTO orders (email, total_cents) VALUES ($1, $2) RETURNING id", [order.email, order.totalCents]);
    const id = rows[0].id as number;
    await client.query("INSERT INTO outbox (topic, key, payload) VALUES ('orders', $1, $2)", [
      String(id),
      { type: "order.placed", orderId: id, email: order.email, totalCents: order.totalCents },
    ]);
    await client.query("COMMIT");
    return id;
  } catch (err) {
    await client.query("ROLLBACK").catch(() => {});
    throw err;
  } finally {
    client.release();
  }
}

export async function relayOutbox(pool: Pool, publish: (row: OutboxRow) => Promise<void>, { batch = 100 } = {}): Promise<number> {
  const client = await pool.connect();
  const sent: number[] = [];
  try {
    await client.query("BEGIN");
    const { rows } = await client.query(
      "SELECT id, topic, key, payload FROM outbox WHERE published_at IS NULL ORDER BY id LIMIT $1 FOR UPDATE SKIP LOCKED",
      [batch],
    );
    let failure: unknown = null;
    for (const r of rows) {
      try {
        await publish({ id: Number(r.id), topic: r.topic, key: r.key, payload: r.payload });
        sent.push(Number(r.id));
      } catch (err) {
        failure = err; // עוצרים: לשמור על הסדר. מה שנשלח — מסמנים
        break;
      }
    }
    if (sent.length) await client.query("UPDATE outbox SET published_at = now() WHERE id = ANY($1::bigint[])", [sent]);
    await client.query("COMMIT");
    if (failure) throw failure;
    return sent.length;
  } catch (err) {
    await client.query("ROLLBACK").catch(() => {});
    throw err;
  } finally {
    client.release();
  }
}
```

```ts save=solutions/P26.5-E18.ts
// P26.5-E18: retry עם השהיה — תור המתנה עם TTL, ו־dead letter חזרה
import type { Channel, ConsumeMessage } from "amqplib";

export interface RetryNames {
  queue: string; // העבודה
  retryQueue: string; // ממתינים delayMs, ואז חוזרים ל־queue
  deadQueue: string; // נכשלו maxAttempts פעמים
}

export async function setupRetry(ch: Channel, n: RetryNames, delayMs: number): Promise<void> {
  // nack ב־queue ← default exchange ("") עם routing key = retryQueue
  await ch.assertQueue(n.queue, { durable: true, deadLetterExchange: "", deadLetterRoutingKey: n.retryQueue });
  // TTL פג ב־retryQueue ← default exchange עם routing key = queue
  await ch.assertQueue(n.retryQueue, { durable: true, messageTtl: delayMs, deadLetterExchange: "", deadLetterRoutingKey: n.queue });
  await ch.assertQueue(n.deadQueue, { durable: true });
}

// כמה פעמים ההודעה כבר נדחתה מ־queue (RabbitMQ סופר ב־x-death)
export function attemptsSoFar(msg: ConsumeMessage, queue: string): number {
  const deaths = (msg.properties.headers?.["x-death"] ?? []) as { queue: string; reason: string; count: number }[];
  return deaths.find((d) => d.queue === queue && d.reason === "rejected")?.count ?? 0;
}

export async function startRetryingWorker(
  ch: Channel,
  n: RetryNames,
  handler: (job: any) => Promise<void>,
  { maxAttempts = 3 } = {},
): Promise<string> {
  await ch.prefetch(10);
  const { consumerTag } = await ch.consume(n.queue, async (msg) => {
    if (!msg) return;
    const attempt = attemptsSoFar(msg, n.queue) + 1;
    try {
      await handler(JSON.parse(msg.content.toString()));
      ch.ack(msg);
    } catch (err) {
      if (attempt >= maxAttempts) {
        ch.sendToQueue(n.deadQueue, msg.content, {
          persistent: true,
          contentType: msg.properties.contentType,
          headers: { "x-attempts": attempt, "x-error": String((err as Error)?.message ?? err) },
        });
        ch.ack(msg);
      } else {
        ch.nack(msg, false, false); // ← retryQueue, ואחרי delayMs חזרה
      }
    }
  });
  return consumerTag;
}
```

```ts save=solutions/P26.5-E19.ts
// P26.5-E19: consumer lag — כמה ה־consumers מאחור, והאם הפער גדל
export interface LagReport {
  perPartition: Record<number, number>;
  total: number;
  worstPartition: number | null;
}

// end: ה־offset הבא שייכתב בכל partition (high watermark); committed: הבא שה־group יקרא (‎-1 = אין commit)
export function consumerLag(end: Record<number, number>, committed: Record<number, number>, earliest: Record<number, number> = {}): LagReport {
  const perPartition: Record<number, number> = {};
  let total = 0;
  let worstPartition: number | null = null;
  for (const [p, hw] of Object.entries(end)) {
    const part = Number(p);
    const c = committed[part];
    const from = c === undefined || c < 0 ? (earliest[part] ?? 0) : c; // בלי commit — כל מה שעוד שמור ב־topic
    const lag = Math.max(0, hw - from);
    perPartition[part] = lag;
    total += lag;
    if (worstPartition === null || lag > perPartition[worstPartition]) worstPartition = part;
  }
  return { perPartition, total, worstPartition };
}

// מתריעים כשה־lag עולה ברציפות — לא על קפיצה אחת
export function isFallingBehind(samples: number[], { window = 3 } = {}): boolean {
  if (samples.length < window + 1) return false;
  const last = samples.slice(-(window + 1));
  return last.every((v, i) => i === 0 || v > last[i - 1]);
}
```

```ts save=solutions/P26.5-E22.ts
export async function publishStatus(producer: Producer<string, string, string, string>, topic: string, orderId: number, status: string) {

export async function syncStatuses(
  opts: { brokers: string[]; groupId: string; topic: string },
  save: (orderId: number, status: string) => Promise<void>,
  max: number,

  const consumer = new Consumer({ groupId: opts.groupId, clientId: "status-sync", bootstrapBrokers: opts.brokers, deserializers: stringDeserializers });
  const stream = await consumer.consume({
    topics: [opts.topic],
    mode: MessagesStreamModes.COMMITTED,
    fallbackMode: MessagesStreamFallbackModes.EARLIEST, // תיקון 3: group חדש מתחיל מההתחלה, לא רק מהודעות חדשות
    autocommit: false,
  });
```

```ts save=solutions/P26.5-E24.ts
// P26.5-E24: צינור ההזמנות של חנות הקפה ב־RabbitMQ
import type { Channel, ConfirmChannel } from "amqplib";

export interface PipelineNames {
  exchange: string;
  emailQueue: string;
  inventoryQueue: string;
  deadLetterExchange: string;
  deadLetterQueue: string;
}

export interface OrderPlaced {
  eventId: string;
  orderId: number;
  email: string;
  items: { productId: number; quantity: number }[];
}

export async function setupOrderPipeline(ch: Channel, n: PipelineNames): Promise<void> {
  await ch.assertExchange(n.exchange, "topic", { durable: true });
  await ch.assertExchange(n.deadLetterExchange, "fanout", { durable: true });
  await ch.assertQueue(n.deadLetterQueue, { durable: true });
  await ch.bindQueue(n.deadLetterQueue, n.deadLetterExchange, "");
  for (const q of [n.emailQueue, n.inventoryQueue]) {
    await ch.assertQueue(q, { durable: true, deadLetterExchange: n.deadLetterExchange });
    await ch.bindQueue(q, n.exchange, "order.placed");
  }
}

// publisher confirms: מחכים שה־broker אישר שקיבל (ושמר) — רק אז "נשלח"
export async function publishOrderPlaced(ch: ConfirmChannel, exchange: string, e: OrderPlaced): Promise<void> {
  ch.publish(exchange, "order.placed", Buffer.from(JSON.stringify(e)), {
    persistent: true,
    contentType: "application/json",
    messageId: e.eventId,
  });
  await ch.waitForConfirms();
}

export interface ProcessedStore {
  // true אם זו הפעם הראשונה (ונרשם), false אם כבר טופל
  markIfNew(consumer: string, eventId: string): Promise<boolean>;
  unmark(consumer: string, eventId: string): Promise<void>;
}

export function memoryStore(): ProcessedStore {
  const seen = new Set<string>();
  return {
    async markIfNew(c, id) {
      const k = `${c}:${id}`;
      if (seen.has(k)) return false;
      seen.add(k);
      return true;
    },
    async unmark(c, id) {
      seen.delete(`${c}:${id}`);
    },
  };
}

function isOrderPlaced(x: any): x is OrderPlaced {
  return (
    x && typeof x.eventId === "string" && Number.isInteger(x.orderId) && typeof x.email === "string" &&
    Array.isArray(x.items) && x.items.every((i: any) => Number.isInteger(i.productId) && Number.isInteger(i.quantity) && i.quantity > 0)
  );
}

// consumer אידמפוטנטי: אותו eventId פעמיים ← פעולה אחת. הודעה פגומה ← dead letter מיד
export async function startConsumer(
  ch: Channel,
  queue: string,
  name: string,
  store: ProcessedStore,
  handle: (e: OrderPlaced) => Promise<void>,
): Promise<string> {
  await ch.prefetch(10);
  const { consumerTag } = await ch.consume(queue, async (msg) => {
    if (!msg) return;
    let e: unknown;
    try {
      e = JSON.parse(msg.content.toString());
    } catch {
      return ch.nack(msg, false, false);
    }
    if (!isOrderPlaced(e)) return ch.nack(msg, false, false);
    if (!(await store.markIfNew(name, e.eventId))) return ch.ack(msg); // כפילות — כבר טופל
    try {
      await handle(e);
      ch.ack(msg);
    } catch {
      await store.unmark(name, e.eventId); // לא הצליח — שיוכל לנסות שוב
      ch.nack(msg, false, false);
    }
  });
  return consumerTag;
}
```

```ts save=solutions/P26.5-E25.ts
// P26.5-E25: אנליטיקה מ־Kafka ל־PostgreSQL — ה־offset נשמר באותה טרנזקציה עם התוצאה
import { Admin, Consumer, MessagesStreamModes, stringDeserializers } from "@platformatic/kafka";
import type { Pool } from "pg";
import { randomUUID } from "node:crypto";

export interface AggregatorOptions {
  brokers: string[];
  topic: string;
  pool: Pool;
  max: number;
}

async function partitionsOf(brokers: string[], topic: string): Promise<number[]> {
  const admin = new Admin({ clientId: "revenue-meta", bootstrapBrokers: brokers });
  try {
    const md = await admin.metadata({ topics: [topic] });
    return Array.from({ length: md.topics.get(topic)!.partitionsCount }, (_, i) => i);
  } finally {
    await admin.close();
  }
}

export async function runRevenueAggregator({ brokers, topic, pool, max }: AggregatorOptions): Promise<number> {
  const { rows } = await pool.query("SELECT partition, next_offset FROM consumer_offsets WHERE topic = $1", [topic]);
  const stored = new Map(rows.map((r: any) => [Number(r.partition), BigInt(r.next_offset)]));
  const offsets = (await partitionsOf(brokers, topic)).map((partition) => ({ topic, partition, offset: stored.get(partition) ?? 0n }));

  // group חד־פעמי לכל ריצה: ה־offsets של Kafka לא נקראים ולא נכתבים — האמת במסד
  const consumer = new Consumer({ groupId: `revenue-${topic}-${randomUUID()}`, clientId: "revenue", bootstrapBrokers: brokers, deserializers: stringDeserializers });
  const stream = await consumer.consume({ topics: [topic], mode: MessagesStreamModes.MANUAL, offsets, autocommit: false });
  let n = 0;
  try {
    for await (const m of stream) {
      const e = JSON.parse(m.value) as { orderId: number; totalCents: number; at: string };
      const client = await pool.connect();
      try {
        await client.query("BEGIN");
        await client.query(
          `INSERT INTO revenue_by_minute (minute, orders, revenue_cents) VALUES (date_trunc('minute', $1::timestamptz), 1, $2)
           ON CONFLICT (minute) DO UPDATE SET orders = revenue_by_minute.orders + 1, revenue_cents = revenue_by_minute.revenue_cents + EXCLUDED.revenue_cents`,
          [e.at, e.totalCents],
        );
        await client.query(
          `INSERT INTO consumer_offsets (topic, partition, next_offset) VALUES ($1, $2, $3)
           ON CONFLICT (topic, partition) DO UPDATE SET next_offset = EXCLUDED.next_offset`,
          [topic, m.partition, String(m.offset + 1n)],
        );
        await client.query("COMMIT"); // התוצאה וה־offset — יחד, או אף אחד
      } catch (err) {
        await client.query("ROLLBACK").catch(() => {});
        throw err;
      } finally {
        client.release();
      }
      if (++n >= max) break;
    }
  } finally {
    await stream.close().catch(() => {});
    await consumer.close();
  }
  return n;
}
```

```ts run save=solutions/P26.5-E26.ts
// P26.5-E26 — להחזיר סדר להודעות
export type PushResult = "delivered" | "buffered" | "duplicate";

export class ReorderBuffer<T> {
  #next: number;
  #waiting = new Map<number, T>();
  #deliver: (seq: number, message: T) => void;

  constructor(deliver: (seq: number, message: T) => void, startSeq = 1) {
    this.#deliver = deliver;
    this.#next = startSeq;
  }

  push(seq: number, message: T): PushResult {
    if (!Number.isInteger(seq)) throw new TypeError(`seq חייב להיות מספר שלם: ${seq}`);
    if (seq < this.#next || this.#waiting.has(seq)) return "duplicate";
    if (seq > this.#next) {
      this.#waiting.set(seq, message);
      return "buffered";
    }
    this.#deliver(seq, message);
    this.#next++;
    while (this.#waiting.has(this.#next)) {
      const waiting = this.#waiting.get(this.#next) as T;
      this.#waiting.delete(this.#next);
      this.#deliver(this.#next, waiting);
      this.#next++;
    }
    return "delivered";
  }

  get nextSeq(): number {
    return this.#next;
  }

  get pending(): number {
    return this.#waiting.size;
  }

  gaps(): number[] {
    if (this.#waiting.size === 0) return [];
    const max = Math.max(...this.#waiting.keys());
    const missing: number[] = [];
    for (let seq = this.#next; seq < max; seq++) if (!this.#waiting.has(seq)) missing.push(seq);
    return missing;
  }
}

const buffer = new ReorderBuffer<string>((seq, status) => console.log(`  ← ${seq}: ${status}`));
for (const [seq, status] of [[1, "placed"], [3, "shipped"], [5, "refunded"], [3, "shipped"], [2, "paid"], [1, "placed"]] as const) {
  const result = buffer.push(seq, status);
  console.log(`push ${seq} → ${result}`, result === "buffered" ? `(חסר: ${buffer.gaps().join(", ")})` : "");
}
console.log("next:", buffer.nextSeq, "pending:", buffer.pending);
```

```sql run db=26w-messaging/files/outbox-next.sql save=solutions/P26.5-E28.sql
WITH pending AS (
  SELECT
    id, topic, key, event_type,
    ROW_NUMBER() OVER (PARTITION BY topic, key ORDER BY id) AS rn,
    COUNT(*) OVER (PARTITION BY topic, key) AS pending
  FROM outbox
  WHERE published_at IS NULL
)
SELECT topic, key, id AS next_id, event_type, pending
FROM pending
WHERE rn = 1
ORDER BY next_id;
```

```ts save=solutions/P26.5-E33.ts
// P26.5-E33: ack לפני או אחרי — לכל נקודת נפילה
type Step = "ack" | "handle";

export function crashOutcomes(order: Step[]) {
  if (order.length !== 2 || !order.includes("ack") || !order.includes("handle")) throw new RangeError("צריך בדיוק ack ו־handle");
  const counts: number[] = [];
  for (const crashAfter of [0, 1, 2, null]) {
    const done = crashAfter === null ? order : order.slice(0, crashAfter);
    let handled = done.filter((s) => s === "handle").length;
    if (crashAfter !== null && !done.includes("ack")) handled += 1; // נמסרת שוב, והפעם הכול עובר
    counts.push(handled);
  }
  const minHandled = Math.min(...counts);
  const maxHandled = Math.max(...counts);
  const guarantee = minHandled === 0 ? "at-most-once" : maxHandled > 1 ? "at-least-once" : "exactly-once";
  return { minHandled, maxHandled, guarantee } as const;
}
```

```ts save=solutions/P26.5-E34.ts
// P26.5-E34: תור בזיכרון — ack, nack ונפילה
export class MemoryQueue<T> {
  #queue: T[] = [];
  #unacked = new Map<number, { consumer: string; msg: T }>();
  #nextTag = 1;

  push(msg: T) {
    this.#queue.push(msg);
  }
  size() {
    return this.#queue.length;
  }
  receive(consumer: string) {
    if (this.#queue.length === 0) return null;
    const msg = this.#queue.shift() as T;
    const tag = this.#nextTag++;
    this.#unacked.set(tag, { consumer, msg });
    return { tag, msg };
  }
  #take(tag: number) {
    const e = this.#unacked.get(tag);
    if (!e) throw new Error(`tag לא מוכר: ${tag}`);
    this.#unacked.delete(tag);
    return e;
  }
  ack(tag: number) {
    this.#take(tag);
  }
  nack(tag: number, requeue: boolean) {
    const e = this.#take(tag);
    if (requeue) this.#queue.unshift(e.msg);
  }
  crash(consumer: string) {
    const mine = [...this.#unacked].filter(([, e]) => e.consumer === consumer).sort(([a], [b]) => a - b);
    for (const [tag] of mine) this.#unacked.delete(tag);
    this.#queue.unshift(...mine.map(([, e]) => e.msg));
    return mine.length;
  }
  unacked(consumer: string) {
    return [...this.#unacked.values()].filter((e) => e.consumer === consumer).length;
  }
}
```

```ts save=solutions/P26.5-E35.ts
type Entry = { key: string; value: string };

export class MemoryLog {
  #parts: Entry[][];
  #committed = new Map<string, number>();

  constructor(partitions: number) {
    this.#parts = Array.from({ length: partitions }, () => []);
  }
  #partitionOf(key: string) {
    let h = 0;
    for (let i = 0; i < key.length; i++) h += key.charCodeAt(i);
    return h % this.#parts.length;
  }
  append(key: string, value: string) {
    const partition = this.#partitionOf(key);
    const offset = this.#parts[partition].push({ key, value }) - 1;
    return { partition, offset };
  }
  read(group: string, partition: number, max: number) {
    const from = this.#committed.get(`${group}:${partition}`) ?? 0;
    return this.#parts[partition].slice(from, from + max).map((e, i) => ({ offset: from + i, ...e }));
  }
  commit(group: string, partition: number, nextOffset: number) {
    this.#committed.set(`${group}:${partition}`, nextOffset);
  }
  lag(group: string) {
    return this.#parts.reduce((sum, p, i) => sum + p.length - (this.#committed.get(`${group}:${i}`) ?? 0), 0);
  }
}
```

```ts save=solutions/P26.5-E36.ts
// P26.5-E36: handler אידמפוטנטי — גם במקביל
type Msg<T> = { id: string } & T;

export function idempotent<T>(handler: (msg: Msg<T>) => Promise<void>) {
  const done = new Set<string>();
  const running = new Map<string, Promise<void>>();
  return async function handle(msg: Msg<T>): Promise<"handled" | "duplicate"> {
    if (done.has(msg.id)) return "duplicate";
    const inFlight = running.get(msg.id);
    if (inFlight) {
      try {
        await inFlight;
        return "duplicate";
      } catch {
        return handle(msg); // הראשון נכשל — ננסה בעצמנו
      }
    }
    const p = handler(msg);
    running.set(msg.id, p);
    try {
      await p;
      done.add(msg.id);
      return "handled";
    } finally {
      running.delete(msg.id);
    }
  };
}
```

```ts save=solutions/P26.5-E37.ts
export async function processAll<T extends { id: string }>(messages: T[], handler: (m: T) => Promise<void>, maxAttempts: number) {
  const queue = messages.map((msg) => ({ msg, attempts: 0 }));
  const done: string[] = [];
  const dead: { id: string; attempts: number; error: string }[] = [];
  while (queue.length) {
    const item = queue.shift()!;
    try {
      await handler(item.msg);
      done.push(item.msg.id);
    } catch (e) {
      item.attempts++;
      if (item.attempts >= maxAttempts) dead.push({ id: item.msg.id, attempts: item.attempts, error: (e as Error).message });
      else queue.push(item);
    }
  }
  return { done, dead };
}
```

```ts save=solutions/P26.5-E45.ts
// P26.5-E45: להודיע לכולם ב־HTTP — ולמה זה כואב
type Service = { name: string; call: (event: unknown) => Promise<void> };

export async function notifyAll(services: Service[], event: unknown, timeoutMs: number) {
  const timers: NodeJS.Timeout[] = [];
  const withTimeout = (p: Promise<void>) =>
    Promise.race([p, new Promise<never>((_, reject) => timers.push(setTimeout(() => reject(new Error("timeout")), timeoutMs)))]);
  const results = await Promise.allSettled(services.map((s) => withTimeout(Promise.resolve().then(() => s.call(event)))));
  timers.forEach(clearTimeout);
  return results.map((r, i) =>
    r.status === "fulfilled" ? { name: services[i].name, ok: true } : { name: services[i].name, ok: false, error: (r.reason as Error).message },
  );
}
```

```ts save=solutions/P26.5-E52.ts
// P26.5-E52: התאמה של topic binding
const words = (s: string) => (s === "" ? [] : s.split("."));

function match(p: string[], k: string[]): boolean {
  if (p.length === 0) return k.length === 0;
  const [head, ...rest] = p;
  if (head === "#") {
    for (let skip = 0; skip <= k.length; skip++) if (match(rest, k.slice(skip))) return true;
    return false;
  }
  if (k.length === 0) return false;
  return (head === "*" || head === k[0]) && match(rest, k.slice(1));
}

export function bindingMatches(pattern: string, routingKey: string) {
  return match(words(pattern), words(routingKey));
}
```

```ts save=solutions/P26.5-E53.ts
// P26.5-E53: fanout — הודעה לכל ה־instances
import type { Channel } from "amqplib";

export async function subscribeInstance(ch: Channel, exchange: string, onMessage: (data: unknown) => void) {
  await ch.assertExchange(exchange, "fanout", { durable: false });
  const { queue } = await ch.assertQueue("", { exclusive: true });
  await ch.bindQueue(queue, exchange, "");
  const { consumerTag } = await ch.consume(queue, (msg) => {
    if (msg) onMessage(JSON.parse(msg.content.toString()));
  }, { noAck: true });
  return {
    queue,
    async cancel() {
      await ch.cancel(consumerTag);
    },
  };
}

export function broadcast(ch: Channel, exchange: string, data: unknown) {
  return ch.publish(exchange, "", Buffer.from(JSON.stringify(data)), { contentType: "application/json" });
}
```

```ts save=solutions/P26.5-E54.ts
export async function startSecondChanceWorker(ch: Channel, queue: string, handler: (job: unknown) => Promise<void>) {
  await ch.prefetch(1);
  const { consumerTag } = await ch.consume(queue, async (msg) => {
```

```ts save=solutions/P26.5-E55.ts
// P26.5-E55: כמה פעמים כבר נכשלה? (x-death)
type Death = { queue?: string; reason?: string; count?: number };

export function deathCount(headers: Record<string, unknown> | undefined, queue: string, reason?: string) {
  const deaths = headers?.["x-death"];
  if (!Array.isArray(deaths)) return 0;
  return (deaths as Death[])
    .filter((d) => d.queue === queue && (reason === undefined || d.reason === reason))
    .reduce((s, d) => s + (typeof d.count === "number" ? d.count : 1), 0);
}
```

```ts save=solutions/P26.5-E59.ts
import express from "express";
import type { Channel } from "amqplib";
import { z } from "zod";

const schema = z.object({
  id: z.string().min(1),
  type: z.enum(["created", "paid", "shipped", "cancelled"]),
  orderId: z.number().int().positive(),
  country: z.string().regex(/^[A-Za-z]{2}$/),

  app.use(express.json());
  app.post("/events", (req, res) => {
```

```ts save=solutions/P26.5-E60.ts
import type { Channel } from "amqplib";

export async function scheduleIn(ch: Channel, { waitPrefix, target }: { waitPrefix: string; target: string }, payload: unknown, delayMs: number) {
```

```ts save=solutions/P26.5-E61.ts
// P26.5-E61: עצירה מסודרת של consumer
import type { Channel, ConsumeMessage } from "amqplib";

export async function startConsumer(ch: Channel, queue: string, handler: (job: unknown) => Promise<void>, prefetch: number) {
  await ch.prefetch(prefetch);
  const inFlight = new Set<Promise<void>>();
  const handle = async (msg: ConsumeMessage) => {
    try {
      await handler(JSON.parse(msg.content.toString()));
      ch.ack(msg);
    } catch {
      ch.nack(msg, false, false);
    }
  };
  const { consumerTag } = await ch.consume(queue, (msg) => {
    if (!msg) return;
    const p = handle(msg).finally(() => inFlight.delete(p));
    inFlight.add(p);
  });
  return {
    async stop() {
      await ch.cancel(consumerTag);
      await Promise.allSettled([...inFlight]);
      await ch.close();
    },
  };
}
```

```ts save=solutions/P26.5-E67.ts
import { ChangeMessageVisibilityCommand, type SQSClient } from "@aws-sdk/client-sqs";

export async function withVisibilityHeartbeat<T>(
  sqs: SQSClient,
  queueUrl: string,
  receiptHandle: string,
  work: () => Promise<T>,
  { visibilitySeconds, everyMs }: { visibilitySeconds: number; everyMs: number },
): Promise<T> {
  const timer = setInterval(() => {
    sqs.send(new ChangeMessageVisibilityCommand({ QueueUrl: queueUrl, ReceiptHandle: receiptHandle, VisibilityTimeout: visibilitySeconds })).catch(() => {});

  try {
    return await work();

    clearInterval(timer);
  }
}
```

```ts save=solutions/P26.5-E68.ts
// P26.5-E68: שליחה ב־batches — וגם מה שנכשל
import { SendMessageBatchCommand, type SQSClient } from "@aws-sdk/client-sqs";

export async function sendAll(sqs: SQSClient, queueUrl: string, messages: { id: string; body: unknown }[]) {
  const ok = new Set<string>();
  const bad = new Map<string, string>();
  for (let i = 0; i < messages.length; i += 10) {
    const chunk = messages.slice(i, i + 10);
    const res = await sqs.send(
      new SendMessageBatchCommand({ QueueUrl: queueUrl, Entries: chunk.map((m) => ({ Id: m.id, MessageBody: JSON.stringify(m.body) })) }),
    );
    for (const s of res.Successful ?? []) ok.add(s.Id!);
    for (const f of res.Failed ?? []) bad.set(f.Id!, f.Code ?? "Unknown");
  }
  return {
    sent: messages.filter((m) => ok.has(m.id)).map((m) => m.id),
    failed: messages.filter((m) => bad.has(m.id)).map((m) => ({ id: m.id, code: bad.get(m.id)! })),
  };
}
```

```ts save=solutions/P26.5-E69.ts
import { DeleteMessageBatchCommand, ReceiveMessageCommand, SendMessageBatchCommand, type SQSClient } from "@aws-sdk/client-sqs";

export async function redrive(sqs: SQSClient, fromUrl: string, toUrl: string, max = Infinity) {
  let moved = 0;
  while (moved < max) {
    const { Messages: msgs = [] } = await sqs.send(
      new ReceiveMessageCommand({ QueueUrl: fromUrl, MaxNumberOfMessages: Math.min(10, max - moved), WaitTimeSeconds: 0 }),
    );
    if (msgs.length === 0) break;
    const sent = await sqs.send(new SendMessageBatchCommand({ QueueUrl: toUrl, Entries: msgs.map((m, i) => ({ Id: String(i), MessageBody: m.Body! })) }));
    const ok = (sent.Successful ?? []).map((s) => msgs[Number(s.Id)]);
    if (ok.length) {
      await sqs.send(new DeleteMessageBatchCommand({ QueueUrl: fromUrl, Entries: ok.map((m, i) => ({ Id: String(i), ReceiptHandle: m.ReceiptHandle! })) }));
    }
    moved += ok.length;
  }
  return moved;
}
```

```ts save=solutions/P26.5-E70.ts
import { GetQueueAttributesCommand, type SQSClient } from "@aws-sdk/client-sqs";

export async function queueDepth(sqs: SQSClient, queueUrl: string) {
  const { Attributes: a = {} } = await sqs.send(
    new GetQueueAttributesCommand({
      QueueUrl: queueUrl,
      AttributeNames: ["ApproximateNumberOfMessages", "ApproximateNumberOfMessagesNotVisible", "ApproximateNumberOfMessagesDelayed"],
    }),
  );
  return {
    visible: Number(a.ApproximateNumberOfMessages ?? 0),
    inFlight: Number(a.ApproximateNumberOfMessagesNotVisible ?? 0),
    delayed: Number(a.ApproximateNumberOfMessagesDelayed ?? 0),
  };
}

export function workersNeeded(depth: { visible: number; inFlight: number }, { perWorker, min, max }: { perWorker: number; min: number; max: number }) {
  return Math.min(max, Math.max(min, Math.ceil((depth.visible + depth.inFlight) / perWorker)));
}
```

```ts save=solutions/P26.5-E74.ts
// P26.5-E74: claim check — הודעה גדולה דרך S3
import { randomUUID } from "node:crypto";
import { GetObjectCommand, PutObjectCommand, type S3Client } from "@aws-sdk/client-s3";
import { SendMessageCommand, type SQSClient } from "@aws-sdk/client-sqs";

export async function sendClaimCheck(
  { s3, sqs }: { s3: S3Client; sqs: SQSClient },
  { bucket, queueUrl }: { bucket: string; queueUrl: string },
  payload: unknown,
  limitBytes = 200_000,
): Promise<"inline" | "s3"> {
  const json = JSON.stringify(payload);
  if (Buffer.byteLength(json) <= limitBytes) {
    await sqs.send(new SendMessageCommand({ QueueUrl: queueUrl, MessageBody: JSON.stringify({ inline: payload }) }));
    return "inline";
  }
  const s3Key = `claims/${randomUUID()}.json`;
  await s3.send(new PutObjectCommand({ Bucket: bucket, Key: s3Key, Body: json, ContentType: "application/json" }));
  await sqs.send(new SendMessageCommand({ QueueUrl: queueUrl, MessageBody: JSON.stringify({ s3Key }) }));
  return "s3";
}

export async function readClaimCheck(s3: S3Client, bucket: string, body: string) {
  const msg = JSON.parse(body) as { inline?: unknown; s3Key?: string };
  if (msg.s3Key) {
    const obj = await s3.send(new GetObjectCommand({ Bucket: bucket, Key: msg.s3Key }));
    return JSON.parse(await obj.Body!.transformToString());
  }
  return msg.inline;
}
```

```ts save=solutions/P26.5-E75.ts
import { DeleteMessageCommand, ReceiveMessageCommand, type SQSClient } from "@aws-sdk/client-sqs";

export function startPoller(sqs: SQSClient, queueUrl: string, handler: (body: unknown) => Promise<void>, { waitSeconds }: { waitSeconds: number }) {
  let running = true;
  const counts = { handled: 0, failed: 0 };
  const loop = (async () => {
    while (running) {
      let msgs;
      try {
        ({ Messages: msgs = [] } = await sqs.send(new ReceiveMessageCommand({ QueueUrl: queueUrl, MaxNumberOfMessages: 10, WaitTimeSeconds: waitSeconds })));
      } catch {
        await new Promise((r) => setTimeout(r, 200));
        continue;
      }
      for (const m of msgs) {
        try {
          await handler(JSON.parse(m.Body!));
          await sqs.send(new DeleteMessageCommand({ QueueUrl: queueUrl, ReceiptHandle: m.ReceiptHandle! }));
          counts.handled++;
        } catch {
          counts.failed++;
        }
      }
    }
  })();
  return {
    async stop() {
      running = false;
      await loop;
    },
    stats: () => ({ ...counts }),
  };
}
```

```ts save=solutions/P26.5-E76.ts
export async function processBatch<M extends { MessageId: string }>(messages: M[], handler: (m: M) => Promise<void>, concurrency: number) {
  const results: boolean[] = [];
  let next = 0;
  async function worker() {
    while (next < messages.length) {
      const i = next++;
      try {
        await handler(messages[i]);
        results[i] = true;
      } catch {
        results[i] = false;
      }
    }
  }
  await Promise.all(Array.from({ length: Math.min(concurrency, messages.length) }, worker));
  return {
    ok: messages.filter((_, i) => results[i]).map((m) => m.MessageId),
    failed: messages.filter((_, i) => !results[i]).map((m) => m.MessageId),
  };
}
```

```ts save=solutions/P26.5-E82.ts
// P26.5-E82: חלוקת partitions בין consumers
export function assignPartitions(partitions: number, consumers: string[]): Record<string, number[]> {
  if (new Set(consumers).size !== consumers.length) throw new Error("שמות כפולים");
  const sorted = [...consumers].sort();
  const n = sorted.length;
  const out: Record<string, number[]> = {};
  if (n === 0) return out;
  const base = Math.floor(partitions / n);
  const extra = partitions % n;
  let next = 0;
  sorted.forEach((c, i) => {
    const count = base + (i < extra ? 1 : 0);
    out[c] = Array.from({ length: count }, (_, k) => next + k);
    next += count;
  });
  return out;
```

```ts save=solutions/P26.5-E83.ts
import type { Admin } from "@platformatic/kafka";

export async function rewindGroup(admin: Admin, groupId: string, topic: string, offsets: Record<number, number>) {
  const entries = Object.entries(offsets).map(([p, o]) => ({ partition: Number(p), offset: o }));
  for (const e of entries) {
    if (!Number.isInteger(e.partition) || e.partition < 0) throw new RangeError(`partition לא תקין: ${e.partition}`);

  await admin.alterConsumerGroupOffsets({
    groupId,

  });
}
```

```ts save=solutions/P26.5-E84.ts
    this.#max = max;
```

```ts save=solutions/P26.5-E90.ts
// P26.5-E90: סדר בתוך key, מקביליות בין keys
export function perKeySerial<M extends { key: string }>(handler: (m: M) => Promise<void>) {
  const tails = new Map<string, Promise<void>>();
  const all = new Set<Promise<void>>();
  return {
    push(m: M) {
      const prev = tails.get(m.key) ?? Promise.resolve();
      const run = prev.then(() => handler(m));
      const tail = run.catch(() => {});
      tails.set(m.key, tail);
      all.add(tail);
      tail.then(() => {
        all.delete(tail);
        if (tails.get(m.key) === tail) tails.delete(m.key);
      });
      return run;
    },
    async idle() {
      while (all.size) await Promise.all([...all]);
    },
  };
}
```

```ts save=solutions/P26.5-E100.ts
// P26.5-E100: upcaster — מ־v1 ל־v2 בקריאה
type V2Data = { orderId: number; email: string; total: { amountCents: number; currency: string } };
type V2 = { type: "order.placed"; version: 2; data: V2Data };

export function upcast(e: { type: string; version: number; data: unknown }): V2 {
  if (e.type !== "order.placed") throw new Error(`סוג לא נתמך: ${e.type}`);
  switch (e.version) {
    case 2:
      return structuredClone(e) as V2;
    case 1: {
      const d = e.data as { orderId: number; email: string; total: number };
      return { ...structuredClone(e), type: "order.placed", version: 2, data: { ...d, total: { amountCents: Math.round(d.total * 100), currency: "ILS" } } };
    }
    default:
      throw new Error(`גרסה לא נתמכת: ${e.version}`);
  }
}
```
<p>כשתגיע v3, מוסיפים <code>case 2</code> שמעלה ל־3, ו־v1 עובר דרך שניהם — כל שלב קטן ונבדק לבד. הלוגיקה העסקית מכירה רק את הגרסה האחרונה, וזה מה שמחזיק את הקוד פשוט שנים.</p>
@mistakes
<ul><li><code>Math.floor(total * 100)</code> — ‏<code>0.29 * 100</code> הוא <code>28.999999999999996</code>, ויוצא 28.</li><li>להעלות גרסה "לא מוכרת" כאילו היא האחרונה — v3 מהעתיד נקרא לא נכון בשקט.</li></ul>
