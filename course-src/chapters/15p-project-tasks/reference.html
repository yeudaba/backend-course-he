🔒 קרא רק אחרי שסיימת

פתרון לדוגמה הוא דרך אחת, לא "התשובה". השווה לשלך: איפה הוא פשוט יותר? איפה שלך? הקבצים המלאים — ב־`15p-project-tasks/solutions/tasks-api`. ואפשר להריץ עליהם את הבודק: `npm run check PR2 -- --dir 15p-project-tasks/solutions/tasks-api` (אחרי `npm install` שם).

החלטות עיצוב שכדאי לשים לב אליהן

  • validate זורק, לא עונה — יש בדיוק מקום אחד שבונה תשובות שגיאה (`errorHandler`). שינוי במבנה השגיאה = שינוי בקובץ אחד.
  • `updateTaskSchema` נכתב מחדש ולא `createTaskSchema.partial()` — אחרת ה־defaults היו דורסים שדות ב־PATCH.
  • ה־repository מחזיר עותקים ו־async — מתנהג כמו מסד נתונים כבר עכשיו, ולכן ההחלפה בפרק 19 לא תפתיע אף שכבה אחרת.
  • `now` מוזרק ל־service — כל חוק שתלוי בזמן (422, overdue, updatedAt) ניתן לבדיקה בלי לחכות.
  • `/stats` לפני `/:id`, ועם הערה בקוד — מי שיבוא אחריך לא יזיז את זה בטעות.
  • CORS עם רשימת origins מוזרקת (נוסף ביחידה 15.5) — לא `*`: רק הדף שלך רשאי לקרוא ל־API מדפדפן.
  • `createApp` מקבל אפשרויות עם ברירות מחדל — `server.ts` נשאר שלוש שורות, והבדיקות שולטות בכל תלות.

src/tasks/tasks.repository.ts

```ts file=src/tasks/tasks.repository.ts import type { Task } from "./tasks.schema.ts"; // החוזה של שכבת האחסון. כל המתודות async — כמו שיהיה מול PostgreSQL (פרק 19). export interface TasksRepository { findAll(): Promise; findById(id: number): Promise; insert(data: Omit): Promise; update(id: number, changes: Partial>): Promise; delete(id: number): Promise; } export class InMemoryTasksRepository implements TasksRepository { #tasks: Task[] = []; #nextId = 1; async findAll() { return this.#tasks.map((t) => ({ ...t })); } async findById(id: number) { const task = this.#tasks.find((t) => t.id === id); return task && { ...task }; } async insert(data: Omit) { const task = { id: this.#nextId++, ...data }; this.#tasks.push(task); return { ...task }; } async update(id: number, changes: Partial>) { const task = this.#tasks.find((t) => t.id === id); if (!task) return undefined; Object.assign(task, changes); return { ...task }; } async delete(id: number) { const before = this.#tasks.length; this.#tasks = this.#tasks.filter((t) => t.id !== id); return this.#tasks.length < before; } } ```

src/tasks/tasks.service.ts

```ts file=src/tasks/tasks.service.ts import { HttpError } from "../errors.ts"; import type { TasksRepository } from "./tasks.repository.ts"; import type { CreateTaskInput, ListQuery, Task, UpdateTaskInput } from "./tasks.schema.ts"; const PRIORITY_RANK = { low: 1, medium: 2, high: 3 }; // "היום" לפי UTC, בפורמט YYYY-MM-DD (אזורי זמן — פרק 24) const today = (now: Date) => now.toISOString().slice(0, 10); export class TasksService { repo: TasksRepository; now: () => Date; constructor(repo: TasksRepository, now: () => Date = () => new Date()) { this.repo = repo; this.now = now; } async list(query: ListQuery) { const { status, priority, q, sort, page, limit } = query; const all = (await this.repo.findAll()).filter( (t) => (!status || t.status === status) && (!priority || t.priority === priority) && (!q || `${t.title} ${t.description ?? ""}`.toLowerCase().includes(q.toLowerCase())) ); const desc = sort.startsWith("-"); const field = (desc ? sort.slice(1) : sort) as "createdAt" | "dueDate" | "priority"; const sorted = all.sort((a, b) => { // משימות בלי dueDate תמיד בסוף, בכל כיוון מיון if (field === "dueDate" && (a.dueDate === null || b.dueDate === null)) { return (a.dueDate === null ? 1 : 0) - (b.dueDate === null ? 1 : 0) || a.id - b.id; } const diff = field === "priority" ? PRIORITY_RANK[a.priority] - PRIORITY_RANK[b.priority] : String(a[field]).localeCompare(String(b[field])); return (desc ? -diff : diff) || a.id - b.id; // שוויון — לפי id עולה, כדי שהסדר יהיה יציב }); const start = (page - 1) * limit; return { data: sorted.slice(start, start + limit), meta: { page, limit, total: all.length, totalPages: Math.ceil(all.length / limit) }, }; } async get(id: number) { const task = await this.repo.findById(id); if (!task) throw new HttpError(404, "TASK_NOT_FOUND", `משימה ${id} לא נמצאה`); return task; } async create(input: CreateTaskInput) { if (input.dueDate !== null && input.dueDate < today(this.now())) { throw new HttpError(422, "DUE_DATE_IN_PAST", "תאריך היעד כבר עבר"); } const at = this.now().toISOString(); return this.repo.insert({ ...input, createdAt: at, updatedAt: at }); } async update(id: number, changes: UpdateTaskInput) { const task = await this.repo.update(id, { ...changes, updatedAt: this.now().toISOString() }); if (!task) throw new HttpError(404, "TASK_NOT_FOUND", `משימה ${id} לא נמצאה`); return task; } async remove(id: number) { if (!(await this.repo.delete(id))) throw new HttpError(404, "TASK_NOT_FOUND", `משימה ${id} לא נמצאה`); } async stats() { const all = await this.repo.findAll(); const byStatus: Record = { todo: 0, in_progress: 0, done: 0 }; for (const t of all) byStatus[t.status]++; const day = today(this.now()); const overdue = all.filter((t) => t.status !== "done" && t.dueDate !== null && t.dueDate < day).length; return { total: all.length, byStatus, overdue }; } } ```

src/tasks/tasks.routes.ts

```ts file=src/tasks/tasks.routes.ts import { Router } from "express"; import { validate } from "../middleware/validate.ts"; import { createTaskSchema, idParamsSchema, listQuerySchema, updateTaskSchema } from "./tasks.schema.ts"; import type { TasksService } from "./tasks.service.ts"; export function createTasksRouter(service: TasksService) { const router = Router(); router.get("/", validate({ query: listQuerySchema }), async (req, res) => { res.json(await service.list(res.locals.query)); }); // לפני /:id — אחרת "stats" היה נתפס כ־id router.get("/stats", async (req, res) => { res.json(await service.stats()); }); router.get("/:id", validate({ params: idParamsSchema }), async (req, res) => { res.json(await service.get(res.locals.params.id)); }); router.post("/", validate({ body: createTaskSchema }), async (req, res) => { const task = await service.create(req.body); res.status(201).location(`/tasks/${task.id}`).json(task); }); router.patch("/:id", validate({ params: idParamsSchema, body: updateTaskSchema }), async (req, res) => { res.json(await service.update(res.locals.params.id, req.body)); }); router.delete("/:id", validate({ params: idParamsSchema }), async (req, res) => { await service.remove(res.locals.params.id); res.sendStatus(204); }); return router; } ```

src/middleware/validate.ts

```ts file=src/middleware/validate.ts import type { Request, Response, NextFunction } from "express"; import type { ZodType } from "zod"; import { HttpError, type ErrorDetail } from "../errors.ts"; type Part = "body" | "query" | "params"; // בודק חלקים של הבקשה. body נקי ← req.body; query ו־params ← res.locals (ב־Express 5 אי אפשר לדרוס אותם) export function validate(schemas: Partial>) { return (req: Request, res: Response, next: NextFunction) => { const details: ErrorDetail[] = []; for (const part of ["params", "query", "body"] as const) { const schema = schemas[part]; if (!schema) continue; const result = schema.safeParse(req[part]); if (!result.success) { for (const issue of result.error.issues) details.push({ path: [part, ...issue.path].join("."), message: issue.message }); } else if (part === "body") { req.body = result.data; } else { res.locals[part] = result.data; } } if (details.length > 0) throw new HttpError(400, "VALIDATION_ERROR", "קלט לא תקין", details); next(); }; } ```

src/errors.ts

```ts file=src/errors.ts import type { Request, Response, NextFunction } from "express"; export interface ErrorDetail { path: string; message: string; } export class HttpError extends Error { status: number; code: string; details?: ErrorDetail[]; constructor(status: number, code: string, message: string, details?: ErrorDetail[]) { super(message); this.status = status; this.code = code; this.details = details; } } export function notFound(req: Request, res: Response) { res.status(404).json({ error: { code: "NOT_FOUND", message: `אין כתובת ${req.method} ${req.path}` } }); } export function errorHandler(err: unknown, req: Request, res: Response, next: NextFunction) { if (err instanceof HttpError) { const { code, message, details } = err; res.status(err.status).json({ error: { code, message, ...(details ? { details } : {}) } }); return; } if (typeof err === "object" && err !== null && "type" in err && err.type === "entity.parse.failed") { res.status(400).json({ error: { code: "INVALID_JSON", message: "JSON לא תקין" } }); return; } console.error(err); res.status(500).json({ error: { code: "INTERNAL", message: "שגיאה פנימית" } }); } ```
