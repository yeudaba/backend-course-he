```ts save=solutions/P28-E06.ts
// P28-E06
// base62: ספרות 0-9, a-z, A-Z. כמו בסיס 10 או 16 — רק עם 62 "ספרות". 7 תווים = 62⁷ ≈ 3.5 טריליון קודים
const ALPHABET = "0123456789abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ";

export function encode(n: number): string {
  if (!Number.isSafeInteger(n) || n < 0) throw new Error("רק מספר שלם אי־שלילי");
  if (n === 0) return "0";
  let out = "";
  while (n > 0) {
    out = ALPHABET[n % 62] + out;
    n = Math.floor(n / 62);
  }
  return out;
}

export function decode(code: string): number {
  if (!/^[0-9a-zA-Z]+$/.test(code)) throw new Error("קוד לא תקין");
  let n = 0;
  for (const ch of code) n = n * 62 + ALPHABET.indexOf(ch);
  if (!Number.isSafeInteger(n)) throw new Error("קוד ארוך מדי");
  return n;
}
```

```ts save=solutions/P28-E07.ts
// P28-E07
// "חישוב על גב מעטפה": מסדרי גודל, לא דיוק. יום = 86,400 שניות ≈ 100,000
export interface Inputs {
  dailyActiveUsers: number;
  requestsPerUserPerDay: number;
  writeRatio: number; // חלק הבקשות שכותבות (0..1)
  bytesPerWrite: number;
  peakFactor: number; // השיא פי כמה מהממוצע
  retentionDays: number;
}

export function estimate(i: Inputs) {
  const perDay = i.dailyActiveUsers * i.requestsPerUserPerDay;
  const avgRps = perDay / 86_400;
  const writesPerDay = perDay * i.writeRatio;
  const storageBytes = writesPerDay * i.bytesPerWrite * i.retentionDays;
  return {
    avgRps: Math.round(avgRps),
    peakRps: Math.round(avgRps * i.peakFactor),
    writesPerSecond: Math.round(writesPerDay / 86_400),
    storageGB: Math.round(storageBytes / 1e9),
  };
}
```

```ts save=solutions/P28-E08.ts
// P28-E08
// שני אלגוריתמים של load balancer: round robin (בתורות) ו־least connections (למי שהכי פנוי)
export function roundRobin(servers: readonly string[]) {
  if (servers.length === 0) throw new Error("אין שרתים");
  let next = 0;
  return () => servers[next++ % servers.length];
}

export function leastConnections(servers: readonly string[]) {
  if (servers.length === 0) throw new Error("אין שרתים");
  const active = new Map(servers.map((s) => [s, 0]));
  return {
    // בתיקו — הראשון ברשימה (סדר קבוע)
    acquire(): string {
      let best = servers[0];
      for (const s of servers) if (active.get(s)! < active.get(best)!) best = s;
      active.set(best, active.get(best)! + 1);
      return best;
    },
    release(server: string) {
      const n = active.get(server);
      if (n === undefined) throw new Error(`שרת לא מוכר: ${server}`);
      if (n > 0) active.set(server, n - 1);
    },
    active: (server: string) => active.get(server) ?? 0,
  };
}
```

```ts save=solutions/P28-E09.ts
// P28-E09
// load balancer עם health checks: אחרי failThreshold כישלונות ברצף — השרת יוצא מהסבב; אחרי recoverAfterMs — מקבל הזדמנות
export function createBalancer(servers: readonly string[], { failThreshold = 3, recoverAfterMs = 10_000, now = () => Date.now() } = {}) {
  const fails = new Map(servers.map((s) => [s, 0]));
  const downSince = new Map<string, number>();
  let next = 0;

  const isUp = (s: string) => {
    const since = downSince.get(s);
    if (since === undefined) return true;
    if (now() - since >= recoverAfterMs) {
      downSince.delete(s); // הזדמנות שנייה; כישלון אחד נוסף יוריד שוב
      fails.set(s, failThreshold - 1);
      return true;
    }
    return false;
  };

  return {
    pick(): string | null {
      for (let i = 0; i < servers.length; i++) {
        const s = servers[next++ % servers.length];
        if (isUp(s)) return s;
      }
      return null; // כולם למטה — 503 ללקוח
    },
    report(server: string, ok: boolean) {
      if (ok) {
        fails.set(server, 0);
        return;
      }
      const n = (fails.get(server) ?? 0) + 1;
      fails.set(server, n);
      if (n >= failThreshold && !downSince.has(server)) downSince.set(server, now());
    },
    healthy: () => servers.filter((s) => !downSince.has(s) || now() - downSince.get(s)! >= recoverAfterMs),
  };
}
```

```ts save=solutions/P28-E11.ts
// P28-E11
// cursor "אטום": הלקוח לא יודע (ולא צריך לדעת) מה בפנים. base64url של JSON — ובדיקה כשהוא חוזר
export interface Cursor {
  createdAt: string; // ISO
  id: number;
}

export function encodeCursor(c: Cursor): string {
  return Buffer.from(JSON.stringify([c.createdAt, c.id])).toString("base64url");
}

export function decodeCursor(raw: string): Cursor {
  try {
    const [createdAt, id] = JSON.parse(Buffer.from(raw, "base64url").toString("utf8"));
    if (typeof createdAt !== "string" || Number.isNaN(Date.parse(createdAt)) || !Number.isSafeInteger(id) || id < 1) throw new Error();
    return { createdAt, id };
  } catch {
    throw new Error("cursor לא תקין");
  }
}

// keyset: "אחרי השורה הזו" במיון (createdAt DESC, id DESC). בלי OFFSET — מהיר בכל עמוד, ולא מדלג/מכפיל כשנוספות שורות
export function nextPage<T extends Cursor>(rows: readonly T[], limit: number, after?: string) {
  const sorted = [...rows].sort((a, b) => (a.createdAt < b.createdAt ? 1 : a.createdAt > b.createdAt ? -1 : b.id - a.id));
  const c = after ? decodeCursor(after) : null;
  const start = c ? sorted.findIndex((r) => r.createdAt < c.createdAt || (r.createdAt === c.createdAt && r.id < c.id)) : 0;
  const page = start === -1 ? [] : sorted.slice(start, start + limit);
  const last = page[page.length - 1];
  const hasMore = start !== -1 && start + limit < sorted.length;
  return { data: page, nextCursor: hasMore && last ? encodeCursor(last) : null };
}
```

```ts save=solutions/P28-E12.ts
// P28-E12
// retry עם exponential backoff ו־jitter: לא כל הלקוחות חוזרים באותה שנייה (thundering herd)
export class HttpError extends Error {
  status: number;
  constructor(status: number, message = `HTTP ${status}`) {
    super(message);
    this.status = status;
  }
}

// מה שווה לנסות שוב: רשת, 5xx, 429. 4xx אחר — הבקשה שגויה, ניסיון נוסף ייכשל באותה דרך
export const isRetryable = (err: unknown) => !(err instanceof HttpError) || err.status >= 500 || err.status === 429;

export async function retry<T>(
  fn: (attempt: number) => Promise<T>,
  {
    retries = 3,
    baseMs = 100,
    maxMs = 5000,
    random = Math.random,
    sleep = (ms: number) => new Promise<void>((r) => setTimeout(r, ms)),
  }: { retries?: number; baseMs?: number; maxMs?: number; random?: () => number; sleep?: (ms: number) => Promise<void> } = {},
): Promise<T> {
  for (let attempt = 0; ; attempt++) {
    try {
      return await fn(attempt);
    } catch (err) {
      if (attempt >= retries || !isRetryable(err)) throw err; // לא ישנים אחרי הניסיון האחרון
      const cap = Math.min(maxMs, baseMs * 2 ** attempt);
      await sleep(Math.floor(random() * cap)); // "full jitter": בין 0 לתקרה
    }
  }
}
```

```ts save=solutions/P28-E13.ts
// P28-E13
// consistent hashing: השרתים והמפתחות על אותו "מעגל" של מספרים; מפתח שייך לשרת הראשון אחריו (עם עטיפה).
// הוספה/הסרה של שרת מזיזה רק את המפתחות שלידו — כ־1/n, לא כמעט הכול כמו ב־hash % n
import { createHash } from "node:crypto";

export function hash32(s: string): number {
  return createHash("md5").update(s).digest().readUInt32BE(0);
}

export function createRing({ vnodes = 100 }: { vnodes?: number } = {}) {
  let points: { at: number; node: string }[] = []; // ממוין לפי at
  const nodes = new Set<string>();

  return {
    add(node: string) {
      if (nodes.has(node)) return;
      nodes.add(node);
      for (let i = 0; i < vnodes; i++) points.push({ at: hash32(`${node}#${i}`), node });
      points.sort((a, b) => a.at - b.at || (a.node < b.node ? -1 : 1));
    },
    remove(node: string) {
      if (!nodes.delete(node)) return;
      points = points.filter((p) => p.node !== node);
    },
    get(key: string): string | null {
      if (points.length === 0) return null;
      const h = hash32(key);
      // חיפוש בינארי: הנקודה הראשונה שה־at שלה >= h; אין כזו — עוטפים להתחלה
      let lo = 0;
      let hi = points.length;
      while (lo < hi) {
        const mid = (lo + hi) >>> 1;
        if (points[mid].at < h) lo = mid + 1;
        else hi = mid;
      }
      return points[lo === points.length ? 0 : lo].node;
    },
    nodes: () => [...nodes],
  };
}
```

```ts save=solutions/P28-E15.ts
// P28-E15
// circuit breaker: שירות שנופל שוב ושוב — מפסיקים לקרוא לו לזמן מה (נכשלים מיד), ואז מנסים בזהירות
export type BreakerState = "closed" | "open" | "half-open";

export function createBreaker<T>(call: () => Promise<T>, { failureThreshold = 5, openMs = 30_000, now = () => Date.now() } = {}) {
  let state: BreakerState = "closed";
  let failures = 0;
  let openedAt = 0;
  let trialRunning = false;

  return {
    get state(): BreakerState {
      if (state === "open" && now() - openedAt >= openMs) state = "half-open";
      return state;
    },
    async exec(): Promise<T> {
      const s = this.state;
      if (s === "open" || (s === "half-open" && trialRunning)) throw new Error("circuit open");
      if (s === "half-open") trialRunning = true; // רק בקשת ניסיון אחת עוברת
      try {
        const result = await call();
        state = "closed";
        failures = 0;
        return result;
      } catch (err) {
        failures++;
        if (s === "half-open" || failures >= failureThreshold) {
          state = "open";
          openedAt = now();
        }
        throw err;
      } finally {
        if (s === "half-open") trialRunning = false;
      }
    },
  };
}
```

```ts save=solutions/P28-E16.ts
// כמה מפתחות זזים כשמוסיפים שרת? hash % n מזיז כמעט הכול; consistent hashing — כ־1/n
import { createHash } from "node:crypto";

export const hash32 = (s: string) => createHash("md5").update(s).digest().readUInt32BE(0);
export const moduloShard = (key: string, n: number) => hash32(key) % n;

export function movedFraction(keys: readonly string[], before: (k: string) => string | number, after: (k: string) => string | number): number {
  if (keys.length === 0) return 0;
  let moved = 0;
  for (const k of keys) if (before(k) !== after(k)) moved++;
  return moved / keys.length;
}

// makeRing — טבעת consistent hashing (כמו E13): { add(node), get(key) }
export function compareResharding(
  keys: readonly string[],
  from: number,
  to: number,
  makeRing: () => { add(node: string): void; get(key: string): string | null },
) {
  const before = makeRing();
  const after = makeRing();
  for (let i = 0; i < from; i++) before.add(`db${i}`);
  for (let i = 0; i < to; i++) after.add(`db${i}`);
  return {
    modulo: movedFraction(keys, (k) => moduloShard(k, from), (k) => moduloShard(k, to)),
    ring: movedFraction(keys, (k) => before.get(k)!, (k) => after.get(k)!),
  };
}
```

```ts save=solutions/P28-E18.ts
// P28-E18
// batching: הרבה בקשות "תביא את X" באותו רגע ← שאילתה אחת עם כולם (כמו DataLoader מול N+1)
export function createBatcher<K, V>(
  loadMany: (keys: K[]) => Promise<Map<K, V>>,
  { maxBatch = 100, waitMs = 0 }: { maxBatch?: number; waitMs?: number } = {},
) {
  let queue: { key: K; resolve: (v: V | undefined) => void; reject: (e: unknown) => void }[] = [];
  let timer: ReturnType<typeof setTimeout> | null = null;

  const flush = () => {
    if (timer) clearTimeout(timer);
    timer = null;
    const batch = queue;
    queue = [];
    const keys = [...new Set(batch.map((b) => b.key))];
    loadMany(keys).then(
      (result) => batch.forEach((b) => b.resolve(result.get(b.key))),
      (err) => batch.forEach((b) => b.reject(err)),
    );
  };

  return (key: K) =>
    new Promise<V | undefined>((resolve, reject) => {
      queue.push({ key, resolve, reject });
      if (new Set(queue.map((q) => q.key)).size >= maxBatch) flush();
      else if (!timer) timer = setTimeout(flush, waitMs);
    });
}
```

```ts save=solutions/P28-E19.ts
export function createFeed({ celebrityThreshold = 1000, timelineCap = 500 }: { celebrityThreshold?: number; timelineCap?: number } = {}) {
  const followers = new Map<string, Set<string>>();
  const following = new Map<string, Set<string>>();
  const timelines = new Map<string, Post[]>(); // החדש ראשון

    follow(user: string, target: string) {

      (postsBy.get(author) ?? postsBy.set(author, []).get(author)!).unshift(p);
```

```ts save=solutions/P28-E20.ts
const ALPHABET = "0123456789abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ";

export function encode(n: number): string {
  if (!Number.isSafeInteger(n) || n < 0) throw new Error("רק מספר שלם אי־שלילי");
  if (n === 0) return "0"; // תקלה 1: בלי זה, 0 ← "" (הלולאה לא רצה), וקוד ריק לא מתפענח
  let out = "";
  while (n > 0) {
    out = ALPHABET[n % 62] + out; // תקלה 2: היה out += … ← הספרות יוצאות הפוך (הקטנה ראשונה)
    n = Math.floor(n / 62); // תקלה 3: היה n / 62 ← שבר, והלולאה ממשיכה עם 0.0something עד שמתקבל אפס
  }
  return out;
}

export function decode(code: string): number {
  if (!/^[0-9a-zA-Z]+$/.test(code)) throw new Error("קוד לא תקין");
  let n = 0;
  for (const ch of code) n = n * 62 + ALPHABET.indexOf(ch);
  return n;
}
```

```ts save=solutions/P28-E23.ts
// P28-E23
export function createBreaker<T>(call: () => Promise<T>, { failureThreshold = 3, openMs = 10_000, now = () => Date.now() } = {}) {
  let state: "closed" | "open" | "half-open" = "closed";
  let failures = 0;
  let openedAt = 0;
  return {
    get state() {
      if (state === "open" && now() - openedAt >= openMs) state = "half-open";
      return state;
    },
    async exec(): Promise<T> {
      const s = this.state;
      // תקלה 2: הבדיקה הייתה רק אחרי שהקריאה נכשלה ← כשהמעגל פתוח, כל בקשה עדיין פנתה לשירות המת (וחיכתה ל־timeout שלו)
      if (s === "open") throw new Error("circuit open");
      try {
        const result = await call();
        failures = 0; // תקלה 1: הצלחה לא איפסה את המונה ← 3 כישלונות "מפוזרים" לאורך יום פותחים את המעגל
        state = "closed";
        return result;
      } catch (err) {
        failures++;
        // תקלה 3: התנאי היה רק s === "closed" ← כישלון של בקשת הניסיון ב־half-open השאיר אותו half-open, וכל הבקשות עברו לשירות
        if (s === "half-open" || failures >= failureThreshold) {
          state = "open";
          openedAt = now();
        }
        throw err;
      }
    },
  };
}
```

```ts save=solutions/P28-E24.ts
// P28-E24
// מזהה בסגנון Snowflake (טוויטר): 64 ביט = 41 זמן (מילישניות מ־epoch משלנו) | 10 worker | 12 מונה.
// כל שרת מייצר לבד, בלי לתאם, והמזהים ממוינים בערך לפי זמן. BigInt — כי Number שומר רק 53 ביט
const EPOCH = 1_767_225_600_000n; // 2026-01-01T00:00:00Z

export function createIdGenerator(workerId: number, { now = () => Date.now() }: { now?: () => number } = {}) {
  if (!Number.isInteger(workerId) || workerId < 0 || workerId > 1023) throw new Error("workerId בין 0 ל־1023");
  let lastMs = -1;
  let seq = 0;
  return {
    next(): bigint {
      let ms = now();
      if (ms < lastMs) throw new Error(`השעון חזר אחורה ב־${lastMs - ms}ms — מסרבים לייצר מזהים כפולים`);
      if (ms === lastMs) {
        seq = (seq + 1) & 4095;
        if (seq === 0) {
          while (ms <= lastMs) ms = now(); // 4096 מזהים באותה מילישנייה — מחכים לבאה
        }
      } else {
        seq = 0;
      }
      lastMs = ms;
      return ((BigInt(ms) - EPOCH) << 22n) | (BigInt(workerId) << 12n) | BigInt(seq);
    },
  };
}

export function parseId(id: bigint) {
  return {
    time: new Date(Number((id >> 22n) + EPOCH)).toISOString(),
    workerId: Number((id >> 12n) & 1023n),
    seq: Number(id & 4095n),
  };
}
```

```ts save=solutions/P28-E25.ts
// P28-E25
// Bloom filter: "בטוח לא קיים" או "אולי קיים" — במעט זיכרון. אין false negatives; false positives בשיעור שבוחרים.
// m ביטים ו־k פונקציות hash, מחושבים מגודל צפוי n ושיעור טעות p
import { createHash } from "node:crypto";

export function createBloom(expected: number, falsePositiveRate: number) {
  const m = Math.ceil((-expected * Math.log(falsePositiveRate)) / Math.LN2 ** 2);
  const k = Math.max(1, Math.round((m / expected) * Math.LN2));
  const bits = new Uint8Array(Math.ceil(m / 8));

  // double hashing: שני hashים מ־sha256, ומהם k מיקומים (h1 + i·h2)
  const positions = (item: string) => {
    const d = createHash("sha256").update(item).digest();
    const h1 = d.readUInt32BE(0);
    const h2 = (d.readUInt32BE(4) | 1) >>> 0; // ">>> 0": בלי זה, | הופך מספר גדול לשלילי — ומיקום שלילי ב־Uint8Array פשוט לא נכתב
    return Array.from({ length: k }, (_, i) => (h1 + i * h2) % m);
  };

  return {
    m,
    k,
    bytes: bits.length,
    add(item: string) {
      for (const p of positions(item)) bits[p >> 3] |= 1 << (p & 7);
    },
    mightContain(item: string): boolean {
      return positions(item).every((p) => (bits[p >> 3] & (1 << (p & 7))) !== 0);
    },
  };
}
```

```ts run save=solutions/P28-E26.ts
// P28-E26 — quorum: כמה העתקים צריכים להסכים
export type Replica = { up: boolean; version: number; value: string | null };

export function isStrong(n: number, r: number, w: number): boolean {
  return r + w > n;
}

export function quorumWrite(replicas: Replica[], w: number, value: string, version: number): boolean {
  let acks = 0;
  for (const replica of replicas) {
    if (!replica.up) continue;
    replica.version = version;
    replica.value = value;
    acks++;
  }
  return acks >= w;
}

export function quorumRead(replicas: Replica[], r: number): { value: string | null; version: number; repaired: number } {
  const answered = replicas.filter((rep) => rep.up).slice(0, r);
  if (answered.length < r) throw new Error(`רק ${answered.length} העתקים זמינים, צריך ${r}`);
  const newest = answered.reduce((best, rep) => (rep.version > best.version ? rep : best));
  let repaired = 0;
  for (const rep of answered) {
    if (rep.version < newest.version) {
      rep.version = newest.version;
      rep.value = newest.value;
      repaired++;
    }
  }
  return { value: newest.value, version: newest.version, repaired };
}

const replicas: Replica[] = [
  { up: true, version: 0, value: null },
  { up: true, version: 0, value: null },
  { up: true, version: 0, value: null },
];
console.log("N=3, R=2, W=2 strong?", isStrong(3, 2, 2), "· R=1, W=1?", isStrong(3, 1, 1));
replicas[2].up = false;
console.log("write v1:", quorumWrite(replicas, 2, "address: Haifa", 1));
replicas[2].up = true;
replicas[0].up = false;
console.log("read:", quorumRead(replicas, 2));
console.log(replicas.map((r) => `${r.version}:${r.value}`));
```

```ts run save=solutions/P28-E28.ts
// P28-E28 — hedged request: בקשה שנייה אם הראשונה מתעכבת
export function hedged<T>(fn: (attempt: number) => Promise<T>, delayMs: number): Promise<T> {
  return new Promise<T>((resolve, reject) => {
    let settled = false;
    let failures = 0;
    let started = 0;
    let timer: ReturnType<typeof setTimeout> | undefined;

    const start = () => {
      started++;
      fn(started).then(
        (value) => {
          if (settled) return;
          settled = true;
          clearTimeout(timer);
          resolve(value);
        },
        (err) => {
          if (settled) return;
          failures++;
          if (started === 1) {
            clearTimeout(timer);
            start(); // הראשונה נכשלה לפני שהשנייה התחילה — מתחילים אותה מיד
          } else if (failures === 2) {
            settled = true;
            reject(err);
          }
        },
      );
    };

    start();
    timer = setTimeout(() => {
      if (!settled && started === 1) start();
    }, delayMs);
  });
}

async function demo() {
  const sleep = (ms: number) => new Promise((r) => setTimeout(r, ms));
  const t0 = Date.now();
  const slowFirst = await hedged(async (attempt) => {
    await sleep(attempt === 1 ? 300 : 40);
    return `תשובה מניסיון ${attempt}`;
  }, 50);
  console.log(slowFirst, "· לפני שהראשונה סיימה?", Date.now() - t0 < 300);
  const fastFirst = await hedged(async (attempt) => `תשובה מניסיון ${attempt}`, 50);
  console.log(fastFirst);
  try {
    await hedged(async (attempt) => {
      throw new Error(`נכשל ${attempt}`);
    }, 50);
  } catch (err) {
    console.log((err as Error).message);
  }
}
demo();
```

```ts save=solutions/P28-E34.ts
export function formatBytes(bytes: number): string {
  if (!Number.isFinite(bytes) || bytes < 0) throw new RangeError(`גודל לא תקין: ${bytes}`);

  let i = 0;
```

```ts save=solutions/P28-E35.ts
type Step = { name: string; ms: number; count?: number };

export function latencyBudget(steps: Step[]) {
  const costs = steps.map((s) => s.ms * (s.count ?? 1));
  const total = costs.reduce((a, b) => a + b, 0);
  if (total === 0) return { totalMs: 0, slowest: null, slowestShare: 0 };
  let best = 0;
  for (let i = 1; i < costs.length; i++) if (costs[i] > costs[best]) best = i;
  return {
    totalMs: Math.round(total * 100) / 100,
    slowest: steps[best].name,
    slowestShare: Math.round((costs[best] / total) * 100),
  };
}
```

```ts save=solutions/P28-E37.ts
export function parseRequirements(text: string) {
  const functional: string[] = [];
  const nonFunctional: string[] = [];
  text.split("\n").forEach((raw, i) => {
    const line = raw.trim();
    if (line === "" || line.startsWith("#")) return;
    const m = line.match(/^(nf|f):(.*)$/i);
    if (!m) throw new Error(`שורה ${i + 1}: חסרה קידומת F: או NF:`);
    const body = m[2].trim();
    if (!body) throw new Error(`שורה ${i + 1}: דרישה ריקה`);
    (m[1].toLowerCase() === "f" ? functional : nonFunctional).push(body);
  });
  return { functional, nonFunctional };
}
```

```ts save=solutions/P28-E44.ts
// P28-E44: שיא מתוך לוגים של JSON
export function trafficFromLogs(lines: string[]) {
  const perMinute = new Map<number, number>();
  let skipped = 0;
  let total = 0;
  let min = Infinity;
  let max = -Infinity;
  for (const line of lines) {
    let ms = NaN;
    try {
      const rec = JSON.parse(line);
      if (typeof rec?.time === "string") ms = Date.parse(rec.time);
    } catch {
      // לא JSON
    }
    if (Number.isNaN(ms)) {
      skipped++;
      continue;
    }
    const minute = Math.floor(ms / 60_000);
    perMinute.set(minute, (perMinute.get(minute) ?? 0) + 1);
    total++;
    min = Math.min(min, minute);
    max = Math.max(max, minute);
  }
  if (total === 0) return { total: 0, skipped, minutes: 0, avgPerMinute: 0, peakPerMinute: 0, peakFactor: 0 };
  const minutes = max - min + 1;
  const avg = total / minutes;
  const peak = Math.max(...perMinute.values());
  return { total, skipped, minutes, avgPerMinute: Math.round(avg * 10) / 10, peakPerMinute: peak, peakFactor: Math.round((peak / avg) * 10) / 10 };
}
```

```ts save=solutions/P28-E45.ts
// P28-E45: Promise.all — כמה זמן באמת?
type Step = { name: string; ms: number; after?: string[] };

export function pageTime(steps: Step[]) {
  const byName = new Map(steps.map((s) => [s.name, s]));
  const done = new Map<string, number>();
  const via = new Map<string, string | null>();
  const visiting = new Set<string>();

  function finish(name: string): number {
    const known = done.get(name);
    if (known !== undefined) return known;
    const step = byName.get(name);
    if (!step) throw new Error(`תלות לא מוכרת: ${name}`);
    if (visiting.has(name)) throw new Error(`מעגל תלויות דרך ${name}`);
    visiting.add(name);
    let start = 0;
    let from: string | null = null;
    for (const dep of step.after ?? []) {
      const t = finish(dep);
      if (t > start) {
        start = t;
        from = dep;
      }
    }
    visiting.delete(name);
    done.set(name, start + step.ms);
    via.set(name, from);
    return start + step.ms;
  }

  let parallel = 0;
  let last: string | null = null;
  for (const s of steps) {
    const t = finish(s.name);
    if (last === null || t > parallel) {
      parallel = t;
      last = s.name;
    }
  }
  const critical: string[] = [];
  for (let cur = last; cur !== null; cur = via.get(cur) ?? null) critical.unshift(cur);
  return { sequential: steps.reduce((a, s) => a + s.ms, 0), parallel, critical };
}
```

```ts save=solutions/P28-E51.ts
// P28-E51: round robin עם משקלים (smooth weighted, כמו nginx)
export function weightedRoundRobin(servers: { name: string; weight: number }[]) {
  if (servers.length === 0) throw new Error("אין שרתים");
  for (const s of servers) {
    if (!Number.isInteger(s.weight) || s.weight <= 0) throw new Error(`משקל לא תקין ל־${s.name}: ${s.weight}`);
  }
  const total = servers.reduce((sum, s) => sum + s.weight, 0);
  const current = servers.map(() => 0);
  return () => {
    let best = 0;
    for (let i = 0; i < servers.length; i++) {
      current[i] += servers[i].weight;
      if (current[i] > current[best]) best = i;
    }
    current[best] -= total;
    return servers[best].name;
  };
}
```

```ts save=solutions/P28-E55.ts
import type { Request, Response, NextFunction } from "express";

export function timeout(ms: number) {
  return (req: Request, res: Response, next: NextFunction) => {
    const timer = setTimeout(() => {
      if (res.headersSent) return;
      (req as Request & { timedOut?: boolean }).timedOut = true;
      res.set("Retry-After", "1").status(503).json({ error: "timeout" });

    res.on("finish", () => clearTimeout(timer));
    res.on("close", () => clearTimeout(timer));
    next();
  };
}
```

```ts save=solutions/P28-E63.ts
import { createHash } from "node:crypto";

const ALPHABET = "0123456789abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ";

function base62(n: number): string {
  let s = "";
  do {
    s = ALPHABET[n % 62] + s;
    n = Math.floor(n / 62);
  } while (n > 0);
  return s;
}

export function hashCode(url: string, length = 7): string {
  const n = createHash("sha256").update(url).digest().readUIntBE(0, 6);
  return base62(n % 62 ** length).padStart(length, "0");
}

export function codeFor(url: string, taken: Set<string>, length = 7): string {
  for (let i = 0; i < 5; i++) {
    const code = hashCode(i === 0 ? url : `${url}#${i}`, length);
    if (!taken.has(code)) return code;
  }
  throw new Error(`לא נמצא קוד פנוי ל־${url} אחרי 5 ניסיונות`);
}
```

```ts save=solutions/P28-E64.ts
  private total = 0;
  private maxPending: number;

  constructor(maxPending = 10_000) {
```

```ts save=solutions/P28-E65.ts
export function normalizeUrl(raw: string): string {
  const url = new URL(raw.trim());
  if (url.protocol !== "http:" && url.protocol !== "https:") throw new Error(`פרוטוקול לא נתמך: ${url.protocol}`);
  url.hash = "";
  const pairs: [string, string][] = [];
  for (const [name, value] of url.searchParams) {
    if (name.startsWith("utm_") || name === "fbclid") continue;
    pairs.push([name, value]);
  }
  pairs.sort((a, b) => (a[0] < b[0] ? -1 : a[0] > b[0] ? 1 : 0));
  url.search = new URLSearchParams(pairs).toString();
  return url.href;
}
```

```ts save=solutions/P28-E69.ts
// P28-E69: redirect עם cache ו־X-Cache
import express from "express";

export type Link = { url: string; expiresAt: number | null };
export type Store = { get(code: string): Promise<Link | null> };
export type Cache = {
  get(key: string): Promise<string | null>;
  set(key: string, value: string, ttlSeconds: number): Promise<void>;
};

export function redirectApp(deps: { store: Store; cache: Cache; now: () => number }) {
  const { store, cache, now } = deps;

  async function lookup(code: string): Promise<{ link: Link | null; hit: boolean }> {
    const key = `link:${code}`;
    try {
      const cached = await cache.get(key);
      if (cached !== null) return { link: JSON.parse(cached), hit: true };
    } catch {
      // cache למטה ← כמו miss
    }
    const link = await store.get(code);
    if (link) {
      try {
        await cache.set(key, JSON.stringify(link), 3600);
      } catch {
        // לא נורא — בפעם הבאה ננסה שוב
      }
    }
    return { link, hit: false };
  }

  const app = express();
  app.get("/:code", async (req, res) => {
    const { link, hit } = await lookup(req.params.code);
    res.set("X-Cache", hit ? "HIT" : "MISS");
    if (!link) return res.status(404).json({ error: "not found" });
    if (link.expiresAt !== null && link.expiresAt <= now()) return res.status(410).json({ error: "expired" });
    res.redirect(302, link.url);
  });
  return app;
}
```

```ts save=solutions/P28-E80.ts
export function keysetQuery(table: string, opts: { limit: number; after?: { createdAt: string; id: number } }) {
  if (!/^[a-z_][a-z0-9_]*$/.test(table)) throw new Error(`שם טבלה לא תקין: ${table}`);
  if (!Number.isInteger(opts.limit) || opts.limit < 1 || opts.limit > 100) throw new RangeError("limit בין 1 ל־100");
  const values: (string | number)[] = [];
  let where = "";
  if (opts.after) {
    values.push(opts.after.createdAt, opts.after.id);
    where = " WHERE (created_at, id) < ($1, $2)";
  }
  values.push(opts.limit);
  return { text: `SELECT * FROM ${table}${where} ORDER BY created_at DESC, id DESC LIMIT $${values.length}`, values };
}
```

```ts save=solutions/P28-E88.ts
type User = { id: string; followers: number; postsPerDay: number };

export function fanoutCost(users: User[], threshold: number) {
  let pushAll = 0;
  let hybrid = 0;
  const celebrities: string[] = [];
  for (const u of users) {
    const writes = u.followers * u.postsPerDay;
    pushAll += writes;
    if (u.followers >= threshold) celebrities.push(u.id);
    else hybrid += writes;
  }
  const saved = pushAll === 0 ? 0 : Math.round(((pushAll - hybrid) / pushAll) * 100);
  return { pushAll, hybrid, celebrities, saved };
}
```

```ts save=solutions/P28-E89.ts
// P28-E89: מיזוג פידים
type Post = { id: number; at: number };

const newer = (a: Post, b: Post) => a.at > b.at || (a.at === b.at && a.id > b.id);

export function mergeFeeds(lists: Post[][], limit: number): Post[] {
  const pos = lists.map(() => 0);
  const seen = new Set<number>();
  const out: Post[] = [];
  while (out.length < limit) {
    let best = -1;
    for (let i = 0; i < lists.length; i++) {
      const head = lists[i][pos[i]];
      if (head && (best === -1 || newer(head, lists[best][pos[best]]))) best = i;
    }
    if (best === -1) break;
    const post = lists[best][pos[best]++];
    if (seen.has(post.id)) continue;
    seen.add(post.id);
    out.push(post);
  }
  return out;
}
```

```ts save=solutions/P28-E90.ts
// P28-E90: פיד עם תקרה
export class FeedStore {
  private feeds = new Map<string, number[]>();
  private cap: number;

  constructor(cap = 500) {
    this.cap = cap;
  }

  add(userId: string, postId: number): void {
    let list = this.feeds.get(userId);
    if (!list) {
      list = [];
      this.feeds.set(userId, list);
    }
    if (list.includes(postId)) return;
    list.unshift(postId);
    if (list.length > this.cap) list.pop();
  }

  read(userId: string, limit = 20, before?: number): number[] {
    const list = this.feeds.get(userId) ?? [];
    let start = 0;
    if (before !== undefined) {
      const i = list.indexOf(before);
      if (i === -1) return [];
      start = i + 1;
    }
    return list.slice(start, start + limit);
  }

  size(userId: string): number {
    return this.feeds.get(userId)?.length ?? 0;
  }
}
```

```ts save=solutions/P28-E91.ts
export function checkDesignDoc(markdown: string) {
  const content = new Map<string, number>();
  let current: string | null = null;
  for (const line of markdown.split("\n")) {
    if (line.startsWith("## ")) {
      const title = line.slice(3).trim();
```

```ts save=solutions/P28-E92.ts
// P28-E92: חיבורים למסד — האם נכנסים?
export function poolBudget(p: { instances: number; poolPerInstance: number; workers?: number; poolPerWorker?: number; maxConnections: number; reserved?: number }) {
```

```ts save=solutions/P28-E93.ts
// P28-E93: מלאי בחלקים
export function splitStock(total: number, parts: number): number[] {
  if (!Number.isInteger(parts) || parts <= 0) throw new RangeError("parts חייב להיות שלם חיובי");
  if (!Number.isInteger(total) || total < 0) throw new RangeError("total חייב להיות שלם אי־שלילי");
  const base = Math.floor(total / parts);
  const extra = total % parts;
  return Array.from({ length: parts }, (_, i) => base + (i < extra ? 1 : 0));
}

export function takeOne(shards: number[], start: number): number {
  const n = shards.length;
  for (let k = 0; k < n; k++) {
    const i = (start + k) % n;
    if (shards[i] > 0) {
      shards[i]--;
      return i;
    }
  }
  return -1;
}
```

```ts save=solutions/P28-E94.ts
// P28-E94: ADR בקובץ
const STATUSES = ["proposed", "accepted", "superseded", "rejected"];

export function adrMarkdown(r: { number: number; title: string; status: string; date: string; context: string; decision: string; consequences: string[] }): string {
  if (!STATUSES.includes(r.status)) throw new Error(`סטטוס לא מוכר: ${r.status}`);
  if (!/^\d{4}-\d{2}-\d{2}$/.test(r.date)) throw new Error(`תאריך לא תקין: ${r.date}`);
  if (!r.title.trim() || !r.decision.trim()) throw new Error("חסרים כותרת או החלטה");
  if (r.consequences.length === 0) throw new Error("צריך לפחות השלכה אחת");
  const lines = [
    `# ADR-${String(r.number).padStart(4, "0")}: ${r.title.trim()}`,
    "",
    `- סטטוס: ${r.status}`,
    `- תאריך: ${r.date}`,
    "",
    "## הקשר",
    r.context,
    "",
    "## החלטה",
    r.decision,
    "",
    "## השלכות",
    ...r.consequences.map((c) => `- ${c}`),
    "",
  ];
  return lines.join("\n");
}
```

```ts save=solutions/P28-E98.ts
// P28-E98: API של פיד ב־Express
import express from "express";

type Post = { id: number; author: string; text: string; at: number };

export function feedApp(opts: { celebrityThreshold: number; now: () => number }) {
  const followers = new Map<string, Set<string>>();
  const following = new Map<string, Set<string>>();
  const feeds = new Map<string, number[]>();
  const posts = new Map<number, Post>();
  const byAuthor = new Map<string, number[]>();
  let nextId = 1;
  let feedWrites = 0;

  const setOf = (m: Map<string, Set<string>>, k: string) => {
    if (!m.has(k)) m.set(k, new Set());
    return m.get(k)!;
  };
  const isCelebrity = (u: string) => (followers.get(u)?.size ?? 0) >= opts.celebrityThreshold;

  const app = express();
  app.use(express.json());

  app.post("/follow", (req, res) => {
    const { follower, followee } = req.body ?? {};
    if (typeof follower !== "string" || typeof followee !== "string" || !follower || !followee || follower === followee) {
      return res.status(400).json({ error: "follower ו־followee שונים" });
    }
    setOf(followers, followee).add(follower);
    setOf(following, follower).add(followee);
    res.status(204).end();
  });

  app.post("/posts", (req, res) => {
    const { author, text } = req.body ?? {};
    if (typeof author !== "string" || !author || typeof text !== "string" || !text.trim()) {
      return res.status(400).json({ error: "author ו־text חובה" });
    }
    const post: Post = { id: nextId++, author, text, at: opts.now() };
    posts.set(post.id, post);
    if (!byAuthor.has(author)) byAuthor.set(author, []);
    byAuthor.get(author)!.push(post.id);
    if (!isCelebrity(author)) {
      for (const f of followers.get(author) ?? []) {
        if (!feeds.has(f)) feeds.set(f, []);
        feeds.get(f)!.push(post.id);
        feedWrites++;
      }
    }
    res.status(201).json(post);
  });

  app.get("/feed/:user", (req, res) => {
    const user = req.params.user;
    const limit = Math.min(Math.max(Number(req.query.limit) || 20, 1), 100);
    const found = new Map<number, Post>();
    for (const id of feeds.get(user) ?? []) found.set(id, posts.get(id)!);
    for (const f of following.get(user) ?? []) {
      if (!isCelebrity(f)) continue;
      for (const id of byAuthor.get(f) ?? []) found.set(id, posts.get(id)!);
    }
    const items = [...found.values()].sort((a, b) => b.at - a.at || b.id - a.id).slice(0, limit);
    res.json({ items });
  });

  app.get("/stats", (req, res) => res.json({ feedWrites }));
  return app;
}
```

```ts save=solutions/P28-E99.ts
// P28-E99: fan-out בתור, בחבילות
export function fanoutJobs(postId: number, followers: string[], batchSize = 1000) {
  if (!Number.isInteger(batchSize) || batchSize <= 0) throw new RangeError("batchSize חייב להיות שלם חיובי");
  const unique = [...new Set(followers)];
  const jobs: { name: string; data: { postId: number; followers: string[] }; opts: { jobId: string } }[] = [];
  for (let i = 0; i < unique.length; i += batchSize) {
    jobs.push({
      name: "fanout",
      data: { postId, followers: unique.slice(i, i + batchSize) },
      opts: { jobId: `fanout:${postId}:${i / batchSize}` },
    });
  }
  return jobs;
}
```

```ts save=solutions/P28-E100.ts
// P28-E100: עמדנו ביעדים?
export function verdict(results: { ms: number; status: number }[], targets: { p99Ms: number; maxErrorRate: number }) {
  const count = results.length;
  if (count === 0) return { count: 0, p50: 0, p99: 0, errorRate: 0, pass: false, reasons: ["no results"] };
  const sorted = results.map((r) => r.ms).sort((a, b) => a - b);
  const pct = (n: number) => sorted[Math.ceil((n / 100) * count) - 1];
  const p50 = pct(50);
  const p99 = pct(99);
  const errorRate = results.filter((r) => r.status >= 500).length / count;
  const asPct = (x: number) => Number((x * 100).toFixed(2));
  const reasons: string[] = [];
  if (p99 > targets.p99Ms) reasons.push(`p99 ${p99}ms > ${targets.p99Ms}ms`);
  if (errorRate > targets.maxErrorRate) reasons.push(`errors ${asPct(errorRate)}% > ${asPct(targets.maxErrorRate)}%`);
  return { count, p50, p99, errorRate, pass: reasons.length === 0, reasons };
}
```
