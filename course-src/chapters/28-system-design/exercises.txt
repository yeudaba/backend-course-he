```ts save=solutions/P28-E06.ts
// P28-E06
// base62: ספרות 0-9, a-z, A-Z. כמו בסיס 10 או 16 — רק עם 62 "ספרות". 7 תווים = 62⁷ ≈ 3.5 טריליון קודים
const ALPHABET = "0123456789abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ";

export function encode(n: number): string {
  if (!Number.isSafeInteger(n) || n < 0) throw new Error("רק מספר שלם אי־שלילי");
  if (n === 0) return "0";
  let out = "";
  while (n > 0) {
    out = ALPHABET[n % 62] + out;
    n = Math.floor(n / 62);
  }
  return out;
}

export function decode(code: string): number {
  if (!/^[0-9a-zA-Z]+$/.test(code)) throw new Error("קוד לא תקין");
  let n = 0;
  for (const ch of code) n = n * 62 + ALPHABET.indexOf(ch);
  if (!Number.isSafeInteger(n)) throw new Error("קוד ארוך מדי");
  return n;
}
```

```ts save=solutions/P28-E07.ts
// P28-E07
// "חישוב על גב מעטפה": מסדרי גודל, לא דיוק. יום = 86,400 שניות ≈ 100,000
export interface Inputs {
  dailyActiveUsers: number;
  requestsPerUserPerDay: number;
  writeRatio: number; // חלק הבקשות שכותבות (0..1)
  bytesPerWrite: number;
  peakFactor: number; // השיא פי כמה מהממוצע
  retentionDays: number;
}

export function estimate(i: Inputs) {
  const perDay = i.dailyActiveUsers * i.requestsPerUserPerDay;
  const avgRps = perDay / 86_400;
  const writesPerDay = perDay * i.writeRatio;
  const storageBytes = writesPerDay * i.bytesPerWrite * i.retentionDays;
  return {
    avgRps: Math.round(avgRps),
    peakRps: Math.round(avgRps * i.peakFactor),
    writesPerSecond: Math.round(writesPerDay / 86_400),
    storageGB: Math.round(storageBytes / 1e9),
  };
}
```

```ts save=solutions/P28-E08.ts
// P28-E08
// שני אלגוריתמים של load balancer: round robin (בתורות) ו־least connections (למי שהכי פנוי)
export function roundRobin(servers: readonly string[]) {
  if (servers.length === 0) throw new Error("אין שרתים");
  let next = 0;
  return () => servers[next++ % servers.length];
}

export function leastConnections(servers: readonly string[]) {
  if (servers.length === 0) throw new Error("אין שרתים");
  const active = new Map(servers.map((s) => [s, 0]));
  return {
    // בתיקו — הראשון ברשימה (סדר קבוע)
    acquire(): string {
      let best = servers[0];
      for (const s of servers) if (active.get(s)! < active.get(best)!) best = s;
      active.set(best, active.get(best)! + 1);
      return best;
    },
    release(server: string) {
      const n = active.get(server);
      if (n === undefined) throw new Error(`שרת לא מוכר: ${server}`);
      if (n > 0) active.set(server, n - 1);
    },
    active: (server: string) => active.get(server) ?? 0,
  };
}
```

```ts save=solutions/P28-E09.ts
// P28-E09
// load balancer עם health checks: אחרי failThreshold כישלונות ברצף — השרת יוצא מהסבב; אחרי recoverAfterMs — מקבל הזדמנות
export function createBalancer(servers: readonly string[], { failThreshold = 3, recoverAfterMs = 10_000, now = () => Date.now() } = {}) {
  const fails = new Map(servers.map((s) => [s, 0]));
  const downSince = new Map<string, number>();
  let next = 0;

  const isUp = (s: string) => {
    const since = downSince.get(s);
    if (since === undefined) return true;
    if (now() - since >= recoverAfterMs) {
      downSince.delete(s); // הזדמנות שנייה; כישלון אחד נוסף יוריד שוב
      fails.set(s, failThreshold - 1);
      return true;
    }
    return false;
  };

  return {
    pick(): string | null {
      for (let i = 0; i < servers.length; i++) {
        const s = servers[next++ % servers.length];
        if (isUp(s)) return s;
      }
      return null; // כולם למטה — 503 ללקוח
    },
    report(server: string, ok: boolean) {
      if (ok) {
        fails.set(server, 0);
        return;
      }
      const n = (fails.get(server) ?? 0) + 1;
      fails.set(server, n);
      if (n >= failThreshold && !downSince.has(server)) downSince.set(server, now());
    },
    healthy: () => servers.filter((s) => !downSince.has(s) || now() - downSince.get(s)! >= recoverAfterMs),
  };
}
```

```ts save=solutions/P28-E11.ts
// P28-E11
// cursor "אטום": הלקוח לא יודע (ולא צריך לדעת) מה בפנים. base64url של JSON — ובדיקה כשהוא חוזר
export interface Cursor {
  createdAt: string; // ISO
  id: number;
}

export function encodeCursor(c: Cursor): string {
  return Buffer.from(JSON.stringify([c.createdAt, c.id])).toString("base64url");
}

export function decodeCursor(raw: string): Cursor {
  try {
    const [createdAt, id] = JSON.parse(Buffer.from(raw, "base64url").toString("utf8"));
    if (typeof createdAt !== "string" || Number.isNaN(Date.parse(createdAt)) || !Number.isSafeInteger(id) || id < 1) throw new Error();
    return { createdAt, id };
  } catch {
    throw new Error("cursor לא תקין");
  }
}

// keyset: "אחרי השורה הזו" במיון (createdAt DESC, id DESC). בלי OFFSET — מהיר בכל עמוד, ולא מדלג/מכפיל כשנוספות שורות
export function nextPage<T extends Cursor>(rows: readonly T[], limit: number, after?: string) {
  const sorted = [...rows].sort((a, b) => (a.createdAt < b.createdAt ? 1 : a.createdAt > b.createdAt ? -1 : b.id - a.id));
  const c = after ? decodeCursor(after) : null;
  const start = c ? sorted.findIndex((r) => r.createdAt < c.createdAt || (r.createdAt === c.createdAt && r.id < c.id)) : 0;
  const page = start === -1 ? [] : sorted.slice(start, start + limit);
  const last = page[page.length - 1];
  const hasMore = start !== -1 && start + limit < sorted.length;
  return { data: page, nextCursor: hasMore && last ? encodeCursor(last) : null };
}
```

```ts save=solutions/P28-E12.ts
// P28-E12
// retry עם exponential backoff ו־jitter: לא כל הלקוחות חוזרים באותה שנייה (thundering herd)
export class HttpError extends Error {
  status: number;
  constructor(status: number, message = `HTTP ${status}`) {
    super(message);
    this.status = status;
  }
}

// מה שווה לנסות שוב: רשת, 5xx, 429. 4xx אחר — הבקשה שגויה, ניסיון נוסף ייכשל באותה דרך
export const isRetryable = (err: unknown) => !(err instanceof HttpError) || err.status >= 500 || err.status === 429;

export async function retry<T>(
  fn: (attempt: number) => Promise<T>,
  {
    retries = 3,
    baseMs = 100,
    maxMs = 5000,
    random = Math.random,
    sleep = (ms: number) => new Promise<void>((r) => setTimeout(r, ms)),
  }: { retries?: number; baseMs?: number; maxMs?: number; random?: () => number; sleep?: (ms: number) => Promise<void> } = {},
): Promise<T> {
  for (let attempt = 0; ; attempt++) {
    try {
      return await fn(attempt);
    } catch (err) {
      if (attempt >= retries || !isRetryable(err)) throw err; // לא ישנים אחרי הניסיון האחרון
      const cap = Math.min(maxMs, baseMs * 2 ** attempt);
      await sleep(Math.floor(random() * cap)); // "full jitter": בין 0 לתקרה
    }
  }
}
```

```ts save=solutions/P28-E13.ts
// P28-E13
// consistent hashing: השרתים והמפתחות על אותו "מעגל" של מספרים; מפתח שייך לשרת הראשון אחריו (עם עטיפה).
// הוספה/הסרה של שרת מזיזה רק את המפתחות שלידו — כ־1/n, לא כמעט הכול כמו ב־hash % n
import { createHash } from "node:crypto";

export function hash32(s: string): number {
  return createHash("md5").update(s).digest().readUInt32BE(0);
}

export function createRing({ vnodes = 100 }: { vnodes?: number } = {}) {
  let points: { at: number; node: string }[] = []; // ממוין לפי at
  const nodes = new Set<string>();

  return {
    add(node: string) {
      if (nodes.has(node)) return;
      nodes.add(node);
      for (let i = 0; i < vnodes; i++) points.push({ at: hash32(`${node}#${i}`), node });
      points.sort((a, b) => a.at - b.at || (a.node < b.node ? -1 : 1));
    },
    remove(node: string) {
      if (!nodes.delete(node)) return;
      points = points.filter((p) => p.node !== node);
    },
    get(key: string): string | null {
      if (points.length === 0) return null;
      const h = hash32(key);
      // חיפוש בינארי: הנקודה הראשונה שה־at שלה >= h; אין כזו — עוטפים להתחלה
      let lo = 0;
      let hi = points.length;
      while (lo < hi) {
        const mid = (lo + hi) >>> 1;
        if (points[mid].at < h) lo = mid + 1;
        else hi = mid;
      }
      return points[lo === points.length ? 0 : lo].node;
    },
    nodes: () => [...nodes],
  };
}
```

```ts save=solutions/P28-E15.ts
// P28-E15
// circuit breaker: שירות שנופל שוב ושוב — מפסיקים לקרוא לו לזמן מה (נכשלים מיד), ואז מנסים בזהירות
export type BreakerState = "closed" | "open" | "half-open";

export function createBreaker<T>(call: () => Promise<T>, { failureThreshold = 5, openMs = 30_000, now = () => Date.now() } = {}) {
  let state: BreakerState = "closed";
  let failures = 0;
  let openedAt = 0;
  let trialRunning = false;

  return {
    get state(): BreakerState {
      if (state === "open" && now() - openedAt >= openMs) state = "half-open";
      return state;
    },
    async exec(): Promise<T> {
      const s = this.state;
      if (s === "open" || (s === "half-open" && trialRunning)) throw new Error("circuit open");
      if (s === "half-open") trialRunning = true; // רק בקשת ניסיון אחת עוברת
      try {
        const result = await call();
        state = "closed";
        failures = 0;
        return result;
      } catch (err) {
        failures++;
        if (s === "half-open" || failures >= failureThreshold) {
          state = "open";
          openedAt = now();
        }
        throw err;
      } finally {
        if (s === "half-open") trialRunning = false;
      }
    },
  };
}
```

```ts save=solutions/P28-E16.ts
// כמה מפתחות זזים כשמוסיפים שרת? hash % n מזיז כמעט הכול; consistent hashing — כ־1/n
import { createHash } from "node:crypto";

export const hash32 = (s: string) => createHash("md5").update(s).digest().readUInt32BE(0);
export const moduloShard = (key: string, n: number) => hash32(key) % n;

export function movedFraction(keys: readonly string[], before: (k: string) => string | number, after: (k: string) => string | number): number {
  if (keys.length === 0) return 0;
  let moved = 0;
  for (const k of keys) if (before(k) !== after(k)) moved++;
  return moved / keys.length;
}

// makeRing — טבעת consistent hashing (כמו E13): { add(node), get(key) }
export function compareResharding(
  keys: readonly string[],
  from: number,
  to: number,
  makeRing: () => { add(node: string): void; get(key: string): string | null },
) {
  const before = makeRing();
  const after = makeRing();
  for (let i = 0; i < from; i++) before.add(`db${i}`);
  for (let i = 0; i < to; i++) after.add(`db${i}`);
  return {
    modulo: movedFraction(keys, (k) => moduloShard(k, from), (k) => moduloShard(k, to)),
    ring: movedFraction(keys, (k) => before.get(k)!, (k) => after.get(k)!),
  };
}
```

```ts save=solutions/P28-E18.ts
// P28-E18
// batching: הרבה בקשות "תביא את X" באותו רגע ← שאילתה אחת עם כולם (כמו DataLoader מול N+1)
export function createBatcher<K, V>(
  loadMany: (keys: K[]) => Promise<Map<K, V>>,
  { maxBatch = 100, waitMs = 0 }: { maxBatch?: number; waitMs?: number } = {},
) {
  let queue: { key: K; resolve: (v: V | undefined) => void; reject: (e: unknown) => void }[] = [];
  let timer: ReturnType<typeof setTimeout> | null = null;

  const flush = () => {
    if (timer) clearTimeout(timer);
    timer = null;
    const batch = queue;
    queue = [];
    const keys = [...new Set(batch.map((b) => b.key))];
    loadMany(keys).then(
      (result) => batch.forEach((b) => b.resolve(result.get(b.key))),
      (err) => batch.forEach((b) => b.reject(err)),
    );
  };

  return (key: K) =>
    new Promise<V | undefined>((resolve, reject) => {
      queue.push({ key, resolve, reject });
      if (new Set(queue.map((q) => q.key)).size >= maxBatch) flush();
      else if (!timer) timer = setTimeout(flush, waitMs);
    });
}
```

```ts save=solutions/P28-E19.ts
export function createFeed({ celebrityThreshold = 1000, timelineCap = 500 }: { celebrityThreshold?: number; timelineCap?: number } = {}) {
  const followers = new Map<string, Set<string>>();
  const following = new Map<string, Set<string>>();
  const timelines = new Map<string, Post[]>(); // החדש ראשון

    follow(user: string, target: string) {

      (postsBy.get(author) ?? postsBy.set(author, []).get(author)!).unshift(p);
```

```ts save=solutions/P28-E20.ts
const ALPHABET = "0123456789abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ";

export function encode(n: number): string {
  if (!Number.isSafeInteger(n) || n < 0) throw new Error("רק מספר שלם אי־שלילי");
  if (n === 0) return "0"; // תקלה 1: בלי זה, 0 ← "" (הלולאה לא רצה), וקוד ריק לא מתפענח
  let out = "";
  while (n > 0) {
    out = ALPHABET[n % 62] + out; // תקלה 2: היה out += … ← הספרות יוצאות הפוך (הקטנה ראשונה)
    n = Math.floor(n / 62); // תקלה 3: היה n / 62 ← שבר, והלולאה ממשיכה עם 0.0something עד שמתקבל אפס
  }
  return out;
}

export function decode(code: string): number {
  if (!/^[0-9a-zA-Z]+$/.test(code)) throw new Error("קוד לא תקין");
  let n = 0;
  for (const ch of code) n = n * 62 + ALPHABET.indexOf(ch);
  return n;
}
```

```ts save=solutions/P28-E22.ts
    },
  };
}
```

```ts save=solutions/P28-E23.ts
// P28-E23
export function createBreaker<T>(call: () => Promise<T>, { failureThreshold = 3, openMs = 10_000, now = () => Date.now() } = {}) {
  let state: "closed" | "open" | "half-open" = "closed";
  let failures = 0;
  let openedAt = 0;
  return {
    get state() {
      if (state === "open" && now() - openedAt >= openMs) state = "half-open";
      return state;
    },
    async exec(): Promise<T> {
      const s = this.state;
      // תקלה 2: הבדיקה הייתה רק אחרי שהקריאה נכשלה ← כשהמעגל פתוח, כל בקשה עדיין פנתה לשירות המת (וחיכתה ל־timeout שלו)
      if (s === "open") throw new Error("circuit open");
      try {
        const result = await call();
        failures = 0; // תקלה 1: הצלחה לא איפסה את המונה ← 3 כישלונות "מפוזרים" לאורך יום פותחים את המעגל
        state = "closed";
        return result;
      } catch (err) {
        failures++;
        // תקלה 3: התנאי היה רק s === "closed" ← כישלון של בקשת הניסיון ב־half-open השאיר אותו half-open, וכל הבקשות עברו לשירות
        if (s === "half-open" || failures >= failureThreshold) {
          state = "open";
          openedAt = now();
        }
        throw err;
      }
    },
  };
}
```
@mistakes

```ts save=solutions/P28-E24.ts
// P28-E24
// מזהה בסגנון Snowflake (טוויטר): 64 ביט = 41 זמן (מילישניות מ־epoch משלנו) | 10 worker | 12 מונה.
// כל שרת מייצר לבד, בלי לתאם, והמזהים ממוינים בערך לפי זמן. BigInt — כי Number שומר רק 53 ביט
const EPOCH = 1_767_225_600_000n; // 2026-01-01T00:00:00Z

export function createIdGenerator(workerId: number, { now = () => Date.now() }: { now?: () => number } = {}) {
  if (!Number.isInteger(workerId) || workerId < 0 || workerId > 1023) throw new Error("workerId בין 0 ל־1023");
  let lastMs = -1;
  let seq = 0;
  return {
    next(): bigint {
      let ms = now();
      if (ms < lastMs) throw new Error(`השעון חזר אחורה ב־${lastMs - ms}ms — מסרבים לייצר מזהים כפולים`);
      if (ms === lastMs) {
        seq = (seq + 1) & 4095;
        if (seq === 0) {
          while (ms <= lastMs) ms = now(); // 4096 מזהים באותה מילישנייה — מחכים לבאה
        }
      } else {
        seq = 0;
      }
      lastMs = ms;
      return ((BigInt(ms) - EPOCH) << 22n) | (BigInt(workerId) << 12n) | BigInt(seq);
    },
  };
}

export function parseId(id: bigint) {
  return {
    time: new Date(Number((id >> 22n) + EPOCH)).toISOString(),
    workerId: Number((id >> 12n) & 1023n),
    seq: Number(id & 4095n),
  };
}
```

```ts save=solutions/P28-E25.ts
// P28-E25
// Bloom filter: "בטוח לא קיים" או "אולי קיים" — במעט זיכרון. אין false negatives; false positives בשיעור שבוחרים.
// m ביטים ו־k פונקציות hash, מחושבים מגודל צפוי n ושיעור טעות p
import { createHash } from "node:crypto";

export function createBloom(expected: number, falsePositiveRate: number) {
  const m = Math.ceil((-expected * Math.log(falsePositiveRate)) / Math.LN2 ** 2);
  const k = Math.max(1, Math.round((m / expected) * Math.LN2));
  const bits = new Uint8Array(Math.ceil(m / 8));

  // double hashing: שני hashים מ־sha256, ומהם k מיקומים (h1 + i·h2)
  const positions = (item: string) => {
    const d = createHash("sha256").update(item).digest();
    const h1 = d.readUInt32BE(0);
    const h2 = (d.readUInt32BE(4) | 1) >>> 0; // ">>> 0": בלי זה, | הופך מספר גדול לשלילי — ומיקום שלילי ב־Uint8Array פשוט לא נכתב
    return Array.from({ length: k }, (_, i) => (h1 + i * h2) % m);
  };

  return {
    m,
    k,
    bytes: bits.length,
    add(item: string) {
      for (const p of positions(item)) bits[p >> 3] |= 1 << (p & 7);
    },
    mightContain(item: string): boolean {
      return positions(item).every((p) => (bits[p >> 3] & (1 << (p & 7))) !== 0);
    },
  };
}
```
<p>על מיליון שמות עם 1% טעות: 1.2MB ו־7 hashים. מדדתי: 1.002% false positives. Set עם אותם מיליון שמות תפס כ־35MB. "לא קיים" — בטוח, ואפשר לענות מיד; "אולי" — שואלים את המסד. כך עושים Cassandra ו־RocksDB (לדלג על קבצים שבטוח אין בהם את המפתח), וכך דפדפנים בדקו פעם אתרים זדוניים.</p>
