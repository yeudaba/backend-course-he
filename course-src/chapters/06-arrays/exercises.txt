```js run save=solutions/P06-E06.js
// P06-E06: גישה בטוחה לאיברים
function lastItem(arr) {
  return arr[arr.length - 1];
}

function firstAndLast(arr) {
  if (arr.length === 0) {
    return [];
  }
  return [arr[0], arr[arr.length - 1]];
}

function safeGet(arr, index, fallback = null) {
  if (index < 0 || index >= arr.length) {
    return fallback;
  }
  return arr[index];
}
```

```js run save=solutions/P06-E09.js
// P06-E09: split ו־join
function parseNumbers(text) {
  if (text.trim() === "") {
    return [];
  }
  const result = [];
  for (const part of text.split(",")) {
    result.push(Number(part.trim()));
  }
  return result;
}

function humanList(items) {
  if (items.length === 0) {
    return "";
  }
  if (items.length === 1) {
    return items[0];
  }
  return `${items.slice(0, -1).join(", ")} ו־${items[items.length - 1]}`;
}
```

```js run save=solutions/P06-E13.js
// P06-E13: ספירה וערכים ייחודיים
function countOccurrences(arr, value) {
  let count = 0;
  for (const x of arr) {
    if (x === value) {
      count++;
    }
  }
  return count;
}

function unique(arr) {
  const result = [];
  for (const x of arr) {
    if (!result.includes(x)) {
      result.push(x);
    }
  }
  return result;
}

function mostFrequent(arr) {
  let best;
  let bestCount = 0;
  for (const value of unique(arr)) {
    const count = countOccurrences(arr, value);
    if (count > bestCount) {
      best = value;
      bestCount = count;
    }
  }
  return best;
}

console.log(unique([3, 1, 3, 2, 1]), mostFrequent(["a", "b", "b", "a", "b"]));

module.exports = { countOccurrences, unique, mostFrequent };
```
```out
[ 3, 1, 2 ] b
```

```js run save=solutions/P06-E17.js
// P06-E17: min, max, average עם reduce
function stats(nums) {
  if (nums.length === 0) {
    return null;
  }
  const min = nums.reduce((m, n) => (n < m ? n : m), nums[0]);
  const max = nums.reduce((m, n) => (n > m ? n : m), nums[0]);
  const total = nums.reduce((s, n) => s + n, 0);
  const average = Math.round((total / nums.length) * 100) / 100;
  return [min, max, average];
}
```

```js run save=solutions/P06-E18.js
// P06-E18: topN וחציון
function topN(nums, n) {
  return [...nums].sort((a, b) => b - a).slice(0, n);
}

function median(nums) {
  if (nums.length === 0) {
    return null;
  }
  const sorted = [...nums].sort((a, b) => a - b);
  const mid = Math.floor(sorted.length / 2);
  if (sorted.length % 2 === 1) {
    return sorted[mid];
  }
  return (sorted[mid - 1] + sorted[mid]) / 2;
}
```

```js run save=solutions/P06-E19.js
// P06-E19: חלוקה לעמודים
function chunk(arr, size) {
  const result = [];
  for (let i = 0; i < arr.length; i += size) {
    result.push(arr.slice(i, i + size));
  }
  return result;
}

function getPage(arr, page, pageSize) {
  if (page < 1) {
    return [];
  }
  return arr.slice((page - 1) * pageSize, page * pageSize);
}

function totalPages(count, pageSize) {
  return Math.max(1, Math.ceil(count / pageSize));
}

console.log(chunk([1, 2, 3, 4, 5], 2), getPage(["a", "b", "c", "d", "e"], 2, 2), totalPages(0, 10));

module.exports = { chunk, getPage, totalPages };
```

```js run save=solutions/P06-E20.js
function summarize(sales) {
  if (sales.length === 0) {
    return "אין מכירות";
  }
  let total = 0;
  let best = sales[0];
  for (const s of sales) {

}
```

```js run save=solutions/P06-E25.js
// P06-E25: twoSum ו־mergeSorted
function twoSum(nums, target) {
  for (let i = 0; i < nums.length; i++) {
    for (let j = i + 1; j < nums.length; j++) {
      if (nums[i] + nums[j] === target) {
        return [i, j];
      }
    }
  }
  return null;
}

function mergeSorted(a, b) {
  const result = [];
  let i = 0;
  let j = 0;
  while (i < a.length && j < b.length) {
    if (a[i] <= b[j]) {
      result.push(a[i]);
      i++;
    } else {
      result.push(b[j]);
      j++;
    }
  }
  return [...result, ...a.slice(i), ...b.slice(j)];
}
```

<p>🔗 <strong>איך זה מתחבר:</strong> בפרק 4 כתבת כמה צוברים באותה לולאה; בפרק 5 למדת ששמות טובים ופונקציות קטנות הם מה שהופך קוד לקריא. ה־<code>reduce</code> כאן "חכם" — אבל קשה לקרוא אותו ולמצוא בו באג. כמו שאמר השיעור: כשהלוגיקה מתארכת — לולאה.</p>
@hint1
<p>הצובר הוא מערך של שלושה מספרים. רק הזמנות חיוביות משנות אותו.</p>
@hint2
<p><code>a[0]</code> — סכום, <code>a[1]</code> — ספירה, <code>a[2]</code> — מקסימום.</p>
@solution
```js run
function summarizeOrders(orders) {
  let total = 0;
  let count = 0;
  let biggest = 0;
  for (const amount of orders) {
    if (amount <= 0) {
      continue;
    }
    total += amount;
    count++;
    if (amount > biggest) {
      biggest = amount;
    }
  }
  return [total, count, biggest];
}

const orders = [120, -5, 300, 0, 80];
const r = orders.reduce((a, o) => (o > 0 ? [a[0] + o, a[1] + 1, o > a[2] ? o : a[2]] : a), [0, 0, 0]);
console.log(r, summarizeOrders(orders));
```
```out
[ 500, 3, 300 ] [ 500, 3, 300 ]
```
<p>אותה תוצאה, אבל בגרסה עם הלולאה כל משתנה אומר מה הוא, ה־<code>continue</code> מסביר מה נזרק ולמה, והפונקציה אומרת מה היא מחשבת. (מקסימום שמתחיל מ־0 בסדר כאן — כי מגיעים אליו רק מספרים חיוביים.) בפרק 7 גם ה־<code>[total, count, biggest]</code> יהפוך לאובייקט עם שמות.</p>
@mistakes
<ul><li>לחשוב ש"שורה אחת" = "קוד טוב". המדד הוא כמה זמן לוקח לאדם אחר להבין אותו.</li></ul>
