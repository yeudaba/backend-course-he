@@ P25.5-E01 | 1 | מילון PostgreSQL ← MySQL | 15 | שיעור 1
@task
<p>לכל דבר שעבד ב־PostgreSQL בקורס — המקבילה ב־MySQL 8, או "אין":</p>
<ol>
<li><code>id integer GENERATED ALWAYS AS IDENTITY</code></li>
<li><code>INSERT … RETURNING id</code></li>
<li><code>ON CONFLICT (…) DO UPDATE</code></li>
<li><code>'a' || 'b'</code></li>
<li><code>ILIKE</code></li>
<li><code>boolean</code>, <code>jsonb</code>, <code>timestamptz</code></li>
<li><code>SELECT DISTINCT ON (…)</code></li>
<li><code>FULL OUTER JOIN</code></li>
<li><code>$1, $2</code> בשאילתה עם פרמטרים</li>
</ol>
@hint1
<p>רוב השפה זהה: SELECT, JOIN, GROUP BY, CTE, window functions (מ־8.0), CHECK (נאכף מ־8.0.16).</p>
@hint2
<p>שלושה מהם "קיימים, אבל עושים משהו אחר".</p>
@solution
<ol>
<li><code>INT UNSIGNED AUTO_INCREMENT PRIMARY KEY</code>.</li>
<li>אין RETURNING. ה־driver מחזיר <code>insertId</code> (mysql2), או <code>SELECT LAST_INSERT_ID()</code> באותו חיבור.</li>
<li><code>INSERT … AS new ON DUPLICATE KEY UPDATE col = new.col</code> — על <em>כל</em> מפתח ייחודי בטבלה, לא על אחד שבוחרים (E08).</li>
<li><strong>מלכודת:</strong> <code>||</code> ב־MySQL הוא OR לוגי — <code>'a' || 'b'</code> מחזיר 0. חיבור מחרוזות: <code>CONCAT('a', 'b')</code>.</li>
<li>אין ILIKE. אבל ברירת המחדל (<code>utf8mb4_0900_ai_ci</code>) כבר לא רגישה לרישיות <em>ולסימנים</em> — <code>LIKE</code> מספיק. לרגישות: <code>COLLATE utf8mb4_bin</code>.</li>
<li><code>BOOLEAN</code> הוא <code>TINYINT(1)</code> (חוזר כ־0/1); <code>JSON</code> (בינארי, דומה ל־jsonb); <code>DATETIME(3)</code> בלי אזור זמן — שומרים UTC בעצמך, או <code>TIMESTAMP</code> (ממיר לפי אזור הזמן של החיבור, טווח עד 2038).</li>
<li>אין. <code>ROW_NUMBER() OVER (PARTITION BY …)</code> בתת־שאילתה (E07, E20).</li>
<li>אין. <code>LEFT JOIN … UNION … RIGHT JOIN</code>.</li>
<li><code>?</code> (mysql2: <code>execute(sql, [a, b])</code>).</li>
</ol>
@mistakes
<ul><li>להעתיק שאילתה מ־PostgreSQL ולהאמין לה כי "לא הייתה שגיאה". <code>||</code> לא זורק — הוא מחזיר 0 בשקט.</li></ul>

@hint1

@mistakes

```ts save=solutions/P25.5-E09.ts
    ]);

      await conn.execute("UPDATE products SET stock = stock - ? WHERE id = ?", [item.quantity, item.productId]);
```

```ts save=solutions/P25.5-E14.ts
// תור עבודות על טבלה: כמה workers לוקחים עבודות במקביל, בלי שאותה עבודה תילקח פעמיים.

export async function claimJobs(pool: Pool, workerId: string, limit: number): Promise<number[]> {
  const conn = await pool.getConnection();

    await conn.beginTransaction();
    const [rows] = await conn.query<any[]>("SELECT id FROM jobs WHERE status = 'queued' ORDER BY id LIMIT ? FOR UPDATE SKIP LOCKED", [limit]);
```

```ts save=solutions/P25.5-E15.ts
export const RETRYABLE = new Set([1213, 1205]);

export async function withDeadlockRetry<T>(
  fn: (attempt: number) => Promise<T>,
  { retries = 3, sleep = (ms: number) => new Promise<void>((r) => setTimeout(r, ms)), random = Math.random } = {},

  }
```

```ts save=solutions/P25.5-E17.ts
// P25.5-E17
import type { Pool } from "mysql2/promise";

export class InProgress extends Error {}

// Idempotency ב־MySQL: INSERT IGNORE על מפתח ראשי — affectedRows 1 = "אני הראשון", 0 = כבר קיים.
// (ב־PostgreSQL: ON CONFLICT DO NOTHING RETURNING.) התוצאה נשמרת כ־JSON, והבקשה החוזרת מקבלת אותה
export async function once<T>(pool: Pool, key: string, fn: () => Promise<T>): Promise<{ result: T; replayed: boolean }> {
  const [ins] = await pool.execute<any>("INSERT IGNORE INTO idempotency (idem_key, status) VALUES (?, 'pending')", [key]);
  if (ins.affectedRows === 1) {
    try {
      const result = await fn();
      await pool.execute("UPDATE idempotency SET status = 'done', result = ? WHERE idem_key = ?", [JSON.stringify(result), key]);
      return { result, replayed: false };
    } catch (err) {
      await pool.execute("DELETE FROM idempotency WHERE idem_key = ?", [key]); // נכשל ← אפשר לנסות שוב עם אותו מפתח
      throw err;
    }
  }
  const [rows] = await pool.execute<any[]>("SELECT status, result FROM idempotency WHERE idem_key = ?", [key]);
  if (!rows[0] || rows[0].status !== "done") throw new InProgress(`בקשה ${key} עדיין בטיפול`);
  const stored = rows[0].result;
  return { result: (typeof stored === "string" ? JSON.parse(stored) : stored) as T, replayed: true };
}
```

```ts save=solutions/P25.5-E19.ts
// P25.5-E19
import type { Pool } from "mysql2/promise";

// כלי migrations בקטן. שני דברים שמיוחדים ל־MySQL:
// (1) DDL עושה COMMIT מרומז — אי אפשר לעטוף CREATE/ALTER בטרנזקציה ולבטל. migration שנכשל באמצע — נשאר באמצע.
// (2) GET_LOCK: נעילה בשם, לכל השרתים — כך שני שרתים שעולים יחד לא מריצים את אותו migration פעמיים
export async function migrate(pool: Pool, migrations: readonly { id: string; sql: string }[], { lockTimeoutSeconds = 10 } = {}): Promise<string[]> {
  const conn = await pool.getConnection();
  try {
    const [[lock]] = await conn.query<any[]>("SELECT GET_LOCK('schema_migrations', ?) AS ok", [lockTimeoutSeconds]);
    if (lock.ok !== 1) throw new Error("לא התקבלה נעילה ל־migrations");
    try {
      await conn.query("CREATE TABLE IF NOT EXISTS schema_migrations (id VARCHAR(100) PRIMARY KEY, applied_at DATETIME(3) NOT NULL DEFAULT CURRENT_TIMESTAMP(3))");
      const [done] = await conn.query<any[]>("SELECT id FROM schema_migrations");
      const applied = new Set(done.map((r) => r.id));
      const ran: string[] = [];
      for (const m of [...migrations].sort((a, b) => (a.id < b.id ? -1 : 1))) {
        if (applied.has(m.id)) continue;
        await conn.query(m.sql);
        await conn.execute("INSERT INTO schema_migrations (id) VALUES (?)", [m.id]);
        ran.push(m.id);
      }
      return ran;
    } finally {
      await conn.query("SELECT RELEASE_LOCK('schema_migrations')");
    }
  } finally {
    conn.release();
  }
}
```

```ts save=solutions/P25.5-E24.ts
import type { Pool } from "mysql2/promise";

export class EmptyCart extends Error {}
export class OutOfStock extends Error {
  productId: number;
  constructor(productId: number) {
    super(`אין מספיק מלאי למוצר ${productId}`);
    this.productId = productId;
  }
}

// "הזמן" של חנות הקפה — ב־MySQL. טרנזקציה אחת: מפתח idempotency, עגלה, נעילת המוצרים (FOR UPDATE, לפי id —
// סדר קבוע מונע deadlocks), בדיקת מלאי, הזמנה ופריטים עם המחיר של עכשיו, הורדת מלאי, ריקון העגלה
export async function createOrderFromCart(pool: Pool, customerId: number, idemKey: string): Promise<{ orderId: number; totalCents: number; replayed: boolean }> {
  const conn = await pool.getConnection();
  try {
    await conn.beginTransaction();
    // אותו מפתח בבקשה מקבילה: ה־INSERT מחכה לנעילה על המפתח, וממשיך כשהראשונה מסתיימת
    const [ins] = await conn.execute<any>("INSERT IGNORE INTO order_requests (idem_key, customer_id) VALUES (?, ?)", [idemKey, customerId]);
    if (ins.affectedRows === 0) {
      // FOR SHARE — קריאה "נועלת" רואה את הגרסה המחויבת האחרונה (לא את ה־snapshot של REPEATABLE READ)
      const [[prev]] = await conn.execute<any[]>("SELECT r.customer_id, r.order_id, o.total FROM order_requests r LEFT JOIN orders o ON o.id = r.order_id WHERE r.idem_key = ? FOR SHARE", [idemKey]);
      await conn.commit();
      if (!prev || prev.customer_id !== customerId || prev.order_id === null) throw new Error("Idempotency-Key בשימוש");
      return { orderId: prev.order_id, totalCents: Math.round(Number(prev.total) * 100), replayed: true };
    }
    const [cart] = await conn.execute<any[]>("SELECT product_id, quantity FROM cart_items WHERE customer_id = ? ORDER BY product_id", [customerId]);
    if (cart.length === 0) throw new EmptyCart("העגלה ריקה");
    const [products] = await conn.query<any[]>("SELECT id, price, stock FROM products WHERE id IN (?) ORDER BY id FOR UPDATE", [cart.map((c) => c.product_id)]);
    const byId = new Map(products.map((p) => [p.id, p]));
    let totalCents = 0;
    for (const c of cart) {
      const p = byId.get(c.product_id);
      if (!p || p.stock < c.quantity) throw new OutOfStock(c.product_id);
      totalCents += Math.round(Number(p.price) * 100) * c.quantity;
    }
    const [order] = await conn.execute<any>("INSERT INTO orders (customer_id, status, total) VALUES (?, 'pending_payment', ?)", [customerId, (totalCents / 100).toFixed(2)]);
    const orderId: number = order.insertId;
    await conn.query("INSERT INTO order_items (order_id, product_id, quantity, unit_price) VALUES ?", [cart.map((c) => [orderId, c.product_id, c.quantity, byId.get(c.product_id).price])]);
    for (const c of cart) await conn.execute("UPDATE products SET stock = stock - ? WHERE id = ?", [c.quantity, c.product_id]);
    await conn.execute("DELETE FROM cart_items WHERE customer_id = ?", [customerId]);
    await conn.execute("UPDATE order_requests SET order_id = ? WHERE idem_key = ?", [orderId, idemKey]);
    await conn.commit();
    return { orderId, totalCents, replayed: false };
  } catch (err) {
    await conn.rollback(); // גם שורת ה־idempotency מתבטלת — אפשר לנסות שוב עם אותו מפתח
    throw err;
  } finally {
    conn.release();
  }
}
```

```ts save=solutions/P25.5-E25.ts
    batches++;

  }

}
```

```ts run save=solutions/P25.5-E27.ts
// P25.5-E27 — INSERT מרובה שורות, עם upsert
type Value = string | number | boolean | null;

const IDENT = /^[a-z_][a-z0-9_]*$/;

function ident(name: string): string {
  if (!IDENT.test(name)) throw new Error(`שם לא חוקי: ${name}`);
  return `\`${name}\``;
}

export function buildBulkInsert(
  table: string,
  rows: Record<string, Value>[],
  options: { add?: string[]; set?: string[] } = {},
): { sql: string; params: Value[] } {
  if (rows.length === 0) throw new Error("אין שורות");
  const columns = Object.keys(rows[0]);
  const key = [...columns].sort().join(",");
  for (const row of rows) {
    if (Object.keys(row).sort().join(",") !== key) throw new Error("לכל השורות צריכות להיות אותן עמודות");
  }

  const placeholders = `(${columns.map(() => "?").join(", ")})`;
  let sql = `INSERT INTO ${ident(table)} (${columns.map(ident).join(", ")}) VALUES ${rows.map(() => placeholders).join(", ")}`;
  const params = rows.flatMap((row) => columns.map((c) => row[c]));

  const updates = [
    ...(options.add ?? []).map((c) => `${ident(c)} = ${ident(table)}.${ident(c)} + new.${ident(c)}`),
    ...(options.set ?? []).map((c) => `${ident(c)} = new.${ident(c)}`),
  ];
  if (updates.length > 0) sql += ` AS new ON DUPLICATE KEY UPDATE ${updates.join(", ")}`;
  return { sql, params };
}

console.log(buildBulkInsert("cart_items", [
  { customer_id: 1, product_id: 3, quantity: 2 },
  { product_id: 4, customer_id: 1, quantity: 1 },
], { add: ["quantity"] }));
try {
  buildBulkInsert("products; DROP TABLE orders", [{ id: 1 }]);
} catch (err) {
  console.log((err as Error).message);
}
```

```ts run save=solutions/P25.5-E28.ts
// P25.5-E28 — שאילתה מ־PostgreSQL ל־MySQL: placeholders ושמות
export function pgToMysql(sql: string, params: unknown[]): { sql: string; params: unknown[] } {
  const out: unknown[] = [];
  // מחרוזת '...' (עם '' בפנים) — כמו שהיא; "שם" ← `שם`; $n ← ?
  const converted = sql.replace(/'(?:[^']|'')*'|"([^"]+)"|\$(\d+)/g, (match, name, n) => {
    if (name !== undefined) return `\`${name}\``;
    if (n !== undefined) {
      const i = Number(n);
      if (i < 1 || i > params.length) throw new RangeError(`$${n} — אין פרמטר כזה (יש ${params.length})`);
      out.push(params[i - 1]);
      return "?";
    }
    return match;
  });
  return { sql: converted, params: out };
}

console.log(pgToMysql("SELECT 'it''s $1 \"x\"' AS note, $1 AS v", [7]));
try {
  pgToMysql("SELECT $3", [1, 2]);
} catch (err) {
  console.log((err as Error).name, (err as Error).message);
}
```

```ts run save=solutions/P25.5-E42.ts
  return { chars: [...value].length, bytes: Buffer.byteLength(value, "utf8") };
```

```ts save=solutions/P25.5-E61.ts
// P25.5-E61: העברה בין ארנקים — בלי deadlock
import type { Pool, RowDataPacket } from "mysql2/promise";

export async function transfer(pool: Pool, fromId: number, toId: number, amount: number) {
  if (!Number.isInteger(amount) || amount <= 0) throw new RangeError(`סכום לא תקין: ${amount}`);
  if (fromId === toId) throw new Error("SAME_WALLET");
  const conn = await pool.getConnection();
  try {
    await conn.beginTransaction();
    // תמיד מהקטן לגדול — כך שתי העברות הפוכות לא נועלות "אחת מכל צד"
    const [rows] = await conn.query<RowDataPacket[]>(
      "SELECT customer_id, balance FROM wallets WHERE customer_id IN (?, ?) ORDER BY customer_id FOR UPDATE",
      [fromId, toId],
    );
    if (rows.length !== 2) throw new Error("WALLET_NOT_FOUND");
    const balance = new Map(rows.map((r) => [Number(r.customer_id), Number(r.balance)]));
    const from = balance.get(fromId)! - amount;
    const to = balance.get(toId)! + amount;
    if (from < 0) throw new Error("INSUFFICIENT_FUNDS");
    await conn.query("UPDATE wallets SET balance = ? WHERE customer_id = ?", [from, fromId]);
    await conn.query("UPDATE wallets SET balance = ? WHERE customer_id = ?", [to, toId]);
    await conn.commit();
    return { from, to };
  } catch (err) {
    await conn.rollback();
    throw err;
  } finally {
    conn.release();
  }
}
```

```ts save=solutions/P25.5-E62.ts
// P25.5-E62: מושב — עכשיו או אף פעם (NOWAIT)
import type { Pool, RowDataPacket } from "mysql2/promise";

const ER_LOCK_NOWAIT = 3572;

export async function reserveSeat(pool: Pool, seatId: number, customerId: number) {
  const conn = await pool.getConnection();
  try {
    await conn.beginTransaction();
    const [rows] = await conn.query<RowDataPacket[]>("SELECT reserved_by FROM seats WHERE id = ? FOR UPDATE NOWAIT", [seatId]);
    let result: "reserved" | "taken" | "not_found";
    if (rows.length === 0) result = "not_found";
    else if (rows[0].reserved_by !== null) result = "taken";
    else {
      await conn.query("UPDATE seats SET reserved_by = ? WHERE id = ?", [customerId, seatId]);
      result = "reserved";
    }
    await conn.commit();
    return result;
  } catch (err) {
    await conn.rollback();
    if ((err as { errno?: number }).errno === ER_LOCK_NOWAIT) return "busy" as const;
    throw err;
  } finally {
    conn.release();
  }
}
```

```ts save=solutions/P25.5-E63.ts
import type { Pool, RowDataPacket } from "mysql2/promise";

export async function salesReport(pool: Pool, between?: () => Promise<void>) {
  const conn = await pool.getConnection();
  try {
    await conn.query("START TRANSACTION READ ONLY, WITH CONSISTENT SNAPSHOT");

      "SELECT COUNT(*) AS orders, COALESCE(SUM(total), 0) AS revenue FROM orders WHERE status IN ('paid', 'shipped')",
```

```ts save=solutions/P25.5-E68.ts
// P25.5-E68: POST /orders עם mysql2 — בלי למכור יותר מדי
import { Router } from "express";
import type { Pool, ResultSetHeader, RowDataPacket } from "mysql2/promise";

const fail = (code: string) => ({ error: { code } });
const isPositiveInt = (v: unknown) => Number.isInteger(v) && (v as number) > 0;

export function ordersRouter(pool: Pool) {
  const router = Router();
  router.post("/orders", async (req, res, next) => {
    const { customerId, productId, quantity } = req.body ?? {};
    if (![customerId, productId, quantity].every(isPositiveInt)) {
      res.status(400).json(fail("VALIDATION_ERROR"));
      return;
    }
    const conn = await pool.getConnection();
    try {
      await conn.beginTransaction();
      const [upd] = await conn.query<ResultSetHeader>(
        "UPDATE products SET stock = stock - ? WHERE id = ? AND is_active AND stock >= ?",
        [quantity, productId, quantity],
      );
      if (upd.affectedRows === 0) {
        const [found] = await conn.query<RowDataPacket[]>("SELECT id FROM products WHERE id = ? AND is_active", [productId]);
        await conn.rollback();
        res.status(found.length ? 409 : 404).json(fail(found.length ? "OUT_OF_STOCK" : "PRODUCT_NOT_FOUND"));
        return;
      }
      const [[p]] = await conn.query<RowDataPacket[]>("SELECT price FROM products WHERE id = ?", [productId]);
      const [order] = await conn.query<ResultSetHeader>(
        "INSERT INTO orders (customer_id, status, total) VALUES (?, 'pending_payment', ? * ?)",
        [customerId, p.price, quantity],
      );
      await conn.query("INSERT INTO order_items (order_id, product_id, quantity, unit_price) VALUES (?, ?, ?, ?)", [order.insertId, productId, quantity, p.price]);
      await conn.commit();
      res.status(201).json({ orderId: order.insertId, totalCents: Math.round(Number(p.price) * 100) * quantity });
    } catch (err) {
      await conn.rollback();
      if ((err as { errno?: number }).errno === 1452) {
        res.status(404).json(fail("CUSTOMER_NOT_FOUND"));
        return;
      }
      next(err);
    } finally {
      conn.release();
    }
  });
  return router;
}
```

```ts save=solutions/P25.5-E69.ts
// P25.5-E69: מחיקת לקוח — בלי לשבור הזמנות
import type { Pool, ResultSetHeader } from "mysql2/promise";

export async function anonymizeCustomer(pool: Pool, customerId: number) {
  const conn = await pool.getConnection();
  try {
    await conn.beginTransaction();
    const [cart] = await conn.query<ResultSetHeader>("DELETE FROM cart_items WHERE customer_id = ?", [customerId]);
    const [upd] = await conn.query<ResultSetHeader>(
      "UPDATE customers SET name = 'Deleted user', email = CONCAT('deleted+', id, '@invalid.local'), city = '' WHERE id = ?",
      [customerId],
    );
    if (upd.affectedRows === 0) throw new Error("NOT_FOUND");
    await conn.commit();
    return { cartItemsRemoved: cart.affectedRows };
  } catch (err) {
    await conn.rollback();
    throw err;
  } finally {
    conn.release();
  }
}
```

```ts save=solutions/P25.5-E76.ts
import type { Pool, RowDataPacket } from "mysql2/promise";

export async function productsByIds(pool: Pool, ids: number[]) {
  const unique = [...new Set(ids)];
  if (unique.length === 0) return [];
  const [rows] = await pool.query<RowDataPacket[]>("SELECT id, name, price FROM products WHERE id IN (?)", [unique]);
  const byId = new Map(rows.map((r) => [Number(r.id), r]));
  return unique
    .filter((id) => byId.has(id))
    .map((id) => {

      return { id, name: String(r.name), priceCents: Math.round(Number(r.price) * 100) };
    });
}
```

```ts save=solutions/P25.5-E77.ts
import type { Pool, ResultSetHeader, RowDataPacket } from "mysql2/promise";

export async function markShipped(pool: Pool, orderId: number) {
  const [res] = await pool.execute<ResultSetHeader>(
    "UPDATE orders SET status = 'shipped' WHERE id = ? AND status IN ('paid', 'shipped')",
    [orderId],
  );
  if (res.changedRows === 1) return "shipped" as const;
  if (res.affectedRows === 1) return "already" as const;

  return rows.length ? ("invalid_status" as const) : ("not_found" as const);
}
```

```ts save=solutions/P25.5-E79.ts
import type { Pool, RowDataPacket } from "mysql2/promise";

const cents = (v: unknown) => Math.round(Number(v) * 100);

export async function orderWithItems(pool: Pool, orderId: number) {
  const [rows] = await pool.query<RowDataPacket[]>(
    `SELECT o.id, o.status, o.total,
            DATE_FORMAT(o.created_at, '%Y-%m-%dT%H:%i:%s.%fZ') AS created_at,
            JSON_ARRAYAGG(JSON_OBJECT('productId', oi.product_id, 'name', p.name, 'quantity', oi.quantity, 'unitPrice', oi.unit_price)) AS items
       FROM orders o
       JOIN order_items oi ON oi.order_id = o.id
       JOIN products p ON p.id = oi.product_id
      WHERE o.id = ?
      GROUP BY o.id`,
    [orderId],
  );
  if (rows.length === 0) return null;
  const r = rows[0];
  const items = (r.items as { productId: number; name: string; quantity: number; unitPrice: number | string }[])
    .map((i) => ({ productId: Number(i.productId), name: i.name, quantity: Number(i.quantity), unitPriceCents: cents(i.unitPrice) }))
    .sort((a, b) => a.productId - b.productId);
  return {
    id: Number(r.id),
    status: String(r.status),
    totalCents: cents(r.total),
    createdAt: String(r.created_at).replace(/(\.\d{3})\d{3}Z$/, "$1Z"),

  };
}
```

```ts save=solutions/P25.5-E83.ts
// P25.5-E83: repository של מוצרים מעל mysql2
import type { Pool, ResultSetHeader, RowDataPacket } from "mysql2/promise";

export type Product = { id: number; name: string; category: string; priceCents: number; stock: number; isActive: boolean; attributes: Record<string, unknown> | null };

export class MysqlProductsRepository {
  #pool: Pool;
  constructor(pool: Pool) {
    this.#pool = pool;
  }

  #toProduct(r: RowDataPacket): Product {
    return {
      id: Number(r.id),
      name: String(r.name),
      category: String(r.category),
      priceCents: Math.round(Number(r.price) * 100),
      stock: Number(r.stock),
      isActive: Boolean(r.is_active),
      attributes: r.attributes ?? null,
    };
  }

  async findById(id: number): Promise<Product | null> {
    const [rows] = await this.#pool.execute<RowDataPacket[]>("SELECT * FROM products WHERE id = ?", [id]);
    return rows.length ? this.#toProduct(rows[0]) : null;
  }

  async list({ category, onlyActive = true, limit = 20 }: { category?: string; onlyActive?: boolean; limit?: number } = {}): Promise<Product[]> {
    const where: string[] = [];
    const params: unknown[] = [];
    if (category !== undefined) {
      where.push("category = ?");
      params.push(category);
    }
    if (onlyActive) where.push("is_active");
    const sql = `SELECT * FROM products ${where.length ? "WHERE " + where.join(" AND ") : ""} ORDER BY id LIMIT ?`;
    const [rows] = await this.#pool.query<RowDataPacket[]>(sql, [...params, limit]);
    return rows.map((r) => this.#toProduct(r));
  }

  async adjustStock(id: number, delta: number): Promise<number> {
    const [res] = await this.#pool.execute<ResultSetHeader>("UPDATE products SET stock = stock + ? WHERE id = ? AND stock + ? >= 0", [delta, id, delta]);
    if (res.affectedRows === 0) {
      const exists = await this.findById(id);
      throw new Error(exists ? "OUT_OF_STOCK" : "NOT_FOUND");
    }
    const p = await this.findById(id);
    return p!.stock;
  }
}
```

```ts save=solutions/P25.5-E84.ts
// P25.5-E84: GET /products עם cursor
import { Router } from "express";
import type { Pool, RowDataPacket } from "mysql2/promise";

const intParam = (v: unknown, def: number | null, min: number, max: number) => {
  if (v === undefined) return def;
  if (typeof v !== "string" || !/^\d+$/.test(v)) return NaN;
  const n = Number(v);
  return n >= min && n <= max ? n : NaN;
};

export function productsRouter(pool: Pool) {
  const router = Router();
  router.get("/products", async (req, res, next) => {
    const limit = intParam(req.query.limit, 10, 1, 50) as number;
    const after = intParam(req.query.after, 0, 0, Number.MAX_SAFE_INTEGER) as number;
    const category = req.query.category;
    if (Number.isNaN(limit) || Number.isNaN(after) || (category !== undefined && typeof category !== "string")) {
      res.status(400).json({ error: { code: "VALIDATION_ERROR" } });
      return;
    }
    try {
      const [rows] = await pool.query<RowDataPacket[]>(
        `SELECT id, name, price FROM products
          WHERE is_active AND id > ? ${category !== undefined ? "AND category = ?" : ""}
          ORDER BY id LIMIT ?`,
        category !== undefined ? [after, category, limit] : [after, limit],
      );
      const items = rows.map((r) => ({ id: Number(r.id), name: String(r.name), priceCents: Math.round(Number(r.price) * 100) }));
      res.json({ items, nextAfter: items.length === limit ? items[items.length - 1].id : null });
    } catch (err) {
      next(err);
    }
  });
  return router;
}
```

```ts save=solutions/P25.5-E85.ts
// P25.5-E85: database זמני לבדיקות
import { randomUUID } from "node:crypto";
import mysql, { type Pool } from "mysql2/promise";

export async function withTempDatabase<T>(url: string, seedSql: string, fn: (pool: Pool) => Promise<T>): Promise<T> {
  const name = `test_${randomUUID().replaceAll("-", "").slice(0, 12)}`;
  const admin = await mysql.createConnection({ uri: url, multipleStatements: true });
  let pool: Pool | undefined;
  try {
    await admin.query(`CREATE DATABASE \`${name}\` CHARACTER SET utf8mb4`);
    pool = mysql.createPool({ uri: url, database: name, multipleStatements: true, connectionLimit: 5 });
    if (seedSql.trim()) await pool.query(seedSql);
    return await fn(pool);
  } finally {
    await pool?.end();
    await admin.query(`DROP DATABASE IF EXISTS \`${name}\``);
    await admin.end();
  }
}
```

```sql run mysql db=25v-mysql/files/indexes.sql save=solutions/P25.5-E92.sql
WITH idx AS (
  SELECT INDEX_NAME AS index_name,
         MAX(NON_UNIQUE) = 0 AS is_unique,
         GROUP_CONCAT(COLUMN_NAME ORDER BY SEQ_IN_INDEX SEPARATOR ',') AS cols
  FROM information_schema.STATISTICS
  WHERE TABLE_SCHEMA = DATABASE() AND TABLE_NAME = 'events_log' AND INDEX_NAME <> 'PRIMARY'

)

       (NOT a.is_unique AND EXISTS (
         SELECT 1 FROM idx b
         WHERE b.index_name <> a.index_name AND CONCAT(b.cols, ',') LIKE CONCAT(a.cols, ',%')
       )) AS redundant
FROM idx a
ORDER BY a.index_name;
```

```ts run save=solutions/P25.5-E93.ts
const UUID = /^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/i;

export function uuidToBin(uuid: string): Buffer {
  if (!UUID.test(uuid)) throw new RangeError(`UUID לא תקין: ${uuid}`);
  return Buffer.from(uuid.replaceAll("-", ""), "hex");
}

export function binToUuid(bin: Buffer): string {
  const h = bin.toString("hex");
  return `${h.slice(0, 8)}-${h.slice(8, 12)}-${h.slice(12, 16)}-${h.slice(16, 20)}-${h.slice(20)}`;
}

  const b = Buffer.from(random.subarray(0, 16));
  b.writeUIntBE(nowMs, 0, 6); // 48 ביט של זמן — ולכן ממוין לפי זמן

  b[8] = (b[8] & 0x3f) | 0x80; // variant 10
  return binToUuid(b);
}
```

```ts save=solutions/P25.5-E94.ts
// P25.5-E94: העברה בקבוצות — עם המשך
export async function copyInBatches<R extends { id: number }>({ readBatch, writeBatch, size, startAfter = 0 }: {
  readBatch: (afterId: number, size: number) => Promise<R[]>;
  writeBatch: (rows: R[]) => Promise<void>;
  size: number;
  startAfter?: number;
}) {
  if (!Number.isInteger(size) || size <= 0) throw new RangeError(`size לא תקין: ${size}`);
  let after = startAfter;
  let batches = 0;
  let rows = 0;
  for (;;) {
    const batch = await readBatch(after, size);
    if (batch.length === 0) break;
    if (!batch.every((r, i) => r.id > (i === 0 ? after : batch[i - 1].id))) throw new Error("NOT_SORTED");
    await writeBatch(batch);
    batches++;
    rows += batch.length;
    after = batch[batch.length - 1].id;
    if (batch.length < size) break;
  }
  return { batches, rows, lastId: after };
}
```

```ts save=solutions/P25.5-E98.ts
// P25.5-E98: שורה מ־PostgreSQL לשורה של MySQL
export function toMysqlValue(v: unknown): string | number | null {
  if (v === null || v === undefined) return null;
  if (typeof v === "boolean") return v ? 1 : 0;
  if (v instanceof Date) return v.toISOString().replace("T", " ").replace("Z", "");
  if (typeof v === "bigint") return v.toString();
  if (typeof v === "object") return JSON.stringify(v);
  return v as string | number;
}

export function pgRowToMysql(row: Record<string, unknown>) {
  return Object.fromEntries(Object.entries(row).map(([k, v]) => [k, toMysqlValue(v)]));
}
```

```ts run save=solutions/P25.5-E99.ts
// P25.5-E99: slow query log — מה הכי איטי?
export function normalizeSql(sql: string): string {
  return sql
    .replace(/'(?:[^']|'')*'/g, "?")
    .replace(/\b\d+(?:\.\d+)?\b/g, "?")
    .replace(/\s+/g, " ")
    .trim()
    .replace(/\s*;$/, "");
}

export function slowLogReport(text: string) {
  const groups = new Map<string, { query: string; count: number; total: number; maxRowsExamined: number }>();
  for (const entry of text.split(/^# Time:/m).slice(1)) {
    const lines = entry.split("\n");
    const qi = lines.findIndex((l) => l.startsWith("# Query_time"));
    if (qi === -1) continue;
    const m = /Query_time:\s*([\d.]+).*Rows_examined:\s*(\d+)/.exec(lines[qi]);
    if (!m) continue;
    const sql = lines
      .slice(qi + 1)
      .filter((l) => !/^\s*(SET timestamp=\d+;|use \S+;)\s*$/i.test(l))
      .join("\n");
    const query = normalizeSql(sql);
    if (!query) continue;
    const g = groups.get(query) ?? { query, count: 0, total: 0, maxRowsExamined: 0 };
    g.count++;
    g.total += Number(m[1]);
    g.maxRowsExamined = Math.max(g.maxRowsExamined, Number(m[2]));
    groups.set(query, g);
  }
  return [...groups.values()]
    .map(({ query, count, total, maxRowsExamined }) => ({ query, count, totalSeconds: Math.round(total * 1000) / 1000, maxRowsExamined }))
    .sort((a, b) => b.totalSeconds - a.totalSeconds);
}
```

```ts save=solutions/P25.5-E100.ts
// P25.5-E100: full table scan — מ־Node
import type { Pool, RowDataPacket } from "mysql2/promise";

export async function fullScans(pool: Pool, sql: string, params: unknown[] = []) {
  const [rows] = await pool.query<RowDataPacket[]>(`EXPLAIN FORMAT=JSON ${sql}`, params);
  const raw = rows[0].EXPLAIN;
  const plan = typeof raw === "string" ? JSON.parse(raw) : raw;
  const found: { table: string; rows: number }[] = [];
  const seen = new Set<string>();
  const walk = (node: unknown): void => {
    if (Array.isArray(node)) return node.forEach(walk);
    if (node === null || typeof node !== "object") return;
    const o = node as Record<string, unknown>;
    if (typeof o.table_name === "string" && o.access_type === "ALL" && !seen.has(o.table_name)) {
      seen.add(o.table_name);
      found.push({ table: o.table_name, rows: Number(o.rows_examined_per_scan) });
    }
    Object.values(o).forEach(walk);
  };
  walk(plan);
  return found;
}
```
