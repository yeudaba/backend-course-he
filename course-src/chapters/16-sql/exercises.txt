@@ P16-E15 | 3 | דוח קטגוריה 🤖 | 15 | שיעור 3
@task
<p>בקובץ <code>16-sql/starter/P16-E15.sql</code>: על כל המוצרים (גם הלא פעילים) בקטגוריה <code>brewing</code> — שורה אחת: <code>products</code> (כמות), <code>min_price</code>, <code>max_price</code>, <code>avg_price</code> (מעוגל לשתי ספרות) ו־<code>out_of_stock</code> (כמה עם מלאי 0).</p>
@hint1
<p>"כמה עם מלאי 0" — <code>COUNT</code> סופר שורות שאינן NULL. איך הופכים תנאי לערך שהוא NULL כשהתנאי לא מתקיים?</p>
@hint2
<p><code>COUNT(*) FILTER (WHERE stock = 0)</code> — תחביר של PostgreSQL בדיוק לזה.</p>
@solution
```sql run db=16-sql/files/shop.sql save=solutions/P16-E15.sql
SELECT
  COUNT(*) AS products,
  MIN(price) AS min_price,
  MAX(price) AS max_price,
  ROUND(AVG(price), 2) AS avg_price,
  COUNT(*) FILTER (WHERE stock = 0) AS out_of_stock
FROM products
WHERE category = 'brewing';
```
```out
 products | min_price | max_price | avg_price | out_of_stock
----------+-----------+-----------+-----------+--------------
        4 |     99.00 |    229.00 |    149.00 |            1
(1 row)
```
<p><code>FILTER</code> הוא דרך נוחה לספור כמה תנאים בשאילתה אחת. (בלעדיו: <code>SUM(CASE WHEN stock = 0 THEN 1 ELSE 0 END)</code> — תראה את זה בקוד ישן.)</p>

// P16-E27: בונה שאילתות עם פרמטרים
export type ProductFilters = {
  category?: string;
  minPrice?: number;
  maxPrice?: number;
  search?: string;
  inStock?: boolean;
  sort?: "price" | "-price" | "name";
  limit?: number;
};

export function buildProductQuery(filters: ProductFilters): { text: string; values: unknown[] } {
  const conditions = ["is_active"];
  const values: unknown[] = [];
  const param = (value: unknown) => {
    values.push(value);
    return `$${values.length}`;
  };

  if (filters.category !== undefined) conditions.push(`category = ${param(filters.category)}`);
  if (filters.minPrice !== undefined) conditions.push(`price >= ${param(filters.minPrice)}`);
  if (filters.maxPrice !== undefined) conditions.push(`price <= ${param(filters.maxPrice)}`);
  if (filters.search !== undefined) conditions.push(`name ILIKE ${param(`%${filters.search}%`)}`);
  if (filters.inStock === true) conditions.push("stock > 0");
  if (filters.inStock === false) conditions.push("stock = 0");

  let orderBy: string;
  switch (filters.sort) {
    case "price":
      orderBy = "price, id";
      break;
    case "-price":
      orderBy = "price DESC, id";
      break;
    case "name":
      orderBy = "name";
      break;
    case undefined:
      orderBy = "id";
      break;
    default:
      throw new Error(`מיון לא מוכר: ${filters.sort}`);
  }

  let text = `SELECT id, name, price FROM products WHERE ${conditions.join(" AND ")} ORDER BY ${orderBy}`;
  if (filters.limit !== undefined) text += ` LIMIT ${param(filters.limit)}`;
  return { text, values };
}

console.log(buildProductQuery({}));
console.log(buildProductQuery({ category: "beans", maxPrice: 90, inStock: true, sort: "-price", limit: 2 }));
console.log(buildProductQuery({ search: "grinder", sort: "name" }));
```

}

}

// P16-E39: SELECT ב־JavaScript
type Row = Record<string, unknown>;

export function select(rows: Row[], columns: string[], where: Row = {}): Row[] {
  if (rows.length > 0) {
    const known = Object.keys(rows[0]);
    for (const name of [...columns, ...Object.keys(where)]) {
      if (!known.includes(name)) throw new Error(`column "${name}" does not exist`);
    }
  }
  return rows
    .filter((row) => Object.entries(where).every(([col, value]) => row[col] === value))
    .map((row) => Object.fromEntries(columns.map((col) => [col, row[col]])));
}

const products = [
  { id: 5, name: "Hand Grinder", category: "grinders", price: 189 },
  { id: 6, name: "Electric Grinder Pro", category: "grinders", price: 649 },
  { id: 7, name: "French Press 1L", category: "brewing", price: 119 },
];
console.log(select(products, ["name", "price"], { category: "grinders" }));
console.log(select(products, ["id"], { category: "grinders", price: 649 }));
try {
  select(products, ["name"], { categroy: "brewing" });
} catch (err) {
  console.log((err as Error).message);
}

// P16-E40: פלט כמו psql
type Row = Record<string, unknown>;

export function formatTable(rows: Row[]): string {
  const footer = `(${rows.length} ${rows.length === 1 ? "row" : "rows"})`;
  if (rows.length === 0) return footer;
  const cols = Object.keys(rows[0]);
  const text = (v: unknown) => (v === null || v === undefined ? "" : String(v));
  const widths = cols.map((c) => Math.max(c.length, ...rows.map((r) => text(r[c]).length)));
  const line = (cells: string[]) =>
    cells
      .map((c, i) => ` ${c.padEnd(widths[i])} `)
      .join("|")
      .trimEnd();
  return [line(cols), widths.map((w) => "-".repeat(w + 2)).join("+"), ...rows.map((r) => line(cols.map((c) => text(r[c])))), footer].join("\n");
}

console.log(formatTable([{ name: "Hand Grinder", price: "189.00" }, { name: "Electric Grinder Pro", price: "649.00" }]));
console.log(formatTable([{ full_name: "Yossi Cohen", phone: null }]));
console.log(formatTable([]));
```

 Electric Grinder Pro | 649.00
(2 rows)
 full_name   | phone
-------------+-------
 Yossi Cohen |
(1 row)
(0 rows)
```

// P16-E41: אילוצים ב־JavaScript
type Row = Record<string, unknown>;
export type Rules = { notNull?: string[]; unique?: string[]; check?: Record<string, (row: Row) => boolean> };

export function checkRows(rows: Row[], rules: Rules): string[] {
  const problems: string[] = [];
  const seen = new Map<string, Set<unknown>>();
  rows.forEach((row, i) => {
    for (const col of rules.notNull ?? []) {
      if (row[col] === null || row[col] === undefined) problems.push(`row ${i}: ${col} is null`);
    }
    for (const col of rules.unique ?? []) {
      const value = row[col];
      if (value === null || value === undefined) continue;
      if (!seen.has(col)) seen.set(col, new Set());
      if (seen.get(col)!.has(value)) problems.push(`row ${i}: duplicate ${col}: ${value}`);
      seen.get(col)!.add(value);
    }
    for (const [name, fn] of Object.entries(rules.check ?? {})) {
      if (!fn(row)) problems.push(`row ${i}: check ${name} failed`);
    }
  });
  return problems;
}

const customers = [
  { email: "dana@example.com", price: 10 },
  { email: "dana@example.com", price: -5 },
  { email: null, price: 3 },
  { email: null, price: 0 },
];
console.log(checkRows(customers, { notNull: ["email"], unique: ["email"], check: { positive_price: (r) => (r.price as number) > 0 } }));
```
```out
[

  'row 1: check positive_price failed',
  'row 2: email is null',

]

// P16-E46: מ־CSV לשורות עם טיפוסים
export type ColumnType = "text" | "integer" | "numeric" | "boolean" | "date";

function convert(raw: string, type: ColumnType, column: string, n: number): unknown {
  const v = raw.trim();
  if (v === "") return null;
  const fail = () => {
    throw new Error(`line ${n}: invalid ${type} for ${column}: "${v}"`);
  };
  switch (type) {
    case "text":
      return v;
    case "integer":
      return /^-?\d+$/.test(v) ? Number(v) : fail();
    case "numeric":
      return Number.isFinite(Number(v)) ? Number(v) : fail();
    case "boolean":
      return v === "true" ? true : v === "false" ? false : fail();
    case "date":
      return /^\d{4}-\d{2}-\d{2}$/.test(v) ? v : fail();
  }
}

export function parseCsv(text: string, types: Record<string, ColumnType>): Record<string, unknown>[] {
  const lines = text.split(/\r?\n/).map((line, i) => ({ line, n: i + 1 }));
  const [header, ...rest] = lines.filter((l) => l.line.trim() !== "");
  if (!header) return [];
  const columns = header.line.split(",").map((c) => c.trim());
  return rest.map(({ line, n }) => {
    const values = line.split(",");
    if (values.length !== columns.length) throw new Error(`line ${n}: expected ${columns.length} values, got ${values.length}`);
    return Object.fromEntries(columns.map((c, i) => [c, convert(values[i], types[c] ?? "text", c, n)]));
  });
}

const csv = "name,price,stock,is_active,added_on\nHand Grinder,189.00,8,true,2026-02-14\n\nMoka Pot , 149, ,false,\n";

try {
  parseCsv("name,stock\nPen,3\nCup,2.5\n", { stock: "integer" });
} catch (err) {
  console.log((err as Error).message);
}

import express from "express";

export type Product = { id: number; name: string; category: string; price: number; stock: number; is_active: boolean };

export function createApp(products: Product[]) {
  const app = express();
  app.get("/products", (req, res) => {
    const known = Object.keys(products[0] ?? {});
    const where = Object.entries(req.query).map(([col, value]) => [col, String(value)] as const);
    const unknown = where.find(([col]) => !known.includes(col));

      .sort((a, b) => a.id - b.id)
      .map(({ id, name, price }) => ({ id, name, price }));

  });

// P16-E48: לנחש טיפוס לכל עמודה
type Row = Record<string, unknown>;
export type Column = { column: string; type: string; nullable: boolean };
const DATE = /^\d{4}-\d{2}-\d{2}$/;

function typeOf(values: unknown[]): string {
  if (values.length === 0) return "unknown";
  if (values.every((v) => Number.isInteger(v))) return "integer";
  if (values.every((v) => typeof v === "number")) return "numeric";
  if (values.every((v) => typeof v === "boolean")) return "boolean";
  if (values.every((v) => typeof v === "string" && DATE.test(v))) return "date";
  if (values.every((v) => typeof v === "string")) return "text";
  return "mixed";
}

export function describe(rows: Row[]): Column[] {
  const columns = new Set<string>();
  for (const row of rows) for (const key of Object.keys(row)) columns.add(key);
  return [...columns].map((column) => {
    const values = rows.map((r) => r[column]);
    const present = values.filter((v) => v !== null && v !== undefined);
    return { column, type: typeOf(present), nullable: present.length < rows.length };
  });
}

    { id: 3, name: 42, price: 7, active: true, added_on: "soon", note: null },

// P16-E56: חיפוש מוצרים מ־Node
export type Db = { query(text: string, values?: unknown[]): Promise<{ rows: any[] }> };
export type Found = { id: number; name: string; price: number };

export async function searchProducts(client: Db, { q, category, limit = 10 }: { q?: string; category?: string; limit?: number }): Promise<Found[]> {
  if (!Number.isInteger(limit) || limit < 1 || limit > 50) throw new Error("limit חייב להיות 1–50");
  const conditions = ["is_active"];
  const values: unknown[] = [];
  if (q && q.trim()) {
    values.push(`%${q.trim().replace(/[\\%_]/g, "\\$&")}%`);
    conditions.push(`name ILIKE $${values.length}`);
  }
  if (category) {
    values.push(category);
    conditions.push(`category = $${values.length}`);
  }
  values.push(limit);
  const text = `SELECT id, name, price FROM products WHERE ${conditions.join(" AND ")} ORDER BY name, id LIMIT $${values.length}`;
  const { rows } = await client.query(text, values);
  return rows.map((r) => ({ id: r.id, name: r.name, price: Number(r.price) }));
}

export type Db = { query(text: string, values?: unknown[]): Promise<{ rows: any[]; fields: { name: string }[] }> };

const cell = (v: unknown) => {
  if (v === null || v === undefined) return "";
  const s = String(v);
  return /[",\r\n]/.test(s) ? `"${s.replaceAll('"', '""')}"` : s;
};

export async function exportCsv(client: Db, text: string, values: unknown[] = []): Promise<string> {
  const { rows, fields } = await client.query(text, values);
  const names = fields.map((f) => f.name);
  const lines = [names.map(cell).join(","), ...rows.map((r) => names.map((n) => cell(r[n])).join(","))];
  return lines.join("\r\n") + "\r\n";
}

export type Db = { query(text: string, values?: unknown[]): Promise<{ rows: any[] }> };
export type Report = { products: number; units: number; cheapest: number | null; priciest: number | null; avgPrice: number | null };

const num = (v: string | number | null) => (v === null ? null : Number(v));

export async function inventoryReport(client: Db, category?: string): Promise<Report> {
  const values: unknown[] = [];
  let where = "is_active";
  if (category !== undefined) {
    values.push(category);
    where += " AND category = $1";

            MIN(price) AS cheapest, MAX(price) AS priciest, ROUND(AVG(price), 2) AS avg_price

  );

// P16-E71: צבירות ב־JavaScript שמתנהגות כמו SQL
type Row = Record<string, unknown>;
type Value = number | string;

export function aggregate(rows: Row[], column: string) {
  const values = rows.map((r) => r[column]).filter((v) => v !== null && v !== undefined) as Value[];
  if (values.length === 0) return { count: 0, countAll: rows.length, sum: null, avg: null, min: null, max: null };
  let min = values[0];
  let max = values[0];
  for (const v of values) {
    if (v < min) min = v;
    if (v > max) max = v;
  }
  const numbers = values as number[];
  const sum = typeof values[0] === "number" ? numbers.reduce((s, x) => s + x, 0) : null;
  return { count: values.length, countAll: rows.length, sum, avg: sum === null ? null : sum / values.length, min, max };
}

console.log(aggregate([], "x"));

// P16-E80: יצירת מוצר — Zod ואז INSERT
import { z } from "zod";

export type Db = { query(text: string, values?: unknown[]): Promise<{ rows: any[] }> };

export const productInput = z.object({
  name: z.string().trim().min(2).max(100),
  category: z.enum(["beans", "grinders", "brewing", "accessories"]),
  price: z.number().positive().max(100000).multipleOf(0.01),
  stock: z.number().int().min(0).default(0),
});

export async function createProduct(client: Db, input: unknown) {
  const data = productInput.parse(input);
  const { rows } = await client.query(
    `INSERT INTO products (name, category, price, stock, added_on)
     VALUES ($1, $2, $3, $4, CURRENT_DATE)
     RETURNING id, name, category, price, stock, is_active`,
    [data.name, data.category, data.price, data.stock],
  );
  const r = rows[0];
  return { id: r.id, name: r.name, category: r.category, price: Number(r.price), stock: r.stock, isActive: r.is_active };
}
```
<p>שלושה גבולות, שלושה תפקידים: Zod — "האם הקלט הגיוני" (עם הודעה למשתמש); המסד — "האם הנתונים עקביים" (CHECK, NOT NULL — גם אם מישהו עקף את הקוד); ו־RETURNING + המרה — "מה הלקוח מקבל". בפרק 19 Prisma יחליף את ה־SQL ואת ההמרה — אבל הגבולות יישארו.</p>
