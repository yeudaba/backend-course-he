export interface Product {
  id: number;
  name: string;
  price: number;
}

@Injectable()
export class ProductsService {
  private readonly products: Product[] = [
    { id: 1, name: "Espresso cup", price: 24.9 },
    { id: 2, name: "Hand grinder", price: 189 },
    { id: 3, name: "Milk jug", price: 59.5 },
  ];

  list(): Product[] {
    return this.products;
  }

  findOne(id: number): Product {
    const product = this.products.find((p) => p.id === id);
    if (!product) throw new NotFoundException(`מוצר ${id} לא נמצא`);
    return product;
  }
}

@Controller("products")
export class ProductsController {
  constructor(private readonly products: ProductsService) {}

  @Get()
  list() {
    return this.products.list();
  }

  @Get(":id")
  findOne(@Param("id", ParseIntPipe) id: number) {
    return this.products.findOne(id);
  }
}

@Module({ controllers: [ProductsController], providers: [ProductsService] })
export class ProductsModule {}

// P20-E08
import { Body, Controller, Get, Injectable, Module, Post } from "@nestjs/common";
import { IsBoolean, IsInt, IsNumber, IsOptional, IsString, Length, Max, Min } from "class-validator";

export class CreateProductDto {
  @IsString()
  @Length(2, 80)
  name: string;

  @IsNumber({ maxDecimalPlaces: 2 })
  @Min(0.01)
  @Max(100_000)
  price: number;

  @IsInt()
  @Min(0)
  stock: number;

  @IsOptional()
  @IsBoolean()
  isActive?: boolean;
}

export interface Product extends CreateProductDto {
  id: number;
  isActive: boolean;
}

@Injectable()
export class ProductsService {
  private readonly products: Product[] = [];
  private nextId = 1;

  list(): Product[] {
    return this.products;
  }

  create(dto: CreateProductDto): Product {
    const product: Product = { id: this.nextId++, name: dto.name, price: dto.price, stock: dto.stock, isActive: dto.isActive ?? true };
    this.products.push(product);
    return product;
  }
}

@Controller("products")
export class ProductsController {
  constructor(private readonly products: ProductsService) {}

  @Get()
  list() {
    return this.products.list();
  }

  @Post()
  create(@Body() dto: CreateProductDto) {
    return this.products.create(dto);
  }
}

@Module({ controllers: [ProductsController], providers: [ProductsService] })
export class ProductsModule {}
```

// P20-E09
import { Body, Controller, Delete, Get, HttpCode, Injectable, Module, NotFoundException, Param, ParseIntPipe, Patch, Post } from "@nestjs/common";
import { PartialType } from "@nestjs/swagger";
import { IsInt, IsNumber, IsString, Length, Min } from "class-validator";

export class CreateProductDto {
  @IsString()
  @Length(2, 80)
  name: string;

  @IsNumber({ maxDecimalPlaces: 2 })
  @Min(0.01)
  price: number;

  @IsInt()
  @Min(0)
  stock: number;
}

export class UpdateProductDto extends PartialType(CreateProductDto) {}

export interface Product extends CreateProductDto {
  id: number;
}

@Injectable()
export class ProductsService {
  private readonly products: Product[] = [];
  private nextId = 1;

  create(dto: CreateProductDto): Product {
    const product = { id: this.nextId++, ...dto };
    this.products.push(product);
    return product;
  }

  findOne(id: number): Product {
    const product = this.products.find((p) => p.id === id);
    if (!product) throw new NotFoundException(`מוצר ${id} לא נמצא`);
    return product;
  }

  update(id: number, dto: UpdateProductDto): Product {
    const product = this.findOne(id);
    Object.assign(product, dto);
    return product;
  }

  remove(id: number): void {
    this.findOne(id);
    this.products.splice(this.products.findIndex((p) => p.id === id), 1);
  }
}

@Controller("products")
export class ProductsController {
  constructor(private readonly products: ProductsService) {}

  @Post()
  create(@Body() dto: CreateProductDto) {
    return this.products.create(dto);
  }

  @Get(":id")
  findOne(@Param("id", ParseIntPipe) id: number) {
    return this.products.findOne(id);
  }

  @Patch(":id")
  update(@Param("id", ParseIntPipe) id: number, @Body() dto: UpdateProductDto) {
    return this.products.update(id, dto);
  }

  @Delete(":id")
  @HttpCode(204)
  remove(@Param("id", ParseIntPipe) id: number) {
    this.products.remove(id);
  }
}

@Module({ controllers: [ProductsController], providers: [ProductsService] })
export class ProductsModule {}

  @Type(() => Number)
  @IsInt()
  @Min(1)

  @IsOptional()
  @Type(() => Number)
  @IsInt()
  @Min(1)
  @Max(50)
  limit: number = 10;

  @IsString()
  q?: string;

  name: string;

export interface Clock {
  now(): Date;
}

@Injectable()

// P20-E12
import {
  ArgumentsHost, BadRequestException, Body, Catch, Controller, ExceptionFilter, Get, HttpException,
  Module, NotFoundException, Param, ParseIntPipe, Post, ValidationPipe, type ValidationError,
} from "@nestjs/common";
import { IsString, Length } from "class-validator";
import type { Response } from "express";

const CODES: Record<number, string> = { 400: "BAD_REQUEST", 401: "UNAUTHORIZED", 403: "FORBIDDEN", 404: "NOT_FOUND", 409: "CONFLICT" };

@Catch(HttpException)
export class ApiExceptionFilter implements ExceptionFilter {
  catch(exception: HttpException, host: ArgumentsHost) {
    const res = host.switchToHttp().getResponse<Response>();
    const status = exception.getStatus();
    const body = exception.getResponse();
    // אם מי שזרק כבר נתן code (למשל ה־exceptionFactory למטה) — משתמשים בו
    if (typeof body === "object" && body !== null && "code" in body) {
      res.status(status).json({ error: body });
      return;
    }
    res.status(status).json({ error: { code: CODES[status] ?? "ERROR", message: exception.message } });
  }
}

export function validationPipe() {
  return new ValidationPipe({
    whitelist: true,
    forbidNonWhitelisted: true,
    transform: true,
    exceptionFactory: (errors: ValidationError[]) =>
      new BadRequestException({
        code: "VALIDATION_ERROR",
        message: "קלט לא תקין",
        details: errors.map((e) => ({ field: e.property, messages: Object.values(e.constraints ?? {}) })),
      }),
  });
}

export class CreateItemDto {
  @IsString()
  @Length(1, 50)
  name: string;
}

@Controller("items")
export class ItemsController {
  private readonly items: { id: number; name: string }[] = [];

  @Post()
  create(@Body() dto: CreateItemDto) {
    const item = { id: this.items.length + 1, name: dto.name };
    this.items.push(item);
    return item;
  }

  @Get(":id")
  findOne(@Param("id", ParseIntPipe) id: number) {
    const item = this.items.find((i) => i.id === id);
    if (!item) throw new NotFoundException(`פריט ${id} לא נמצא`);
    return item;
  }
}

@Module({ controllers: [ItemsController] })
export class ItemsModule {}
```

import {
  CanActivate, Controller, ExecutionContext, ForbiddenException, Get, Inject, Injectable, Module,
  UnauthorizedException, UseGuards,
} from "@nestjs/common";
import type { Request } from "express";

export const API_KEYS = "API_KEYS";

@Injectable()
export class ApiKeyGuard implements CanActivate {
  constructor(@Inject(API_KEYS) private readonly keys: Set<string>) {}

  canActivate(context: ExecutionContext): boolean {
    const req = context.switchToHttp().getRequest<Request>();
    const key = req.get("x-api-key");
    if (!key) throw new UnauthorizedException("חסר X-Api-Key");
    if (!this.keys.has(key)) throw new ForbiddenException("מפתח לא מוכר");
    return true;
  }
}

@Controller()
export class SecretController {
  @Get("public")
  public() {
    return { ok: true };
  }

  @Get("secret")
  @UseGuards(ApiKeyGuard)
  secret() {
    return { secret: 42 };
  }
}

@Module({
  controllers: [SecretController],
  providers: [ApiKeyGuard, { provide: API_KEYS, useValue: new Set(["dev-key"]) }],
})
export class SecretModule {}

import { CallHandler, Controller, ExecutionContext, Get, Inject, Injectable, Module, NestInterceptor } from "@nestjs/common";
import { APP_INTERCEPTOR } from "@nestjs/core";
import type { Request, Response } from "express";
import { Observable, tap } from "rxjs";

export const LOGGER = "LOGGER";
export interface Logger {
  log(line: string): void;
}

@Injectable()
export class TimingInterceptor implements NestInterceptor {
  constructor(@Inject(LOGGER) private readonly logger: Logger) {}

  intercept(context: ExecutionContext, next: CallHandler): Observable<unknown> {
    const start = Date.now();
    const http = context.switchToHttp();
    const req = http.getRequest<Request>();
    const res = http.getResponse<Response>();
    return next.handle().pipe(
      tap(() => {
        // רץ אחרי ה־handler, לפני ש־Nest שולח את התשובה — לכן עוד אפשר להוסיף header
        const ms = Date.now() - start;
        res.setHeader("X-Response-Time", `${ms}ms`);
        this.logger.log(`${req.method} ${req.originalUrl} ${res.statusCode} ${ms}ms`);
      }),
    );
  }
}

@Controller()
export class PingController {
  @Get("ping")
  ping() {
    return { pong: true };
  }

  @Get("slow")
  async slow() {
    await new Promise((resolve) => setTimeout(resolve, 60));
    return { done: true };
  }
}

@Module({
  controllers: [PingController],
  providers: [
    { provide: LOGGER, useValue: console },
    { provide: APP_INTERCEPTOR, useClass: TimingInterceptor },
  ],
})
export class PingModule {}

@Injectable()
export class ParsePositiveIntPipe implements PipeTransform<string, number> {
  transform(value: string, metadata: ArgumentMetadata): number {

import { Controller, Get, Inject, Module } from "@nestjs/common";
import { z } from "zod";

export const ENV = "ENV";
export const CONFIG = "CONFIG";

export const configSchema = z.object({
  APP_NAME: z.string().min(1),
  PORT: z.coerce.number().int().min(1).max(65535).default(3000),
  DEBUG: z.enum(["true", "false"]).transform((v) => v === "true").default(false),
});
export type Config = z.infer<typeof configSchema>;

export function loadConfig(env: Record<string, string | undefined>): Config {
  const result = configSchema.safeParse(env);
  if (!result.success) {
    const problems = result.error.issues.map((i) => `${i.path.join(".")}: ${i.message}`).join("; ");
    throw new Error(`הגדרות לא תקינות — ${problems}`);
  }
  return result.data;
}

  @Get()
  info() {
    return { name: this.config.APP_NAME, port: this.config.PORT, debug: this.config.DEBUG };
  }
}

import {
  ArgumentsHost, Catch, Controller, ExceptionFilter, Get, HttpException, Injectable, Module,
  NotFoundException, Param, ParseIntPipe,
} from "@nestjs/common";
import { APP_FILTER } from "@nestjs/core";
import type { Response } from "express";

@Catch(HttpException) // ← תקלה 3: היה @Catch(NotFoundException), אז ה־400 של ParseIntPipe יצא במבנה של Nest
export class ApiExceptionFilter implements ExceptionFilter {
  catch(exception: HttpException, host: ArgumentsHost) {
    const res = host.switchToHttp().getResponse<Response>();
    const status = exception.getStatus();
    const code = status === 404 ? "NOT_FOUND" : status === 400 ? "BAD_REQUEST" : "ERROR";
    res.status(status).json({ error: { code, message: exception.message } }); // ← תקלה 2: בלי status — 200
  }
}

@Injectable()
export class OrdersService {
  private readonly orders = [
    { id: 1, item: "Espresso cup" },
    { id: 2, item: "Milk jug" },
  ];

    const order = this.orders.find((o) => o.id === id);
    if (!order) throw new NotFoundException(`הזמנה ${id} לא נמצאה`); // ← תקלה 1: Error רגיל הופך ל־500

  }
}

// P20-E24
import {
  ArgumentsHost, BadRequestException, Body, Catch, ConflictException, Controller, ExceptionFilter, Get, HttpCode,
  HttpException, Inject, Injectable, Module, NotFoundException, Param, ParseIntPipe, Post, Query, ValidationPipe,
  type ValidationError,
} from "@nestjs/common";
import { APP_FILTER, APP_PIPE } from "@nestjs/core";
import { Type } from "class-transformer";
import { IsIn, IsInt, IsOptional, IsString, Length, Max, Min } from "class-validator";
import type { Response } from "express";

// ---------- tokens ----------
export const CLOCK = "CLOCK";
export interface Clock {
  now(): Date;
}

// ---------- שגיאות: { error: { code, message, details? } } ----------
const apiError = (code: string, message: string, details?: unknown) => ({ code, message, ...(details ? { details } : {}) });
const CODES: Record<number, string> = { 400: "BAD_REQUEST", 404: "NOT_FOUND", 409: "CONFLICT" };

@Catch(HttpException)
export class ApiExceptionFilter implements ExceptionFilter {
  catch(exception: HttpException, host: ArgumentsHost) {
    const res = host.switchToHttp().getResponse<Response>();
    const status = exception.getStatus();
    const body = exception.getResponse();
    const error = typeof body === "object" && body !== null && "code" in body ? body : apiError(CODES[status] ?? "ERROR", exception.message);
    res.status(status).json({ error });
  }
}

export function createValidationPipe() {
  return new ValidationPipe({
    whitelist: true,
    forbidNonWhitelisted: true,
    transform: true,
    exceptionFactory: (errors: ValidationError[]) =>
      new BadRequestException(apiError("VALIDATION_ERROR", "קלט לא תקין", errors.map((e) => ({ field: e.property, messages: Object.values(e.constraints ?? {}) })))),
  });
}

export class CreateBookDto {
  @IsString() @Length(1, 200) title: string;
  @IsString() @Length(1, 200) author: string;
  @IsInt() @Min(1) @Max(100) copies: number;
}

export class ListBooksQuery {
  @IsOptional() @IsString() author?: string;
  @IsOptional() @IsIn(["true", "false"]) available?: string;
  @IsOptional() @Type(() => Number) @IsInt() @Min(1) page: number = 1;
  @IsOptional() @Type(() => Number) @IsInt() @Min(1) @Max(50) limit: number = 10;
}

export class CreateLoanDto {
  @IsString() @Length(1, 100) member: string;
}

// ---------- service ----------
export interface Book {
  id: number;
  title: string;
  author: string;
  copies: number;
  available: number;
}
export interface Loan {
  id: number;
  bookId: number;
  member: string;
  loanedAt: string;
  dueAt: string;
  returnedAt: string | null;
}

const LOAN_DAYS = 14;

@Injectable()
export class LibraryService {
  private readonly books: Book[] = [];
  private readonly loans: Loan[] = [];

  constructor(@Inject(CLOCK) private readonly clock: Clock) {}

  createBook(dto: CreateBookDto): Book {
    const book = { id: this.books.length + 1, title: dto.title, author: dto.author, copies: dto.copies, available: dto.copies };
    this.books.push(book);
    return book;
  }

  listBooks({ author, available, page, limit }: ListBooksQuery) {
    let books = this.books;
    if (author) books = books.filter((b) => b.author.toLowerCase().includes(author.toLowerCase()));
    if (available === "true") books = books.filter((b) => b.available > 0);
    return { data: books.slice((page - 1) * limit, page * limit), meta: { page, limit, total: books.length, totalPages: Math.ceil(books.length / limit) } };
  }

  getBook(id: number): Book {
    const book = this.books.find((b) => b.id === id);
    if (!book) throw new NotFoundException(apiError("BOOK_NOT_FOUND", `ספר ${id} לא נמצא`));
    return book;
  }

  lend(bookId: number, member: string): Loan {
    const book = this.getBook(bookId);
    if (this.loans.some((l) => l.bookId === bookId && l.member === member && l.returnedAt === null)) {
      throw new ConflictException(apiError("ALREADY_BORROWED", `${member} כבר מחזיק/ה את הספר`));
    }
    if (book.available === 0) throw new ConflictException(apiError("NO_COPIES", "אין עותק פנוי"));
    const now = this.clock.now();
    const loan = {
      id: this.loans.length + 1,
      bookId,
      member,
      loanedAt: now.toISOString(),
      dueAt: new Date(now.getTime() + LOAN_DAYS * 24 * 60 * 60 * 1000).toISOString(),
      returnedAt: null,
    };
    this.loans.push(loan);
    book.available--;
    return loan;
  }

  returnLoan(id: number): Loan {
    const loan = this.loans.find((l) => l.id === id);
    if (!loan) throw new NotFoundException(apiError("LOAN_NOT_FOUND", `השאלה ${id} לא נמצאה`));
    if (loan.returnedAt) throw new ConflictException(apiError("ALREADY_RETURNED", "ההשאלה כבר הוחזרה"));
    loan.returnedAt = this.clock.now().toISOString();
    this.getBook(loan.bookId).available++;
    return loan;
  }

  overdue(): Loan[] {
    const now = this.clock.now().getTime();
    return this.loans.filter((l) => l.returnedAt === null && new Date(l.dueAt).getTime() < now);
  }
}

// ---------- controllers ----------
@Controller("books")
export class BooksController {
  constructor(private readonly library: LibraryService) {}

  @Post()
  create(@Body() dto: CreateBookDto) {
    return this.library.createBook(dto);
  }

  @Get()
  list(@Query() query: ListBooksQuery) {
    return this.library.listBooks(query);
  }

  @Get(":id")
  findOne(@Param("id", ParseIntPipe) id: number) {
    return this.library.getBook(id);
  }

  @Post(":id/loans")
  lend(@Param("id", ParseIntPipe) id: number, @Body() dto: CreateLoanDto) {
    return this.library.lend(id, dto.member);
  }
}

@Controller("loans")
export class LoansController {
  constructor(private readonly library: LibraryService) {}

  @Get("overdue") // לפני כל ":id"
  overdue() {
    return this.library.overdue();
  }

  @Post(":id/return")
  @HttpCode(200)
  returnLoan(@Param("id", ParseIntPipe) id: number) {
    return this.library.returnLoan(id);
  }
}

@Module({
  controllers: [BooksController, LoansController],
  providers: [
    LibraryService,
    { provide: CLOCK, useValue: { now: () => new Date() } },
    { provide: APP_FILTER, useClass: ApiExceptionFilter },
    { provide: APP_PIPE, useFactory: createValidationPipe },
  ],
})
export class LibraryModule {}

// P20-E25
import {
  CanActivate, Controller, ExecutionContext, Get, HttpException, HttpStatus, Inject, Injectable, Module, UseGuards,
} from "@nestjs/common";
import type { Request, Response } from "express";

export const CLOCK = "CLOCK";
export const RATE_LIMIT = "RATE_LIMIT";
export interface Clock {
  now(): Date;
}
export interface RateLimitOptions {
  limit: number; // כמה בקשות מותר בחלון
  windowMs: number; // אורך החלון
}

@Injectable()
export class RateLimitGuard implements CanActivate {
  // לכל לקוח: מתי התחיל החלון שלו, וכמה בקשות היו בו
  private readonly windows = new Map<string, { start: number; count: number }>();

  constructor(
    @Inject(RATE_LIMIT) private readonly options: RateLimitOptions,
    @Inject(CLOCK) private readonly clock: Clock,
  ) {}

  canActivate(context: ExecutionContext): boolean {
    const http = context.switchToHttp();
    const req = http.getRequest<Request>();
    const res = http.getResponse<Response>();
    const key = req.get("x-client-id") ?? req.ip ?? "unknown";
    const now = this.clock.now().getTime();

    let w = this.windows.get(key);
    if (!w || now >= w.start + this.options.windowMs) {
      w = { start: now, count: 0 };
      this.windows.set(key, w);
    }
    if (w.count >= this.options.limit) {
      const seconds = Math.ceil((w.start + this.options.windowMs - now) / 1000);
      res.setHeader("Retry-After", String(seconds)); // לפני ה־throw: ה־header נשאר על ה־response
      throw new HttpException({ error: { code: "RATE_LIMITED", message: "יותר מדי בקשות — נסה שוב מאוחר יותר" } }, HttpStatus.TOO_MANY_REQUESTS);
    }
    w.count++;
    res.setHeader("X-RateLimit-Remaining", String(this.options.limit - w.count));
    return true;
  }
}

@Controller("data")
@UseGuards(RateLimitGuard)
export class DataController {
  @Get()
  get() {
    return { data: [1, 2, 3] };
  }
}

@Module({
  controllers: [DataController],
  providers: [
    RateLimitGuard,
    { provide: RATE_LIMIT, useValue: { limit: 5, windowMs: 60_000 } },
    { provide: CLOCK, useValue: { now: () => new Date() } },
  ],
})
export class DataModule {}
```

// P20-E26: מודול סימניות — מקובץ ריק
import { Body, Controller, Delete, Get, HttpCode, Injectable, Module, NotFoundException, Param, ParseIntPipe, Post, Query } from "@nestjs/common";
import { ArrayMaxSize, IsArray, IsOptional, IsString, IsUrl, Length } from "class-validator";

export interface Bookmark {
  id: number;
  url: string;
  title: string;
  tags: string[];
}

export class CreateBookmarkDto {
  @IsUrl()
  url: string;

  @IsString()
  @Length(1, 100)
  title: string;

  @IsOptional()
  @IsArray()
  @ArrayMaxSize(5)
  @IsString({ each: true })
  tags?: string[];
}

@Injectable()
export class BookmarksService {
  private readonly bookmarks: Bookmark[] = [];
  private nextId = 1;

  list(tag?: string): Bookmark[] {
    return tag === undefined ? this.bookmarks : this.bookmarks.filter((b) => b.tags.includes(tag));
  }

  get(id: number): Bookmark {
    const bookmark = this.bookmarks.find((b) => b.id === id);
    if (!bookmark) throw new NotFoundException(`סימנייה ${id} לא נמצאה`);
    return bookmark;
  }

  create(dto: CreateBookmarkDto): Bookmark {
    const bookmark = { id: this.nextId++, url: dto.url, title: dto.title.trim(), tags: dto.tags ?? [] };
    this.bookmarks.push(bookmark);
    return bookmark;
  }

  remove(id: number): void {
    const bookmark = this.get(id);
    this.bookmarks.splice(this.bookmarks.indexOf(bookmark), 1);
  }
}

@Controller("bookmarks")
export class BookmarksController {
  constructor(private readonly bookmarks: BookmarksService) {}

  @Get()
  list(@Query("tag") tag?: string) {
    return this.bookmarks.list(tag);
  }

  @Get(":id")
  get(@Param("id", ParseIntPipe) id: number) {
    return this.bookmarks.get(id);
  }

  @Post()
  create(@Body() dto: CreateBookmarkDto) {
    return this.bookmarks.create(dto);
  }

  @Delete(":id")
  @HttpCode(204)
  remove(@Param("id", ParseIntPipe) id: number) {
    this.bookmarks.remove(id);
  }
}

@Module({ controllers: [BookmarksController], providers: [BookmarksService] })
export class BookmarksModule {}

// P20-E27: slug — regex בפונקציה וב־pipe
import { BadRequestException, Controller, Get, Injectable, Module, Param, type PipeTransform, Query } from "@nestjs/common";

const SLUG = /^[a-z0-9]+(?:-[a-z0-9]+)*$/;

export function slugify(title: string): string {
  return title
    .toLowerCase()
    .replace(/[^a-z0-9]+/g, "-")
    .replace(/^-+|-+$/g, "");
}

@Injectable()
export class SlugPipe implements PipeTransform<string, string> {
  transform(value: string): string {
    if (typeof value !== "string" || value.length > 60 || !SLUG.test(value)) {
      throw new BadRequestException(`slug לא תקין: ${value}`);
    }
    return value;
  }
}

@Controller("slugs")
export class SlugsController {
  @Get("preview")
  preview(@Query("title") title = "") {
    return { slug: slugify(title) };
  }

  @Get(":slug")
  show(@Param("slug", SlugPipe) slug: string) {
    return { slug };
  }
}

@Module({ controllers: [SlugsController] })
export class SlugsModule {}

import { Controller, Get, Inject, Injectable, Module, NotFoundException, Param, ParseIntPipe } from "@nestjs/common";

export interface Customer {
  id: number;
  fullName: string;
}

export interface Order {
  id: number;
  customerId: number;
  total: number;
  status: "paid" | "cancelled";
}

export interface CustomersRepository {
  findById(id: number): Promise<Customer | null>;
}

export interface OrdersRepository {
  findByCustomer(customerId: number): Promise<Order[]>;
}

export const CUSTOMERS_REPOSITORY = "CUSTOMERS_REPOSITORY";
export const ORDERS_REPOSITORY = "ORDERS_REPOSITORY";

export class InMemoryCustomersRepository implements CustomersRepository {
  private readonly customers: Customer[] = [{ id: 1, fullName: "Dana Levi" }];

  async findById(id: number) {
    return this.customers.find((c) => c.id === id) ?? null;
  }
}

export class InMemoryOrdersRepository implements OrdersRepository {
  private readonly orders: Order[] = [
    { id: 1, customerId: 1, total: 120.5, status: "paid" },
    { id: 2, customerId: 1, total: 80, status: "cancelled" },
  ];

  async findByCustomer(customerId: number) {
    return this.orders.filter((o) => o.customerId === customerId);
  }
}

export class GreetingsService {
  private texts: Record<string, string> = { he: "שלום", en: "Hello", ar: "مرحبا" };

// P20-E42: DI container קטן משלך
type Entry = { deps: string[]; factory: (...args: any[]) => unknown };

export class Container {
  private registry = new Map<string, Entry>();
  private instances = new Map<string, unknown>();
  private building: string[] = [];

  register(token: string, deps: string[], factory: (...args: any[]) => unknown): void {
    this.registry.set(token, { deps, factory });
  }

  get<T>(token: string): T {
    if (this.instances.has(token)) return this.instances.get(token) as T;
    const entry = this.registry.get(token);
    if (!entry) throw new Error(`Cannot resolve ${token}`);
    if (this.building.includes(token)) {
      const cycle = [...this.building.slice(this.building.indexOf(token)), token];
      throw new Error(`Circular dependency: ${cycle.join(" -> ")}`);
    }
    this.building.push(token);
    try {
      const args = entry.deps.map((dep) => {
        if (!this.registry.has(dep)) throw new Error(`Cannot resolve ${token}: missing ${dep}`);
        return this.get(dep);
      });
      const instance = entry.factory(...args);
      this.instances.set(token, instance);
      return instance as T;
    } finally {
      this.building.pop();
    }
  }
}
```

// P20-E43: decorators משלך — @Route ו־routesOf
type RouteInfo = { method: string; path: string; handler: string };

const prefixes = new Map<Function, string>();
const routes = new Map<Function, RouteInfo[]>();

export function Prefix(path: string): ClassDecorator {
  return (target) => {
    prefixes.set(target, path);
  };
}

export function Route(method: "GET" | "POST" | "DELETE", path = ""): MethodDecorator {
  return (target, propertyKey) => {
    const cls = target.constructor;
    if (!routes.has(cls)) routes.set(cls, []);
    routes.get(cls)!.push({ method, path, handler: String(propertyKey) });
  };
}

export function routesOf(cls: Function): string[] {
  const prefix = prefixes.get(cls) ?? "";
  return (routes.get(cls) ?? []).map((r) => {
    const full = "/" + [prefix, r.path].filter((p) => p !== "").join("/");
    return `${r.method} ${full} -> ${r.handler}`;
  });
}
```

import { Body, ConflictException, Controller, Injectable, Module, Post } from "@nestjs/common";
import { IsEmail, IsOptional, IsString, Length, Matches, MinLength } from "class-validator";

export class SignupDto {
  @IsEmail()
  email: string;

  @IsString()
  @MinLength(8)
  @Matches(/\d/, { message: "password חייב לכלול ספרה" })
  password: string;

  @IsOptional()
  @IsString()
  @Length(2, 40)
  displayName?: string;
}

type User = { id: number; email: string; displayName: string | null; password: string };

@Injectable()
export class UsersService {
  private users: User[] = [];

  signup(dto: SignupDto) {
    const email = dto.email.toLowerCase();
    if (this.users.some((u) => u.email === email)) throw new ConflictException("האימייל כבר רשום");
    const user: User = { id: this.users.length + 1, email, displayName: dto.displayName ?? null, password: dto.password };
    this.users.push(user);
    const { password, ...pub } = user;
    return pub;
  }
}

@Controller("signup")
export class SignupController {
  constructor(private readonly users: UsersService) {}
  @Post()
  signup(@Body() dto: SignupDto) {
    return this.users.signup(dto);
  }
}

@Module({ controllers: [SignupController], providers: [UsersService] })
export class SignupModule {}

import { BadRequestException, Controller, Get, Module, Query } from "@nestjs/common";
import { Type } from "class-transformer";
import { IsDateString, IsIn, IsInt, IsOptional, Max, Min } from "class-validator";

const EVENTS = [
  { id: 1, title: "Latte Art", city: "tel-aviv", date: "2026-12-01" },
  { id: 2, title: "Cupping", city: "haifa", date: "2026-11-20" },
  { id: 3, title: "Roasting 101", city: "tel-aviv", date: "2027-01-15" },
  { id: 4, title: "Espresso Lab", city: "jerusalem", date: "2026-12-10" },
  { id: 5, title: "Pour Over", city: "haifa", date: "2027-02-02" },
];

export class EventsQuery {
  @IsOptional()
  @IsIn(["tel-aviv", "haifa", "jerusalem"])
  city?: string;

  @IsOptional()
  @IsDateString()
  from?: string;

  @IsOptional()
  @IsDateString()
  to?: string;

  @IsOptional()
  @Type(() => Number)
  @IsInt()
  @Min(1)
  @Max(20)
  limit = 10;
}

@Controller("events")
export class EventsController {
  @Get()
  list(@Query() q: EventsQuery) {
    if (q.from && q.to && q.from > q.to) throw new BadRequestException("from חייב להיות לפני to");
    return EVENTS.filter((e) => (!q.city || e.city === q.city) && (!q.from || e.date >= q.from) && (!q.to || e.date <= q.to))
      .sort((a, b) => a.date.localeCompare(b.date))
      .slice(0, q.limit);
  }
}

@Module({ controllers: [EventsController] })
export class EventsModule {}

// P20-E55: הזמנה עם פריטים — DTO מקונן
import { BadRequestException, Body, Controller, Module, Post } from "@nestjs/common";
import { Type } from "class-transformer";
import { ArrayMaxSize, ArrayMinSize, IsArray, IsEmail, IsInt, IsNumber, Matches, Max, Min, ValidateNested } from "class-validator";

export class OrderItemDto {
  @Matches(/^[A-Z]{3}-\d{3}$/)
  sku: string;

  @IsInt()
  @Min(1)
  @Max(99)
  qty: number;

  @IsNumber({ maxDecimalPlaces: 2 })
  @Min(0.01)
  price: number;
}

export class CreateOrderDto {
  @IsEmail()
  customerEmail: string;

  @IsArray()
  @ArrayMinSize(1)
  @ArrayMaxSize(20)
  @ValidateNested({ each: true })
  @Type(() => OrderItemDto)
  items: OrderItemDto[];
}

@Controller("orders")
export class OrdersController {
  private nextId = 1;

  @Post()
  create(@Body() dto: CreateOrderDto) {
    const seen = new Set<string>();
    for (const i of dto.items) {
      if (seen.has(i.sku)) throw new BadRequestException(`sku כפול: ${i.sku}`);
      seen.add(i.sku);
    }
    const cents = dto.items.reduce((sum, i) => sum + Math.round(i.price * 100) * i.qty, 0);
    const items = dto.items.map((i) => ({ sku: i.sku, qty: i.qty, price: i.price }));
    return { id: this.nextId++, customerEmail: dto.customerEmail, items, total: cents / 100 };
  }
}

@Module({ controllers: [OrdersController] })
export class OrdersModule {}
```

// P20-E56: עמודים עם קישורים — next ו־prev
import { Controller, Get, Module, Query } from "@nestjs/common";
import { Type } from "class-transformer";
import { IsInt, IsOptional, IsString, Max, Min } from "class-validator";

const ARTICLES = Array.from({ length: 30 }, (_, i) => {
  const id = i + 1;
  const special: Record<number, string> = { 7: "Nest Tips", 21: "More Nest" };
  return { id, title: special[id] ?? `Article ${id}` };
});

export class ListArticlesQuery {
  @IsOptional() @Type(() => Number) @IsInt() @Min(1)
  page = 1;

  @IsOptional() @Type(() => Number) @IsInt() @Min(1) @Max(20)
  limit = 5;

  @IsOptional() @IsString()
  q?: string;
}

@Controller("articles")
export class ArticlesController {
  @Get()
  list(@Query() { page, limit, q }: ListArticlesQuery) {
    const all = q ? ARTICLES.filter((a) => a.title.toLowerCase().includes(q.toLowerCase())) : ARTICLES;
    const total = all.length;
    const pages = Math.ceil(total / limit);
    const link = (p: number) => "/articles?" + new URLSearchParams({ ...(q ? { q } : {}), page: String(p), limit: String(limit) });
    return {
      data: all.slice((page - 1) * limit, page * limit),
      meta: { page, limit, total, pages },
      links: { self: link(page), next: page < pages ? link(page + 1) : null, prev: page > 1 ? link(page - 1) : null },
    };
  }
}

@Module({ controllers: [ArticlesController] })
export class ArticlesModule {}

// P20-E64: interceptor של idempotency
import { Body, CallHandler, Controller, ExecutionContext, Get, Inject, Injectable, Module, NestInterceptor, Post, UseInterceptors } from "@nestjs/common";
import type { Request, Response } from "express";
import { Observable, of, tap } from "rxjs";

export const IDEMPOTENCY_STORE = "IDEMPOTENCY_STORE";

@Injectable()
export class IdempotencyInterceptor implements NestInterceptor {
  constructor(@Inject(IDEMPOTENCY_STORE) private readonly store: Map<string, unknown>) {}

  intercept(context: ExecutionContext, next: CallHandler): Observable<unknown> {
    const http = context.switchToHttp();
    const key = http.getRequest<Request>().get("idempotency-key");
    if (!key) return next.handle();
    if (this.store.has(key)) {
      http.getResponse<Response>().setHeader("Idempotent-Replayed", "true");
      return of(this.store.get(key));
    }
    return next.handle().pipe(tap((body) => this.store.set(key, body)));
  }
}

@Controller("payments")
export class PaymentsController {
  private payments: { id: number; amount: number }[] = [];

  @Post()
  @UseInterceptors(IdempotencyInterceptor)
  create(@Body() body: { amount: number }) {
    const p = { id: this.payments.length + 1, amount: body.amount };
    this.payments.push(p);
    return p;
  }

  @Get("count")
  count() {
    return { count: this.payments.length };
  }
}

@Module({
  controllers: [PaymentsController],
  providers: [IdempotencyInterceptor, { provide: IDEMPOTENCY_STORE, useFactory: () => new Map() }],
})
export class PaymentsModule {}
```

import { BadRequestException, Controller, Get, Inject, Module, Param, ParseFloatPipe } from "@nestjs/common";

export const CONFIG = "CONFIG";
export const PRICING = "PRICING";

export interface Pricing {
  name: string;
  price(base: number): number;
}

export class RegularPricing implements Pricing {
  name = "regular";
  price(base: number) {
    return base;
  }
}

export class HolidayPricing implements Pricing {
  name = "holiday";
  price(base: number) {
    return Math.round(Math.round(base * 100) * 0.85) / 100;
  }
}

@Controller("price")
export class PriceController {
  constructor(@Inject(PRICING) private readonly pricing: Pricing) {}

  @Get(":base")
  get(@Param("base", ParseFloatPipe) base: number) {
    if (!(base > 0)) throw new BadRequestException("base חייב להיות חיובי");
    return { base, final: this.pricing.price(base), strategy: this.pricing.name };
  }
}

  providers: [
    { provide: CONFIG, useValue: { holiday: false } },
    {
      provide: PRICING,
      useFactory: (cfg: { holiday: boolean }) => (cfg.holiday ? new HolidayPricing() : new RegularPricing()),
      inject: [CONFIG],
    },
  ],
})
export class PricingModule {}

// P20-E72: configureApp — prefix, pipe ו־filter
import { ArgumentsHost, Body, Catch, Controller, ExceptionFilter, Get, HttpException, INestApplication, Module, NotFoundException, Param, ParseIntPipe, Post, ValidationPipe } from "@nestjs/common";
import { IsString, Length } from "class-validator";
import type { Response } from "express";

export class CreateNoteDto {
  @IsString()
  @Length(1, 200)
  text: string;
}

@Controller("notes")
export class NotesController {
  private notes: { id: number; text: string }[] = [];

  @Post()
  create(@Body() dto: CreateNoteDto) {
    const note = { id: this.notes.length + 1, text: dto.text };
    this.notes.push(note);
    return note;
  }

  @Get(":id")
  findOne(@Param("id", ParseIntPipe) id: number) {
    const note = this.notes.find((n) => n.id === id);
    if (!note) throw new NotFoundException(`פתק ${id} לא נמצא`);
    return note;
  }
}

@Module({ controllers: [NotesController] })
export class NotesModule {}

const CODES: Record<number, string> = { 400: "BAD_REQUEST", 404: "NOT_FOUND" };

@Catch(HttpException)
export class ApiErrorFilter implements ExceptionFilter {
  catch(exception: HttpException, host: ArgumentsHost) {
    const res = host.switchToHttp().getResponse<Response>();
    const status = exception.getStatus();
    const body = exception.getResponse();
    const msg = typeof body === "object" && body !== null ? (body as { message?: unknown }).message : body;
    const error = Array.isArray(msg)
      ? { code: CODES[status] ?? "ERROR", message: "Validation failed", details: msg }
      : { code: CODES[status] ?? "ERROR", message: exception.message };
    res.status(status).json({ error });
  }
}

export function configureApp(app: INestApplication) {
  app.setGlobalPrefix("api");
  app.useGlobalPipes(new ValidationPipe({ whitelist: true, forbidNonWhitelisted: true, transform: true }));
  app.useGlobalFilters(new ApiErrorFilter());
}

// P20-E75: filter לכל שגיאה — גם JSON שבור ו־500
import { ArgumentsHost, BadRequestException, Body, Catch, Controller, ExceptionFilter, Get, HttpException, Module, NotFoundException, Param, Post } from "@nestjs/common";
import { APP_FILTER } from "@nestjs/core";
import type { Response } from "express";

const CODES: Record<number, string> = { 400: "BAD_REQUEST", 401: "UNAUTHORIZED", 403: "FORBIDDEN", 404: "NOT_FOUND", 409: "CONFLICT" };

@Catch()
export class AllErrorsFilter implements ExceptionFilter {
  catch(exception: unknown, host: ArgumentsHost) {
    const res = host.switchToHttp().getResponse<Response>();
    if (exception instanceof BadRequestException && /JSON/i.test(exception.message)) {
      return res.status(400).json({ error: { code: "INVALID_JSON", message: "גוף הבקשה אינו JSON תקין" } });
    }
    if (exception instanceof HttpException) {
      const status = exception.getStatus();
      return res.status(status).json({ error: { code: CODES[status] ?? "ERROR", message: exception.message } });
    }
    res.status(500).json({ error: { code: "INTERNAL", message: "שגיאה פנימית" } });
  }
}

@Controller("items")
export class ItemsController {
  @Get("boom/now")
  boom() {
    throw new Error("secret: db password");
  }

  @Get(":id")
  findOne(@Param("id") id: string) {
    if (id !== "1") throw new NotFoundException(`פריט ${id} לא נמצא`);
    return { id: 1 };
  }

  @Post()
  create(@Body() body: unknown) {
    return body;
  }
}

@Module({ controllers: [ItemsController], providers: [{ provide: APP_FILTER, useClass: AllErrorsFilter }] })
export class ItemsModule {}

// P20-E76: ZodValidationPipe עם נתיבים
import { ArgumentMetadata, BadRequestException, Body, Controller, Get, Module, Param, PipeTransform, Post, Query } from "@nestjs/common";
import { z } from "zod";

export class ZodValidationPipe implements PipeTransform {
  constructor(private readonly schema: z.ZodType) {}

  transform(value: unknown, metadata: ArgumentMetadata) {
    const r = this.schema.safeParse(value);
    if (r.success) return r.data;
    const type = metadata.type === "param" ? "params" : metadata.type;
    const details = r.error.issues.map((issue) => ({
      path: [type, ...(metadata.data ? [metadata.data] : []), ...issue.path.map(String)].join("."),
      message: issue.message,
    }));
    throw new BadRequestException({ error: { code: "VALIDATION_ERROR", message: "קלט לא תקין", details } });
  }
}

const CreateTask = z.object({
  title: z.string().trim().min(1).max(100),
  priority: z.enum(["low", "medium", "high"]).default("medium"),
});
const Id = z.coerce.number().int().positive();
const ListQuery = z.object({ page: z.coerce.number().int().min(1).default(1) });

@Controller("tasks")
export class TasksController {
  @Post()
  create(@Body(new ZodValidationPipe(CreateTask)) body: z.infer<typeof CreateTask>) {
    return body;
  }

  @Get(":id")
  findOne(@Param("id", new ZodValidationPipe(Id)) id: number) {
    return { id };
  }

  @Get()
  list(@Query(new ZodValidationPipe(ListQuery)) query: z.infer<typeof ListQuery>) {
    return query;
  }
}

@Module({ controllers: [TasksController] })
export class TasksModule {}
```

// P20-E78: אפליקציה לבדיקות — createTestApp
import { Controller, Get, Inject, Injectable, Module } from "@nestjs/common";
import { Test } from "@nestjs/testing";

export const CLOCK = "CLOCK";
export const TASKS = "TASKS";
export type Task = { id: number; title: string; dueDate: string | null; done: boolean };

@Injectable()
export class TasksService {
  constructor(
    @Inject(TASKS) private readonly tasks: Task[],
    @Inject(CLOCK) private readonly clock: { now(): Date },
  ) {}

  overdue() {
    const today = this.clock.now().toISOString().slice(0, 10);
    return this.tasks
      .filter((t) => !t.done && t.dueDate !== null && t.dueDate < today)
      .sort((a, b) => a.dueDate!.localeCompare(b.dueDate!));
  }
}

@Controller("tasks")
export class TasksController {
  constructor(private readonly service: TasksService) {}
  @Get("overdue")
  overdue() {
    return this.service.overdue();
  }
}

@Module({
  controllers: [TasksController],
  providers: [TasksService, { provide: TASKS, useValue: [] }, { provide: CLOCK, useValue: { now: () => new Date() } }],
})
export class TasksModule {}

export async function createTestApp({ now, tasks }: { now: Date; tasks: Task[] }) {
  const moduleRef = await Test.createTestingModule({ imports: [TasksModule] })
    .overrideProvider(CLOCK)
    .useValue({ now: () => now })
    .overrideProvider(TASKS)
    .useValue(tasks)
    .compile();
  const app = moduleRef.createNestApplication({ logger: false });
  await app.init();
  return app;
}

// P20-E82: לקרוא מסמך OpenAPI בקוד
const ORDER = ["get", "post", "put", "patch", "delete"];

export function listEndpoints(doc: any): string[] {
  const rows = Object.entries(doc.paths ?? {}).flatMap(([path, ops]) =>
    Object.keys(ops as object)
      .filter((m) => ORDER.includes(m))
      .map((m) => ({ path, m })),
  );
  rows.sort((a, b) => (a.path === b.path ? ORDER.indexOf(a.m) - ORDER.indexOf(b.m) : a.path < b.path ? -1 : 1));
  return rows.map((r) => `${r.m.toUpperCase()} ${r.path}`);
}

export function requiredFields(doc: any, schema: string): string[] {
  const s = doc.components?.schemas?.[schema];
  if (!s) throw new Error(`אין סכמה ${schema}`);
  return [...(s.required ?? [])].sort();
}
```
<p>מסמך OpenAPI הוא סתם JSON — ואפשר לבדוק אותו בקוד: "כל endpoint מתועד", "כל DTO מסמן שדות חובה". זו ההתחלה של בדיקות חוזה.</p>

// P20-E83: /health עם בדיקת מסד ו־timeout
import { Controller, Get, HttpException, Inject, Module } from "@nestjs/common";

export const DB_PING = "DB_PING";

@Controller("health")
export class HealthController {
  constructor(@Inject(DB_PING) private readonly ping: () => Promise<void>) {}

  @Get()
  async check() {
    let timer: NodeJS.Timeout | undefined;
    const timeout = new Promise<never>((_, reject) => {
      timer = setTimeout(() => reject(new Error("timeout")), 200);
    });
    try {
      await Promise.race([this.ping(), timeout]);
      return { status: "ok", checks: { db: "up" } };
    } catch {
      throw new HttpException({ status: "error", checks: { db: "down" } }, 503);
    } finally {
      clearTimeout(timer);
    }
  }
}

@Module({ controllers: [HealthController], providers: [{ provide: DB_PING, useValue: async () => {} }] })
export class HealthModule {}
```
<p>בלי timeout, מסד "תקוע" מחזיק את <code>/health</code> עד ש־Docker מוותר (שניות ארוכות), ובינתיים כל בדיקה חדשה תופסת עוד חיבור. ‏503 (ולא 500) אומר למאזן העומסים "אל תשלח לי תנועה עכשיו" — וזה מה שהוא בודק.</p>
