```js run mjs save=solutions/P15.5-E06.mjs
// P15.5-E06
const ENTITIES = { "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;", "'": "&#39;" };

export function escapeHtml(value) {
  if (value === null || value === undefined) return "";
  return String(value).replace(/[&<>"']/g, (ch) => ENTITIES[ch]);
}

export function renderTaskItem(task) {
  return `<li class="task ${escapeHtml(task.priority)}">${escapeHtml(task.title)}</li>`;
}

console.log(renderTaskItem({ title: '<img src=x onerror="alert(1)">', priority: "high" }));
console.log(renderTaskItem({ title: "Tom & Jerry", priority: 'x" onclick="alert(1)' }));
```

```js save=solutions/P15.5-E08.mjs
// P15.5-E08
export class ApiError extends Error {
  constructor(status, code, message, details = []) {
    super(message);
    this.name = "ApiError";
    this.status = status;
    this.code = code;
    this.details = details;
  }
}

export function createApiClient(baseUrl) {
  async function request(method, path, body) {
    const res = await fetch(baseUrl + path, {
      method,
      headers: body === undefined ? {} : { "Content-Type": "application/json" },
      body: body === undefined ? undefined : JSON.stringify(body),
    });
    if (res.status === 204) return undefined;
    const data = await res.json().catch(() => null);
    if (!res.ok) {
      const e = data?.error;
      throw new ApiError(res.status, e?.code ?? "UNKNOWN", e?.message ?? `HTTP ${res.status}`, e?.details ?? []);
    }
    return data;
  }

  return {
    listTasks(params = {}) {
      const sp = new URLSearchParams();
      for (const [k, v] of Object.entries(params)) if (v !== "" && v !== undefined && v !== null) sp.append(k, String(v));
      const qs = sp.toString();
      return request("GET", `/tasks${qs ? `?${qs}` : ""}`);
    },
    createTask: (body) => request("POST", "/tasks", body),
    updateTask: (id, changes) => request("PATCH", `/tasks/${id}`, changes),
    deleteTask: (id) => request("DELETE", `/tasks/${id}`),
  };
}
```

```ts save=solutions/P15.5-E09.ts
// P15.5-E09
import type { Request, Response, NextFunction } from "express";

export function corsMiddleware(origins: string[]) {
  const allowed = new Set(origins);
  return (req: Request, res: Response, next: NextFunction) => {
    const origin = req.get("origin");
    const ok = origin !== undefined && allowed.has(origin);
    res.vary("Origin");
    if (ok) res.set("Access-Control-Allow-Origin", origin);

    if (req.method === "OPTIONS") {
```

```ts save=solutions/P15.5-E12.ts
// P15.5-E12
import type { Request, Response, NextFunction } from "express";

function matcher(pattern: string): (origin: string) => boolean {
  if (!pattern.includes("*")) return (origin) => origin === pattern;
  const source = pattern
    .split("*")
    .map((part) => part.replace(/[.*+?^${}()|[\]\\/]/g, "\\$&"))
    .join("[a-z0-9-]+");
  const re = new RegExp(`^${source}$`);
  return (origin) => re.test(origin);
}

export function createCors({ allow, credentials = false, maxAge = 600 }: { allow: string[]; credentials?: boolean; maxAge?: number }) {
  const matchers = allow.map(matcher);
  return (req: Request, res: Response, next: NextFunction) => {
    const origin = req.get("origin");
    if (origin === undefined) return next();
    res.vary("Origin");
    const ok = matchers.some((m) => m(origin));
    const isPreflight = req.method === "OPTIONS" && req.get("access-control-request-method") !== undefined;

    if (!ok) {
      if (isPreflight) {
        res.status(403).json({ error: { code: "CORS_ORIGIN_NOT_ALLOWED", message: `origin לא מורשה: ${origin}` } });
        return;
      }
      return next();
    }

    res.set("Access-Control-Allow-Origin", origin);
    if (credentials) res.set("Access-Control-Allow-Credentials", "true");
    if (isPreflight) {
      res.set({
        "Access-Control-Allow-Methods": "GET,POST,PUT,PATCH,DELETE",
        "Access-Control-Allow-Headers": req.get("access-control-request-headers") ?? "Content-Type",
        "Access-Control-Max-Age": String(maxAge),
      });
      res.sendStatus(204);
      return;
    }
    next();
  };
}
```
<p>למה לא <code>*</code> עם credentials? הדפדפן עצמו אוסר את השילוב — ובצדק: "כל אתר רשאי לקרוא, כולל עם העוגיות של המשתמש" זה בדיוק מה שה־Same-Origin Policy נועד למנוע. ותבניות preview (Vercel, Netlify) הן המקרה הנפוץ ביותר ל־wildcard — וגם המקום שבו regex רשלני (בלי <code>^...$</code>, בלי קידוד נקודות) פותח פרצה.</p>

```js run mjs save=solutions/P15.5-E14.mjs
// P15.5-E14: debounce לשדה חיפוש
export function debounce(fn, ms) {
  let timer = null;
  function debounced(...args) {
    clearTimeout(timer);
    timer = setTimeout(() => {
      timer = null;
      fn(...args);
    }, ms);
  }
  debounced.cancel = () => {
    clearTimeout(timer);
    timer = null;
  };
  return debounced;
}

// המשתמש מקליד "c", "co", "cor", "cors" — 30ms בין הקשות. החיפוש רץ פעם אחת, 100ms אחרי ההקשה האחרונה

const sleep = (ms) => new Promise((resolve) => setTimeout(resolve, ms));

  search(q);
  await sleep(30);
}
await sleep(150);

search.cancel();
await sleep(150);
console.log("סוף");
```

```js run mjs save=solutions/P15.5-E15.mjs
export async function loadDashboard(getJson) {
  const [me, tasks, stats] = await Promise.allSettled([getJson("/me"), getJson("/tasks"), getJson("/stats")]);
  if (me.status === "rejected") {
    throw me.reason;
  }
  const errors = [];

    if (result.status === "fulfilled") {
      return result.value;
    }
    errors.push(`${name}: ${result.reason.message}`);
    return null;
  };
  return { me: me.value, tasks: valueOf(tasks, "tasks"), stats: valueOf(stats, "stats"), errors };
}
```

```js run mjs save=solutions/P15.5-E30.mjs
const esc = (s) => String(s).replaceAll("&", "&amp;").replaceAll("<", "&lt;").replaceAll(">", "&gt;").replaceAll('"', "&quot;").replaceAll("'", "&#39;");

export function renderOptions(options, selected, placeholder) {
  const isSel = (v) => selected !== undefined && selected !== null && String(v) === String(selected);
  const option = (value, label, sel) => `<option value="${esc(value)}"${sel ? " selected" : ""}>${esc(label)}</option>`;
  const items = options.map((o) => option(o.value, o.label, isSel(o.value)));
  if (placeholder !== undefined) items.unshift(option("", placeholder, !options.some((o) => isSel(o.value))));
  return items.join("");
}
```

```js run mjs save=solutions/P15.5-E31.mjs
// P15.5-E31: בודק תוויות לטופס
function tags(html) {
  return [...html.matchAll(/<(\w+)([^>]*)>/g)].map((m) => {
    const attrs = {};
    for (const a of m[2].matchAll(/([\w-]+)(?:\s*=\s*("[^"]*"|'[^']*'|[^\s>]+))?/g)) {
      attrs[a[1].toLowerCase()] = (a[2] ?? "").replace(/^["']|["']$/g, "");
    }
    return { tag: m[1].toLowerCase(), attrs };
  });
}

export function checkForm(html) {
  const list = tags(html);
  const problems = [];
  const counts = new Map();
  for (const t of list) if (t.attrs.id) counts.set(t.attrs.id, (counts.get(t.attrs.id) ?? 0) + 1);
  for (const [id, n] of counts) if (n > 1) problems.push(`id כפול: ${id}`);

  const labelsFor = new Set(list.filter((t) => t.tag === "label" && t.attrs.for).map((t) => t.attrs.for));
  for (const t of list) {
    if (!["input", "select", "textarea"].includes(t.tag)) continue;
    if (["submit", "hidden", "button"].includes((t.attrs.type ?? "").toLowerCase())) continue;
    if (!t.attrs.name) problems.push(`שדה בלי name: ${t.attrs.id || "?"}`);
    if (!t.attrs.id || !labelsFor.has(t.attrs.id)) problems.push(`שדה בלי label: ${t.attrs.name || "?"}`);
  }
  for (const t of list) {
    if (t.tag === "label" && t.attrs.for && !counts.has(t.attrs.for)) problems.push(`label יתום: ${t.attrs.for}`);
  }
  return problems;
}
```

```js run mjs save=solutions/P15.5-E44.mjs
// P15.5-E44: לחשב specificity
export function specificity(selector) {
  const ids = (selector.match(/#[\w-]+/g) ?? []).length;
  const classes = (selector.match(/\.[\w-]+|\[[^\]]+\]|:[\w-]+/g) ?? []).length;
  const tags = selector
    .trim()
    .split(/\s+/)
    .filter((part) => /^[a-z][\w-]*/i.test(part)).length;
  return [ids, classes, tags];
}

function compare(a, b) {
  for (let i = 0; i < 3; i++) if (a[i] !== b[i]) return a[i] - b[i];
  return 0;
}

export function winner(element, rules) {
  let best;
  for (const rule of rules) {
    if (!best || compare(specificity(rule.selector), specificity(best.selector)) >= 0) best = rule;
  }
  return best?.value;
}
```

```js run mjs save=solutions/P15.5-E49.mjs
// P15.5-E49: יחס ניגודיות
function rgb(hex) {
  let h = hex.slice(1);
  if (h.length === 3) h = [...h].map((c) => c + c).join("");
  return [0, 2, 4].map((i) => parseInt(h.slice(i, i + 2), 16));
}

export function luminance(hex) {
  const [R, G, B] = rgb(hex).map((c) => {
    const s = c / 255;
    return s <= 0.03928 ? s / 12.92 : ((s + 0.055) / 1.055) ** 2.4;
  });
  return 0.2126 * R + 0.7152 * G + 0.0722 * B;
}

export function contrast(hex1, hex2) {
  const a = luminance(hex1);
  const b = luminance(hex2);
  const ratio = (Math.max(a, b) + 0.05) / (Math.min(a, b) + 0.05);
  return Math.round(ratio * 100) / 100;
}

export function readable(fg, bg) {
  return contrast(fg, bg) >= 4.5;
}

for (const [fg, bg] of [["#000", "#fff"], ["#8c959f", "#ffffff"], ["#0969da", "#ffffff"], ["#82071e", "#ffebe9"], ["#fff", "#fff"]]) {
  console.log(fg, bg, contrast(fg, bg), readable(fg, bg));
}
```

```js run mjs save=solutions/P15.5-E50.mjs
export function minifyCss(css) {
```

```js save=solutions/P15.5-E57.mjs
export function bindCounter(doc, { max = 10 } = {}) {
  const dec = doc.querySelector("#dec");
  const inc = doc.querySelector("#inc");
  const out = doc.querySelector("#count");
  let value = 0;

    out.textContent = String(value);
    dec.disabled = value === 0;
    inc.disabled = value === max;

    if (value < max) value++;

}
```

```js run mjs save=solutions/P15.5-E72.mjs
// P15.5-E72: לפרק CORS_ORIGINS
function isOrigin(x) {
  try {
    const u = new URL(x);
    return /^https?:$/.test(u.protocol) && u.pathname === "/" && !u.search && !u.hash && !x.endsWith("?") && !x.endsWith("#");
  } catch {
    return false;
  }
}

export function parseCorsOrigins(value, fallback = ["http://localhost:5500"]) {
  if (value === undefined || value.trim() === "") return [...fallback];
  if (value.trim() === "*") throw new Error("CORS_ORIGINS=* אסור — כתוב רשימה מפורשת");
  const items = value.split(",").map((s) => s.trim()).filter(Boolean);
  const bad = items.filter((x) => !isOrigin(x));
  if (bad.length) throw new Error(`CORS_ORIGINS לא תקין: ${bad.join(", ")}`);
  return [...new Set(items.map((x) => new URL(x).origin))];
}

  try {
    parseCorsOrigins(v);
  } catch (err) {
    console.log(err.message);
  }
}
```

```js run mjs save=solutions/P15.5-E74.mjs
// P15.5-E74: רק התשובה האחרונה
export function latestOnly(fn) {
  let latest = 0;
  return async (...args) => {
    const my = ++latest;
    try {
      const value = await fn(...args);
      return my === latest ? { stale: false, value } : { stale: true };
    } catch (err) {
      if (my !== latest) return { stale: true };
      throw err;
    }
```

```js run mjs save=solutions/P15.5-E79.mjs
export function summarizeTasks(tasks, today) {
  const open = tasks.filter((t) => t.status !== "done");
  return {
    total: tasks.length,
    done: tasks.length - open.length,
    overdue: open.filter((t) => t.dueDate && t.dueDate < today).length,
    dueToday: open.filter((t) => t.dueDate === today).length,
  };
}

export function summaryText({ total, done, overdue, dueToday }) {
  if (total === 0) return "אין משימות";
  const parts = [total === 1 ? "משימה אחת" : `${total} משימות`];
  if (done) parts.push(`${done} בוצעו`);
  if (overdue) parts.push(`${overdue} באיחור`);
  if (dueToday) parts.push(`${dueToday} להיום`);
  return parts.join(" · ");
}

];
```

```js run mjs save=solutions/P15.5-E80.mjs
const ORDER = ["high", "medium", "low", "other"];
const due = (t) => t.dueDate ?? "9999-12-31";

function compare(a, b) {
  return (a.status === "done") - (b.status === "done") || due(a).localeCompare(due(b)) || a.id - b.id;
}

export function groupByPriority(tasks) {
  const groups = new Map(ORDER.map((p) => [p, []]));
  for (const t of tasks) groups.get(ORDER.includes(t.priority) && t.priority !== "other" ? t.priority : "other").push(t);
  for (const [key, list] of groups) {
    if (list.length === 0) groups.delete(key);
    else list.sort(compare);
  }
  return groups;
}

for (const [priority, list] of groupByPriority(tasks)) console.log(priority, list.map((t) => t.id));
```

```out
high [ 4, 3, 6, 2 ]
low [ 1 ]
other [ 5 ]
```
<p><code>Map</code> שנבנתה עם המפתחות בסדר הנכון — ואז "ממלאים" — שומרת את הסדר גם כשהמשימה הראשונה במערך היא <code>low</code>. ‏<code>true - false</code> הוא 1: כך "בוצעה" עוברת לסוף בשורה אחת. ו־<code>"9999-12-31"</code> במקום <code>null</code> הופך "בלי תאריך" ל"הכי מאוחר" — בלי <code>if</code>־ים.</p>
@mistakes
