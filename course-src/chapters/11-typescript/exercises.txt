// P11-E07: אותה לוגיקה מפרק 6, עם טיפוסים
export function cleanTags(input: string, max = 5): string[] {
  return input
    .split(",")
    .map((tag) => tag.trim().toLowerCase())
    .filter((tag) => tag !== "")
    .filter((tag, i, all) => all.indexOf(tag) === i)
    .slice(0, max);
}

export function chunk<T>(arr: T[], size: number): T[][] {
  const result: T[][] = [];
  for (let i = 0; i < arr.length; i += size) {
    result.push(arr.slice(i, i + size));
  }
  return result;
}

export function median(nums: number[]): number | null {
  if (nums.length === 0) {
    return null;
  }
  const sorted = [...nums].sort((a, b) => a - b);
  const mid = Math.floor(sorted.length / 2);
  return sorted.length % 2 === 1 ? sorted[mid] : (sorted[mid - 1] + sorted[mid]) / 2;
}

export type Product = {
  readonly id: number;
  name: string;
  price: number;
  tags: string[];
  stock?: number;
};

export function createProduct(id: number, name: string, price: number): Product {
  return { id, name, price, tags: [] };
}

// P11-E12
export function toNumber(value: unknown): number | null {
  if (typeof value === "number") {
    return Number.isFinite(value) ? value : null;
  }
  if (typeof value === "string") {
    const text = value.trim();
    if (text === "") {
      return null;
    }
    const n = Number(text);
    return Number.isFinite(n) ? n : null;
  }
  return null;
}

// P11-E14
export function groupBy<T>(items: T[], keyFn: (item: T) => string): Record<string, T[]> {
  const result: Record<string, T[]> = {};
  for (const item of items) {
    const key = keyFn(item);
    (result[key] ??= []).push(item);
  }
  return result;
}

export function indexBy<T, K>(items: T[], keyFn: (item: T) => K): Map<K, T> {
  return new Map(items.map((item) => [keyFn(item), item]));
}

export function sumBy<T>(items: T[], fn: (item: T) => number): number {
  return items.reduce((s, item) => s + fn(item), 0);
}

  | { kind: "circle"; radius: number }
  | { kind: "rect"; width: number; height: number }
  | { kind: "triangle"; base: number; height: number };

function unreachable(x: never): never {

      a = shape.width * shape.height;

export type Result<T> = { ok: true; value: T } | { ok: false; error: string };

export function ok<T>(value: T): Result<T> {
  return { ok: true, value };
}

  return { ok: false, error };
}

export function parseAge(input: string): Result<number> {

export type User = { id: number; email: string; roles: string[] };

export function isUser(value: unknown): value is User {
  return (
    typeof value === "object" &&
    value !== null &&
    "id" in value &&
    typeof value.id === "number" &&
    "email" in value &&
    typeof value.email === "string" &&
    "roles" in value &&
    Array.isArray(value.roles) &&
    value.roles.every((r: unknown) => typeof r === "string")
  );
}

export function parseUsers(json: string): { users: User[]; invalid: number } {
  let data: unknown;
  try {
    data = JSON.parse(json);
  } catch {
    throw new Error("הקלט אינו מערך JSON");
  }
  if (!Array.isArray(data)) {
    throw new Error("הקלט אינו מערך JSON");
  }
  const users = data.filter(isUser);
  return { users, invalid: data.length - users.length };
}

export type Inventory = Map<string, number>;

export function restock(inv: Inventory, code: string, qty: number): Inventory {
  const next = new Map(inv);
  next.set(code, (next.get(code) ?? 0) + qty);
  return next;

export function reserve(inv: Inventory, code: string, qty: number): Result<Inventory> {
  const current = inv.get(code);

// P11-E24: event bus מוקלד
export type AppEvent =
  | { type: "order.created"; orderId: number; total: number }
  | { type: "order.paid"; orderId: number; amount: number }
  | { type: "user.signedUp"; userId: number; email: string };

type EventOf<T extends AppEvent["type"]> = Extract<AppEvent, { type: T }>;

export function createBus() {
  const handlers = new Map<AppEvent["type"], ((e: AppEvent) => void)[]>();
  const log: AppEvent[] = [];

  function on<T extends AppEvent["type"]>(type: T, handler: (e: EventOf<T>) => void): void {
    const list = handlers.get(type) ?? [];
    list.push((e) => handler(e as EventOf<T>));
    handlers.set(type, list);
  }

  function emit(event: AppEvent): void {
    log.push(event);
    for (const handler of handlers.get(event.type) ?? []) {
      handler(event);
    }
  }

  function history(): AppEvent[] {
    return [...log];
  }

  return { on, emit, history };
}

const bus = createBus();
bus.on("order.paid", (e) => console.log(`הזמנה ${e.orderId} שולמה: ${e.amount}`));
bus.emit({ type: "order.created", orderId: 7, total: 120 });
bus.emit({ type: "order.paid", orderId: 7, amount: 120 });
console.log(bus.history().length);

// P11-E25: mini validation library
export type Result<T> = { ok: true; value: T } | { ok: false; error: string };
export type Validator<T> = (value: unknown, path?: string) => Result<T>;

const fail = (path: string | undefined, msg: string): { ok: false; error: string } => ({
  ok: false,
  error: path ? `${path}: ${msg}` : msg,
});

export function string(opts: { min?: number; max?: number } = {}): Validator<string> {
  return (value, path) => {
    if (typeof value !== "string") return fail(path, "חייב להיות מחרוזת");
    if (opts.min !== undefined && value.length < opts.min) return fail(path, `לפחות ${opts.min} תווים`);
    if (opts.max !== undefined && value.length > opts.max) return fail(path, `לכל היותר ${opts.max} תווים`);
    return { ok: true, value };
  };
}

export function number(opts: { min?: number; max?: number; integer?: boolean } = {}): Validator<number> {
  return (value, path) => {
    if (typeof value !== "number" || !Number.isFinite(value)) return fail(path, "חייב להיות מספר");
    if (opts.integer && !Number.isInteger(value)) return fail(path, "חייב להיות מספר שלם");
    if (opts.min !== undefined && value < opts.min) return fail(path, `לפחות ${opts.min}`);
    if (opts.max !== undefined && value > opts.max) return fail(path, `לכל היותר ${opts.max}`);
    return { ok: true, value };
  };
}

export function boolean(): Validator<boolean> {
  return (value, path) => (typeof value === "boolean" ? { ok: true, value } : fail(path, "חייב להיות true או false"));
}

export function optional<T>(v: Validator<T>): Validator<T | undefined> {
  return (value, path) => (value === undefined ? { ok: true, value: undefined } : v(value, path));
}

export function array<T>(v: Validator<T>): Validator<T[]> {
  return (value, path) => {
    if (!Array.isArray(value)) return fail(path, "חייב להיות מערך");
    const out: T[] = [];
    for (let i = 0; i < value.length; i++) {
      const r = v(value[i], `${path ?? ""}[${i}]`);
      if (!r.ok) return r;
      out.push(r.value);
    }
    return { ok: true, value: out };
  };
}

export function object<S extends Record<string, Validator<unknown>>>(
  shape: S
): Validator<{ [K in keyof S]: S[K] extends Validator<infer T> ? T : never }> {
  return (value, path) => {
    if (typeof value !== "object" || value === null || Array.isArray(value)) return fail(path, "חייב להיות אובייקט");
    const input = value as Record<string, unknown>;
    const out: Record<string, unknown> = {};
    for (const key of Object.keys(shape)) {
      const r = shape[key](input[key], path ? `${path}.${key}` : key);
      if (!r.ok) return r;
      if (r.value !== undefined) out[key] = r.value;
    }
    return { ok: true, value: out as { [K in keyof S]: S[K] extends Validator<infer T> ? T : never } };
  };
}

  email: string({ min: 3 }),
  note: optional(string({ max: 20 })),
  items: array(object({ sku: string(), qty: number({ min: 1, integer: true }) })),
});
console.log(JSON.stringify(orderSchema({ email: "a@b.c", items: [{ sku: "A", qty: 2 }], hacker: true })));

export type User = {
  id: number;
  name: string;
  email: string;
  passwordHash: string;
  role: "admin" | "member";
};

export type PublicUser = Omit<User, "passwordHash">;
export type UserPatch = Partial<Pick<User, "name" | "email">>;
export type RoleCounts = Record<User["role"], number>;

export function toPublic(u: User): PublicUser {
  const { passwordHash, ...rest } = u;
  return rest;
}

export function applyPatch(u: User, patch: UserPatch): User {
  return { ...u, ...patch };
}

export function countRoles(users: User[]): RoleCounts {
  const counts: RoleCounts = { admin: 0, member: 0 };
  for (const u of users) {
    counts[u.role]++;
  }
  return counts;
}

console.log(applyPatch(dana, { name: "דנה כ." }).name);
console.log(countRoles([dana]));

// P11-E27: משך זמן
export function parseDuration(text: string): number | null {
  const words = text.split(" ").filter((w) => w !== "");
  if (words.length === 0) return null;

  let total = 0;
  for (const w of words) {
    const unit = w.slice(-1);
    const digits = w.slice(0, -1);
    const n = Number(digits);
    if (digits === "" || !Number.isInteger(n) || n < 0) return null;
    if (unit === "h") {
      total += n * 60;
    } else if (unit === "m") {
      total += n;
    } else {
      return null;
    }
  }
  return total;
}

export function formatDuration(minutes: number): string {
  const h = Math.floor(minutes / 60);
  const m = minutes % 60;
  const parts: string[] = [];
  if (h > 0) parts.push(`${h}h`);
  if (m > 0 || h === 0) parts.push(`${m}m`);
  return parts.join(" ");
}

for (const t of ["1h 30m", "15m 1h", "1.5h", ""]) {
  const minutes = parseDuration(t);
  console.log(JSON.stringify(t), "→", minutes === null ? "לא תקין" : formatDuration(minutes));
}

// P11-E28: שליפת משתמשים
export type User = { id: number; name: string };

export async function findUsers(
  ids: number[],
  getUser: (id: number) => Promise<User | undefined>
): Promise<{ found: User[]; missing: number[] }> {
  const unique = [...new Set(ids)];
  const results = await Promise.all(unique.map(getUser));
  const found: User[] = [];
  const missing: number[] = [];
  unique.forEach((id, i) => {
    const user = results[i];
    if (user) {
      found.push(user);
    } else {
      missing.push(id);
    }
  });
  return { found, missing };
}

  [3, { id: 3, name: "עומר" }],
]);

export function byLength(words: string[]): Map<number, string[]> {
  const groups = new Map<number, string[]>();
  for (const w of words) {
    const list = groups.get(w.length) ?? [];

export function receipt(names: string[], prices: number[]): string {
  if (names.length !== prices.length) throw new Error("names ו־prices באורך שונה");

export type Product = { id: number; name: string; price: number };

export async function loadProducts(ids: number[], getOne: (id: number) => Promise<Product>): Promise<Product[]> {
  return Promise.all(ids.map((id) => getOne(id)));

  let best: Product | null = null;

  }
  return best;

};

export type Product = { id: number; name: string; price: number };

export function indexById(products: Product[]): Map<number, Product> {
  const index = new Map<number, Product>();
  for (const p of products) {
    if (index.has(p.id)) throw new Error(`מזהה כפול: ${p.id}`);
    index.set(p.id, p);
  }
  return index;
}

export function missingIds(ids: number[], index: Map<number, Product>): number[] {
  const missing = new Set<number>();
  for (const id of ids) {
    if (!index.has(id)) missing.add(id);
  }
  return [...missing];
}

export function orderTotal(lines: { id: number; qty: number }[], index: Map<number, Product>): number {
  let total = 0;
  for (const line of lines) {
    const p = index.get(line.id);
    if (!p) throw new Error(`אין מוצר ${line.id}`);
    total += p.price * line.qty;
  }
  return total;
}

]);

    case "L":

export type AccountEvent =
  | { type: "deposit"; amount: number }
  | { type: "withdraw"; amount: number }
  | { type: "fee"; amount: number; reason: string }
  | { type: "freeze" };

export function balance(events: AccountEvent[]): number {
  let total = 0;
  for (const e of events) {
    switch (e.type) {
      case "deposit":
        total += e.amount;
        break;
      case "withdraw":
      case "fee":
        total -= e.amount;
        break;
      case "freeze":

// P11-E68: פקודות CLI כ־union
export type Command =
  | { kind: "add"; name: string; amount: number }
  | { kind: "list" }
  | { kind: "remove"; id: number }
  | { kind: "error"; message: string };

const error = (message: string): Command => ({ kind: "error", message });

export function parseCommand(line: string): Command {
  const [cmd, ...args] = line.trim().split(" ").filter((w) => w !== "");
  switch (cmd) {
    case "add": {
      if (args.length !== 2) return error("שימוש: add <name> <amount>");
      const amount = Number(args[1]);
      if (!Number.isFinite(amount) || amount <= 0) return error("סכום לא תקין");
      return { kind: "add", name: args[0], amount };
    }
    case "list":
      return args.length === 0 ? { kind: "list" } : error("שימוש: list");
    case "remove": {
      if (args.length !== 1) return error("שימוש: remove <id>");
      const id = Number(args[0]);
      if (!Number.isInteger(id) || id <= 0) return error("מזהה לא תקין");
      return { kind: "remove", id };
    }
    default:
      return error("פקודה לא מוכרת");
  }
}

for (const line of ["add קפה 12.5", "  list ", "remove 0", "add קפה", "fly"]) {
  console.log(JSON.stringify(parseCommand(line)));
}

export type Todo = { title: string; done: boolean; priority?: "low" | "high" };

export function parseTodo(json: string): Todo | null {
  let data: unknown;
  try {
    data = JSON.parse(json);
  } catch {
    return null;
  }
  if (typeof data !== "object" || data === null || Array.isArray(data)) return null;
  if (!("title" in data) || typeof data.title !== "string" || data.title.trim() === "") return null;
  if (!("done" in data) || typeof data.done !== "boolean") return null;
  const todo: Todo = { title: data.title, done: data.done };
  if ("priority" in data) {
    if (data.priority !== "low" && data.priority !== "high") return null;
    todo.priority = data.priority;
  }
  return todo;
}

export type Config = { host: string; port: number; debug: boolean };

export function parseConfig(raw: unknown): Config {
  if (typeof raw !== "object" || raw === null || Array.isArray(raw)) {
    throw new Error("ההגדרות חייבות להיות אובייקט");
  }
  if (!("host" in raw) || typeof raw.host !== "string" || raw.host.trim() === "") {
    throw new Error("host לא תקין");
  }

  if ("debug" in raw) {
    if (typeof raw.debug !== "boolean") throw new Error("debug לא תקין");
    debug = raw.debug;
  }
  return { host: raw.host, port: raw.port, debug };
}
