```js run mjs save=solutions/P14-E06.mjs
// P14-E06
const DEFAULT_PORTS = { "http:": 80, "https:": 443 };

export function parseUrl(text) {
  let url;
  try {
    url = new URL(text);
  } catch {
    throw new Error(`כתובת לא תקינה: ${text}`);
  }
  return {
    protocol: url.protocol.slice(0, -1),
    host: url.hostname,
    port: url.port ? Number(url.port) : (DEFAULT_PORTS[url.protocol] ?? null),
    path: url.pathname,
    query: Object.fromEntries(url.searchParams),
  };
}

console.log(JSON.stringify(parseUrl("http://localhost:4000/products?maxPrice=100&sort=price")));
console.log(JSON.stringify(parseUrl("https://shop.co.il")));
```

```js run mjs save=solutions/P14-E07.mjs
// P14-E07
const CATEGORIES = { 2: "success", 3: "redirect", 4: "client-error", 5: "server-error" };
const RETRYABLE = new Set([408, 429, 500, 502, 503, 504]);

export function statusCategory(code) {
  if (!Number.isInteger(code)) return "unknown";
  return CATEGORIES[Math.floor(code / 100)] ?? "unknown";
}

export function shouldRetry(code) {

}
```

```js run mjs save=solutions/P14-E08.mjs
// P14-E08
export function parseHeaders(lines) {
  const result = {};
  for (const line of lines) {
    const i = line.indexOf(":");
    if (i <= 0) throw new Error(`header לא תקין: ${line}`);
    const name = line.slice(0, i).trim().toLowerCase();
    const value = line.slice(i + 1).trim();
    result[name] = name in result ? `${result[name]}, ${value}` : value;
  }
  return result;
}
```

```js run mjs save=solutions/P14-E09.mjs
// P14-E09
export function buildQuery(params) {
  const sp = new URLSearchParams();
  for (const [key, value] of Object.entries(params)) {
    if (value === undefined || value === null) continue;
    for (const v of Array.isArray(value) ? value : [value]) sp.append(key, String(v));
  }
  const s = sp.toString();
  return s ? `?${s}` : "";
}

console.log(buildQuery({ page: 2, tag: ["a", "b"], q: "red & blue", skip: undefined }));
console.log(JSON.stringify(buildQuery({ skip: null })));
```

```js run mjs save=solutions/P14-E10.mjs
export function toCurl({ method = "GET", url, headers = {}, body }) {
```

```js run mjs save=solutions/P14-E13.mjs
// P14-E13
export function parseRequest(raw) {
  const text = raw.replaceAll("\r\n", "\n");
  const split = text.indexOf("\n\n");
  const head = split === -1 ? text : text.slice(0, split);
  const rawBody = split === -1 ? "" : text.slice(split + 2);
  const [requestLine, ...headerLines] = head.split("\n");

  const parts = requestLine.split(" ");
  if (parts.length !== 3) throw new Error("שורת בקשה לא תקינה");
  const [method, target] = parts;
  const url = new URL(target, "http://x");

  const headers = {};
  for (const line of headerLines) {
    const i = line.indexOf(":");
    headers[line.slice(0, i).trim().toLowerCase()] = line.slice(i + 1).trim();
  }

  let body;
  if (rawBody !== "") {
    body = headers["content-type"]?.includes("json") ? JSON.parse(rawBody) : rawBody;
  }
  return { method, path: url.pathname, query: Object.fromEntries(url.searchParams), headers, body };
}
```

```js run mjs save=solutions/P14-E15.mjs
function substitute(s, vars) {
  return s.replace(/\{\{(\w+)\}\}/g, (_, name) => {
    if (!(name in vars)) throw new Error(`משתנה לא מוגדר: ${name}`);
    return vars[name];
  });
}

export function parseHttpFile(text) {
  const vars = {};
  const blocks = [];
  let current = null;
  for (const line of text.replaceAll("\r\n", "\n").split("\n")) {
    if (line.startsWith("###")) {
      current = { name: line.slice(3).trim(), lines: [] };
      blocks.push(current);
    } else if (current) {
      current.lines.push(line);
    } else {
      const m = line.match(/^@(\w+)\s*=\s*(.*)$/);
      if (m) vars[m[1]] = m[2].trim();
    }
  }

  const requests = [];
  for (const { name, lines } of blocks) {
    let i = 0;
    const isSkippable = (l) => l.trim() === "" || l.startsWith("#") || l.startsWith("//");
    while (i < lines.length && isSkippable(lines[i])) i++;
    if (i === lines.length) continue;
    const [method, url] = lines[i].trim().split(/\s+/);
    i++;
    const headers = {};
    for (; i < lines.length && lines[i].trim() !== ""; i++) {
      const c = lines[i].indexOf(":");
      headers[lines[i].slice(0, c).trim()] = substitute(lines[i].slice(c + 1).trim(), vars);
    }
    const body = lines.slice(i).join("\n").trim();
    requests.push({ name, method, url: substitute(url, vars), headers, body: body === "" ? undefined : substitute(body, vars) });
  }
  return requests;
}
```

```js run mjs save=solutions/P14-E16.mjs
// P14-E16
const segments = (p) => p.split("/").filter(Boolean);

function matchParts(pattern, parts) {
  if (pattern.length !== parts.length) return null;
  const params = {};
  for (let i = 0; i < pattern.length; i++) {
    if (pattern[i].startsWith(":")) params[pattern[i].slice(1)] = decodeURIComponent(parts[i]);
    else if (pattern[i] !== parts[i]) return null;
  }
  return params;
}

export function createRouter() {
  const routes = [];
  return {
    add(method, pattern, handler) {
      routes.push({ method, pattern: segments(pattern), handler });
    },
    match(method, path) {
      const parts = segments(path);
      const allow = [];
      for (const route of routes) {
        const params = matchParts(route.pattern, parts);
        if (!params) continue;
        if (route.method === method) return { status: 200, handler: route.handler, params };
        allow.push(route.method);
      }
      return allow.length > 0 ? { status: 405, allow } : { status: 404 };
    },
  };
}
```

```js run mjs save=solutions/P14-E17.mjs
export class HttpError extends Error {
  constructor(status, message) {
    super(message);
    this.name = "HttpError";
    this.status = status;
  }
}

export async function readJsonBody(req, { limit = 1_000_000 } = {}) {
  if (!req.headers["content-type"]?.includes("application/json")) {
    throw new HttpError(415, "Content-Type חייב להיות application/json");
  }
  const chunks = [];
  let size = 0;
  for await (const chunk of req) {
    const buf = Buffer.from(chunk);
    size += buf.length;
    if (size > limit) throw new HttpError(413, "הגוף גדול מדי");
    chunks.push(buf);
  }
  const text = Buffer.concat(chunks).toString("utf8");
  if (text.trim() === "") throw new HttpError(400, "גוף ריק");
  try {
    return JSON.parse(text);
  } catch {
    throw new HttpError(400, "JSON לא תקין");
  }
}
```

```js save=solutions/P14-E18.mjs
import http from "node:http";

function send(res, status, body, headers = {}) {
  if (status === 204) {
    res.writeHead(204, headers);
    return res.end();
  }
  res.writeHead(status, { "Content-Type": "application/json; charset=utf-8", ...headers });
  res.end(JSON.stringify(body));
}

async function readJson(req) {
  const chunks = [];
  for await (const chunk of req) chunks.push(Buffer.from(chunk));
  return JSON.parse(Buffer.concat(chunks).toString("utf8") || "null");
}

// מחזירה הודעת שגיאה, או null אם תקין
function validateNote(body, { partial }) {
  if (typeof body !== "object" || body === null || Array.isArray(body)) return "הגוף חייב להיות אובייקט";
  if (partial && body.text === undefined && body.done === undefined) return "צריך text או done";

    return "text: מחרוזת של 1–200 תווים";
  }
  if (body.done !== undefined && typeof body.done !== "boolean") return "done: true או false";
```

```js save=solutions/P14-E19.mjs
// P14-E19
const RETRYABLE = new Set([408, 429, 500, 502, 503, 504]);
const sleep = (ms) => new Promise((r) => setTimeout(r, ms));

export async function fetchWithRetry(url, init = {}, { retries = 2, delayMs = 200 } = {}) {
  let lastError;
  let lastResponse;
  for (let attempt = 0; attempt <= retries; attempt++) {
    if (attempt > 0) await sleep(delayMs * 2 ** (attempt - 1));
    try {
      lastResponse = await fetch(url, init);
      lastError = undefined;
      if (!RETRYABLE.has(lastResponse.status)) return lastResponse;
    } catch (err) {
      lastError = err;
      lastResponse = undefined;
    }
  }
  if (lastError) throw lastError;
  return lastResponse;
}
```

```js run mjs save=solutions/P14-E24.mjs
async function readBody(req) {
  if (!req.headers["content-type"]?.includes("application/json")) return undefined;
  const chunks = [];
  for await (const chunk of req) chunks.push(Buffer.from(chunk));
  const text = Buffer.concat(chunks).toString("utf8");
  if (text === "") return undefined;
  try {
    return JSON.parse(text);
  } catch {
    throw Object.assign(new Error("JSON לא תקין"), { status: 400 });
  }
}

function enhance(res) {
  res.status = (code) => {
    res.statusCode = code;
    return res;
  };
  res.json = (body) => {
    if (res.statusCode === 204) return res.end();
    res.setHeader("Content-Type", "application/json; charset=utf-8");
    res.end(JSON.stringify(body));
  };
}

export function createApp() {
  const middleware = [];
  const routes = [];

  async function route(req, res) {
    const parts = segments(req.path);
    const allow = [];
    for (const r of routes) {
      const params = matchParts(r.pattern, parts);
      if (!params) continue;
      if (r.method !== req.method) {
        allow.push(r.method);
        continue;
      }
      req.params = params;
      return r.handler(req, res);
    }
    if (allow.length > 0) {
      res.setHeader("Allow", allow.join(", "));
      return res.status(405).json({ error: "Method Not Allowed" });
    }
    res.status(404).json({ error: "לא נמצא" });
  }

  async function handle(req, res) {
    enhance(res);
    try {
      const url = new URL(req.url, "http://localhost");
      req.path = url.pathname;
      req.query = Object.fromEntries(url.searchParams);
      req.params = {};
      req.body = await readBody(req);
      let i = 0;
      const next = async () => {
        const fn = middleware[i++];
        if (fn) await fn(req, res, next);
        else await route(req, res);
      };
      await next();
    } catch (err) {
      if (res.headersSent) return res.end();
      if (err.status) return res.status(err.status).json({ error: err.message });
      console.error(err);
      res.status(500).json({ error: "שגיאה פנימית" });
    }
  }

  const app = {
    use(fn) {
      middleware.push(fn);
    },
    listen(port, cb) {
      return http.createServer(handle).listen(port, cb);
    },
  };
  for (const method of ["get", "post", "put", "patch", "delete"]) {
    app[method] = (pattern, handler) => routes.push({ method: method.toUpperCase(), pattern: segments(pattern), handler });
  }
  return app;
}
```

```js save=solutions/P14-E25.mjs
// P14-E25
import http from "node:http";

export function createLimitedServer({ limit, windowMs, now = Date.now }) {
  const windows = new Map();

  return http.createServer((req, res) => {
    const client = req.headers["x-client-id"] ?? "anonymous";
    const t = now();
    let w = windows.get(client);
    if (!w || t - w.start >= windowMs) {
      w = { start: t, count: 0 };
      windows.set(client, w);
    }
    w.count++;

    const headers = { "Content-Type": "application/json; charset=utf-8" };
    if (w.count > limit) {
      headers["Retry-After"] = String(Math.ceil((w.start + windowMs - t) / 1000));
      headers["X-RateLimit-Remaining"] = "0";
      res.writeHead(429, headers);
      return res.end(JSON.stringify({ error: "יותר מדי בקשות" }));
    }
    headers["X-RateLimit-Remaining"] = String(limit - w.count);
    res.writeHead(200, headers);
    res.end(JSON.stringify({ ok: true }));
  });
}
```

```js run save=solutions/P14-E26.mjs
// P14-E26: פירוק header של Cookie
export function parseCookies(header) {
  const cookies = {};
  if (!header) {
    return cookies;
  }
  for (const part of header.split(";")) {
    const i = part.indexOf("=");
    if (i === -1) {
      continue;
    }
    const name = part.slice(0, i).trim();
    const value = part.slice(i + 1).trim();
    if (name) {
      cookies[name] = decodeURIComponent(value);
    }
  }
  return cookies;
}
```

```js run save=solutions/P14-E27.mjs
// P14-E27: סטטיסטיקת בקשות כמחלקה
export class RequestStats {
  #routes = new Map();

  record(method, path, status, ms) {
    const route = `${method} ${path}`;
    const s = this.#routes.get(route) ?? { count: 0, totalMs: 0, errors: 0 };
    s.count++;
    s.totalMs += ms;
    if (status >= 500) {
      s.errors++;
    }
    this.#routes.set(route, s);
  }

  get total() {
    let sum = 0;
    for (const s of this.#routes.values()) {
      sum += s.count;
    }
    return sum;
  }

  summary() {
    return [...this.#routes]
      .map(([route, s]) => ({ route, count: s.count, avgMs: Math.round(s.totalMs / s.count), errors: s.errors }))
      .sort((a, b) => b.count - a.count || a.route.localeCompare(b.route));
  }
}

console.log(stats.summary());
```

```js run save=solutions/P14-E28.mjs
// P14-E28: לחכות שהשרת יעלה
import http from "node:http";

const sleep = (ms) => new Promise((resolve) => setTimeout(resolve, ms));

export async function waitUntilReady(url, { tries = 10, delayMs = 100 } = {}) {
  let attempt = 0;
  while (attempt < tries) {
    attempt++;
    try {
      const res = await fetch(url);
      if (res.ok) {
        return attempt;
      }
    } catch {
      // השרת עוד לא מקשיב — ננסה שוב
    }
    if (attempt < tries) {
      await sleep(delayMs);
    }
  }
  throw new Error(`השרת לא מוכן אחרי ${tries} ניסיונות`);
}

  calls++;

});

  const url = `http://localhost:${server.address().port}/health`;
  console.log("מוכן בניסיון", await waitUntilReady(url, { delayMs: 20 }));
  try {
    await waitUntilReady("http://localhost:1/health", { tries: 3, delayMs: 20 });
  } catch (err) {
    console.log(err.message);
  }
  server.close();
});
```

```js run mjs save=solutions/P14-E33.mjs
  const m = String(method).trim().toUpperCase();
  switch (m) {
    case "GET":
    case "DELETE":

}
```

```js run mjs save=solutions/P14-E37.mjs
// P14-E37: לבחור status לפי הסדר
export function decideStatus({ method, loggedIn, allowed, exists, valid, duplicate }) {
  if (!loggedIn) return 401;
  if (!allowed) return 403;
  if (method !== "POST" && !exists) return 404;
  if (!valid) return 400;
  if (duplicate) return 409;
  if (method === "POST") return 201;
  if (method === "DELETE") return 204;
  return 200;
}
```

```js run mjs save=solutions/P14-E41.mjs
export function countByCategory(lines) {
  const counts = { "2xx": 0, "3xx": 0, "4xx": 0, "5xx": 0, invalid: 0 };
  for (const line of lines) {
    if (line.trim() === "") continue;
    const parts = line.trim().split(/\s+/);
    const status = Number(parts[2]);
    if (parts.length !== 4 || !Number.isInteger(status) || status < 100 || status > 599) {
      counts.invalid++;
      continue;
    }
    const key = `${Math.floor(status / 100)}xx`;
    if (key in counts) counts[key]++;
  }
  return counts;
```

```js run mjs save=solutions/P14-E43.mjs
// P14-E43: קישורי עמודים
function positiveInt(text, fallback) {
  const n = Number(text);
  return Number.isInteger(n) && n > 0 ? n : fallback;
}

export function pageLinks(url, total) {
  const params = new URL(url).searchParams;
  const page = positiveInt(params.get("page"), 1);
  const limit = Math.min(positiveInt(params.get("limit"), 10), 100);
  const pages = Math.max(1, Math.ceil(total / limit));
  const link = (p) => {
    const u = new URL(url);
    u.searchParams.set("page", String(p));
    return u.href;
  };
  return {
    page,
    limit,
    pages,
    prev: page > 1 ? link(page - 1) : null,
    next: page < pages ? link(page + 1) : null,
  };
}

console.log(pageLinks("http://shop.test/products", 0));
console.log(pageLinks("http://shop.test/products?page=-3&limit=abc", 25));
```

```js run mjs save=solutions/P14-E50.mjs
// P14-E50: לקרוא Content-Type
export function parseContentType(value) {
  if (typeof value !== "string" || value.trim() === "") return null;
  const [type, ...params] = value.split(";");
  let charset = null;
  for (const param of params) {
    const i = param.indexOf("=");
    if (i === -1) continue;
    const name = param.slice(0, i).trim().toLowerCase();
    if (name === "charset") charset = param.slice(i + 1).trim().replace(/^"|"$/g, "").toLowerCase();
  }
  return { type: type.trim().toLowerCase(), charset };
}

export function isJson(value) {
  const parsed = parseContentType(value);
  if (!parsed) return false;
  return parsed.type === "application/json" || parsed.type.endsWith("+json");
}
```

```js run mjs save=solutions/P14-E51.mjs
// P14-E51: לבחור פורמט לפי Accept
function parseAccept(accept) {
  return accept.split(",").map((part) => {
    const [type, ...params] = part.split(";").map((s) => s.trim().toLowerCase());
    let q = 1;
    for (const p of params) if (p.startsWith("q=")) q = Number(p.slice(2));
    return { type, q };
  });
}

export function chooseFormat(accept, available) {
  if (!accept || accept.trim() === "") return available[0] ?? null;
  const list = parseAccept(accept);
  let best = null;
  let bestQ = 0;
  for (const format of available) {
    const f = format.toLowerCase();
    const rule =
      list.find((r) => r.type === f) ?? list.find((r) => r.type === f.split("/")[0] + "/*") ?? list.find((r) => r.type === "*/*");
    const q = rule?.q ?? 0;
    if (q > bestQ) {
      best = format;
      bestQ = q;
    }
  }
  return best;
}

console.log(chooseFormat("image/png", both));
console.log(chooseFormat("*/*, application/json;q=0", both));
```

```js run mjs save=solutions/P14-E57.mjs
// P14-E57: sessions בזיכרון
export class SessionStore {
  #sessions = new Map();

  login(user) {
    const token = crypto.randomUUID();
    this.#sessions.set(token, user);
    return token;
  }

  userFor(headers) {
    const [scheme, token, ...rest] = (headers.authorization ?? "").split(" ");
    if (scheme !== "Bearer" || !token || rest.length > 0) return null;
    return this.#sessions.get(token) ?? null;
  }

  logout(token) {
    return this.#sessions.delete(token);
  }

  get size() {
    return this.#sessions.size;
  }
}
```

```js run mjs save=solutions/P14-E63.mjs
// P14-E63: מפקודת curl לבקשה
export function parseCurlArgs(args) {
  let method;
  let url;
  let body;
  const headers = {};
  for (let i = 0; i < args.length; i++) {
    const arg = args[i];
    switch (arg) {
      case "-X":
      case "--request":
        method = args[++i].toUpperCase();
        break;
      case "-H":
      case "--header": {
        const h = args[++i];
        const colon = h.indexOf(":");
        headers[h.slice(0, colon).trim()] = h.slice(colon + 1).trim();
        break;
      }
      case "-d":
      case "--data":
        body = args[++i];
        break;
      case "-i":
      case "-v":
      case "-s":
        break;
      default:
        if (arg.startsWith("-")) throw new Error(`דגל לא מוכר: ${arg}`);
        url = arg;
    }
  }
  if (url === undefined) throw new Error("חסרה כתובת");
  return { method: method ?? (body === undefined ? "GET" : "POST"), url, headers, body };
}
```

```js run mjs save=solutions/P14-E64.mjs
// P14-E64: לפרק שורת פקודה עם מירכאות
export function splitArgs(cmd) {
  const args = [];
  let current = "";
  let started = false;
  let quote = null;
  const finish = () => {
    if (started) args.push(current);
    current = "";
    started = false;
  };
  for (let i = 0; i < cmd.length; i++) {
    const ch = cmd[i];
    if (quote === "'") {
      if (ch === "'") quote = null;
      else current += ch;
    } else if (quote === '"') {
      if (ch === "\\" && (cmd[i + 1] === '"' || cmd[i + 1] === "\\")) current += cmd[++i];
      else if (ch === '"') quote = null;
      else current += ch;
    } else if (ch === "'" || ch === '"') {
      quote = ch;
      started = true;
    } else if (ch === "\\" && cmd[i + 1] === "\n") {
      i++;
      finish();
    } else if (ch === " " || ch === "\n" || ch === "\t") {
      finish();
    } else {
      current += ch;
      started = true;
    }
  }
  if (quote) throw new Error("מירכאה לא נסגרה");
  finish();
  return args;
}
```

```js run mjs save=solutions/P14-E65.mjs
function headersOf(lines) {
  const headers = {};
  for (const line of lines.slice(1)) {
    const i = line.indexOf(":");
    if (line.trim() === "" || i === -1) continue;
    headers[line.slice(0, i).trim().toLowerCase()] = line.slice(i + 1).trim();
  }
  return headers;
}

export function parseVerbose(text) {
  const lines = text.split(/\r?\n/);
  const sent = lines.filter((l) => l.startsWith("> ") || l === ">").map((l) => l.slice(2));
  const got = lines.filter((l) => l.startsWith("< ") || l === "<").map((l) => l.slice(2));
  let request = null;
  let response = null;
  if (sent.length) {
    const [method, path] = sent[0].split(" ");
    request = { method, path, headers: headersOf(sent) };
  }
  if (got.length) {
    response = { status: Number(got[0].split(" ")[1]), headers: headersOf(got) };
  }
  return { request, response };
}
```

```js run mjs save=solutions/P14-E66.mjs
// P14-E66: מבקשות לקובץ .http
export function toHttpFile(requests) {
  const origins = new Set(requests.map((r) => new URL(r.url).origin));
  const base = requests.length >= 2 && origins.size === 1 ? [...origins][0] : null;
  const lines = [];
  if (base) lines.push(`@base = ${base}`, "");
  requests.forEach((r, i) => {
    if (i > 0) lines.push("");
    const url = base && r.url.startsWith(base) ? "{{base}}" + r.url.slice(base.length) : r.url;
    lines.push(`### ${r.name}`, `${r.method} ${url}`);
    for (const [name, value] of Object.entries(r.headers ?? {})) lines.push(`${name}: ${value}`);
    if (r.body !== undefined) {
      lines.push("", typeof r.body === "string" ? r.body : JSON.stringify(r.body, null, 2));
    }
  });
  return lines.join("\n") + "\n";
}
```

```js run mjs save=solutions/P14-E67.mjs
  let best = null;
  const byHost = {};
  for (const e of entries) {
    if (!best || e.ms > best.ms) best = e;
```

```js run mjs save=solutions/P14-E68.mjs
// P14-E68: בודק לקובצי .http
export function lintHttpFile(text) {
  const lines = text.split(/\r?\n/);
  const defined = new Set();
  for (const line of lines) {
    const m = line.match(/^@(\w+)\s*=/);
    if (m) defined.add(m[1]);
  }
  const warnings = [];
  const warn = (line, text) => warnings.push({ line, text });
  let state = "before";
  let req = null;
  const endRequest = () => {
    if (req && req.hasBody && !req.hasType) warn(req.line, "יש גוף בלי Content-Type");
    req = null;
  };
  lines.forEach((line, i) => {
    const n = i + 1;
    for (const m of line.matchAll(/\{\{(\w+)\}\}/g)) {
      if (!defined.has(m[1])) warn(n, `המשתנה ${m[1]} לא הוגדר`);
    }
    if (line.startsWith("###")) {
      endRequest();
      state = "before";
      return;
    }
    const isComment = line.startsWith("#") || line.startsWith("//");
    if (state === "before") {
      if (line.trim() === "" || isComment || line.startsWith("@")) return;
      req = { line: n, hasBody: false, hasType: false };
      state = "headers";
    } else if (state === "headers") {
      if (line.trim() === "") state = "body";
      else if (isComment) return;
      else if (!/^[^\s:]+:\s*\S/.test(line)) warn(n, "header לא תקין");
      else if (line.split(":")[0].toLowerCase() === "content-type") req.hasType = true;
    } else if (line.trim() !== "") {
      req.hasBody = true;
    }
  });
  endRequest();
  return warnings.sort((a, b) => a.line - b.line).map((w) => `שורה ${w.line}: ${w.text}`);
}
```

```js run mjs save=solutions/P14-E72.mjs
// P14-E72: להסתיר סודות בפקודת curl
const SECRET_PARAMS = new Set(["token", "key", "apikey", "password", "secret"]);

function redactHeader(header) {
  const colon = header.indexOf(":");
  if (colon === -1) return header;
  const name = header.slice(0, colon);
  const value = header.slice(colon + 1).trim();
  switch (name.trim().toLowerCase()) {
    case "authorization": {
      const space = value.indexOf(" ");
      return `${name}: ${space === -1 ? "***" : value.slice(0, space) + " ***"}`;
    }
    case "cookie":
      return `${name}: ${value
        .split(";")
        .filter((pair) => pair.trim() !== "")
        .map((pair) => pair.trim().split("=")[0] + "=***")
        .join("; ")}`;
    default:
      return header;
  }
}

function redactUrl(text) {
  const u = new URL(text);
  for (const key of [...u.searchParams.keys()]) {
    if (SECRET_PARAMS.has(key.toLowerCase())) u.searchParams.set(key, "***");
  }
  return u.href;
}

export function redactCurl(args) {
  return args.map((arg, i) => {
    if (args[i - 1] === "-H" || args[i - 1] === "--header") return redactHeader(arg);
    if (arg.startsWith("http://") || arg.startsWith("https://")) return redactUrl(arg);
    return arg;
  });
}
```

```js run mjs save=solutions/P14-E73.mjs
// P14-E73: סיכום קובצי .http בתיקייה
import fs from "node:fs/promises";
import os from "node:os";
import path from "node:path";

export async function summarizeHttpDir(dir) {
  let names;
  try {
    names = await fs.readdir(dir);
  } catch (err) {
    if (err.code === "ENOENT") throw new Error(`התיקייה לא קיימת: ${dir}`);
    throw err;
  }
  const files = names.filter((n) => n.endsWith(".http")).sort();
  return Promise.all(
    files.map(async (file) => {
      const text = await fs.readFile(path.join(dir, file), "utf8");
      const methods = {};
      let requests = 0;
      for (const line of text.split(/\r?\n/)) {
        const m = line.match(/^(GET|POST|PUT|PATCH|DELETE) /);
        if (!m) continue;
        requests++;
        methods[m[1]] = (methods[m[1]] ?? 0) + 1;
      }
      return { file, requests, methods };
    }),
  );
}

const dir = await fs.mkdtemp(path.join(os.tmpdir(), "http-"));

await fs.writeFile(path.join(dir, "notes.txt"), "GET http://x/ignored\n");
console.log(await summarizeHttpDir(dir));
await fs.rm(dir, { recursive: true });
```

```js run mjs save=solutions/P14-E74.mjs
// P14-E74: ממה שנכשל ב־Network — לפקודות curl
export function failedAsCurl(entries) {
  const failed = entries.filter((e) => e.status === 0 || e.status >= 400).sort((a, b) => a.startedAt - b.startedAt);
  const seen = new Set();
  const out = [];
  for (const e of failed) {
    const key = `${e.method} ${e.url}`;
    if (seen.has(key)) continue;
    seen.add(key);
    const parts = ["curl"];
    if (e.method !== "GET") parts.push("-X", e.method);
    parts.push(`'${e.url}'`);
    for (const [name, value] of Object.entries(e.headers ?? {})) {
      if (name.toLowerCase() !== "cookie") parts.push("-H", `'${name}: ${value}'`);
    }
    out.push(parts.join(" "));
  }
  return out;
}

const entries = [
  { method: "GET", url: "http://shop.test/api/me", status: 401, startedAt: 30, headers: { Accept: "application/json" } },

  { method: "GET", url: "http://shop.test/", status: 200, startedAt: 1 },
  { method: "GET", url: "http://shop.test/api/me", status: 401, startedAt: 20 },
  { method: "GET", url: "http://ads.test/pixel", status: 0, startedAt: 5 },
];
```

```js run mjs
  console.log("לקוח:", err.name);
}
server.closeAllConnections();
server.close();
```
```out
unhandledRejection: המסד לא זמין
לקוח: TimeoutError
```
<p>ה־<code>catch</code> לא רץ: <code>createServer</code> רק <strong>רשם</strong> את ה־handler ויצא. השגיאה נזרקה מאוחר יותר, בתוך Promise שאיש לא מחכה לו — ‏unhandled rejection. הלקוח לא קיבל כלום (אף אחד לא קרא ל־<code>res.end</code>) ונפל ב־timeout. בלי <code>process.on</code>, ‏Node (מגרסה 15) <strong>מפיל את כל התהליך</strong> על unhandled rejection — כל המשתמשים מתנתקים בגלל בקשה אחת. הפתרון: <code>try/catch</code> <strong>בתוך</strong> ה־handler האסינכרוני, שמחזיר 500. ‏Express 5 (פרק 15) עושה את זה בשבילך.</p>
@mistakes
<ul><li>לפתור עם <code>process.on("unhandledRejection")</code> בקוד אמיתי ולהמשיך כרגיל — השרת לא קורס, אבל הלקוח עדיין תקוע בלי תשובה.</li></ul>
