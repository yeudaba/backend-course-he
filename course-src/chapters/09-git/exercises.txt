```js run save=solutions/P09-E27.js
// P09-E27: ניתוח של git log
function parseLog(text) {
  return text
    .split("\n")
    .filter((line) => line.trim() !== "")
    .map((line) => {
      const i = line.indexOf(" ");
      return { hash: line.slice(0, i), message: line.slice(i + 1) };
    });
}

function typeOf(message) {
  const i = message.indexOf(":");
  if (i <= 0) return "other";
  const type = message.slice(0, i);
  return type.includes(" ") ? "other" : type.toLowerCase();
}

function countTypes(commits) {
  const counts = {};
  for (const { message } of commits) {
    const type = typeOf(message);
    counts[type] = (counts[type] ?? 0) + 1;
  }
  return counts;
}

const log = `a1b2c3d feat: add login
9f8e7d6 fix: wrong total
5a4b3c2 Update README
1122334 feat: add logout
`;
const commits = parseLog(log);
console.log(commits[2]);
console.log(countTypes(commits));

module.exports = { parseLog, countTypes };
```

```js run save=solutions/P09-E28.js
function matches(filePath, rule) {
  if (rule.endsWith("/")) {
    return filePath.startsWith(rule) || filePath.includes("/" + rule);
  }
  if (rule.startsWith("*.")) {
    return filePath.endsWith(rule.slice(1));
  }
  return filePath === rule || filePath.endsWith("/" + rule);
}

function isIgnored(filePath, gitignore) {
  const rules = gitignore
    .split("\n")
    .map((line) => line.trim())
    .filter((line) => line !== "" && !line.startsWith("#"));
  return rules.some((rule) => matches(filePath, rule));
}

const gitignore = `
# תלויות
node_modules/
*.log
.env
`;
for (const p of ["node_modules/lodash/index.js", "src/app.log", "config/.env", ".env.example", "src/index.js"]) {
  console.log(p.padEnd(30), isIgnored(p, gitignore));
}

module.exports = { isIgnored };
```

```js run file=09-git/starter/P09-E42.js save=solutions/P09-E42.js
  const result = { staged: [], modified: [], untracked: [] };
  for (const line of text.split("\n")) {
    if (line === "") {
      continue;
    }
    const file = line.slice(3);

    if (line[1] !== " ") {

    }
  }
```

```js run file=09-git/starter/P09-E43.js save=solutions/P09-E43.js
function shortlog(text) {
  const counts = new Map();
  for (const line of text.split("\n")) {
    if (line.trim() === "") {
```

```js run file=09-git/starter/P09-E62.js save=solutions/P09-E62.js
function resolve(text, choice) {
  const out = [];
  let state = "out";
  let ours = [];
  let theirs = [];
  for (const line of text.split("\n")) {
    if (line.startsWith("<<<<<<<")) {
      state = "ours";
    } else if (line === "=======" && state === "ours") {
      state = "theirs";
    } else if (line.startsWith(">>>>>>>") && state === "theirs") {
      if (choice === "ours" || choice === "both") {
        out.push(...ours);
      }
      if (choice === "theirs" || choice === "both") {
        out.push(...theirs);
      }
      ours = [];
      theirs = [];
      state = "out";
    } else if (state === "ours") {
      ours.push(line);
    } else if (state === "theirs") {
      theirs.push(line);
    } else {
      out.push(line);
    }
  }
  return out.join("\n");
}
```

```js run file=09-git/starter/P09-E76.js save=solutions/P09-E76.js
const WORDS = ["PASSWORD", "SECRET", "TOKEN", "API_KEY"];

function findSecrets(text) {
  const found = [];
  const lines = text.split("\n");
  for (let i = 0; i < lines.length; i++) {
    const line = lines[i];
    if (line.includes("sk_live_")) {
      found.push({ line: i + 1, kind: "stripe" });
      continue;
    }
    if (line.includes("PRIVATE KEY-----")) {
      found.push({ line: i + 1, kind: "private-key" });
      continue;
    }
    const eq = line.indexOf("=");
    if (eq === -1) {
      continue;
    }
    const name = line.slice(0, eq).trim().split(" ").at(-1).toUpperCase();
    let value = line.slice(eq + 1).trim();
    if (value.endsWith(";")) {
      value = value.slice(0, -1);
    }
    value = value.replaceAll('"', "").replaceAll("'", "").trim();
    if (WORDS.some((w) => name.includes(w)) && value !== "" && !value.startsWith("process.env")) {
      found.push({ line: i + 1, kind: "assignment" });
    }
  }
  return found;
}
```

<p>סקריפט שמדמה את זרימת העבודה עם Pull Request, עם שרת מקומי (P09-E74):</p>
<ol><li><code>server.git</code> (bare) ו־<code>git clone server.git app</code>. ב־<code>app</code>: commit <code>Add README</code> (<code>README.md</code>: <code># App</code>) ו־<code>git push -u origin main</code>.</li><li>ענף <code>feature/usage</code>: הוסף ל־<code>README.md</code> שורה <code>Run: node index.js</code> (בסוף), commit <code>Add usage to README</code>, ו־<code>git push -u origin feature/usage</code>.</li><li>"מיזוג ה־PR": חזרה ל־<code>main</code>, ו־<code>git merge --no-ff --no-edit feature/usage</code> — <code>--no-ff</code> מכריח merge commit גם כשאפשר fast-forward, בדיוק כמו כפתור Merge pull request ב־GitHub. ‏<code>git push</code>.</li><li>מחיקת הענף המקומי.</li></ol>
<p>📄 כותבים בקובץ <code>09-git/starter/P09-E80.sh</code> (יש בו רק הערה). מנסים בתיקייה ריקה, ובודקים:</p>
```term
% mkdir -p playground/e80 && cd playground/e80
% bash ../../09-git/starter/P09-E80.sh
% git -C server.git log --oneline --graph --all
% cd ../..
% npm run check P09-E80
```

@hint1
<p>כל הפקודות אחרי ה־clone — בתוך <code>app</code>. להוספת שורה: <code>&gt;&gt;</code>.</p>
@hint2
<p>מחיקה: <code>git branch -d feature/usage</code> — אחרי <code>git switch main</code> והמיזוג (אחרת Git יסרב: הענף לא מוזג).</p>
@solution
```bash file=09-git/starter/P09-E80.sh save=solutions/P09-E80.sh
git init --bare server.git
git clone server.git app
cd app
echo "# App" > README.md
git add README.md
git commit -m "Add README"
git push -u origin main

git switch -c feature/usage
echo "Run: node index.js" >> README.md
git commit -am "Add usage to README"
git push -u origin feature/usage

git switch main
git merge --no-ff --no-edit feature/usage
git push
git branch -d feature/usage
```
```term
% git -C server.git log --oneline --graph --all
*   a7d2e51 (HEAD -> main) Merge branch 'feature/usage'
|\
| * 3f9c0b8 (feature/usage) Add usage to README
|/
* 0e41d2c Add README
```
<p>(ה־hashes אצלך יהיו שונים.) הענף <code>feature/usage</code> עדיין קיים <strong>בשרת</strong> — ב־GitHub יש כפתור "Delete branch" אחרי המיזוג (או <code>git push origin --delete feature/usage</code>). ה־merge commit משאיר בהיסטוריה סימן ברור: "כאן נכנסה התכונה הזו" — ואפשר לבטל את כולה עם <code>revert</code> אחד.</p>
@mistakes
<ul><li><code>git merge</code> בלי <code>--no-ff</code> — fast-forward, ובהיסטוריה לא רואים שהיה כאן PR.</li></ul>
