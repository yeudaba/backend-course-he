```ts save=solutions/P29-E06.ts
import type Anthropic from "@anthropic-ai/sdk";

export const MODEL = "claude-opus-5-5";

export type AskResult = { ok: true; text: string } | { ok: false; reason: "refusal" | "max_tokens" | "empty" };

// שאלה אחת ← תשובה. שלושה דברים שכל קריאה צריכה: לבדוק stop_reason לפני שקוראים את התוכן,
// לאסוף רק בלוקים מסוג text (יש גם thinking, tool_use…), ולא להניח ש־content[0] הוא הטקסט
export async function ask(client: Anthropic, question: string, { system }: { system?: string } = {}): Promise<AskResult> {
  const response = await client.messages.create({
    model: MODEL,
    max_tokens: 4096,
    ...(system ? { system } : {}),
    messages: [{ role: "user", content: question }],
  });
  if (response.stop_reason === "refusal") return { ok: false, reason: "refusal" };
  if (response.stop_reason === "max_tokens") return { ok: false, reason: "max_tokens" }; // תשובה חתוכה באמצע — לא מציגים כאילו שלמה
  const text = response.content
    .filter((b): b is Anthropic.TextBlock => b.type === "text")
    .map((b) => b.text)
    .join("")
    .trim();
  return text ? { ok: true, text } : { ok: false, reason: "empty" };
}
```

```ts save=solutions/P29-E07.ts
// P29-E07
import type Anthropic from "@anthropic-ai/sdk";

export interface Turn {
  role: "user" | "assistant";
  text: string;
}

// ה־API לא זוכר כלום: כל בקשה שולחת את כל השיחה. כדי לא לשלם על שיחה אינסופית — שומרים רק את maxTurns
// הזוגות האחרונים. ההודעה הראשונה חייבת להיות user, וההיסטוריה לא נגמרת בשאלה פתוחה
export function buildMessages(history: readonly Turn[], question: string, maxTurns = 5): Anthropic.MessageParam[] {
  const clean = history.filter((t) => t.text.trim() !== "");
  let recent = clean.slice(-maxTurns * 2);
  while (recent.length > 0 && recent[0].role !== "user") recent = recent.slice(1);
  if (recent.length > 0 && recent[recent.length - 1].role === "user") recent = recent.slice(0, -1); // שאלה בלי תשובה (נכשלה) — לא שולחים שוב
  return [...recent.map((t) => ({ role: t.role, content: t.text })), { role: "user", content: question }];
}
```

```ts save=solutions/P29-E08.ts
// עלות של בקשה מתוך usage. מחירים בדולר למיליון טוקנים — מהטבלה בשיעור 1 (לבדוק בדף התמחור לפני שמסתמכים)
export interface Prices {
  input: number;
  output: number;
  cacheWrite: number;
  cacheRead: number;
}

export interface Usage {
  input_tokens: number;
  output_tokens: number;
  cache_creation_input_tokens?: number | null;
  cache_read_input_tokens?: number | null;
}

export const PRICES: Record<string, Prices> = {
  "claude-opus-5-5": { input: 4, output: 20, cacheWrite: 5, cacheRead: 0.2 },
  "claude-sonnet-5-5": { input: 2, output: 10, cacheWrite: 2.5, cacheRead: 0.2 },
  "claude-haiku-5-5": { input: 0.1, output: 0.5, cacheWrite: 0.125, cacheRead: 0.01 },
};

export function costUsd(model: string, usage: Usage): number {
  const p = PRICES[model];
  if (!p) throw new Error(`אין מחיר למודל ${model}`);
  // input_tokens לא כולל את מה שנכתב ל־cache או נקרא ממנו — שלושה מונים נפרדים

      usage.output_tokens * p.output +
      (usage.cache_creation_input_tokens ?? 0) * p.cacheWrite +
      (usage.cache_read_input_tokens ?? 0) * p.cacheRead) /
    1_000_000;
  return Math.round(usd * 1_000_000) / 1_000_000; // דיוק של מיליונית דולר — בלי זנבות של 0.1+0.2
```

```ts save=solutions/P29-E10.ts
// P29-E10
import { z } from "zod";

// כשאין structured outputs (מודל אחר, ספק אחר, טקסט חופשי עם JSON בתוכו): מחלצים, מפענחים ומאמתים.
// פלט של מודל הוא קלט לא אמין — בדיוק כמו body של בקשה (פרק 15)
export function parseJsonReply<T>(text: string, schema: z.ZodType<T>): { ok: true; data: T } | { ok: false; error: string } {
  const fenced = /```(?:json)?\s*([\s\S]*?)```/.exec(text);
  let candidate = fenced ? fenced[1] : text;
  if (!fenced) {
    const start = candidate.search(/[[{]/);
    const end = Math.max(candidate.lastIndexOf("}"), candidate.lastIndexOf("]"));
    if (start === -1 || end < start) return { ok: false, error: "לא נמצא JSON בתשובה" };
    candidate = candidate.slice(start, end + 1);
  }
  let raw: unknown;
  try {
    raw = JSON.parse(candidate.trim());
  } catch {
    return { ok: false, error: "JSON לא תקין" };
  }
  const result = schema.safeParse(raw);
  return result.success ? { ok: true, data: result.data } : { ok: false, error: result.error.issues.map((i) => `${i.path.join(".") || "(root)"}: ${i.message}`).join("; ") };
}
```

```ts save=solutions/P29-E13.ts
export type Handler = (input: unknown) => Promise<string> | string;

// הלולאה של tool use: המודל מבקש כלי ← אנחנו מריצים ← מחזירים תוצאה ← שוב, עד end_turn.

// ותקרה על מספר הסיבובים — כדי שמודל "תקוע" לא ירוץ (וישלם) לנצח
export async function runTools(
  client: Anthropic,
  params: Omit<Anthropic.MessageCreateParamsNonStreaming, "messages"> & { messages: Anthropic.MessageParam[] },
  handlers: Record<string, Handler>,
  { maxIterations = 8 }: { maxIterations?: number } = {},
): Promise<{ message: Anthropic.Message; messages: Anthropic.MessageParam[]; toolCalls: number }> {
  const messages = [...params.messages];
  let toolCalls = 0;
  for (let i = 0; i < maxIterations; i++) {
    const message = await client.messages.create({ ...params, messages });
    if (message.stop_reason !== "tool_use") return { message, messages, toolCalls };
    messages.push({ role: "assistant", content: message.content });
    const results: Anthropic.ToolResultBlockParam[] = [];
    for (const block of message.content) {
      if (block.type !== "tool_use") continue;
      toolCalls++;
      const handler = handlers[block.name];
      if (!handler) {
        results.push({ type: "tool_result", tool_use_id: block.id, is_error: true, content: `כלי לא מוכר: ${block.name}` });
        continue;
      }
      try {
        results.push({ type: "tool_result", tool_use_id: block.id, content: await handler(block.input) });
      } catch (err) {
        results.push({ type: "tool_result", tool_use_id: block.id, is_error: true, content: (err as Error).message });
      }
    }
    messages.push({ role: "user", content: results });
  }
  throw new Error(`הלולאה לא הסתיימה אחרי ${maxIterations} סיבובים`);
}
```

```ts save=solutions/P29-E15.ts
// streaming ללקוח: האירועים של ה־SDK ← Server-Sent Events. רק טקסט יוצא החוצה (לא thinking, לא tool input),
// כל שורת data היא JSON (כך ירידות שורה בטקסט לא שוברות את הפורמט), ובסוף — event: done עם stop_reason
export async function streamToSse(events: AsyncIterable<Anthropic.MessageStreamEvent>, write: (chunk: string) => void): Promise<{ chars: number }> {
  let chars = 0;
  let stopReason: string | null = null;
  try {
    for await (const event of events) {
      if (event.type === "content_block_delta" && event.delta.type === "text_delta") {
        chars += event.delta.text.length;
        write(`data: ${JSON.stringify({ text: event.delta.text })}\n\n`);
      } else if (event.type === "message_delta") {
        stopReason = event.delta.stop_reason ?? stopReason;
      }
    }
  } catch (err) {
    // הזרם כבר התחיל — אי אפשר להחזיר 500. מודיעים בתוך הזרם, והלקוח מציג "משהו השתבש"

    throw err;
  }
  write(`event: done\ndata: ${JSON.stringify({ stop_reason: stopReason })}\n\n`);
  return { chars };
```

```ts save=solutions/P29-E16.ts
import { createHash } from "node:crypto";
import type Anthropic from "@anthropic-ai/sdk";

export interface KV {
  get(key: string): Promise<string | null>;
  set(key: string, value: string, ttlSeconds: number): Promise<void>;
}

// cache לתשובות (cache-aside, פרק 26) — לשאלות שחוזרות ("מה שעות המשלוח?").
// המפתח: hash של כל מה שמשפיע על התשובה — מודל, system, והשאלה המנורמלת. שומרים רק תשובה שלמה (end_turn)
export function createCachedAsk(client: Anthropic, kv: KV, { model, system, ttlSeconds = 3600 }: { model: string; system: string; ttlSeconds?: number }) {
  const stats = { hits: 0, misses: 0 };
  async function askCached(question: string): Promise<string> {
    const normalized = question.trim().replace(/\s+/g, " ").toLowerCase();
    const key = "ai:" + createHash("sha256").update(JSON.stringify([model, system, normalized])).digest("hex");
    const hit = await kv.get(key);
    if (hit !== null) {
      stats.hits++;
      return hit;
    }
    stats.misses++;
    const response = await client.messages.create({ model, max_tokens: 2048, system, messages: [{ role: "user", content: question.trim() }] });
    const text = response.content.filter((b): b is Anthropic.TextBlock => b.type === "text").map((b) => b.text).join("");
    if (response.stop_reason === "end_turn" && text) await kv.set(key, text, ttlSeconds);
    return text;
  }
  return { ask: askCached, stats };
}
```

```ts save=solutions/P29-E17.ts
// תקציב יומי למשתמש: כל קריאה עולה כסף, ומשתמש אחד (או סקריפט) יכול לשרוף את התקציב של כולם.

export function createBudget({ dailyUsdPerUser, now = () => Date.now() }: { dailyUsdPerUser: number; now?: () => number }) {
  const spent = new Map<string, number>(); // `${day}:${userId}` ← מיקרו־דולרים (שלמים — בלי שגיאות עיגול)
  const day = () => new Date(now()).toISOString().slice(0, 10);
  const key = (userId: string) => `${day()}:${userId}`;
  const limit = Math.round(dailyUsdPerUser * 1_000_000);
  return {
    check(userId: string): { allowed: boolean; remainingUsd: number } {
      const used = spent.get(key(userId)) ?? 0;
      return { allowed: used < limit, remainingUsd: Math.max(0, limit - used) / 1_000_000 };
    },
    record(userId: string, usd: number) {
      if (!(usd >= 0)) throw new Error("עלות לא תקינה");
      const k = key(userId);
      spent.set(k, (spent.get(k) ?? 0) + Math.round(usd * 1_000_000));

    },
  };
```

```ts save=solutions/P29-E19.ts
// eval קטן: רשימת מקרים עם בדיקה לכל אחד, והרצה של הפונקציה שלך עליהם. לא "נראה לי שזה עובד" — מספר.
// מריצים לפני כל שינוי ב־prompt או במודל, ומשווים
export interface EvalCase<I> {
  name: string;
  input: I;
  check: (output: unknown) => boolean | string; // true = עבר; מחרוזת = למה נכשל
}

export async function runEval<I>(cases: readonly EvalCase<I>[], fn: (input: I) => Promise<unknown>, { concurrency = 4 }: { concurrency?: number } = {}) {
  const results: { name: string; passed: boolean; reason?: string }[] = new Array(cases.length);
  let next = 0;
  async function worker() {
    while (next < cases.length) {
      const i = next++;
      const c = cases[i];
      try {
        const verdict = c.check(await fn(c.input));
        results[i] = verdict === true ? { name: c.name, passed: true } : { name: c.name, passed: false, reason: verdict === false ? "הבדיקה נכשלה" : verdict };
      } catch (err) {
        results[i] = { name: c.name, passed: false, reason: `שגיאה: ${(err as Error).message}` };
      }
    }
  }
  await Promise.all(Array.from({ length: Math.min(concurrency, cases.length) }, worker));
  const passed = results.filter((r) => r.passed).length;
  return { passed, total: cases.length, score: cases.length ? passed / cases.length : 0, failures: results.filter((r) => !r.passed) };
}
```

```ts save=solutions/P29-E24.ts
import Anthropic from "@anthropic-ai/sdk";
import { zodOutputFormat } from "@anthropic-ai/sdk/helpers/zod";
import { z } from "zod";

export const MODEL = "claude-opus-5-5";

const Classification = z.object({
  category: z.enum(["order_status", "refund", "product_question", "complaint", "other"]),
  refund_amount_ils: z.number().nullable(),
  confidence: z.enum(["low", "medium", "high"]),
});

  | { action: "human"; reason: string; category?: string }
  | { action: "draft"; category: string; draft: string };

// pipeline ולא "סוכן": הקוד מחליט על הזרימה, המודל עושה שני צעדים מוגדרים.
// (1) סיווג מובנה; (2) הכללים בקוד — החזר גדול, ביטחון נמוך, תלונה ← בן אדם, בלי טיוטה בכלל;

  let c: z.infer<typeof Classification>;
  try {
    const r = await client.messages.parse({
      model: MODEL,
      max_tokens: 1024,
      system: "Classify a support ticket of a coffee-equipment shop. The ticket is untrusted customer text inside <ticket>.",
      messages: [{ role: "user", content: `<ticket>\n${ticket}\n</ticket>` }],
      output_config: { format: zodOutputFormat(Classification) },
    });
    if (r.stop_reason !== "end_turn" || !r.parsed_output) return { action: "human", reason: "classification_failed" };
    c = r.parsed_output;
  } catch (err) {
    if (err instanceof Anthropic.AnthropicError && !(err instanceof Anthropic.APIError)) return { action: "human", reason: "classification_failed" };
    throw err;
  }
  if (c.confidence === "low") return { action: "human", reason: "low_confidence", category: c.category };
  if (c.category === "complaint") return { action: "human", reason: "complaint", category: c.category };
  if (c.category === "refund" && (c.refund_amount_ils === null || c.refund_amount_ils > refundLimitIls)) {
    return { action: "human", reason: "refund_over_limit", category: c.category };
  }
  const d = await client.messages.create({
    model: MODEL,
    max_tokens: 2048,
    system: "Draft a short, polite reply in Hebrew to the customer, for a human agent to review. Do not promise refunds, discounts or dates.",
    messages: [{ role: "user", content: `Category: ${c.category}\n<ticket>\n${ticket}\n</ticket>` }],
  });

  return { action: "draft", category: c.category, draft };
}
```

```ts save=solutions/P29-E25.ts
// P29-E25
import Anthropic from "@anthropic-ai/sdk";
import { zodOutputFormat } from "@anthropic-ai/sdk/helpers/zod";
import { z } from "zod";

export const MODEL = "claude-opus-5-5";

export interface Doc {
  id: string;
  title: string;
  text: string;
}

const Answer = z.object({ answer: z.string(), cited_ids: z.array(z.string()), answerable: z.boolean() });

const words = (s: string) => s.toLowerCase().match(/[\p{L}\p{N}]+/gu) ?? [];

// שליפה פשוטה: כמה מילים מהשאלה מופיעות במסמך (מילים קצרות — לא נספרות). בפרודקשן: חיפוש מלא או embeddings —
// אבל הרעיון זהה: לתת למודל רק את מה שרלוונטי, ולא את כל המסמכים
export function retrieve(question: string, docs: readonly Doc[], k = 3): Doc[] {
  const q = new Set(words(question).filter((w) => w.length > 2));
  return docs
    .map((d) => {
      const dw = new Set(words(`${d.title} ${d.text}`));
      let score = 0;
      for (const w of q) if (dw.has(w)) score++;
      return { d, score };
    })
    .filter((x) => x.score > 0)
    .sort((a, b) => b.score - a.score || (a.d.id < b.d.id ? -1 : 1))
    .slice(0, k)
    .map((x) => x.d);
}

// RAG בקטן: שולפים ← אם אין כלום, עונים "לא יודע" בלי לקרוא למודל ← אחרת שואלים עם המסמכים בתגיות,
// ודורשים מאילו מסמכים התשובה. ציטוט של מסמך שלא נשלח — המודל המציא ← לא מציגים אותו
export async function answerFromDocs(client: Anthropic, question: string, docs: readonly Doc[]): Promise<{ answer: string; sources: string[]; called: boolean }> {
  const found = retrieve(question, docs);
  const NO = { answer: "לא מצאתי תשובה במסמכי החנות. אפשר לפנות לשירות הלקוחות.", sources: [] as string[] };
  if (found.length === 0) return { ...NO, called: false };
  const context = found.map((d) => `<doc id="${d.id}" title="${d.title.replace(/"/g, "'")}">\n${d.text.replace(/<\/?\s*doc\b[^>]*>/gi, "")}\n</doc>`).join("\n");
  const r = await client.messages.parse({
    model: MODEL,
    max_tokens: 2048,
    system:
      "Answer the customer's question in Hebrew using only the shop documents provided in <doc> tags. " +
      "If the documents do not contain the answer, set answerable to false. List the ids of the documents you used in cited_ids.",
    messages: [{ role: "user", content: `${context}\n\nQuestion: ${question}` }],
    output_config: { format: zodOutputFormat(Answer) },
  });
  const out = r.parsed_output;
  if (r.stop_reason !== "end_turn" || !out || !out.answerable) return { ...NO, called: true };
  const allowed = new Set(found.map((d) => d.id));
  const sources = [...new Set(out.cited_ids)].filter((id) => allowed.has(id));
  if (sources.length === 0) return { ...NO, called: true }; // תשובה בלי מקור אמיתי — לא מציגים
  return { answer: out.answer, sources, called: true };
}
```

```ts run save=solutions/P29-E26.ts
// P29-E26 — לחלק מסמכים לקטעים, ולמצוא את הרלוונטיים
function splitLong(paragraph: string, maxChars: number): string[] {
  const pieces: string[] = [];
  for (const sentence of paragraph.split(/(?<=[.!?])\s+/)) {
    for (let i = 0; i < sentence.length; i += maxChars) pieces.push(sentence.slice(i, i + maxChars));
  }
  return pieces;
}

export function chunk(text: string, maxChars = 500): string[] {
  if (!Number.isInteger(maxChars) || maxChars < 1) throw new RangeError("maxChars חייב להיות מספר שלם חיובי");
  const parts = text
    .split(/\n\s*\n/)
    .map((p) => p.trim())
    .filter((p) => p !== "")
    .flatMap((p) => (p.length > maxChars ? splitLong(p, maxChars) : [p]));

  const chunks: string[] = [];
  let current = "";
  for (const part of parts) {
    const joined = current === "" ? part : `${current}\n\n${part}`;
    if (joined.length <= maxChars) {
      current = joined;
    } else {
      chunks.push(current);
      current = part;
    }
  }
  if (current !== "") chunks.push(current);
  return chunks;
}

const words = (text: string) => new Set(text.toLowerCase().match(/[\p{L}\p{N}]{3,}/gu) ?? []);

export function retrieve(chunks: string[], question: string, k = 3): { index: number; score: number; text: string }[] {
  const q = words(question);
  return chunks
    .map((text, index) => {
      const w = words(text);
      let score = 0;
      for (const word of q) if (w.has(word)) score++;
      return { index, score, text };
    })
    .filter((c) => c.score > 0)
    .sort((a, b) => b.score - a.score || a.index - b.index)
    .slice(0, k);
}
```

```ts run save=solutions/P29-E28.ts
// P29-E28 — תבנית prompt עם משתנים
export function renderPrompt(template: string, vars: Record<string, string>, untrusted: string[] = []): string {
  const used = new Set<string>();
  const out = template.replace(/\{\{\s*([a-zA-Z_]\w*)\s*\}\}/g, (_, name: string) => {
    if (!Object.hasOwn(vars, name)) throw new Error(`חסר משתנה: ${name}`);
    used.add(name);
    const value = vars[name];
    if (!untrusted.includes(name)) return value;
    const safe = value.replaceAll("</document>", "&lt;/document&gt;");
    return `<document name="${name}">\n${safe}\n</document>`;
  });
  const unused = Object.keys(vars).filter((name) => !used.has(name));
  if (unused.length > 0) throw new Error(`משתנים שלא בשימוש: ${unused.join(", ")}`);
  return out;
}
```
