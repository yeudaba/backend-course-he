```js run save=solutions/P10-E06.js
// P10-E06
function delay(ms, value) {
  return new Promise((resolve) => setTimeout(() => resolve(value), ms));
}

async function countdown(from, ms, log = console.log) {
  for (let n = from; n >= 1; n--) {
    log(n);
    await delay(ms);
  }
  log("🚀");
}

module.exports = { delay, countdown };
```

@hint2
<p>JSON.parse בתוך try/catch. היזהר לא לקרוא ל־callback בתוך ה־try (אם הוא עצמו יזרוק — ייקרא שוב מה־catch).</p>

```js run save=solutions/P10-E07.js
// P10-E07: callback בסגנון error-first
const fs = require("node:fs");

function readJson(file, callback) {
  fs.readFile(file, "utf8", (err, text) => {
    if (err) {
      callback(err);
      return;
    }
    let data;
    try {
      data = JSON.parse(text);
    } catch (parseError) {
      callback(parseError);
      return;
    }
    callback(null, data);
  });
}

module.exports = { readJson };
```

```js run save=solutions/P10-E08.js
function legacyGetPrice(code, callback) {
```

```js run save=solutions/P10-E10.js
function slowSquareRoot(n, ms = 10) {
  return new Promise((resolve, reject) => {
    setTimeout(() => {
      if (typeof n !== "number" || Number.isNaN(n)) {
        reject(new Error("לא מספר"));
        return;
      }
      if (n < 0) {
        reject(new Error("אין שורש למספר שלילי"));
        return;
      }
      resolve(Math.sqrt(n));
    }, ms);
  });
}
```

```js run save=solutions/P10-E15.js
function withTimeout(promise, ms) {
```

```js run save=solutions/P10-E25.js
// P10-E25: cache עם שיתוף בקשות בטיסה
function memoizeAsync(fn, ttlMs) {
  const cache = new Map();
  return function (key) {
    const hit = cache.get(key);
    if (hit && hit.expires > Date.now()) {
      return hit.promise;
    }
    const promise = fn(key);
    cache.set(key, { promise, expires: Date.now() + ttlMs });
    promise.catch(() => cache.delete(key));
    return promise;
  };
}

  calls++;
  await new Promise((r) => setTimeout(r, 10));
  return n * 2;
};
const cached = memoizeAsync(slowDouble, 1000);
Promise.all([cached(5), cached(5), cached(5)]).then((r) => console.log(r, "calls:", calls));
```

```js run save=solutions/P10-E28.js
  const all = [];
  let page = 1;
  let count = 0;
  while (page !== null) {
    if (count === maxPages) {
      throw new Error("יותר מדי עמודים");
```

```js run save=solutions/P10-E44.js
// P10-E44: מנבא ה־event loop
function predictOrder(steps) {
  const sync = [];
  const micro = [];
  const timers = [];
  for (const step of steps) {
    switch (step.kind) {
      case "sync":
        sync.push(step.label);
        break;
      case "promise":
        micro.push(step.label);
        break;
      case "timeout":
        timers.push(step);
        break;
      default:
        throw new Error(`סוג לא מוכר: ${step.kind}`);
    }
  }
  timers.sort((a, b) => a.ms - b.ms);
  return [...sync, ...micro, ...timers.map((t) => t.label)];
}
```

```js run save=solutions/P10-E54.js
// P10-E54: לעטוף readFile ב־Promise
const fs = require("node:fs");

function readText(file) {
  return new Promise((resolve, reject) => {
    fs.readFile(file, "utf8", (err, text) => {
      if (err) {
        reject(err);
      } else {
        resolve(text);
      }
    });
  });
}

function readUpper(file) {
  return readText(file).then((t) => t.toUpperCase());
}

const os = require("node:os");
const path = require("node:path");
const file = path.join(os.tmpdir(), "p10-e54.txt");
fs.writeFileSync(file, "hello");
readText(file)
  .then((t) => console.log(t))
  .then(() => readUpper(file))
  .then((t) => console.log(t))
  .then(() => readText("/no/such.txt"))
  .catch((e) => console.log(e.code));

module.exports = { readText, readUpper };
```

```js run save=solutions/P10-E58.js
// P10-E58: לקרוא את כל הקבצים במקביל
const fs = require("node:fs");

function readAll(files, callback) {
  if (files.length === 0) {
    setTimeout(() => callback(null, []), 0);
    return;
  }
  const texts = [];
  let left = files.length;
  let finished = false;
  files.forEach((file, i) => {
    fs.readFile(file, "utf8", (err, text) => {
      if (finished) return;
      if (err) {
        finished = true;
        callback(err);
        return;
      }
      texts[i] = text;
      left--;
      if (left === 0) {
        finished = true;
        callback(null, texts);
      }
    });
  });
}

const os = require("node:os");
const path = require("node:path");
const a = path.join(os.tmpdir(), "p10-e58-a.txt");
const b = path.join(os.tmpdir(), "p10-e58-b.txt");
fs.writeFileSync(a, "ראשון");
fs.writeFileSync(b, "שני");
readAll([a, b, a], (err, texts) => {
  console.log(err, texts);
  readAll([a, "/no/such.txt", b], (err) => console.log("נכשל:", err.code));
});

module.exports = { readAll };
```

```js run save=solutions/P10-E59.js
const fs = require("node:fs");

function loadUsers(file, callback) {
  fs.readFile(file, "utf8", (err, text) => {
    if (err) {
      callback(err);
      return;
    }
    const lines = text.split("\n");
    const users = [];
    for (let i = 1; i < lines.length; i++) {
      if (lines[i].trim() === "") continue;
      const parts = lines[i].split(",").map((p) => p.trim());

        callback(new Error(`שורה ${i + 1} לא תקינה`));
        return;
      }
      users.push({ name: parts[0], age, city: parts[2] });
    }
    callback(null, users);
  });
}
```

```js run save=solutions/P10-E60.js
// P10-E60: משימות ברצף בלי פירמידה
function series(tasks, callback) {
  const results = [];
  if (tasks.length === 0) {
    setTimeout(() => callback(null, results), 0);
    return;
  }
  function next(i) {
    if (i === tasks.length) {
      callback(null, results);
      return;
    }
    tasks[i]((err, result) => {
      if (err) {
        callback(err);
        return;
      }
      results.push(result);
      next(i + 1);
    });
  }
  next(0);
}

const log = [];
series([
  (done) => setTimeout(() => { log.push("config"); done(null, "config"); }, 10),
  (done) => setTimeout(() => { log.push("user"); done(null, "user"); }, 5),
], (err, results) => {
  console.log(err, results, log);
  series([
    (done) => done(new Error("אין הגדרות")),
    (done) => done(null, "לא אמור לרוץ"),
  ], (err) => console.log("נכשל:", err.message));
});

module.exports = { series };
```


```js run save=solutions/P10-E72.js
// P10-E72: promisify משלך
function promisify(fn) {
  return (...args) =>
    new Promise((resolve, reject) => {
      fn(...args, (err, value) => {
        if (err) {
          reject(err);
        } else {
          resolve(value);
        }
      });
    });
}

function addLater(a, b, cb) {
  setTimeout(() => (typeof a === "number" ? cb(null, a + b) : cb(new TypeError("לא מספר"))), 5);
}
const add = promisify(addLater);
add(2, 3)
  .then(console.log)
  .then(() => add("x", 1))
  .catch((e) => console.log(e.message))
  .then(() => promisify(require("node:fs").readFile)("/no/such.txt", "utf8"))
  .catch((e) => console.log(e.code));

module.exports = { promisify };
```

```js run save=solutions/P10-E73.js
const fs = require("node:fs/promises");

function sumFile(file) {
  return fs.readFile(file, "utf8").then((text) => {
    const lines = text.split("\n");
    let sum = 0;
    for (let i = 0; i < lines.length; i++) {
      const t = lines[i].trim();
      if (t === "") continue;
      const n = Number(t);
      if (Number.isNaN(n)) throw new Error(`שורה ${i + 1}: "${t}" אינו מספר`);
      sum += n;
    }
    return sum;
  });
```

```js run save=solutions/P10-E74.js
// P10-E74: הגדרות עם ברירות מחדל
const fs = require("node:fs/promises");

function loadSettings(file, defaults) {
  return fs
    .readFile(file, "utf8")
    .then((text) => ({ ...defaults, ...JSON.parse(text) }))
    .catch((err) => {
      if (err.code === "ENOENT") return { ...defaults };
      throw err;
    });
}

const path = require("node:path");
const file = path.join(os.tmpdir(), "p10-e74.json");
require("node:fs").writeFileSync(file, '{"port": 8080, "name": "shop"}');
const defaults = { port: 3000, debug: false };
loadSettings(file, defaults)
  .then((s) => console.log(s))
  .then(() => loadSettings("/no/such.json", defaults))
  .then((s) => console.log(s, s === defaults))
```

<ol><li>ארבע שורות, ו־<code>catch</code> אחד לכל השלבים — כולל שגיאה שנזרקת בקוד שלנו (למשל <code>config.userId</code> על <code>undefined</code>).</li><li>מקום אחד: <code>finally</code>. בגרסת ה־callbacks היו צריכים חמישה.</li><li>לא כמו שהקוד כתוב: כל שלב צריך את התוצאה של הקודם (<code>config</code> ← <code>user</code> ← <code>orders</code>). אילו <code>readOrdersP</code> הייתה מקבלת את <code>config.userId</code> ישירות — אפשר היה להביא את המשתמש וההזמנות יחד: <code>await Promise.all([readUserP(id), readOrdersP(id)])</code>.</li></ol>
@mistakes
<ul><li>לשים <code>try/catch</code> נפרד סביב כל שורה — חוזרים לפירמידה, רק בלי ההזחה.</li><li>לשכוח את <code>await</code> לפני <code>sendReportP</code> — "done" מודפס לפני שהדוח נשלח, וכישלון שלו לא נתפס.</li></ul>
