// P27-E06
// Set: "ראיתי כבר?" ב־O(1). מעבר אחד על המערך ← O(n) זמן, O(n) זיכרון
export function firstDuplicate<T>(items: readonly T[]): T | undefined {
  const seen = new Set<T>();
  for (const item of items) {
    if (seen.has(item)) return item;
    seen.add(item);
  }
  return undefined;
}

export function hasDuplicates<T>(items: readonly T[]): boolean {
  return new Set(items).size !== items.length;
}
```

// P27-E07
// Map של "ערך ← אינדקס" למה שכבר ראינו: לכל מחיר בודקים אם ההשלמה שלו כבר הופיעה. O(n)
export function twoSum(prices: readonly number[], target: number): [number, number] | null {
  const seenAt = new Map<number, number>();
  for (let j = 0; j < prices.length; j++) {
    const i = seenAt.get(target - prices[j]);
    if (i !== undefined) return [i, j];
    if (!seenAt.has(prices[j])) seenAt.set(prices[j], j); // הראשון — כדי שהזוג יהיה הכי מוקדם
  }
  return null;
}
```
<p>זו כנראה שאלת הראיון הנפוצה בעולם ("Two Sum"). הדפוס שמאחוריה — "Map של מה שכבר ראיתי, ובדיקה מול השלמה" — חוזר בעשרות שאלות.</p>

```ts save=solutions/P27-E08.ts
// P27-E08
// מפתח קנוני לכל קבוצה: האותיות ממוינות. Map שומר את סדר ההוספה — כלומר סדר ההופעה הראשונה
export function groupAnagrams(words: readonly string[]): string[][] {
  const groups = new Map<string, string[]>();
  for (const word of words) {
    const key = [...word.toLowerCase()].sort().join("");
    const group = groups.get(key);
    if (group) group.push(word);
    else groups.set(key, [word]);
  }
  return [...groups.values()];
}
```

// P27-E09
// מחסנית: כל פותח נדחף; כל סוגר חייב להתאים לפותח האחרון. מחסנית לא ריקה בסוף ← משהו לא נסגר
const PAIRS: Record<string, string> = { ")": "(", "]": "[", "}": "{" };

export function checkBrackets(text: string): { ok: true } | { ok: false; index: number } {
  const stack: number[] = []; // אינדקסים של הפותחים — כדי לדעת איפה הבעיה
  for (let i = 0; i < text.length; i++) {
    const ch = text[i];
    if (ch === "(" || ch === "[" || ch === "{") stack.push(i);
    else if (ch in PAIRS) {
      if (stack.length === 0 || text[stack[stack.length - 1]] !== PAIRS[ch]) return { ok: false, index: i };
      stack.pop();
    }
  }
  return stack.length === 0 ? { ok: true } : { ok: false, index: stack[0] };
}

// P27-E10
// חיפוש בינארי: הטווח [lo, hi) תמיד מכיל את התשובה, וחוצים אותו בכל צעד ← O(log n)
export function lowerBound(sorted: readonly number[], x: number): number {
  let lo = 0;
  let hi = sorted.length; // התשובה יכולה להיות length: "כולם קטנים מ־x"
  while (lo < hi) {
    const mid = (lo + hi) >>> 1;
    if (sorted[mid] < x) lo = mid + 1;
    else hi = mid;
  }
  return lo;
}

// כמה ערכים ב־[from, to] — שני חיפושים בינאריים
export function countInRange(sorted: readonly number[], from: number, to: number): number {
  if (from > to) return 0;
  return lowerBound(sorted, to + 1) - lowerBound(sorted, from);
}
```

// P27-E11
// חלון נע עם שני מצביעים: right מתקדם, left "מדביק" כשהחלון ארוך מדי. כל אינדקס נכנס ויוצא פעם אחת ← O(n)
export function peakInWindow(timestamps: readonly number[], windowMs: number): { count: number; start: number } {
  let best = { count: 0, start: timestamps[0] ?? 0 };
  let left = 0;
  for (let right = 0; right < timestamps.length; right++) {
    while (timestamps[right] - timestamps[left] >= windowMs) left++;
    const count = right - left + 1;
    if (count > best.count) best = { count, start: timestamps[left] };
  }
  return best;
}

export function mergeIntervals(intervals: readonly (readonly [number, number])[]): [number, number][] {
  const sorted = [...intervals].sort((a, b) => a[0] - b[0]);
  const merged: [number, number][] = [];
  for (const [start, end] of sorted) {
    const last = merged[merged.length - 1];
    if (last && start <= last[1]) last[1] = Math.max(last[1], end);
    else merged.push([start, end]);

  return merged;

// ספירה ב־Map ← O(n); מיון של הערכים השונים ← O(m log m). (עם heap בגודל k: O(m log k) — לרוב לא שווה את המורכבות)
export function topK(items: readonly string[], k: number): { item: string; count: number }[] {
  const counts = new Map<string, number>();
  for (const item of items) counts.set(item, (counts.get(item) ?? 0) + 1);

    .map(([item, count]) => ({ item, count }))

// P27-E15
// BFS: מתקדמים "בגלים" — כל התאים במרחק 1, אחר כך 2... הפעם הראשונה שמגיעים ל־E היא הקצרה ביותר
export function shortestPath(grid: readonly string[]): number {
  const h = grid.length;
  const w = grid[0]?.length ?? 0;
  let start: [number, number] | null = null;
  for (let y = 0; y < h; y++) {
    const x = grid[y].indexOf("S");
    if (x !== -1) start = [x, y];
  }
  if (!start) return -1;
  const dist = new Map<number, number>([[start[1] * w + start[0], 0]]); // מסמנים כשנכנסים לתור, לא כשיוצאים
  const queue: [number, number][] = [start];
  for (let head = 0; head < queue.length; head++) { // אינדקס במקום shift(): לא מזיזים כלום, גם כשהתור ענק
    const [x, y] = queue[head];
    const d = dist.get(y * w + x)!;
    if (grid[y][x] === "E") return d;
    for (const [dx, dy] of [[1, 0], [-1, 0], [0, 1], [0, -1]]) {
      const nx = x + dx;
      const ny = y + dy;
      if (nx < 0 || ny < 0 || nx >= w || ny >= h || grid[ny][nx] === "#" || dist.has(ny * w + nx)) continue;
      dist.set(ny * w + nx, d + 1);
      queue.push([nx, ny]);
    }
  }
  return -1;
}

// P27-E16
// Map ב־JavaScript שומר סדר הכנסה: הראשון = הכי פחות בשימוש. "שימוש" = מוחקים ומכניסים מחדש (לסוף). הכול O(1)
export class LRUCache<K, V> {
  private readonly map = new Map<K, V>();
  private readonly capacity: number;

  // (בלי "constructor(private capacity)": Node מריץ TypeScript בלי קומפילציה, וזה תחביר שאינו JavaScript)
  constructor(capacity: number) {
    this.capacity = capacity;
    if (!Number.isInteger(capacity) || capacity < 1) throw new Error("capacity חייב להיות שלם חיובי");
  }

  get size() {
    return this.map.size;
  }

  get(key: K): V | undefined {
    if (!this.map.has(key)) return undefined;
    const value = this.map.get(key)!;
    this.map.delete(key);
    this.map.set(key, value);
    return value;
  }

  set(key: K, value: V): void {
    this.map.delete(key);
    this.map.set(key, value);
    if (this.map.size > this.capacity) this.map.delete(this.map.keys().next().value!);
  }

  keys(): K[] {
    return [...this.map.keys()]; // מהישן לחדש
  }
}
```

// P27-E17
// מיון טופולוגי (Kahn): מתחילים ממי שאין לו תלויות, ו"משחררים" את מי שחיכה לו.
// נשארו צמתים שלא שוחררו ← יש מעגל. בכל שלב — האלפביתי הראשון, כדי שהסדר יהיה קבוע
export function buildOrder(deps: Readonly<Record<string, readonly string[]>>): string[] {
  const names = new Set<string>(Object.keys(deps));
  for (const list of Object.values(deps)) for (const d of list) names.add(d);
  const waitingFor = new Map<string, number>();
  const dependents = new Map<string, string[]>();
  for (const name of names) {
    waitingFor.set(name, 0);
    dependents.set(name, []);
  }
  for (const [name, list] of Object.entries(deps)) {
    for (const d of new Set(list)) {
      waitingFor.set(name, waitingFor.get(name)! + 1);
      dependents.get(d)!.push(name);
    }
  }
  const ready = [...names].filter((n) => waitingFor.get(n) === 0).sort();
  const order: string[] = [];
  while (ready.length) {
    const next = ready.shift()!;
    order.push(next);
    for (const dep of dependents.get(next)!) {
      const left = waitingFor.get(dep)! - 1;
      waitingFor.set(dep, left);
      if (left === 0) {
        ready.push(dep);
        ready.sort();
      }
    }
  }
  if (order.length !== names.size) {
    const stuck = [...names].filter((n) => !order.includes(n)).sort();
    throw new Error(`יש מעגל בתלויות: ${stuck.join(", ")}`);
  }
  return order;
}

// sweep line: כל סשן = שני אירועים (+1 בהתחלה, −1 בסוף). ממיינים לפי זמן — וסוף לפני התחלה באותו רגע
// (end לא כולל: סשן שנגמר ב־10 וסשן שמתחיל ב־10 לא חופפים). O(n log n)
export function peakConcurrency(sessions: readonly { start: number; end: number }[]): { peak: number; at: number | null } {
  const events: [number, number][] = [];
  for (const s of sessions) {
    if (s.end <= s.start) continue;
    events.push([s.start, 1], [s.end, -1]);
  }
  events.sort((a, b) => a[0] - b[0] || a[1] - b[1]);
  let current = 0;
  let best = { peak: 0, at: null as number | null };
  for (const [time, delta] of events) {
    current += delta;
    if (current > best.peak) best = { peak: current, at: time };
  }
  return best;
}

// P27-E20
export function lowerBound(sorted: readonly number[], x: number): number {
  let lo = 0;
  let hi = sorted.length; // תקלה 1: היה length − 1 ← "כולם קטנים" החזיר את האינדקס האחרון במקום length
  while (lo < hi) {
    const mid = (lo + hi) >>> 1;
    if (sorted[mid] < x) lo = mid + 1; // תקלה 2: היה <= ← זה upper bound (הראשון שגדול מ־x), לא הראשון שגדול או שווה
    else hi = mid;
  }
  return lo;
}

export function countInRange(sorted: readonly number[], from: number, to: number): number {
  if (from > to) return 0;
  return lowerBound(sorted, to + 1) - lowerBound(sorted, from); // תקלה 3: היה lowerBound(to) ← ערכים ששווים ל־to לא נספרו
}

// P27-E22
export function shortestPath(grid: readonly string[]): number {
  const h = grid.length;
  const w = grid[0]?.length ?? 0;
  let start: [number, number] | null = null;
  for (let y = 0; y < h; y++) {
    const x = grid[y].indexOf("S");
    if (x !== -1) start = [x, y];
  }
  if (!start) return -1;
  const dist = new Map<string, number>([[`${start[0]},${start[1]}`, 0]]);
  const queue: [number, number][] = [start];
  while (queue.length) {
    const [x, y] = queue.shift()!; // תקלה 2: היה pop() ← מחסנית = DFS, שמוצא מסלול — לא בהכרח הקצר
    const d = dist.get(`${x},${y}`)!;
    if (grid[y][x] === "E") return d;
    for (const [dx, dy] of [[1, 0], [-1, 0], [0, 1], [0, -1]]) {
      const nx = x + dx;
      const ny = y + dy;
      // תקלה 1: היה nx <= w ו־ny <= h ← grid[y][w] הוא undefined, "לא קיר" — והמסלול יצא מחוץ למפה
      if (nx < 0 || ny < 0 || nx >= w || ny >= h || grid[ny][nx] === "#" || dist.has(`${nx},${ny}`)) continue;
      dist.set(`${nx},${ny}`, d + 1);
      queue.push([nx, ny]);
    }
  }
  return -1; // תקלה 3: היה return 0 ← "אין דרך" נראה כמו "כבר שם"
}
```
<p>(כאן נשאר <code>shift()</code> — ברור יותר לקריאה. ב־BFS על רשת התור נשאר קטן (רק ה"חזית"), אז גם על 400×400 מדדתי שזה מהיר; בתור שמגיע למאות אלפים — אינדקס, כמו ב־E15.)</p>

// P27-E24
// תכנון דינמי (DP): התשובה לסכום a נבנית מהתשובות לסכומים קטנים יותר — כל אחד מחושב פעם אחת. O(amount × coins)
export function minCoins(coins: readonly number[], amount: number): number {
  const best = new Array<number>(amount + 1).fill(Infinity);
  best[0] = 0;
  for (let a = 1; a <= amount; a++) {
    for (const c of coins) {
      if (c <= a && best[a - c] + 1 < best[a]) best[a] = best[a - c] + 1;
    }
  }
  return best[amount] === Infinity ? -1 : best[amount];
}

// כמה צירופים (בלי חשיבות לסדר): הלולאה החיצונית על המטבעות — כך 1+2 ו־2+1 נספרים פעם אחת
export function countWays(coins: readonly number[], amount: number): number {
  const ways = new Array<number>(amount + 1).fill(0);
  ways[0] = 1;
  for (const c of coins) {
    for (let a = c; a <= amount; a++) ways[a] += ways[a - c];
  }
  return ways[amount];
}
```

// P27-E25
// Trie: עץ של אותיות. כל צומת שומר גם את ה־k הכי פופולריים מתחתיו — כך suggest לא סורק את כל תת־העץ.
// add מעדכן את הרשימות לאורך המסלול: O(L × k). suggest: O(L) — הליכה לצומת, והרשימה מוכנה
interface Node {
  children: Map<string, Node>;
  top: string[]; // עד MAX מונחים, ממוינים
}

const MAX = 10;

export function createAutocomplete() {
  const root: Node = { children: new Map(), top: [] };
  const weights = new Map<string, number>(); // מונח (lowercase) ← משקל מצטבר
  const display = new Map<string, string>(); // מונח (lowercase) ← הכתיב הראשון שנוסף
  const better = (a: string, b: string) => weights.get(b)! - weights.get(a)! || (a < b ? -1 : a > b ? 1 : 0);

  return {
    add(term: string, weight = 1) {
      const key = term.trim().toLowerCase();
      if (!key) return;
      weights.set(key, (weights.get(key) ?? 0) + weight);
      if (!display.has(key)) display.set(key, term.trim());
      let node = root;
      const path = [root];
      for (const ch of key) {
        let next = node.children.get(ch);
        if (!next) {
          next = { children: new Map(), top: [] };
          node.children.set(ch, next);
        }
        node = next;
        path.push(node);
      }
      for (const n of path) {
        if (!n.top.includes(key)) n.top.push(key);
        n.top.sort(better);
        if (n.top.length > MAX) n.top.length = MAX;
      }
    },
    suggest(prefix: string, k = 5): string[] {
      let node: Node | undefined = root;
      for (const ch of prefix.trim().toLowerCase()) {
        node = node.children.get(ch);
        if (!node) return [];
      }
      return node.top.slice(0, Math.min(k, MAX)).map((key) => display.get(key)!);
    },
    get size() {
      return weights.size;
    },
  };
}
```

export function firstUnique<T>(items: T[]): T | undefined {
  const counts = new Map<T, number>();
  for (const item of items) counts.set(item, (counts.get(item) ?? 0) + 1);
  for (const item of items) if (counts.get(item) === 1) return item;
  return undefined;
}

// P27-E27 — מיזוג רשימות ממוינות
export function mergeSorted(a: number[], b: number[]): number[] {
  const out: number[] = [];
  let i = 0;
  let j = 0;
  while (i < a.length && j < b.length) {
    if (a[i] <= b[j]) out.push(a[i++]);
    else out.push(b[j++]);
  }
  while (i < a.length) out.push(a[i++]);
  while (j < b.length) out.push(b[j++]);
  return out;
}

export function mergeAll(lists: number[][]): number[] {
  if (lists.length === 0) return [];
  let round = lists;
  while (round.length > 1) {
    const next: number[][] = [];
    for (let k = 0; k < round.length; k += 2) {
      next.push(k + 1 < round.length ? mergeSorted(round[k], round[k + 1]) : round[k]);
    }
    round = next;
  }
  return round[0];
}

// P27-E28 — דחיסת רצפים (run-length encoding)
export function compress(text: string): string {
  if (/\d/.test(text)) throw new TypeError("אי אפשר לדחוס טקסט עם ספרות — לא יהיה אפשר לפתוח אותו");
  let out = "";
  let i = 0;
  while (i < text.length) {
    let j = i;
    while (j < text.length && text[j] === text[i]) j++;
    const run = j - i;
    out += run === 1 ? text[i] : `${text[i]}${run}`;
    i = j;
  }
  return out;
}

export function decompress(code: string): string {
  if (/\d/.test(code[0] ?? "")) throw new SyntaxError("קוד לא יכול להתחיל בספרה");
  let out = "";
  for (const [, ch, count] of code.matchAll(/(\D)(\d*)/g)) {
    if (count.startsWith("0")) throw new SyntaxError(`מספר לא תקין אחרי ${ch}: ${count}`);
    out += ch.repeat(count === "" ? 1 : Number(count));
  }
  return out;
}

const samples = ["aaabccdddd", "abc", "", "zzzzzzzzzzzz"];
for (const s of samples) {
  const c = compress(s);
  console.log(JSON.stringify(s), "→", JSON.stringify(c), "→", decompress(c) === s ? "✓" : "✗");
}

export function canFulfill(order: string[], stock: string[]): boolean {
  const left = new Map<string, number>();
  for (const s of stock) left.set(s, (left.get(s) ?? 0) + 1);

// P27-E38: תור בלי shift
export class FastQueue<T> {
  #items: T[] = [];
  #head = 0;

  push(x: T) {
    this.#items.push(x);
  }
  shift(): T | undefined {
    if (this.#head >= this.#items.length) return undefined;
    const x = this.#items[this.#head];
    this.#items[this.#head++] = undefined as T; // לא להחזיק הפניה
    if (this.#head * 2 >= this.#items.length) {
      this.#items = this.#items.slice(this.#head);
      this.#head = 0;
    }
    return x;
  }
  peek(): T | undefined {
    return this.#head < this.#items.length ? this.#items[this.#head] : undefined;
  }
  get size() {
    return this.#items.length - this.#head;
  }
}
```
<p>"O(1) בממוצע" (amortized): לפעמים פעולה יקרה, אבל היא קורית רק אחרי הרבה זולות. ‏<code>push</code> של מערך עובד בדיוק כך (מכפיל את הקיבולת מדי פעם).</p>

export function usernamesApp(usernames: string[]) {

  const app = express();
  app.use(express.json());
  app.get("/available", (req, res) => {
    const name = typeof req.query.name === "string" ? norm(req.query.name) : "";
    if (!name) return res.status(400).json({ error: "חסר name" });
    res.json({ available: !taken.has(name) });
  });
  app.post("/usernames", (req, res) => {
    const name = typeof req.body?.name === "string" ? norm(req.body.name) : "";

    taken.add(name);
    res.status(201).json({ name });
  });
  return app;
}

// P27-E49: הרצף הארוך ביותר בלי חזרה
export function longestUnique<T>(items: T[]) {
  const lastAt = new Map<T, number>();
  let left = 0;
  let best = { length: 0, start: 0 };
  for (let right = 0; right < items.length; right++) {
    const last = lastAt.get(items[right]);
    if (last !== undefined && last >= left) left = last + 1;
    lastAt.set(items[right], right);
    if (right - left + 1 > best.length) best = { length: right - left + 1, start: left };
  }
  return best;
}
```

// P27-E54: הגבלת קצב בחלון נע — זיכרון קבוע לכל לקוח
export function createSlidingLimiter({ limit, windowMs }: { limit: number; windowMs: number }) {
  const log = new Map<string, number[]>();
  const prune = (clientId: string, now: number) => {
    const times = log.get(clientId);
    if (!times) return [];
    while (times.length && times[0] <= now - windowMs) times.shift();
    if (times.length === 0) log.delete(clientId);
    return times;
  };
  return {
    hit(clientId: string, now: number) {
      const times = prune(clientId, now);
      if (times.length >= limit) return false;
      if (times.length === 0) log.set(clientId, times);
      times.push(now);
      return true;
    },
    tracked(clientId: string) {
      return log.get(clientId)?.length ?? 0;
    },
  };
}
```

// P27-E55: LRU עם TTL
export function createTtlLru<V>(capacity: number, ttlMs: number, now: () => number = Date.now) {
  const map = new Map<string, { value: V; expiresAt: number }>();
  return {
    get(key: string): V | undefined {
      const e = map.get(key);
      if (!e) return undefined;
      map.delete(key);
      if (now() >= e.expiresAt) return undefined;
      map.set(key, e);
      return e.value;
    },
    set(key: string, value: V) {
      map.delete(key);
      map.set(key, { value, expiresAt: now() + ttlMs });
      while (map.size > capacity) {
        map.delete(map.keys().next().value as string);
      }
    },
    size() {
      for (const [k, e] of map) if (now() >= e.expiresAt) map.delete(k);
      return map.size;
    },
  };
}

export function diffObjects(before: Record<string, unknown>, after: Record<string, unknown>) {
  const added: string[] = [];
  const removed: string[] = [];
  const changed: { key: string; from: unknown; to: unknown }[] = [];
  for (const k of Object.keys(before)) {
    if (!Object.hasOwn(after, k)) removed.push(k);
    else if (JSON.stringify(before[k]) !== JSON.stringify(after[k])) changed.push({ key: k, from: before[k], to: after[k] });
  }
  for (const k of Object.keys(after)) if (!Object.hasOwn(before, k)) added.push(k);
  const byKey = (a: string, b: string) => (a < b ? -1 : a > b ? 1 : 0);

}

// P27-E67: משורות לעץ — עם זיהוי מעגל
type Row = { id: number; parentId: number | null; name: string };
type Node = { id: number; name: string; children: Node[] };

const byName = (a: Node, b: Node) => (a.name < b.name ? -1 : a.name > b.name ? 1 : 0);

export function buildTree(rows: Row[]): Node[] {
  const nodes = new Map<number, Node>();
  for (const r of rows) nodes.set(r.id, { id: r.id, name: r.name, children: [] });
  const roots: Node[] = [];
  for (const r of rows) {
    const node = nodes.get(r.id)!;
    if (r.parentId === null) roots.push(node);
    else {
      const parent = nodes.get(r.parentId);
      if (!parent) throw new Error(`ORPHAN ${r.id}`);
      parent.children.push(node);
    }
  }
  let reached = 0;
  const queue = [...roots];
  for (let head = 0; head < queue.length; head++) {
    reached++;
    queue[head].children.sort(byName);
    queue.push(...queue[head].children);
  }
  if (reached !== rows.length) throw new Error("CYCLE");
  return roots.sort(byName);
}
```

// P27-E68: BFS אסינכרוני — סריקת קישורים עד עומק
export async function crawl(start: string, getLinks: (url: string) => Promise<string[]>, maxDepth: number) {
  const out = [{ url: start, depth: 0 }];
  const seen = new Set([start]);
  let level = [start];
  for (let depth = 0; depth < maxDepth && level.length; depth++) {
    const results = await Promise.all(level.map((u) => getLinks(u).catch(() => [] as string[])));
    const next: string[] = [];
    for (const links of results) {
      for (const link of links) {
        if (seen.has(link)) continue;
        seen.add(link);
        next.push(link);
        out.push({ url: link, depth: depth + 1 });
      }
    }
    level = next;
  }
  return out;
}

// P27-E74: הרווח המקסימלי — מעבר אחד
export function maxProfit(prices: number[]) {
  let best: { profit: number; buy: number; sell: number } | null = null;
  let minAt = 0;
  for (let i = 1; i < prices.length; i++) {
    const profit = prices[i] - prices[minAt];
    if (profit > 0 && (!best || profit > best.profit)) best = { profit, buy: minAt, sell: i };
    if (prices[i] < prices[minAt]) minAt = i;
  }
  return best;
}

export function productExceptSelf(nums: number[]): number[] {
  const out = new Array<number>(nums.length);
  let left = 1;
  for (let i = 0; i < nums.length; i++) {
    out[i] = left;
    left *= nums[i];
  }
  let right = 1;
  for (let i = nums.length - 1; i >= 0; i--) {
    out[i] *= right;
    right *= nums[i];
  }
  return out;
}

export function isPalindromeSentence(s: string): boolean {
  let i = 0;
  let j = s.length - 1;
  while (i < j) {
    if (!ok(s[i])) i++;
    else if (!ok(s[j])) j--;

      i++;
      j--;
    }
  }

}

export function groupBy<T, K>(items: T[], keyOf: (item: T) => K): Map<K, T[]> {
  const out = new Map<K, T[]>();
  for (const item of items) {
    const k = keyOf(item);
    const list = out.get(k);
    if (list) list.push(item);
    else out.set(k, [item]);
  }
  return out;
}

export function countBy<T, K>(items: T[], keyOf: (item: T) => K): Map<K, number> {
  const out = new Map<K, number>();

    out.set(k, (out.get(k) ?? 0) + 1);
  }
  return out;
}

// P27-E89: סיסמאות עם scrypt ו־salt
import { randomBytes, scryptSync, timingSafeEqual } from "node:crypto";

export function hashPassword(password: string): string {
  const salt = randomBytes(16);
  const hash = scryptSync(password, salt, 64);
  return `scrypt$${salt.toString("hex")}$${hash.toString("hex")}`;
}

export function verifyPassword(password: string, stored: string): boolean {
  const [kind, saltHex, hashHex] = stored.split("$");
  if (kind !== "scrypt" || !/^[0-9a-f]{32}$/.test(saltHex ?? "") || !/^[0-9a-f]{128}$/.test(hashHex ?? "")) return false;
  const expected = Buffer.from(hashHex, "hex");
  const actual = scryptSync(password, Buffer.from(saltHex, "hex"), 64);
  return actual.length === expected.length && timingSafeEqual(actual, expected);
}
```

// P27-E91: cursor לעימוד — לקודד, לפענח, להמשיך
type Cursor = { createdAt: string; id: number };

export function encodeCursor(c: Cursor): string {
  return Buffer.from(JSON.stringify({ createdAt: c.createdAt, id: c.id })).toString("base64url");
}

export function decodeCursor(s: string): Cursor | null {
  if (!/^[A-Za-z0-9_-]+$/.test(s)) return null;
  try {
    const v = JSON.parse(Buffer.from(s, "base64url").toString("utf8"));
    if (typeof v?.createdAt !== "string" || !Number.isInteger(v?.id)) return null;
    return { createdAt: v.createdAt, id: v.id };
  } catch {
    return null;
  }
}

// האם a "אחרי" b בסדר יורד (createdAt, id)
const after = (a: Cursor, b: Cursor) => a.createdAt < b.createdAt || (a.createdAt === b.createdAt && a.id < b.id);

export function pageAfter<T extends Cursor>(sortedDesc: T[], cursor: string | null, limit: number) {
  let start = 0;
  const c = cursor === null ? null : decodeCursor(cursor);
  if (c) {
    let lo = 0;
    let hi = sortedDesc.length;
    while (lo < hi) {
      const mid = (lo + hi) >>> 1;
      if (after(sortedDesc[mid], c)) hi = mid;
      else lo = mid + 1;
    }
    start = lo;
  }
  const items = sortedDesc.slice(start, start + limit);
  const more = start + limit < sortedDesc.length;
  const last = items.at(-1);
  return { items, nextCursor: more && last ? encodeCursor(last) : null };
}

export function chooseStatus(s: S): number {
  if (s.rateLimited) return 429; // לא מבזבזים עבודה (גם לא בדיקת טוקן) על מי שחרג

}

// P27-E93: Idempotency-Key בזיכרון
import { createHash } from "node:crypto";

type Entry = { hash: string; promise: Promise<unknown>; at: number };

export function createIdempotency(ttlMs: number, now: () => number = Date.now) {
  const entries = new Map<string, Entry>();
  return {
    async run<T>(key: string, body: unknown, fn: () => Promise<T>) {
      const hash = createHash("sha256").update(JSON.stringify(body)).digest("hex");
      const e = entries.get(key);
      if (e && now() - e.at < ttlMs) {
        if (e.hash !== hash) throw new Error("KEY_REUSED");
        return { status: "replayed" as const, result: (await e.promise) as T };
      }
      const promise = fn();
      entries.set(key, { hash, promise, at: now() });
      promise.catch(() => {
        if (entries.get(key)?.promise === promise) entries.delete(key);
      });
      return { status: "fresh" as const, result: await promise };
    },
  };
}
```

// P27-E95: heap — k הגדולים מתוך זרם
export class MinHeap {
  #a: number[] = [];

  get size() {
    return this.#a.length;
  }
  peek() {
    return this.#a[0];
  }
  push(x: number) {
    const a = this.#a;
    a.push(x);
    let i = a.length - 1;
    while (i > 0) {
      const p = (i - 1) >> 1;
      if (a[p] <= a[i]) break;
      [a[p], a[i]] = [a[i], a[p]];
      i = p;
    }
  }
  pop() {
    const a = this.#a;
    if (a.length === 0) return undefined;
    const top = a[0];
    const last = a.pop()!;
    if (a.length) {
      a[0] = last;
      let i = 0;
      for (;;) {
        const l = 2 * i + 1;
        const r = l + 1;
        let m = i;
        if (l < a.length && a[l] < a[m]) m = l;
        if (r < a.length && a[r] < a[m]) m = r;
        if (m === i) break;
        [a[m], a[i]] = [a[i], a[m]];
        i = m;
      }
    }
    return top;
  }
}

export function topK(values: Iterable<number>, k: number): number[] {
  if (k <= 0) return [];
  const heap = new MinHeap();
  for (const x of values) {
    if (heap.size < k) heap.push(x);
    else if (x > heap.peek()!) {
      heap.pop();
      heap.push(x);
    }
  }
  const out: number[] = [];
  while (heap.size) out.push(heap.pop()!);
  return out.reverse();
}
```

// P27-E101: smoke test אחרי deploy
type Check = { path: string; status: number; contains?: string };

async function runCheck(baseUrl: string, c: Check, timeoutMs: number): Promise<string | null> {
  try {
    const res = await fetch(baseUrl + c.path, { signal: AbortSignal.timeout(timeoutMs) });
    const body = await res.text();
    if (res.status !== c.status) return `status ${res.status}`;
    if (c.contains !== undefined && !body.includes(c.contains)) return "missing text";
    return null;
  } catch (err) {
    return (err as Error).name === "TimeoutError" ? "timeout" : "network";
  }
}

export async function smokeTest(baseUrl: string, checks: Check[], timeoutMs: number) {
  const reasons = await Promise.all(checks.map((c) => runCheck(baseUrl, c, timeoutMs)));
  const failures = checks.flatMap((c, i) => (reasons[i] ? [{ path: c.path, reason: reasons[i]! }] : []));
  return { ok: failures.length === 0, failures };
}
```
<p>בדיקת עשן לא מחליפה בדיקות — היא עונה על שאלה אחת: "הגרסה שעלתה עכשיו, בסביבה האמיתית, עונה?" (‏health, דף ראשי, endpoint אחד שנוגע במסד). נכשלה ← rollback אוטומטי, לפני שלקוח שם לב.</p>

// P27-E102: p50, p95, p99 — לא ממוצע
export function percentiles(latencies: number[], ps: number[]) {
  for (const p of ps) if (!(p > 0 && p <= 100)) throw new RangeError(`p לא תקין: ${p}`);
  const sorted = Float64Array.from(latencies).sort();
  const n = sorted.length;
  const out: Record<string, number | null> = {};
  for (const p of ps) out[`p${p}`] = n === 0 ? null : sorted[Math.ceil((p / 100) * n) - 1];
  return out;
}
```
<p>למה לא ממוצע: 99 בקשות של 20ms ובקשה אחת של 5 שניות — ממוצע 70ms ("בסדר"), ‏p99 — 5 שניות. ואחד מכל מאה לקוחות מחכה 5 שניות. ‏p95/p99 הם מה שמשתמשים מרגישים, ועליהם שמים התראות.</p>
@mistakes
<ul><li>‏<code>latencies.sort()</code> — ממיין כמחרוזות (‏E32), ומשנה את המערך של מי שקרא.</li><li>‏<code>Math.floor</code> — ‏p100 יוצא מחוץ למערך, ו־p50 של 2 ערכים לוקח את השני.</li></ul>
