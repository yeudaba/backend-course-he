```ts save=solutions/P27-E06.ts
// P27-E06
// Set: "ראיתי כבר?" ב־O(1). מעבר אחד על המערך ← O(n) זמן, O(n) זיכרון
export function firstDuplicate<T>(items: readonly T[]): T | undefined {
  const seen = new Set<T>();
  for (const item of items) {
    if (seen.has(item)) return item;
    seen.add(item);
  }
  return undefined;
}

export function hasDuplicates<T>(items: readonly T[]): boolean {
  return new Set(items).size !== items.length;
}
```

```ts save=solutions/P27-E07.ts
// P27-E07
// Map של "ערך ← אינדקס" למה שכבר ראינו: לכל מחיר בודקים אם ההשלמה שלו כבר הופיעה. O(n)
export function twoSum(prices: readonly number[], target: number): [number, number] | null {
  const seenAt = new Map<number, number>();
  for (let j = 0; j < prices.length; j++) {
    const i = seenAt.get(target - prices[j]);
    if (i !== undefined) return [i, j];
    if (!seenAt.has(prices[j])) seenAt.set(prices[j], j); // הראשון — כדי שהזוג יהיה הכי מוקדם
  }
  return null;
}
```
<p>זו כנראה שאלת הראיון הנפוצה בעולם ("Two Sum"). הדפוס שמאחוריה — "Map של מה שכבר ראיתי, ובדיקה מול השלמה" — חוזר בעשרות שאלות.</p>

```ts save=solutions/P27-E08.ts
// P27-E08
// מפתח קנוני לכל קבוצה: האותיות ממוינות. Map שומר את סדר ההוספה — כלומר סדר ההופעה הראשונה
export function groupAnagrams(words: readonly string[]): string[][] {
  const groups = new Map<string, string[]>();
  for (const word of words) {
    const key = [...word.toLowerCase()].sort().join("");
    const group = groups.get(key);
    if (group) group.push(word);
    else groups.set(key, [word]);
  }
  return [...groups.values()];
}
```

```ts save=solutions/P27-E09.ts
// P27-E09
// מחסנית: כל פותח נדחף; כל סוגר חייב להתאים לפותח האחרון. מחסנית לא ריקה בסוף ← משהו לא נסגר
const PAIRS: Record<string, string> = { ")": "(", "]": "[", "}": "{" };

export function checkBrackets(text: string): { ok: true } | { ok: false; index: number } {
  const stack: number[] = []; // אינדקסים של הפותחים — כדי לדעת איפה הבעיה
  for (let i = 0; i < text.length; i++) {
    const ch = text[i];
    if (ch === "(" || ch === "[" || ch === "{") stack.push(i);
    else if (ch in PAIRS) {
      if (stack.length === 0 || text[stack[stack.length - 1]] !== PAIRS[ch]) return { ok: false, index: i };
      stack.pop();
    }
  }
  return stack.length === 0 ? { ok: true } : { ok: false, index: stack[0] };
}
```

```ts save=solutions/P27-E10.ts
// P27-E10
// חיפוש בינארי: הטווח [lo, hi) תמיד מכיל את התשובה, וחוצים אותו בכל צעד ← O(log n)
export function lowerBound(sorted: readonly number[], x: number): number {
  let lo = 0;
  let hi = sorted.length; // התשובה יכולה להיות length: "כולם קטנים מ־x"
  while (lo < hi) {
    const mid = (lo + hi) >>> 1;
    if (sorted[mid] < x) lo = mid + 1;
    else hi = mid;
  }
  return lo;
}

// כמה ערכים ב־[from, to] — שני חיפושים בינאריים
export function countInRange(sorted: readonly number[], from: number, to: number): number {
  if (from > to) return 0;
  return lowerBound(sorted, to + 1) - lowerBound(sorted, from);
}
```

```ts save=solutions/P27-E11.ts
// P27-E11
// חלון נע עם שני מצביעים: right מתקדם, left "מדביק" כשהחלון ארוך מדי. כל אינדקס נכנס ויוצא פעם אחת ← O(n)
export function peakInWindow(timestamps: readonly number[], windowMs: number): { count: number; start: number } {
  let best = { count: 0, start: timestamps[0] ?? 0 };
  let left = 0;
  for (let right = 0; right < timestamps.length; right++) {
    while (timestamps[right] - timestamps[left] >= windowMs) left++;
    const count = right - left + 1;
    if (count > best.count) best = { count, start: timestamps[left] };
  }
  return best;
}
```

```ts save=solutions/P27-E13.ts
export function mergeIntervals(intervals: readonly (readonly [number, number])[]): [number, number][] {
  const sorted = [...intervals].sort((a, b) => a[0] - b[0]);
  const merged: [number, number][] = [];
  for (const [start, end] of sorted) {
    const last = merged[merged.length - 1];
    if (last && start <= last[1]) last[1] = Math.max(last[1], end);
    else merged.push([start, end]);

  return merged;
```

```ts save=solutions/P27-E14.ts
// ספירה ב־Map ← O(n); מיון של הערכים השונים ← O(m log m). (עם heap בגודל k: O(m log k) — לרוב לא שווה את המורכבות)
export function topK(items: readonly string[], k: number): { item: string; count: number }[] {
  const counts = new Map<string, number>();
  for (const item of items) counts.set(item, (counts.get(item) ?? 0) + 1);

    .map(([item, count]) => ({ item, count }))
```

```ts save=solutions/P27-E15.ts
// P27-E15
// BFS: מתקדמים "בגלים" — כל התאים במרחק 1, אחר כך 2... הפעם הראשונה שמגיעים ל־E היא הקצרה ביותר
export function shortestPath(grid: readonly string[]): number {
  const h = grid.length;
  const w = grid[0]?.length ?? 0;
  let start: [number, number] | null = null;
  for (let y = 0; y < h; y++) {
    const x = grid[y].indexOf("S");
    if (x !== -1) start = [x, y];
  }
  if (!start) return -1;
  const dist = new Map<number, number>([[start[1] * w + start[0], 0]]); // מסמנים כשנכנסים לתור, לא כשיוצאים
  const queue: [number, number][] = [start];
  for (let head = 0; head < queue.length; head++) { // אינדקס במקום shift(): לא מזיזים כלום, גם כשהתור ענק
    const [x, y] = queue[head];
    const d = dist.get(y * w + x)!;
    if (grid[y][x] === "E") return d;
    for (const [dx, dy] of [[1, 0], [-1, 0], [0, 1], [0, -1]]) {
      const nx = x + dx;
      const ny = y + dy;
      if (nx < 0 || ny < 0 || nx >= w || ny >= h || grid[ny][nx] === "#" || dist.has(ny * w + nx)) continue;
      dist.set(ny * w + nx, d + 1);
      queue.push([nx, ny]);
    }
  }
  return -1;
}
```

```ts save=solutions/P27-E16.ts
// P27-E16
// Map ב־JavaScript שומר סדר הכנסה: הראשון = הכי פחות בשימוש. "שימוש" = מוחקים ומכניסים מחדש (לסוף). הכול O(1)
export class LRUCache<K, V> {
  private readonly map = new Map<K, V>();
  private readonly capacity: number;

  // (בלי "constructor(private capacity)": Node מריץ TypeScript בלי קומפילציה, וזה תחביר שאינו JavaScript)
  constructor(capacity: number) {
    this.capacity = capacity;
    if (!Number.isInteger(capacity) || capacity < 1) throw new Error("capacity חייב להיות שלם חיובי");
  }

  get size() {
    return this.map.size;
  }

  get(key: K): V | undefined {
    if (!this.map.has(key)) return undefined;
    const value = this.map.get(key)!;
    this.map.delete(key);
    this.map.set(key, value);
    return value;
  }

  set(key: K, value: V): void {
    this.map.delete(key);
    this.map.set(key, value);
    if (this.map.size > this.capacity) this.map.delete(this.map.keys().next().value!);
  }

  keys(): K[] {
    return [...this.map.keys()]; // מהישן לחדש
  }
}
```

```ts save=solutions/P27-E17.ts
// P27-E17
// מיון טופולוגי (Kahn): מתחילים ממי שאין לו תלויות, ו"משחררים" את מי שחיכה לו.
// נשארו צמתים שלא שוחררו ← יש מעגל. בכל שלב — האלפביתי הראשון, כדי שהסדר יהיה קבוע
export function buildOrder(deps: Readonly<Record<string, readonly string[]>>): string[] {
  const names = new Set<string>(Object.keys(deps));
  for (const list of Object.values(deps)) for (const d of list) names.add(d);
  const waitingFor = new Map<string, number>();
  const dependents = new Map<string, string[]>();
  for (const name of names) {
    waitingFor.set(name, 0);
    dependents.set(name, []);
  }
  for (const [name, list] of Object.entries(deps)) {
    for (const d of new Set(list)) {
      waitingFor.set(name, waitingFor.get(name)! + 1);
      dependents.get(d)!.push(name);
    }
  }
  const ready = [...names].filter((n) => waitingFor.get(n) === 0).sort();
  const order: string[] = [];
  while (ready.length) {
    const next = ready.shift()!;
    order.push(next);
    for (const dep of dependents.get(next)!) {
      const left = waitingFor.get(dep)! - 1;
      waitingFor.set(dep, left);
      if (left === 0) {
        ready.push(dep);
        ready.sort();
      }
    }
  }
  if (order.length !== names.size) {
    const stuck = [...names].filter((n) => !order.includes(n)).sort();
    throw new Error(`יש מעגל בתלויות: ${stuck.join(", ")}`);
  }
  return order;
}
```

```ts save=solutions/P27-E19.ts
// sweep line: כל סשן = שני אירועים (+1 בהתחלה, −1 בסוף). ממיינים לפי זמן — וסוף לפני התחלה באותו רגע
// (end לא כולל: סשן שנגמר ב־10 וסשן שמתחיל ב־10 לא חופפים). O(n log n)
export function peakConcurrency(sessions: readonly { start: number; end: number }[]): { peak: number; at: number | null } {
  const events: [number, number][] = [];
  for (const s of sessions) {
    if (s.end <= s.start) continue;
    events.push([s.start, 1], [s.end, -1]);
  }
  events.sort((a, b) => a[0] - b[0] || a[1] - b[1]);
  let current = 0;
  let best = { peak: 0, at: null as number | null };
  for (const [time, delta] of events) {
    current += delta;
    if (current > best.peak) best = { peak: current, at: time };
  }
  return best;
}
```

```ts save=solutions/P27-E20.ts
// P27-E20
export function lowerBound(sorted: readonly number[], x: number): number {
  let lo = 0;
  let hi = sorted.length; // תקלה 1: היה length − 1 ← "כולם קטנים" החזיר את האינדקס האחרון במקום length
  while (lo < hi) {
    const mid = (lo + hi) >>> 1;
    if (sorted[mid] < x) lo = mid + 1; // תקלה 2: היה <= ← זה upper bound (הראשון שגדול מ־x), לא הראשון שגדול או שווה
    else hi = mid;
  }
  return lo;
}

export function countInRange(sorted: readonly number[], from: number, to: number): number {
  if (from > to) return 0;
  return lowerBound(sorted, to + 1) - lowerBound(sorted, from); // תקלה 3: היה lowerBound(to) ← ערכים ששווים ל־to לא נספרו
}
```

```ts save=solutions/P27-E22.ts
// P27-E22
export function shortestPath(grid: readonly string[]): number {
  const h = grid.length;
  const w = grid[0]?.length ?? 0;
  let start: [number, number] | null = null;
  for (let y = 0; y < h; y++) {
    const x = grid[y].indexOf("S");
    if (x !== -1) start = [x, y];
  }
  if (!start) return -1;
  const dist = new Map<string, number>([[`${start[0]},${start[1]}`, 0]]);
  const queue: [number, number][] = [start];
  while (queue.length) {
    const [x, y] = queue.shift()!; // תקלה 2: היה pop() ← מחסנית = DFS, שמוצא מסלול — לא בהכרח הקצר
    const d = dist.get(`${x},${y}`)!;
    if (grid[y][x] === "E") return d;
    for (const [dx, dy] of [[1, 0], [-1, 0], [0, 1], [0, -1]]) {
      const nx = x + dx;
      const ny = y + dy;
      // תקלה 1: היה nx <= w ו־ny <= h ← grid[y][w] הוא undefined, "לא קיר" — והמסלול יצא מחוץ למפה
      if (nx < 0 || ny < 0 || nx >= w || ny >= h || grid[ny][nx] === "#" || dist.has(`${nx},${ny}`)) continue;
      dist.set(`${nx},${ny}`, d + 1);
      queue.push([nx, ny]);
    }
  }
  return -1; // תקלה 3: היה return 0 ← "אין דרך" נראה כמו "כבר שם"
}
```
<p>(כאן נשאר <code>shift()</code> — ברור יותר לקריאה. ב־BFS על רשת התור נשאר קטן (רק ה"חזית"), אז גם על 400×400 מדדתי שזה מהיר; בתור שמגיע למאות אלפים — אינדקס, כמו ב־E15.)</p>

```ts save=solutions/P27-E24.ts
// P27-E24
// תכנון דינמי (DP): התשובה לסכום a נבנית מהתשובות לסכומים קטנים יותר — כל אחד מחושב פעם אחת. O(amount × coins)
export function minCoins(coins: readonly number[], amount: number): number {
  const best = new Array<number>(amount + 1).fill(Infinity);
  best[0] = 0;
  for (let a = 1; a <= amount; a++) {
    for (const c of coins) {
      if (c <= a && best[a - c] + 1 < best[a]) best[a] = best[a - c] + 1;
    }
  }
  return best[amount] === Infinity ? -1 : best[amount];
}

// כמה צירופים (בלי חשיבות לסדר): הלולאה החיצונית על המטבעות — כך 1+2 ו־2+1 נספרים פעם אחת
export function countWays(coins: readonly number[], amount: number): number {
  const ways = new Array<number>(amount + 1).fill(0);
  ways[0] = 1;
  for (const c of coins) {
    for (let a = c; a <= amount; a++) ways[a] += ways[a - c];
  }
  return ways[amount];
}
```

```ts save=solutions/P27-E25.ts
// P27-E25
// Trie: עץ של אותיות. כל צומת שומר גם את ה־k הכי פופולריים מתחתיו — כך suggest לא סורק את כל תת־העץ.
// add מעדכן את הרשימות לאורך המסלול: O(L × k). suggest: O(L) — הליכה לצומת, והרשימה מוכנה
interface Node {
  children: Map<string, Node>;
  top: string[]; // עד MAX מונחים, ממוינים
}

const MAX = 10;

export function createAutocomplete() {
  const root: Node = { children: new Map(), top: [] };
  const weights = new Map<string, number>(); // מונח (lowercase) ← משקל מצטבר
  const display = new Map<string, string>(); // מונח (lowercase) ← הכתיב הראשון שנוסף
  const better = (a: string, b: string) => weights.get(b)! - weights.get(a)! || (a < b ? -1 : a > b ? 1 : 0);

  return {
    add(term: string, weight = 1) {
      const key = term.trim().toLowerCase();
      if (!key) return;
      weights.set(key, (weights.get(key) ?? 0) + weight);
      if (!display.has(key)) display.set(key, term.trim());
      let node = root;
      const path = [root];
      for (const ch of key) {
        let next = node.children.get(ch);
        if (!next) {
          next = { children: new Map(), top: [] };
          node.children.set(ch, next);
        }
        node = next;
        path.push(node);
      }
      for (const n of path) {
        if (!n.top.includes(key)) n.top.push(key);
        n.top.sort(better);
        if (n.top.length > MAX) n.top.length = MAX;
      }
    },
    suggest(prefix: string, k = 5): string[] {
      let node: Node | undefined = root;
      for (const ch of prefix.trim().toLowerCase()) {
        node = node.children.get(ch);
        if (!node) return [];
      }
      return node.top.slice(0, Math.min(k, MAX)).map((key) => display.get(key)!);
    },
    get size() {
      return weights.size;
    },
  };
}
```

```ts run save=solutions/P27-E26.ts
// P27-E26 — הראשון שמופיע פעם אחת בלבד
export function firstUnique<T>(items: T[]): T | undefined {
  const counts = new Map<T, number>();
  for (const item of items) counts.set(item, (counts.get(item) ?? 0) + 1);
  for (const item of items) if (counts.get(item) === 1) return item;
  return undefined;
}
```

```ts run save=solutions/P27-E27.ts
// P27-E27 — מיזוג רשימות ממוינות
export function mergeSorted(a: number[], b: number[]): number[] {
  const out: number[] = [];
  let i = 0;
  let j = 0;
  while (i < a.length && j < b.length) {
    if (a[i] <= b[j]) out.push(a[i++]);
    else out.push(b[j++]);
  }
  while (i < a.length) out.push(a[i++]);
  while (j < b.length) out.push(b[j++]);
  return out;
}

export function mergeAll(lists: number[][]): number[] {
  if (lists.length === 0) return [];
  let round = lists;
  while (round.length > 1) {
    const next: number[][] = [];
    for (let k = 0; k < round.length; k += 2) {
      next.push(k + 1 < round.length ? mergeSorted(round[k], round[k + 1]) : round[k]);
    }
    round = next;
  }
  return round[0];
}
```

```ts run save=solutions/P27-E28.ts
// P27-E28 — דחיסת רצפים (run-length encoding)
export function compress(text: string): string {
  if (/\d/.test(text)) throw new TypeError("אי אפשר לדחוס טקסט עם ספרות — לא יהיה אפשר לפתוח אותו");
  let out = "";
  let i = 0;
  while (i < text.length) {
    let j = i;
    while (j < text.length && text[j] === text[i]) j++;
    const run = j - i;
    out += run === 1 ? text[i] : `${text[i]}${run}`;
    i = j;
  }
  return out;
}

export function decompress(code: string): string {
  if (/\d/.test(code[0] ?? "")) throw new SyntaxError("קוד לא יכול להתחיל בספרה");
  let out = "";
  for (const [, ch, count] of code.matchAll(/(\D)(\d*)/g)) {
    if (count.startsWith("0")) throw new SyntaxError(`מספר לא תקין אחרי ${ch}: ${count}`);
    out += ch.repeat(count === "" ? 1 : Number(count));
  }
  return out;
}

const samples = ["aaabccdddd", "abc", "", "zzzzzzzzzzzz"];
for (const s of samples) {
  const c = compress(s);
  console.log(JSON.stringify(s), "→", JSON.stringify(c), "→", decompress(c) === s ? "✓" : "✗");
}
```
