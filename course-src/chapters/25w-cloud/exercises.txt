```ts save=solutions/P25.5-E06.ts
import { S3Client, PutObjectCommand, GetObjectCommand, NoSuchKey } from "@aws-sdk/client-s3";

export async function putJson(s3: S3Client, bucket: string, key: string, value: unknown): Promise<void> {
  await s3.send(
    new PutObjectCommand({
      Bucket: bucket,
      Key: key,
      Body: JSON.stringify(value),
      ContentType: "application/json; charset=utf-8",
    }),
  );
}

export async function getJson<T = unknown>(s3: S3Client, bucket: string, key: string): Promise<T | null> {
  try {
    const res = await s3.send(new GetObjectCommand({ Bucket: bucket, Key: key }));
    const text = await res.Body!.transformToString("utf-8");
    return JSON.parse(text) as T;
  } catch (err) {
    if (err instanceof NoSuchKey) return null; // רק "אין כזה" הוא null; כל שגיאה אחרת — למעלה
    throw err;
  }
}
```

```ts save=solutions/P25.5-E07.ts
// P25.5-E07: כתובת חתומה (presigned URL) להעלאת קובץ ישירות ל־S3
import { S3Client, PutObjectCommand } from "@aws-sdk/client-s3";
import { getSignedUrl } from "@aws-sdk/s3-request-presigner";
import { randomUUID } from "node:crypto";

export const ALLOWED_TYPES = ["image/png", "image/jpeg", "application/pdf"];
export const MAX_BYTES = 5 * 1024 * 1024;
export const EXPIRES_IN = 300;

export interface UploadRequest {
  userId: number;
  filename: string;
  contentType: string;
  size: number;
}

export function safeName(filename: string): string {
  const base = filename.split(/[\\/]/).pop() ?? "";
  const clean = base.replace(/[^A-Za-z0-9._-]/g, "_").replace(/^\.+/, "").slice(-100);
  return clean || "file";
}

export async function createUploadUrl(s3: S3Client, bucket: string, req: UploadRequest) {
  if (!ALLOWED_TYPES.includes(req.contentType)) throw new Error(`סוג קובץ לא מותר: ${req.contentType}`);
  if (!Number.isInteger(req.size) || req.size <= 0 || req.size > MAX_BYTES) throw new Error("גודל קובץ לא תקין (עד 5MB)");
  if (!Number.isInteger(req.userId) || req.userId <= 0) throw new Error("userId לא תקין");
  const key = `uploads/${req.userId}/${randomUUID()}-${safeName(req.filename)}`;
  const url = await getSignedUrl(
    s3,
    new PutObjectCommand({ Bucket: bucket, Key: key, ContentType: req.contentType, ContentLength: req.size }),
    { expiresIn: EXPIRES_IN },
  );
  return { url, key, expiresIn: EXPIRES_IN, headers: { "content-type": req.contentType } };
```

```ts save=solutions/P25.5-E12.ts
export interface AppConfig {
  databaseUrl: string;
  jwtSecret: string;
  redisUrl: string;
  port: number;
}

type ReadFile = (path: string) => string;

const REQUIRED = { databaseUrl: "DATABASE_URL", jwtSecret: "JWT_SECRET", redisUrl: "REDIS_URL" } as const;

export function readSetting(env: Record<string, string | undefined>, name: string, readFile: ReadFile): string | undefined {
  const direct = env[name];
  const file = env[`${name}_FILE`];
  if (direct !== undefined && file !== undefined) throw new Error(`${name} ו־${name}_FILE מוגדרים שניהם — בחר אחד`);
  if (file !== undefined) {
    let text: string;
    try {
      text = readFile(file);
    } catch {
      throw new Error(`${name}_FILE: אי אפשר לקרוא את ${file}`);
    }
    const value = text.replace(/\r?\n$/, "");
    if (!value) throw new Error(`${name}_FILE: הקובץ ${file} ריק`);
    return value;
  }
  return direct === "" ? undefined : direct;
}

export function loadConfig(env: Record<string, string | undefined>, readFile: ReadFile): AppConfig {
  const missing: string[] = [];
  const out: Record<string, string> = {};
  for (const [key, name] of Object.entries(REQUIRED)) {
    const v = readSetting(env, name, readFile);
    if (v === undefined) missing.push(name);
    else out[key] = v;
  }
  if (missing.length) throw new Error(`חסרות הגדרות: ${missing.join(", ")}`);
  const port = Number(env.PORT ?? "3000");
  if (!Number.isInteger(port) || port < 1 || port > 65535) throw new Error(`PORT לא תקין: ${env.PORT}`);
  return { databaseUrl: out.databaseUrl, jwtSecret: out.jwtSecret, redisUrl: out.redisUrl, port };
}
```

```ts save=solutions/P25.5-E13.ts
// P25.5-E13: להעריך IAM policy — כמו ש־AWS מחליט
export interface Statement {
  Effect: "Allow" | "Deny";
  Action: string | string[];
  Resource: string | string[];
}
export interface Policy {
  Version: string;
  Statement: Statement[];
}
export type Decision = "allow" | "explicit-deny" | "implicit-deny";

const list = (x: string | string[]) => (Array.isArray(x) ? x : [x]);

export function wildcardMatch(pattern: string, value: string, ignoreCase = false): boolean {
  const escaped = pattern.replace(/[.+^${}()|[\]\\]/g, "\\$&").replace(/\*/g, ".*").replace(/\?/g, ".");
  return new RegExp(`^${escaped}$`, ignoreCase ? "i" : "").test(value);
}

export function evaluate(policies: Policy[], action: string, resource: string): Decision {
  let allowed = false;
  for (const p of policies) {
    for (const s of p.Statement) {
      const hit =
        list(s.Action).some((a) => wildcardMatch(a, action, true)) && // שמות פעולות לא רגישים לרישיות
        list(s.Resource).some((r) => wildcardMatch(r, resource)); // ARN רגיש
      if (!hit) continue;
      if (s.Effect === "Deny") return "explicit-deny"; // Deny מפורש מנצח תמיד
      allowed = true;
    }
  }
  return allowed ? "allow" : "implicit-deny";
}
```
<p>באמת יש עוד שכבות: policies של ה־bucket עצמו (resource-based), Permission Boundaries, Service Control Policies של הארגון, ו־Conditions. העיקרון לא משתנה: Deny מפורש בכל שכבה מנצח.</p>

```ts save=solutions/P25.5-E14.ts
// P25.5-E14: כמויות של Kubernetes — CPU ו־memory
const BINARY: Record<string, number> = { Ki: 2 ** 10, Mi: 2 ** 20, Gi: 2 ** 30, Ti: 2 ** 40 };
const DECIMAL: Record<string, number> = { k: 1e3, M: 1e6, G: 1e9, T: 1e12 };

// מחזיר millicores: "250m" ← 250, "1" ← 1000, "0.5" ← 500
export function parseCpu(q: string | number): number {
  const s = String(q).trim();
  const m = /^(\d+(?:\.\d+)?)(m?)$/.exec(s);
  if (!m) throw new Error(`CPU לא תקין: ${q}`);
  const n = Number(m[1]);
  const milli = m[2] ? n : n * 1000;
  if (!Number.isInteger(milli)) throw new Error(`CPU לא תקין (פחות ממילי־ליבה): ${q}`);
  return milli;
}

// מחזיר בתים: "512Mi" ← 536870912, "1G" ← 1e9
export function parseMemory(q: string | number): number {
  const s = String(q).trim();
  const m = /^(\d+(?:\.\d+)?)(Ki|Mi|Gi|Ti|k|M|G|T)?$/.exec(s);
  if (!m) throw new Error(`memory לא תקין: ${q}`);
  const unit = m[2] ? (BINARY[m[2]] ?? DECIMAL[m[2]]) : 1;
  return Math.round(Number(m[1]) * unit);
}

export function formatMemory(bytes: number): string {
  for (const u of ["Gi", "Mi", "Ki"]) if (bytes >= BINARY[u] && bytes % BINARY[u] === 0) return `${bytes / BINARY[u]}${u}`;
  return String(bytes);
}
```

```ts save=solutions/P25.5-E17.ts
// P25.5-E17: rolling update — כמה pods מותר, ואיך ה־rollout מתקדם
export type Amount = number | `${number}%`;

function resolve(amount: Amount, replicas: number, round: (n: number) => number, name: string): number {
  if (typeof amount === "number") {
    if (!Number.isInteger(amount) || amount < 0) throw new Error(`${name} לא תקין: ${amount}`);
    return amount;
  }
  const m = /^(\d+)%$/.exec(amount);
  if (!m) throw new Error(`${name} לא תקין: ${amount}`);
  return round((replicas * Number(m[1])) / 100);
}

export function rolloutBounds(replicas: number, maxSurge: Amount, maxUnavailable: Amount) {
  if (!Number.isInteger(replicas) || replicas < 1) throw new Error("replicas חייב להיות לפחות 1");
  const surge = resolve(maxSurge, replicas, Math.ceil, "maxSurge"); // Kubernetes: surge מעגל למעלה
  const unavailable = resolve(maxUnavailable, replicas, Math.floor, "maxUnavailable"); // unavailable — למטה
  if (surge === 0 && unavailable === 0) throw new Error("maxSurge ו־maxUnavailable לא יכולים להיות שניהם 0 — ה־rollout לא יתקדם");
  return { maxPods: replicas + surge, minAvailable: replicas - unavailable };
}

export interface Snapshot {
  old: number;
  new: number;
}

// מודל פשוט: pods חדשים שנוצרו בצעד מוכנים (ready) בסוף הצעד
export function simulateRollout(replicas: number, maxSurge: Amount, maxUnavailable: Amount) {
  const { maxPods, minAvailable } = rolloutBounds(replicas, maxSurge, maxUnavailable);
  let oldPods = replicas;
  let newPods = 0;
  const steps: Snapshot[] = [{ old: oldPods, new: newPods }];
  while (oldPods > 0 || newPods < replicas) {
    const remove = Math.max(0, Math.min(oldPods, oldPods + newPods - minAvailable));
    oldPods -= remove;
    const add = Math.max(0, Math.min(replicas - newPods, maxPods - oldPods - newPods));
    newPods += add;
    if (remove === 0 && add === 0) throw new Error("ה־rollout תקוע");
    steps.push({ old: oldPods, new: newPods });
  }
  return { maxPods, minAvailable, steps };
}
```
<p>הבחירה בפועל: <code>maxUnavailable: 0</code> + <code>maxSurge: 1</code> — אף פעם לא פחות קיבולת, אבל איטי ודורש מקום ל־pod נוסף. <code>25%/25%</code> (ברירת המחדל של Kubernetes) — מהר יותר. <code>maxSurge: 0</code> — כשאין מקום פנוי ב־cluster.</p>

```ts save=solutions/P25.5-E18.ts
// P25.5-E18: scheduler קטן — לאיזה node כל pod הולך (לפי requests)
export interface NodeSpec {
  name: string;
  cpu: string;
  memory: string;
}
export interface PodSpec {
  name: string;
  cpu: string;
  memory: string;
}

// מוכנות (מ־E14)
export const parseCpu = (q: string) => (q.endsWith("m") ? Number(q.slice(0, -1)) : Number(q) * 1000);
const UNITS: Record<string, number> = { Ki: 2 ** 10, Mi: 2 ** 20, Gi: 2 ** 30, k: 1e3, M: 1e6, G: 1e9 };
export const parseMemory = (q: string) => {
  const m = /^(\d+(?:\.\d+)?)(Ki|Mi|Gi|k|M|G)?$/.exec(q)!;
  return Math.round(Number(m[1]) * (m[2] ? UNITS[m[2]] : 1));
};

export function schedule(nodes: NodeSpec[], pods: PodSpec[]) {
  const free = new Map(nodes.map((n) => [n.name, { cpu: parseCpu(n.cpu), memory: parseMemory(n.memory) }]));
  const order = pods
    .map((p) => ({ name: p.name, cpu: parseCpu(p.cpu), memory: parseMemory(p.memory) }))
    .sort((a, b) => b.cpu - a.cpu || b.memory - a.memory || a.name.localeCompare(b.name)); // הגדולים קודם
  const placements: Record<string, string> = {};
  const pending: string[] = [];
  for (const pod of order) {
    const node = nodes.find((n) => {
      const f = free.get(n.name)!;
      return f.cpu >= pod.cpu && f.memory >= pod.memory;
    });
    if (!node) {
      pending.push(pod.name);
      continue;
    }
    const f = free.get(node.name)!;
    f.cpu -= pod.cpu;
    f.memory -= pod.memory;
    placements[pod.name] = node.name;
  }
  return { placements, pending, free: Object.fromEntries(free) };
}
```

```ts save=solutions/P25.5-E19.ts
import { S3Client, ListObjectsV2Command, DeleteObjectsCommand } from "@aws-sdk/client-s3";

export async function listAllKeys(s3: S3Client, bucket: string, prefix: string, pageSize = 1000): Promise<string[]> {
  const keys: string[] = [];
  let token: string | undefined;
  do {
    const page = await s3.send(
      new ListObjectsV2Command({ Bucket: bucket, Prefix: prefix, MaxKeys: pageSize, ContinuationToken: token }),
    );
    for (const o of page.Contents ?? []) keys.push(o.Key!);
    token = page.IsTruncated ? page.NextContinuationToken : undefined;
  } while (token);
  return keys;
}

export async function deletePrefix(s3: S3Client, bucket: string, prefix: string): Promise<number> {
  if (!prefix || !prefix.endsWith("/")) throw new Error('prefix חייב להיות תיקייה שמסתיימת ב־"/" (ולא ריק — זה כל ה־bucket)');
  const keys = await listAllKeys(s3, bucket, prefix);
  for (let i = 0; i < keys.length; i += 1000) {
    const batch = keys.slice(i, i + 1000);
    const res = await s3.send(
      new DeleteObjectsCommand({ Bucket: bucket, Delete: { Objects: batch.map((Key) => ({ Key })), Quiet: true } }),
    );
    if (res.Errors?.length) throw new Error(`מחיקה נכשלה: ${res.Errors[0].Key} — ${res.Errors[0].Message}`);
  }
  return keys.length;
}
```

```ts save=solutions/P25.5-E25.ts
// P25.5-E25: GitOps — לעדכן את ה־tag של image ב־repo התצורה, בלי להרוס את הקובץ
import { parseAllDocuments, isMap, isSeq, YAMLMap } from "yaml";

const TAG_RE = /^[A-Za-z0-9_][A-Za-z0-9._-]{0,127}$/;

// "ghcr.io/x/app:1.2" ← "ghcr.io/x/app"; גם @sha256:…; ולא מתבלבל עם port של registry (localhost:5000/app)
export function repositoryOf(image: string): string {
  const at = image.indexOf("@");
  const noDigest = at >= 0 ? image.slice(0, at) : image;
  const slash = noDigest.lastIndexOf("/");
  const colon = noDigest.lastIndexOf(":");
  return colon > slash ? noDigest.slice(0, colon) : noDigest;
}

function podSpecs(doc: any): YAMLMap[] {
  const out: YAMLMap[] = [];
  const kind = doc.get("kind");
  const spec = doc.get("spec");
  if (!isMap(spec)) return out;
  const tpl = kind === "CronJob" ? (spec.getIn(["jobTemplate", "spec", "template", "spec"]) as unknown) : spec.getIn(["template", "spec"]);
  if (isMap(tpl)) out.push(tpl);
  return out;
}

export function bumpImageTag(text: string, repository: string, tag: string): { text: string; changed: number } {
  if (!TAG_RE.test(tag)) throw new Error(`tag לא תקין: ${tag}`);
  const docs = parseAllDocuments(text);
  let changed = 0;
  for (const doc of docs) {
    if ("errors" in doc && doc.errors.length) throw new Error(`YAML לא תקין: ${doc.errors[0].message}`);
    for (const pod of podSpecs(doc)) {
      for (const field of ["initContainers", "containers"]) {
        const list = pod.get(field);
        if (!isSeq(list)) continue;
        for (const c of list.items) {
          if (!isMap(c)) continue;
          const image = c.get("image");
          if (typeof image !== "string" || repositoryOf(image) !== repository) continue;
          const next = `${repository}:${tag}`;
          if (image !== next) {
            c.set("image", next);
            changed++;
          }
        }
      }
    }
  }
  if (changed === 0 && !text.includes(`${repository}:${tag}`)) throw new Error(`לא נמצא image של ${repository}`);
  return { text: docs.map((d) => d.toString()).join(""), changed };
}
```
