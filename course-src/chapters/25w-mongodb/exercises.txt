// בונים את ה־filter רק מהשדות שהגיעו: undefined לא נכנס, אחרת הוא היה מחפש "שדה שערכו null"
export async function findProducts(db: Db, { category, maxPrice, inStock, tag }: { category?: string; maxPrice?: number; inStock?: boolean; tag?: string } = {}) {
  const filter: Record<string, unknown> = {};
  if (category !== undefined) filter.category = category;
  if (maxPrice !== undefined) filter.price = { $lte: maxPrice };
  if (inStock) filter.stock = { $gt: 0 };

  name: string;

  stock: number;

}

import type { Db } from "mongodb";

// collection עם טיפוס: TypeScript יודע ש־tags הוא מערך של מחרוזות, ובודק את $addToSet ו־$pull
interface Product {
  _id: string;
  name: string;
  price: number;
  stock: number;
  tags: string[];
}

const norm = (t: string) => t.trim().toLowerCase();

// $addToSet — מוסיף רק אם לא קיים. $push היה מוסיף כפילות
export async function addTag(db: Db, productId: string, tag: string) {
  const r = await db.collection<Product>("products").updateOne({ _id: productId }, { $addToSet: { tags: norm(tag) } });
  return r.matchedCount === 1;
}

export async function removeTag(db: Db, productId: string, tag: string) {
  const r = await db.collection<Product>("products").updateOne({ _id: productId }, { $pull: { tags: norm(tag) } });
  return r.matchedCount === 1;
}
```

// P25.7-E17
import type { Db, MongoClient } from "mongodb";

interface Product {
  _id: string;
  name: string;
  price: number;
  stock: number;
}

export class OutOfStockError extends Error {}

// הזמנה = הורדת מלאי בכמה מוצרים + מסמך הזמנה. או הכול, או כלום — טרנזקציה
export async function placeOrder(client: MongoClient, db: Db, customerId: string, cart: { productId: string; qty: number }[]) {
  if (!cart.length) throw new RangeError("עגלה ריקה");
  const session = client.startSession();
  try {
    return await session.withTransaction(async () => {
      const items = [];
      for (const { productId, qty } of cart) {
        // כל פעולה בתוך הטרנזקציה — עם { session }. בלי זה היא רצה מחוץ לטרנזקציה ולא מתבטלת
        const p = await db.collection<Product>("products").findOneAndUpdate(
          { _id: productId, stock: { $gte: qty } },
          { $inc: { stock: -qty } },
          { session, returnDocument: "after", projection: { name: 1, price: 1 } }
        );
        if (!p) throw new OutOfStockError(`אין מספיק ${productId}`); // זריקה ← withTransaction מבטל הכול
        items.push({ productId, name: p.name, price: p.price, qty }); // מחיר מהמסד, לא מהלקוח
      }
      const order = { customerId, status: "pending", createdAt: new Date(), items, total: items.reduce((s, i) => s + i.price * i.qty, 0) };
      const { insertedId } = await db.collection("orders").insertOne(order, { session });
      return { _id: insertedId, ...order };
    });
  } finally {
    await session.endSession();
  }
}

// P25.7-E25
import type { Db } from "mongodb";
import type pg from "pg";

// מעבר מ־PostgreSQL ל־MongoDB: הזמנה + השורות שלה (order_items) ← מסמך אחד עם מערך items.
// אפשר להריץ שוב ושוב (idempotent): upsert לפי ה־id המקורי
export async function migrateOrders(pool: pg.Pool, db: Db) {
  const { rows } = await pool.query(`
    SELECT o.id, o.customer_id, o.status, o.created_at,
           json_agg(json_build_object('productId', i.product_id, 'name', p.name, 'price', i.unit_price, 'qty', i.quantity) ORDER BY i.id) AS items
    FROM orders o
    JOIN order_items i ON i.order_id = o.id
    JOIN products p ON p.id = i.product_id
    GROUP BY o.id
    ORDER BY o.id`);
  if (rows.length === 0) return { migrated: 0 };
  const ops = rows.map((r) => {
    const items = r.items.map((i: { productId: number; name: string; price: string | number; qty: number }) => ({ productId: `p${i.productId}`, name: i.name, price: Number(i.price), qty: i.qty }));
    const doc = {
      customerId: `c${r.customer_id}`,
      status: r.status,
      createdAt: r.created_at, // timestamptz ← Date (pg ממיר לבד) ← BSON Date
      items,
      total: Math.round(items.reduce((s: number, i: { price: number; qty: number }) => s + i.price * i.qty, 0) * 100) / 100,
    };
    return { replaceOne: { filter: { _id: `o${r.id}` }, replacement: doc, upsert: true } } as const;
  });
  // bulkWrite: פקודה אחת לשרת במקום אלף
  await db.collection<{ _id: string }>("orders").bulkWrite(ops, { ordered: false });
  return { migrated: rows.length };
}

// P25.7-E26 — חיפוש מוצרים מ־query string
import type { Db } from "mongodb";

type Query = Record<string, string | undefined>;

const escapeRegex = (s: string) => s.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");

function price(value: string, name: string): number {
  const n = Number(value);
  if (value.trim() === "" || !Number.isFinite(n) || n < 0) throw new RangeError(`${name} לא תקין: ${value}`);
  return n;
}

export function buildProductFilter(query: Query): Record<string, unknown> {
  const filter: Record<string, unknown> = {};
  if (query.q) filter.name = { $regex: escapeRegex(query.q.trim()), $options: "i" };
  if (query.category) filter.category = query.category;

  const range: Record<string, number> = {};
  if (query.minPrice !== undefined) range.$gte = price(query.minPrice, "minPrice");
  if (query.maxPrice !== undefined) range.$lte = price(query.maxPrice, "maxPrice");
  if (range.$gte !== undefined && range.$lte !== undefined && range.$gte > range.$lte) throw new RangeError("minPrice גדול מ־maxPrice");
  if (Object.keys(range).length > 0) filter.price = range;

  if (query.tags) {
    const tags = query.tags.split(",").map((t) => t.trim()).filter((t) => t !== "");
    if (tags.length > 0) filter.tags = { $all: tags };
  }

  switch (query.inStock) {
    case undefined:
      break;
    case "true":
      filter.stock = { $gt: 0 };
      break;
    case "false":
      filter.stock = 0;
      break;
    default:
      throw new RangeError(`inStock חייב להיות true או false: ${query.inStock}`);
  }
  return filter;
}

export async function searchProducts(db: Db, query: Query): Promise<string[]> {
  const docs = await db.collection("products").find(buildProductFilter(query)).sort({ name: 1 }).project({ _id: 0, name: 1 }).toArray();
  return docs.map((d) => d.name);
}

// P25.7-E28 — מה מסתתר ב־ObjectId
const OBJECT_ID = /^[0-9a-f]{24}$/i;

export function objectIdDate(id: string): Date {
  if (!OBJECT_ID.test(id)) throw new TypeError(`ObjectId לא תקין: ${id}`);
  const seconds = parseInt(id.slice(0, 8), 16);
  return new Date(seconds * 1000);
}

export function objectIdFromDate(date: Date): string {
  const seconds = Math.floor(date.getTime() / 1000);
  return seconds.toString(16).padStart(8, "0") + "0".repeat(16);
}

export function idRangeForDay(day: string): { $gte: string; $lt: string } {
  const match = day.match(/^(\d{4})-(\d{2})-(\d{2})$/);
  if (!match) throw new TypeError(`תאריך לא תקין: ${day}`);
  const start = new Date(Date.UTC(Number(match[1]), Number(match[2]) - 1, Number(match[3])));
  const end = new Date(start.getTime() + 24 * 60 * 60 * 1000);
  return { $gte: objectIdFromDate(start), $lt: objectIdFromDate(end) };
}

// P25.7-E36: מוצר חדש — עם Zod לפני insert
import type { Db } from "mongodb";
import { z } from "zod";

const schema = z.strictObject({
  name: z.string().trim().min(1),
  category: z.enum(["beans", "gear", "merch"]),
  price: z.number().int().positive(),
  stock: z.number().int().min(0).default(0),
  tags: z.array(z.string()).default([]),
});

export async function addProduct(db: Db, input: unknown) {
  const r = schema.safeParse(input);
  if (!r.success) throw new Error("VALIDATION_ERROR");
  const { insertedId } = await db.collection("products").insertOne({ ...r.data, createdAt: new Date() });
  return { id: insertedId.toHexString() };
}

// P25.7-E38: ממסמך ל־JSON של API
import { ObjectId } from "mongodb";

export function toApi(value: unknown): unknown {
  if (value instanceof ObjectId) return value.toHexString();
  if (value instanceof Date) return value.toISOString();
  if (Array.isArray(value)) return value.map(toApi);
  if (value !== null && typeof value === "object") {
    return Object.fromEntries(Object.entries(value).map(([k, x]) => [k === "_id" ? "id" : k, toApi(x)]));
  }
  return value;
}

import type { Db } from "mongodb";

export async function loadSeed(db: Db, jsonText: string) {
  let data: Record<string, Record<string, unknown>[]>;
  try {
    data = JSON.parse(jsonText, (key, value) => {
      if (!key.endsWith("At") || typeof value !== "string") return value;
      const d = new Date(value);
      if (Number.isNaN(d.getTime())) throw new Error(`תאריך לא תקין בשדה ${key}`);
      return d;
    });
  } catch (err) {
    if (err instanceof SyntaxError) throw new Error("JSON לא תקין");
    throw err;
  }
  const counts: Record<string, number> = {};
  for (const [name, docs] of Object.entries(data)) {
    if (docs.length) await db.collection(name).insertMany(docs);
    counts[name] = docs.length;
  }
  return counts;
}

import type { Db } from "mongodb";

export async function updatePrice(db: Db, id: string, expectedPrice: number, newPrice: number) {
  if (!Number.isInteger(newPrice) || newPrice <= 0) throw new RangeError(`מחיר לא תקין: ${newPrice}`);
  const products = db.collection<{ _id: string; price: number }>("products");
  const r = await products.updateOne({ _id: id, price: expectedPrice }, { $set: { price: newPrice } });
  if (r.matchedCount === 1) return "updated" as const;
  return (await products.countDocuments({ _id: id }, { limit: 1 })) ? ("conflict" as const) : ("not_found" as const);
}

// P25.7-E53: עמוד הבא של הזמנות — keyset
import type { Db, Filter } from "mongodb";

type Order = { _id: string; createdAt: Date; total: number };

export async function ordersPage(db: Db, { limit, after }: { limit: number; after?: string }) {
  let filter: Filter<Order> = {};
  if (after !== undefined) {
    const parts = after.split("|");
    const t = new Date(parts[0]);
    if (parts.length !== 2 || !parts[1] || Number.isNaN(t.getTime())) throw new Error("BAD_CURSOR");
    filter = { $or: [{ createdAt: { $lt: t } }, { createdAt: t, _id: { $lt: parts[1] } }] };
  }
  const docs = await db.collection<Order>("orders")
    .find(filter, { projection: { createdAt: 1, total: 1 } })
    .sort({ createdAt: -1, _id: -1 })
    .limit(limit)
    .toArray();
  const items = docs.map((o) => ({ id: o._id, createdAt: o.createdAt.toISOString(), total: o.total }));
  const last = items.at(-1);
  return { items, next: items.length === limit && last ? `${last.createdAt}|${last.id}` : null };
}
```

import type { Db } from "mongodb";
import { z } from "zod";

const schema = z.strictObject({
  name: z.string().trim().min(1).optional(),
  price: z.number().int().positive().optional(),
  tags: z.array(z.string()).optional(),
});

  const r = schema.safeParse(body);

// P25.7-E60: הזמנה מהעגלה — עם עותק של המחיר
type Product = { _id: string; name: string; price: number; stock: number };
type Item = { productId: string; name: string; price: number; qty: number };

export function buildOrder(products: Product[], cart: { productId: string; qty: number }[], customerId: string, now: Date) {
  if (cart.length === 0) throw new Error("EMPTY_CART");
  const byId = new Map(products.map((p) => [p._id, p]));
  const items = new Map<string, Item>();
  for (const { productId, qty } of cart) {
    if (!Number.isInteger(qty) || qty <= 0) throw new RangeError(`כמות לא תקינה: ${qty}`);
    const p = byId.get(productId);
    if (!p) throw new Error(`UNKNOWN_PRODUCT ${productId}`);
    const existing = items.get(productId);
    if (existing) existing.qty += qty;
    else items.set(productId, { productId, name: p.name, price: p.price, qty });
  }
  const list = [...items.values()];
  return {
    customerId,
    status: "pending" as const,
    createdAt: now,
    items: list,
    total: list.reduce((s, i) => s + i.price * i.qty, 0),
  };
}
```

import type { Db } from "mongodb";

type P = { _id: string; reviews?: { rating: number }[]; ratingCount?: number; ratingSum?: number };

export async function recomputeRatings(db: Db): Promise<string[]> {
  const products = db.collection<P>("products");
  const fixed: string[] = [];
  for (const p of await products.find({}, { projection: { reviews: 1, ratingCount: 1, ratingSum: 1 } }).toArray()) {
    const reviews = p.reviews ?? [];
    const count = reviews.length;
    const sum = reviews.reduce((s, r) => s + r.rating, 0);
    if (p.ratingCount !== count || p.ratingSum !== sum) {
      await products.updateOne({ _id: p._id }, { $set: { ratingCount: count, ratingSum: sum } });
      fixed.push(p._id);
    }
  }
  return fixed.sort();
}

import type { Db } from "mongodb";

type P = { _id: string; reviews?: { by: string; rating: number }[] };

export async function extractReviews(db: Db) {
  const reviews = db.collection("reviews");
  await reviews.createIndex({ productId: 1, by: 1 }, { unique: true });
  const products = db.collection<P>("products");
  let moved = 0;
  for (const p of await products.find({ reviews: { $exists: true } }, { projection: { reviews: 1 } }).toArray()) {
    for (const r of p.reviews ?? []) {
      const res = await reviews.updateOne({ productId: p._id, by: r.by }, { $set: { rating: r.rating } }, { upsert: true });
      moved += res.upsertedCount;

    await products.updateOne({ _id: p._id }, { $unset: { reviews: "" } }); // רק אחרי שכל הביקורות נשמרו

}

export async function ordersWithCustomer(db: Db, status: string) {
  return (await db.collection("orders").aggregate([

    { $sort: { createdAt: -1 } },

import type { Db, MongoClient } from "mongodb";

export async function moveStock(client: MongoClient, db: Db, { from, to, productId, qty }: { from: string; to: string; productId: string; qty: number }) {
  if (!Number.isInteger(qty) || qty <= 0) throw new RangeError(`כמות לא תקינה: ${qty}`);
  const warehouses = db.collection<{ _id: string }>("warehouses");
  const field = `stock.${productId}`;
  const session = client.startSession();
  try {
    await session.withTransaction(async () => {
      const out = await warehouses.updateOne({ _id: from, [field]: { $gte: qty } }, { $inc: { [field]: -qty } }, { session });
      if (out.modifiedCount === 0) throw new Error("INSUFFICIENT_STOCK");
      const inn = await warehouses.updateOne({ _id: to }, { $inc: { [field]: qty } }, { session });
      if (inn.matchedCount === 0) throw new Error("NO_SUCH_WAREHOUSE"); // abort — גם ההורדה מתבטלת
    });
  } finally {
    await session.endSession();
  }
}

// P25.7-E94: מחירון חדש — ב־bulkWrite אחד
import type { Db } from "mongodb";

export async function applyPriceList(db: Db, list: { productId: string; price: number }[]) {
  for (const { price } of list) if (!Number.isInteger(price) || price <= 0) throw new RangeError(`מחיר לא תקין: ${price}`);
  if (list.length === 0) return { matched: 0, modified: 0, missing: [] };
  const products = db.collection<{ _id: string; price: number }>("products");
  const ids = [...new Set(list.map((l) => l.productId))];
  const found = new Set((await products.find({ _id: { $in: ids } }, { projection: { _id: 1 } }).toArray()).map((p) => p._id));
  const r = await products.bulkWrite(list.map(({ productId, price }) => ({ updateOne: { filter: { _id: productId }, update: { $set: { price } } } })));
  return { matched: r.matchedCount, modified: r.modifiedCount, missing: ids.filter((id) => !found.has(id)).sort() };
}
```

import type { Db } from "mongodb";
import { z } from "zod";

const schema = z.strictObject({ email: z.string().trim().toLowerCase(), password: z.string() });

export function loginRouter(db: Db) {
  const users = db.collection<{ _id: string; email: string; passwordHash: string }>("users");
  const router = Router();
  router.post("/login", async (req, res) => {
    const r = schema.safeParse(req.body);
    if (!r.success) {
      res.status(400).json({ error: { code: "VALIDATION_ERROR" } });
      return;
    }

    if (!ok) {
      res.status(401).json({ error: { code: "INVALID_CREDENTIALS" } });

  });

// P25.7-E99: Idempotency-Key ב־Mongo
import { MongoServerError, type Db } from "mongodb";

type Rec = { _id: { userId: string; key: string }; fingerprint: string; status: "pending" | "done"; result?: unknown; createdAt: Date };

export async function runOnce<T>(db: Db, userId: string, key: string, fingerprint: string, fn: () => Promise<T>) {
  const col = db.collection<Rec>("idempotency");
  const _id = { userId, key };
  try {
    await col.insertOne({ _id, fingerprint, status: "pending", createdAt: new Date() });
  } catch (err) {
    if (!(err instanceof MongoServerError && err.code === 11000)) throw err;
    const existing = await col.findOne({ _id });
    if (!existing) throw new Error("IN_PROGRESS"); // נמחק בדיוק עכשיו — הלקוח ינסה שוב
    if (existing.fingerprint !== fingerprint) throw new Error("KEY_REUSED");
    if (existing.status === "pending") throw new Error("IN_PROGRESS");
    return { replayed: true, result: existing.result as T };
  }
  try {
    const result = await fn();
    await col.updateOne({ _id }, { $set: { status: "done", result } });
    return { replayed: false, result };
  } catch (err) {
    await col.deleteOne({ _id, status: "pending" });
    throw err;
  }
}
```

import type { Db } from "mongodb";

const WEEK_MS = 7 * 86_400_000;

export async function ensureCartTtl(db: Db): Promise<void> {
  await db.collection("carts").createIndex({ expiresAt: 1 }, { expireAfterSeconds: 0 });
}

export async function touchCart(db: Db, customerId: string, now: Date): Promise<Date> {
  const expiresAt = new Date(now.getTime() + WEEK_MS);
  await db.collection<{ _id: string }>("carts").updateOne(
    { _id: customerId },
    { $set: { updatedAt: now, expiresAt }, $setOnInsert: { items: [] } },
    { upsert: true },
  );
  return expiresAt;
}
```
