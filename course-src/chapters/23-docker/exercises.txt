@@ P23-E25 | 5 | אתגר: wait-for 🤖 | 25 | שיעור 1
@task
<p><code>bash P23-E25.sh &lt;url&gt; &lt;שניות&gt;</code> — מחכה עד שהכתובת עונה 2xx: מנסה כל שנייה; הצלחה ← <code>✔ &lt;url&gt; עונה</code> ו־exit 0; נגמר הזמן ← <code>✖ &lt;url&gt; לא ענה תוך N שניות</code> ל־stderr ו־exit 1. בלי שני ארגומנטים ← exit 2. (שימושי ב־CI: "אחרי docker compose up — חכה שה־API באמת עונה".)</p>
@hint1
<p><code>curl -fs -o /dev/null --max-time 2 "$url"</code> — מצליח רק על 2xx/3xx, בשקט, ולא נתקע.</p>
@hint2
<p><code>$SECONDS</code> ב־bash — כמה שניות הסקריפט רץ. לולאת <code>while</code> עם <code>sleep 1</code>.</p>
@solution
```bash save=solutions/P23-E25.sh
#!/usr/bin/env bash
# P23-E25: מחכה עד שכתובת עונה 2xx, או עד שנגמר הזמן
set -uo pipefail

if [ $# -ne 2 ]; then
  echo "שימוש: wait-for.sh <url> <שניות>" >&2
  exit 2
fi

url="$1"
timeout="$2"
echo "מחכה ל־$url..."
start=$SECONDS
while [ $((SECONDS - start)) -lt "$timeout" ]; do
  # -f: קוד 4xx/5xx ← כישלון.  -s: בלי פס התקדמות.  --max-time: לא להיתקע על חיבור תקוע
  if curl -fs -o /dev/null --max-time 2 "$url"; then
    echo "✔ $url עונה"
    exit 0
  fi
  sleep 1
done

echo "✖ $url לא ענה תוך $timeout שניות" >&2
exit 1
```
<p>שים לב ל־<code>--max-time</code>: בלעדיו, שרת שמקבל חיבור ולא עונה תוקע את הסקריפט לנצח, והזמן שקבעת לא שווה כלום. ו־<code>-f</code>: בלעדיו curl מחשיב 503 כהצלחה (הוא קיבל תשובה, הרי).</p>

# P23-E26 — rotate-backups.sh <dir> <keep>
set -euo pipefail

if [ $# -ne 2 ]; then
  echo "שימוש: rotate-backups.sh <dir> <keep>" >&2
  exit 2
fi
dir=$1
keep=$2

if ! [[ $keep =~ ^[1-9][0-9]*$ ]]; then
  echo "keep חייב להיות מספר שלם חיובי: $keep" >&2
  exit 2
fi
if [ ! -d "$dir" ]; then
  echo "התיקייה לא קיימת: $dir" >&2
  exit 1
fi

# רק גיבויים בשם הנכון; התאריך בפורמט YYYY-MM-DD ממוין נכון גם כטקסט
backups=$(find "$dir" -maxdepth 1 -type f -name 'db-[0-9][0-9][0-9][0-9]-[0-9][0-9]-[0-9][0-9].sql.gz' -exec basename {} \; | sort)
count=$(printf '%s\n' "$backups" | grep -c . || true)

if [ "$count" -le "$keep" ]; then
  exit 0
fi

printf '%s\n' "$backups" | head -n $((count - keep)) | while read -r name; do
  rm -- "$dir/$name"
  echo "$name"
done

// P23-E27 — קורא קובץ .env
export function parseEnv(text: string): Record<string, string> {
  const result: Record<string, string> = {};
  const lines = text.split("\n");
  for (let i = 0; i < lines.length; i++) {
    let line = lines[i].trim();
    if (line === "" || line.startsWith("#")) continue;
    if (line.startsWith("export ")) line = line.slice("export ".length).trim();

    const eq = line.indexOf("=");
    if (eq === -1) throw new SyntaxError(`שורה ${i + 1}: חסר =`);
    const key = line.slice(0, eq).trim();
    if (!/^[A-Za-z_][A-Za-z0-9_]*$/.test(key)) throw new SyntaxError(`שורה ${i + 1}: שם לא חוקי: ${key}`);

    let value = line.slice(eq + 1).trim();
    if (value.length >= 2 && value.startsWith('"') && value.endsWith('"')) {
      value = value.slice(1, -1).replaceAll("\\n", "\n");
    } else if (value.length >= 2 && value.startsWith("'") && value.endsWith("'")) {
      value = value.slice(1, -1);
    } else {
      const hash = value.indexOf(" #");
      if (hash !== -1) value = value.slice(0, hash).trim();
    }
    result[key] = value;
  }
  return result;
}

// P23-E28 — גדלים של images
const UNITS = ["B", "kB", "MB", "GB", "TB"];

export function parseSize(text: string): number {
  const match = text.trim().match(/^(\d+(?:\.\d+)?)\s*([a-zA-Z]+)$/);
  if (!match) throw new Error(`גודל לא תקין: ${text}`);
  const value = Number(match[1]);
  switch (match[2]) {
    case "B":
      return value;
    case "kB":
    case "KB":
      return value * 1000;
    case "MB":
      return value * 1000 ** 2;
    case "GB":
      return value * 1000 ** 3;
    case "TB":
      return value * 1000 ** 4;
    default:
      throw new Error(`יחידה לא מוכרת: ${match[2]}`);
  }
}

export function formatSize(bytes: number): string {
  let unit = 0;
  let value = bytes;
  while (value >= 1000 && unit < UNITS.length - 1) {
    value /= 1000;
    unit++;
  }
  const text = unit === 0 ? String(Math.round(value)) : value.toFixed(1).replace(/\.0$/, "");
  return text + UNITS[unit];
}

export function totalSize(dockerImages: string): string {
  const [header, ...rows] = dockerImages.trim().split("\n");
  const sizeColumn = header.split(/\s{2,}/).indexOf("SIZE");
  if (sizeColumn === -1) throw new Error("אין עמודה SIZE");
  const bytes = rows
    .filter((row) => row.trim() !== "")
    .reduce((sum, row) => sum + parseSize(row.trim().split(/\s{2,}/)[sizeColumn]), 0);
  return formatSize(bytes);
}

export function toOctal(perms: string): string {
  if (!/^[-dl]([r-][w-][x-]){3}$/.test(perms)) throw new Error(`הרשאות לא תקינות: ${perms}`);
  let out = "";
  for (let i = 0; i < 3; i++) {
    const t = perms.slice(1 + i * 3, 4 + i * 3);
    out += (t[0] === "r" ? 4 : 0) + (t[1] === "w" ? 2 : 0) + (t[2] === "x" ? 1 : 0);

}

export type PortMap = { hostIp: string; hostPort: number; containerPort: number; protocol: "tcp" | "udp" };

const RE = /^(\[[^\]]+\]|[\d.]+):(\d+)->(\d+)\/(tcp|udp)$/;

export function parsePorts(text: string): PortMap[] {
  if (text.trim() === "") return [];
  const out: PortMap[] = [];
  for (const item of text.split(",").map((s) => s.trim())) {
    if (!item.includes("->")) {
      if (/^\d+\/(tcp|udp)$/.test(item)) continue;
      throw new Error(`לא מובן: ${item}`);
    }
    const m = RE.exec(item);
    if (!m) throw new Error(`לא מובן: ${item}`);
    out.push({ hostIp: m[1].replace(/^\[|\]$/g, ""), hostPort: Number(m[2]), containerPort: Number(m[3]), protocol: m[4] as "tcp" | "udp" });
  }
  return out;
}

// P23-E44: registry/repository:tag@digest
export type ImageRef = { registry: string; repository: string; tag: string | null; digest: string | null };

export function parseImageRef(ref: string): ImageRef {
  const bad = () => new Error(`שם לא תקין: ${ref}`);
  if (ref === "" || /\s/.test(ref)) throw bad();
  const [rest, digest = null] = ref.split("@");
  let name = rest;
  let tag: string | null = null;
  const colon = rest.lastIndexOf(":");
  if (colon !== -1 && !rest.slice(colon).includes("/")) {
    name = rest.slice(0, colon);
    tag = rest.slice(colon + 1);
  }
  if (tag === null && digest === null) tag = "latest";
  const parts = name.split("/");
  let registry = "docker.io";
  if (parts.length > 1 && (/[.:]/.test(parts[0]) || parts[0] === "localhost")) registry = parts.shift()!;
  let repository = parts.join("/");
  if (!repository || /[A-Z]/.test(repository) || tag === "") throw bad();
  if (registry === "docker.io" && !repository.includes("/")) repository = `library/${repository}`;
  return { registry, repository, tag, digest };
}

  image: string;
  name?: string;
  detach?: boolean;
  rm?: boolean;
  restart?: "no" | "always" | "unless-stopped";
  env?: Record<string, string>;
  ports?: string[];
  volumes?: string[];
  command?: string[];

export function runArgs(spec: RunSpec): string[] {
  const last = spec.image.split("/").at(-1)!;
  if (!last.includes(":") && !spec.image.includes("@")) throw new Error(`צריך tag מפורש: ${spec.image}`);
  if (spec.rm && spec.restart && spec.restart !== "no") throw new Error("--rm ו־--restart לא הולכים יחד");

// P23-E51: מה חשוב ב־docker inspect — בלי סודות
const SENSITIVE = /PASSWORD|SECRET|TOKEN|KEY/i;

export function summarize(inspect: any[]) {
  const c = inspect[0];
  if (!c) throw new Error("אין קונטיינר");
  const ports = Object.entries(c.HostConfig?.PortBindings ?? {})
    .flatMap(([port, binds]) => ((binds as any[]) ?? []).map((b) => `${b.HostIp || "0.0.0.0"}:${b.HostPort}->${port}`))
    .sort();
  const mounts = (c.Mounts ?? []).map((m: any) => (m.Type === "volume" ? `volume ${m.Name} → ${m.Destination}` : `bind ${m.Source} → ${m.Destination}`));
  const env: Record<string, string> = {};
  for (const s of c.Config?.Env ?? []) {
    const i = s.indexOf("=");
    const k = s.slice(0, i);
    env[k] = SENSITIVE.test(k) ? "***" : s.slice(i + 1);
  }
  return {
    name: String(c.Name).replace(/^\//, ""),
    image: c.Config.Image,
    running: c.State.Running,
    health: c.State.Health?.Status ?? null,
    ports,
    mounts,
    env,
  };
}

// P23-E63: lint ל־Dockerfile
type Step = { cmd: string; args: string; stage: number };

function parse(text: string): Step[] {
  const out: Step[] = [];
  let stage = -1;
  for (const raw of text.replace(/\\\r?\n/g, " ").split(/\r?\n/)) {
    const line = raw.trim();
    if (!line || line.startsWith("#")) continue;
    const m = /^(\w+)\s+(.*)$/.exec(line);
    if (!m) continue;
    const cmd = m[1].toUpperCase();
    if (cmd === "FROM") stage++;
    out.push({ cmd, args: m[2].trim(), stage });
  }
  return out;
}

export function lintDockerfile(text: string): string[] {
  const steps = parse(text);
  const out: string[] = [];
  const froms = steps.filter((s) => s.cmd === "FROM").map((s) => s.args.split(/\s+/)[0]);
  if (froms.some((img) => (!img.includes(":") && !img.includes("@")) || img.endsWith(":latest"))) out.push("FROM בלי tag");
  if (steps.some((s) => s.cmd === "RUN" && /npm install\b/.test(s.args))) out.push("npm install במקום npm ci");
  const lastStage = Math.max(...steps.map((s) => s.stage));
  for (let st = 0; st <= lastStage; st++) {
    const inStage = steps.filter((s) => s.stage === st);
    const copyAll = inStage.findIndex((s) => s.cmd === "COPY" && /^\.\s+\.\/?$/.test(s.args));
    const install = inStage.findIndex((s) => s.cmd === "RUN" && /npm (ci|install)\b/.test(s.args));
    if (copyAll !== -1 && install !== -1 && copyAll < install) {
      out.push("COPY . . לפני התקנת התלויות");
      break;
    }
  }
  for (const s of steps) {
    if (s.cmd !== "ENV" && s.cmd !== "ARG") continue;

function toRegExp(pattern: string): RegExp {
  let re = "";
  for (let i = 0; i < pattern.length; i++) {
    const c = pattern[i];
    if (c === "*" && pattern[i + 1] === "*") {
      if (pattern[i + 2] === "/") {
        re += "(?:.*/)?";
        i += 2;
      } else {
        re += ".*";
        i += 1;
      }
    } else if (c === "*") re += "[^/]*";
    else if (c === "?") re += "[^/]";
    else re += c.replace(/[.+^${}()|[\]\\]/g, "\\$&");
  }
  return new RegExp(`^${re}(?:/.*)?$`);
}

export function isIgnored(file: string, patterns: string[]): boolean {
  let ignored = false;
  for (const raw of patterns) {
    let p = raw.trim();
    if (!p || p.startsWith("#")) continue;
    const negate = p.startsWith("!");
    if (negate) p = p.slice(1);
    p = p.replace(/^\/+|\/+$/g, "");
    if (toRegExp(p).test(file)) ignored = !negate;
  }
  return ignored;
}

// P23-E72: interpolation בסגנון compose
const RE = /\$\$|\$\{([A-Za-z_]\w*)(?:(:?[-?])([^}]*))?\}|\$([A-Za-z_]\w*)/g;

export function interpolate(text: string, env: Record<string, string | undefined>): string {
  return text.replace(RE, (all, braced?: string, op?: string, arg?: string, bare?: string) => {
    if (all === "$$") return "$";
    const name = (braced ?? bare)!;
    const value = env[name];
    const empty = value === undefined || value === "";
    switch (op) {
      case ":-":
        return empty ? arg! : value!;
      case "-":
        return value === undefined ? arg! : value;
      case ":?":
        if (empty) throw new Error(`${name}: ${arg}`);
        return value!;
      case "?":
        if (value === undefined) throw new Error(`${name}: ${arg}`);
        return value;
      default:
        return value ?? "";
    }
  });
}
```

export type DbConfig = { user: string; password: string; host: string; port: number; database: string };

export function toDatabaseUrl(c: DbConfig): string {
  const e = encodeURIComponent;
  return `postgresql://${e(c.user)}:${e(c.password)}@${c.host}:${c.port}/${e(c.database)}`;
}

export function parseDatabaseUrl(url: string): DbConfig {
  let u: URL;
  try {
    u = new URL(url);
  } catch {
    throw new Error("DATABASE_URL לא תקין");
  }
  const database = decodeURIComponent(u.pathname.slice(1));
  if (!["postgresql:", "postgres:"].includes(u.protocol) || !database) throw new Error("DATABASE_URL לא תקין");
  return {
    user: decodeURIComponent(u.username),
    password: decodeURIComponent(u.password),
    host: u.hostname,
    port: u.port ? Number(u.port) : 5432,
    database,
  };
}

// P23-E89: גלים של עלייה לפי depends_on
export function startupWaves(doc: any): string[][] {
  const services: Record<string, any> = doc.services ?? {};
  const names = Object.keys(services);
  const deps = (s: any): string[] => (Array.isArray(s.depends_on) ? s.depends_on : Object.keys(s.depends_on ?? {}));
  for (const name of names) {
    for (const dep of deps(services[name])) {
      if (!(dep in services)) throw new Error(`${name} תלוי ב־${dep} שלא קיים`);
    }
  }
  const done = new Set<string>();
  const waves: string[][] = [];
  while (done.size < names.length) {
    const wave = names.filter((n) => !done.has(n) && deps(services[n]).every((d) => done.has(d)));
    if (wave.length === 0) throw new Error(`מעגל תלויות: ${names.filter((n) => !done.has(n)).join(", ")}`);
    wave.forEach((n) => done.add(n));
    waves.push(wave);
  }
  return waves;
}
```

type Row = { Service: string; State: string; Health: string; ExitCode: number };

export function problems(jsonLines: string, oneShot: string[]): string[] {
  const rows: Row[] = jsonLines.split("\n").filter((l) => l.trim()).map((l) => JSON.parse(l));
  const out: string[] = [];
  for (const r of rows) {
    if (oneShot.includes(r.Service)) {
      if (r.State === "exited" && r.ExitCode !== 0) out.push(`${r.Service}: נכשל (exit ${r.ExitCode})`);

// P23-E95: להשלים .env מתוך .env.example
function parse(text: string): [string, string][] {
  const out: [string, string][] = [];
  for (const raw of text.split(/\r?\n/)) {
    const line = raw.trim();
    if (!line || line.startsWith("#")) continue;
    const i = line.indexOf("=");
    if (i === -1) continue;
    out.push([line.slice(0, i).trim(), line.slice(i + 1).trim()]);
  }
  return out;
}

export function syncEnv(example: string, current: string) {
  const have = new Set(parse(current).map(([k]) => k));
  const missing = parse(example).filter(([k]) => !have.has(k));
  let text = current;
  if (text !== "" && !text.endsWith("\n")) text += "\n";
  if (missing.length) text += "# נוסף מ־.env.example\n" + missing.map(([k, v]) => `${k}=${v}`).join("\n") + "\n";
  const placeholders = parse(text).filter(([, v]) => v === "" || v.includes("change-me")).map(([k]) => k);
  return { text, added: missing.map(([k]) => k), placeholders };
}

  const out: string[] = [];
  for (const [name, s] of Object.entries(services)) {
    const exposed = (s.ports ?? []).length > 0;
    for (const [k, v] of envOf(s.environment)) {
      if (fromVar(v)) continue;
      if (SENSITIVE.test(k)) {
        if (k === "POSTGRES_PASSWORD" && !exposed) continue;
        out.push(`${name}.${k}`);
      } else if (k === "DATABASE_URL" && /:\/\/[^:/@]+:[^@]+@/.test(v)) {
        let host = "";
        try {
          host = new URL(v).hostname;
        } catch {}
        if ((services[host]?.ports ?? []).length > 0) out.push(`${name}.DATABASE_URL`);
      }
    }
  }
  return out;
}
```

// P23-E101: מדיניות שמירה — יומיים ושבועיים
const RE = /^[\w-]+-(\d{4})-(\d{2})-(\d{2})\.sql\.gz$/;
const dayNumber = (y: string, m: string, d: string) => Date.UTC(Number(y), Number(m) - 1, Number(d)) / 86_400_000;

export function toDelete(files: string[], today: string, { daily = 7, weekly = 4 }: { daily?: number; weekly?: number } = {}): string[] {
  const [ty, tm, td] = today.split("-");
  const todayN = dayNumber(ty, tm, td);
  const backups = files
    .map((f) => ({ f, m: RE.exec(f) }))
    .filter((x) => x.m)
    .map(({ f, m }) => ({ f, age: todayN - dayNumber(m![1], m![2], m![3]) }));
  const keep = new Set<string>();
  const newestInWeek = new Map<number, { f: string; age: number }>();
  for (const b of backups) {
    if (b.age < daily) {
      keep.add(b.f);
      continue;
    }
    const week = Math.floor((b.age - daily) / 7);
    if (week >= weekly) continue;
    const cur = newestInWeek.get(week);
    if (!cur || b.age < cur.age) newestInWeek.set(week, b);
  }
  for (const b of newestInWeek.values()) keep.add(b.f);
  return backups
    .filter((b) => !keep.has(b.f))
    .sort((a, b) => b.age - a.age)
    .map((b) => b.f);
}
```
<p>‏"age" (כמה ימים אחורה) הופך את כל החישוב לחשבון של מספרים שלמים, בלי חודשים באורכים שונים. ‏<code>Date.UTC</code> — כדי ששעון קיץ לא יזיז יום (פרק 22). גיבוי מהעתיד מקבל age שלילי — ונשמר.</p>
@mistakes
