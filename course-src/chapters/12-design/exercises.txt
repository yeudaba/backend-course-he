export class Counter {
  #count: number;
  #start: number;
  #max: number;

  constructor(start = 0, max = Infinity) {
    this.#start = start;
    this.#max = max;
    this.#count = start;
  }

  get value(): number {
    return this.#count;
  }

    return this;
  }

}

// P12-E07
export class Temperature {
  static readonly ABSOLUTE_ZERO = -273.15;
  readonly celsius: number;

  constructor(celsius: number) {
    if (celsius < Temperature.ABSOLUTE_ZERO) {
      throw new RangeError("מתחת לאפס המוחלט");
    }
    this.celsius = celsius;
  }

  static fromFahrenheit(f: number): Temperature {
    return new Temperature(((f - 32) * 5) / 9);
  }

  get fahrenheit(): number {
    return (this.celsius * 9) / 5 + 32;
  }

  isFreezing(): boolean {
    return this.celsius <= 0;
  }

  toString(): string {
    return `${this.celsius}°C`;
  }
}

// P12-E08
export type CartLine = { sku: string; price: number; qty: number };

export class ShoppingCart {
  #lines = new Map<string, CartLine>();

  add(sku: string, price: number, qty = 1): void {
    if (qty <= 0 || price <= 0) {
      throw new Error("כמות ומחיר חייבים להיות חיוביים");
    }
    const existing = this.#lines.get(sku);
    this.#lines.set(sku, { sku, price, qty: (existing?.qty ?? 0) + qty });
  }

  remove(sku: string): boolean {
    return this.#lines.delete(sku);
  }

  get total(): number {
    let sum = 0;
    for (const l of this.#lines.values()) {
      sum += l.price * l.qty;
    }
    return Math.round(sum * 100) / 100;
  }

  get itemCount(): number {
    let n = 0;
    for (const l of this.#lines.values()) {
      n += l.qty;
    }
    return n;
  }

  items(): CartLine[] {
    return [...this.#lines.values()].map((l) => ({ ...l }));
  }
}

const cart = new ShoppingCart();
cart.add("A", 2.5, 2);
cart.add("A", 2.5);
cart.add("B", 10);
console.log(cart.total, cart.itemCount, cart.items().length);

// P12-E09
export class AppError extends Error {
  readonly status: number;
  readonly code: string;
  constructor(message: string, status: number, code: string) {
    super(message);
    this.name = new.target.name;
    this.status = status;
    this.code = code;
  }
}

export class NotFoundError extends AppError {
  constructor(entity: string, id: string | number) {
    super(`${entity} ${id} לא נמצא`, 404, "NOT_FOUND");
  }
}

export class ValidationError extends AppError {
  readonly field: string;
  constructor(field: string, message: string) {
    super(message, 400, "VALIDATION");
    this.field = field;
  }
}

export class ConflictError extends AppError {
  constructor(message: string) {
    super(message, 409, "CONFLICT");
  }
}

export function toHttp(err: unknown): { status: number; body: { code: string; message: string; field?: string } } {
  if (err instanceof AppError) {
    return {
      status: err.status,
      body: { code: err.code, message: err.message, ...(err instanceof ValidationError ? { field: err.field } : {}) },
    };
  }
  return { status: 500, body: { code: "INTERNAL", message: "שגיאה פנימית" } };
}

// P12-E10
export interface Logger {
  info(msg: string): void;
  error(msg: string): void;
}

export class MemoryLogger implements Logger {
  #lines: string[] = [];
  info(msg: string): void {
    this.#lines.push(`INFO: ${msg}`);
  }
  error(msg: string): void {
    this.#lines.push(`ERROR: ${msg}`);
  }
  get lines(): string[] {
    return [...this.#lines];
  }
}

export class PrefixLogger implements Logger {
  #inner: Logger;
  #prefix: string;
  constructor(inner: Logger, prefix: string) {
    this.#inner = inner;
    this.#prefix = prefix;
  }
  info(msg: string): void {
    this.#inner.info(`${this.#prefix} ${msg}`);
  }
  error(msg: string): void {
    this.#inner.error(`${this.#prefix} ${msg}`);
  }
}

export function processOrders(ids: number[], logger: Logger): number {
  let ok = 0;
  for (const id of ids) {
    if (id > 0) {
      logger.info(`order ${id} ok`);
      ok++;
    } else {
      logger.error(`order ${id} invalid`);
    }
  }
  return ok;
}

const mem = new MemoryLogger();
console.log(processOrders([1, -2], new PrefixLogger(mem, "[orders]")), mem.lines);

// P12-E11
export class InMemoryRepository<T extends { id: number }> {
  #items = new Map<number, T>();

  async findById(id: number): Promise<T | null> {
    const item = this.#items.get(id);
    return item ? { ...item } : null;
  }

  async findAll(): Promise<T[]> {
    return [...this.#items.values()].sort((a, b) => a.id - b.id).map((x) => ({ ...x }));
  }

  async save(item: T): Promise<void> {
    this.#items.set(item.id, { ...item });
  }

  async delete(id: number): Promise<boolean> {
    return this.#items.delete(id);
  }

  async count(): Promise<number> {
    return this.#items.size;
  }
}

const repo = new InMemoryRepository<{ id: number; name: string }>();

export class ValidationError extends Error {

  constructor(field: string, message: string) {

export type CreateUserDto = { email: string; name: string; age?: number };

export function parseCreateUser(input: unknown): CreateUserDto {
  if (typeof input !== "object" || input === null || Array.isArray(input)) {
    throw new ValidationError("body", "חייב להיות אובייקט");
  }
  const { email, name, age } = input as Record<string, unknown>;
  if (typeof email !== "string" || !email.includes("@")) {

// P12-E13
export class Inventory {
  #stock = new Map<string, number>();
  #reserved = new Map<string, number>();

  #assertQty(qty: number): void {
    if (!Number.isInteger(qty) || qty <= 0) {
      throw new Error("כמות לא תקינה");
    }
  }

  addStock(sku: string, qty: number): void {
    this.#assertQty(qty);
    this.#stock.set(sku, (this.#stock.get(sku) ?? 0) + qty);
  }

  available(sku: string): number {
    return (this.#stock.get(sku) ?? 0) - (this.#reserved.get(sku) ?? 0);
  }

  reserve(sku: string, qty: number): void {
    this.#assertQty(qty);
    if (this.available(sku) < qty) {
      throw new Error(`אין מספיק מלאי של ${sku}`);
    }
    this.#reserved.set(sku, (this.#reserved.get(sku) ?? 0) + qty);
  }

  release(sku: string, qty: number): void {
    this.#assertQty(qty);
    const reserved = this.#reserved.get(sku) ?? 0;
    if (qty > reserved) {
      throw new Error("שחרור גדול מהשריון");
    }
    this.#reserved.set(sku, reserved - qty);
  }

  commit(sku: string, qty: number): void {
    this.release(sku, qty);
    this.#stock.set(sku, (this.#stock.get(sku) ?? 0) - qty);
  }
}

// P12-E14
export class RateLimiter {
  #limit: number;
  #windowMs: number;
  #now: () => number;
  #hits = new Map<string, number[]>();

  constructor({ limit, windowMs, now = Date.now }: { limit: number; windowMs: number; now?: () => number }) {
    this.#limit = limit;
    this.#windowMs = windowMs;
    this.#now = now;
  }

  #recent(key: string): number[] {
    const cutoff = this.#now() - this.#windowMs;
    const recent = (this.#hits.get(key) ?? []).filter((t) => t > cutoff);
    this.#hits.set(key, recent);
    return recent;
  }

  allow(key: string): boolean {
    const recent = this.#recent(key);
    if (recent.length >= this.#limit) {
      return false;
    }
    recent.push(this.#now());
    return true;
  }

  remaining(key: string): number {
    return Math.max(0, this.#limit - this.#recent(key).length);
  }
}

// P12-E17
export interface Repo<T extends { id: number }> {
  findById(id: number): Promise<T | null>;
  save(item: T): Promise<void>;
}

export class CachedRepository<T extends { id: number }> implements Repo<T> {
  #inner: Repo<T>;
  #cache = new Map<number, T | null>();
  #hits = 0;
  #misses = 0;

  constructor(inner: Repo<T>) {
    this.#inner = inner;
  }

  get hits(): number {
    return this.#hits;
  }
  get misses(): number {
    return this.#misses;
  }

  async findById(id: number): Promise<T | null> {
    if (this.#cache.has(id)) {
      this.#hits++;
      const cached = this.#cache.get(id) ?? null;
      return cached ? { ...cached } : null;
    }
    this.#misses++;
    const item = await this.#inner.findById(id);
    this.#cache.set(id, item ? { ...item } : null);
    return item;
  }

  async save(item: T): Promise<void> {
    await this.#inner.save(item);
    this.#cache.set(item.id, { ...item });
  }
}

  async save() {},
};

export class ValidationError extends Error {
  readonly field: string;
  constructor(field: string, message: string) {
    super(message);
    this.field = field;
  }
}

function parseIntStrict(text: string): number | null {
  return /^\d+$/.test(text.trim()) ? Number(text.trim()) : null;
}

export function parsePagination(query: Record<string, string | undefined>): { page: number; pageSize: number; offset: number } {
  let page = 1;
  if (query.page !== undefined) {
    const p = parseIntStrict(query.page);
    if (p === null || p < 1) {
      throw new ValidationError("page", "page חייב להיות מספר שלם חיובי");
    }
    page = p;
  }

    if (s === null || s < 1 || s > 100) {
      throw new ValidationError("pageSize", "pageSize חייב להיות בין 1 ל־100");
    }
    pageSize = s;
  }

export type Wallet = { id: string; balanceAgorot: number };
export interface WalletRepo {
  find(id: string): Promise<Wallet | null>;
  save(w: Wallet): Promise<void>;
}
export class NotFoundError extends Error {}
export class ValidationError extends Error {}
export class InsufficientFundsError extends Error {}

type TransferResult = { transferId: number; fromBalance: number; toBalance: number };

export class TransferService {
  #wallets: WalletRepo;
  #done = new Map<string, TransferResult>();
  #nextId = 1;

  constructor(wallets: WalletRepo) {
    this.#wallets = wallets;
  }

  async transfer(req: { from: string; to: string; amount: number; idempotencyKey: string }): Promise<TransferResult> {
    const previous = this.#done.get(req.idempotencyKey);
    if (previous) {
      return previous;
    }
    const agorot = Math.round(req.amount * 100);
    if (!(req.amount > 0) || Math.abs(req.amount * 100 - agorot) > 1e-9) {
      throw new ValidationError("סכום לא תקין");
    }
    if (req.from === req.to) {
      throw new ValidationError("לא ניתן להעביר לאותו ארנק");
    }
    const from = await this.#wallets.find(req.from);
    const to = await this.#wallets.find(req.to);
    if (!from || !to) {
      throw new NotFoundError("ארנק לא נמצא");
    }
    if (from.balanceAgorot < agorot) {
      throw new InsufficientFundsError("אין מספיק יתרה");
    }
    const newFrom = { ...from, balanceAgorot: from.balanceAgorot - agorot };
    const newTo = { ...to, balanceAgorot: to.balanceAgorot + agorot };
    await this.#wallets.save(newFrom);
    await this.#wallets.save(newTo);
    const result = { transferId: this.#nextId++, fromBalance: newFrom.balanceAgorot / 100, toBalance: newTo.balanceAgorot / 100 };
    this.#done.set(req.idempotencyKey, result);
    return result;
  }
}

// P12-E25: מיכל DI מינימלי
export class Container {
  #factories = new Map<string, (c: Container) => unknown>();
  #instances = new Map<string, unknown>();
  #resolving: string[] = [];

  register(name: string, factory: (c: Container) => unknown): void {
    if (this.#factories.has(name)) {
      throw new Error(`השירות ${name} כבר רשום`);
    }
    this.#factories.set(name, factory);
  }

  resolve<T = unknown>(name: string): T {
    if (this.#instances.has(name)) {
      return this.#instances.get(name) as T;
    }
    const factory = this.#factories.get(name);
    if (!factory) {
      throw new Error(`השירות ${name} לא רשום`);
    }
    if (this.#resolving.includes(name)) {
      const cycle = [...this.#resolving.slice(this.#resolving.indexOf(name)), name];
      throw new Error(`תלות מעגלית: ${cycle.join(" -> ")}`);
    }
    this.#resolving.push(name);
    try {
      const instance = factory(this);
      this.#instances.set(name, instance);
      return instance as T;
    } finally {
      this.#resolving.pop();
    }
  }
}

class Repo {}
class Service {
  repo: Repo;
  constructor(repo: Repo) {
    this.repo = repo;
  }
}
const c = new Container();
c.register("repo", () => new Repo());
c.register("service", (ct) => new Service(ct.resolve<Repo>("repo")));
const s = c.resolve<Service>("service");
console.log(s === c.resolve("service"), s.repo === c.resolve("repo"));

c.register("A", (ct) => ct.resolve("B"));
c.register("B", (ct) => ct.resolve("A"));
try {
  c.resolve("A");
} catch (err) {
  console.log((err as Error).message);
}
