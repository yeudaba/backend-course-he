```ts run save=solutions/P12-E06.ts
export class Counter {
  #count: number;
  #start: number;
  #max: number;

  constructor(start = 0, max = Infinity) {
    this.#start = start;
    this.#max = max;
    this.#count = start;
  }

  get value(): number {
    return this.#count;
  }

    return this;
  }

}
```

```ts run save=solutions/P12-E07.ts
// P12-E07
export class Temperature {
  static readonly ABSOLUTE_ZERO = -273.15;
  readonly celsius: number;

  constructor(celsius: number) {
    if (celsius < Temperature.ABSOLUTE_ZERO) {
      throw new RangeError("מתחת לאפס המוחלט");
    }
    this.celsius = celsius;
  }

  static fromFahrenheit(f: number): Temperature {
    return new Temperature(((f - 32) * 5) / 9);
  }

  get fahrenheit(): number {
    return (this.celsius * 9) / 5 + 32;
  }

  isFreezing(): boolean {
    return this.celsius <= 0;
  }

  toString(): string {
    return `${this.celsius}°C`;
  }
}
```

```ts run save=solutions/P12-E08.ts
// P12-E08
export type CartLine = { sku: string; price: number; qty: number };

export class ShoppingCart {
  #lines = new Map<string, CartLine>();

  add(sku: string, price: number, qty = 1): void {
    if (qty <= 0 || price <= 0) {
      throw new Error("כמות ומחיר חייבים להיות חיוביים");
    }
    const existing = this.#lines.get(sku);
    this.#lines.set(sku, { sku, price, qty: (existing?.qty ?? 0) + qty });
  }

  remove(sku: string): boolean {
    return this.#lines.delete(sku);
  }

  get total(): number {
    let sum = 0;
    for (const l of this.#lines.values()) {
      sum += l.price * l.qty;
    }
    return Math.round(sum * 100) / 100;
  }

  get itemCount(): number {
    let n = 0;
    for (const l of this.#lines.values()) {
      n += l.qty;
    }
    return n;
  }

  items(): CartLine[] {
    return [...this.#lines.values()].map((l) => ({ ...l }));
  }
}

const cart = new ShoppingCart();
cart.add("A", 2.5, 2);
cart.add("A", 2.5);
cart.add("B", 10);
console.log(cart.total, cart.itemCount, cart.items().length);
```

```ts run save=solutions/P12-E09.ts
// P12-E09
export class AppError extends Error {
  readonly status: number;
  readonly code: string;
  constructor(message: string, status: number, code: string) {
    super(message);
    this.name = new.target.name;
    this.status = status;
    this.code = code;
  }
}

export class NotFoundError extends AppError {
  constructor(entity: string, id: string | number) {
    super(`${entity} ${id} לא נמצא`, 404, "NOT_FOUND");
  }
}

export class ValidationError extends AppError {
  readonly field: string;
  constructor(field: string, message: string) {
    super(message, 400, "VALIDATION");
    this.field = field;
  }
}

export class ConflictError extends AppError {
  constructor(message: string) {
    super(message, 409, "CONFLICT");
  }
}

export function toHttp(err: unknown): { status: number; body: { code: string; message: string; field?: string } } {
  if (err instanceof AppError) {
    return {
      status: err.status,
      body: { code: err.code, message: err.message, ...(err instanceof ValidationError ? { field: err.field } : {}) },
    };
  }
  return { status: 500, body: { code: "INTERNAL", message: "שגיאה פנימית" } };
}
```

```ts run save=solutions/P12-E10.ts
// P12-E10
export interface Logger {
  info(msg: string): void;
  error(msg: string): void;
}

export class MemoryLogger implements Logger {
  #lines: string[] = [];
  info(msg: string): void {
    this.#lines.push(`INFO: ${msg}`);
  }
  error(msg: string): void {
    this.#lines.push(`ERROR: ${msg}`);
  }
  get lines(): string[] {
    return [...this.#lines];
  }
}

export class PrefixLogger implements Logger {
  #inner: Logger;
  #prefix: string;
  constructor(inner: Logger, prefix: string) {
    this.#inner = inner;
    this.#prefix = prefix;
  }
  info(msg: string): void {
    this.#inner.info(`${this.#prefix} ${msg}`);
  }
  error(msg: string): void {
    this.#inner.error(`${this.#prefix} ${msg}`);
  }
}

export function processOrders(ids: number[], logger: Logger): number {
  let ok = 0;
  for (const id of ids) {
    if (id > 0) {
      logger.info(`order ${id} ok`);
      ok++;
    } else {
      logger.error(`order ${id} invalid`);
    }
  }
  return ok;
}

const mem = new MemoryLogger();
console.log(processOrders([1, -2], new PrefixLogger(mem, "[orders]")), mem.lines);
```

```ts run save=solutions/P12-E11.ts
// P12-E11
export class InMemoryRepository<T extends { id: number }> {
  #items = new Map<number, T>();

  async findById(id: number): Promise<T | null> {
    const item = this.#items.get(id);
    return item ? { ...item } : null;
  }

  async findAll(): Promise<T[]> {
    return [...this.#items.values()].sort((a, b) => a.id - b.id).map((x) => ({ ...x }));
  }

  async save(item: T): Promise<void> {
    this.#items.set(item.id, { ...item });
  }

  async delete(id: number): Promise<boolean> {
    return this.#items.delete(id);
  }

  async count(): Promise<number> {
    return this.#items.size;
  }
}

const repo = new InMemoryRepository<{ id: number; name: string }>();
```

```ts run save=solutions/P12-E12.ts
export class ValidationError extends Error {

  constructor(field: string, message: string) {

export type CreateUserDto = { email: string; name: string; age?: number };

export function parseCreateUser(input: unknown): CreateUserDto {
  if (typeof input !== "object" || input === null || Array.isArray(input)) {
    throw new ValidationError("body", "חייב להיות אובייקט");
  }
  const { email, name, age } = input as Record<string, unknown>;
  if (typeof email !== "string" || !email.includes("@")) {
```

```ts run save=solutions/P12-E13.ts
// P12-E13
export class Inventory {
  #stock = new Map<string, number>();
  #reserved = new Map<string, number>();

  #assertQty(qty: number): void {
    if (!Number.isInteger(qty) || qty <= 0) {
      throw new Error("כמות לא תקינה");
    }
  }

  addStock(sku: string, qty: number): void {
    this.#assertQty(qty);
    this.#stock.set(sku, (this.#stock.get(sku) ?? 0) + qty);
  }

  available(sku: string): number {
    return (this.#stock.get(sku) ?? 0) - (this.#reserved.get(sku) ?? 0);
  }

  reserve(sku: string, qty: number): void {
    this.#assertQty(qty);
    if (this.available(sku) < qty) {
      throw new Error(`אין מספיק מלאי של ${sku}`);
    }
    this.#reserved.set(sku, (this.#reserved.get(sku) ?? 0) + qty);
  }

  release(sku: string, qty: number): void {
    this.#assertQty(qty);
    const reserved = this.#reserved.get(sku) ?? 0;
    if (qty > reserved) {
      throw new Error("שחרור גדול מהשריון");
    }
    this.#reserved.set(sku, reserved - qty);
  }

  commit(sku: string, qty: number): void {
    this.release(sku, qty);
    this.#stock.set(sku, (this.#stock.get(sku) ?? 0) - qty);
  }
}
```

```ts run save=solutions/P12-E14.ts
// P12-E14
export class RateLimiter {
  #limit: number;
  #windowMs: number;
  #now: () => number;
  #hits = new Map<string, number[]>();

  constructor({ limit, windowMs, now = Date.now }: { limit: number; windowMs: number; now?: () => number }) {
    this.#limit = limit;
    this.#windowMs = windowMs;
    this.#now = now;
  }

  #recent(key: string): number[] {
    const cutoff = this.#now() - this.#windowMs;
    const recent = (this.#hits.get(key) ?? []).filter((t) => t > cutoff);
    this.#hits.set(key, recent);
    return recent;
  }

  allow(key: string): boolean {
    const recent = this.#recent(key);
    if (recent.length >= this.#limit) {
      return false;
    }
    recent.push(this.#now());
    return true;
  }

  remaining(key: string): number {
    return Math.max(0, this.#limit - this.#recent(key).length);
  }
}
```

```ts run save=solutions/P12-E17.ts
// P12-E17
export interface Repo<T extends { id: number }> {
  findById(id: number): Promise<T | null>;
  save(item: T): Promise<void>;
}

export class CachedRepository<T extends { id: number }> implements Repo<T> {
  #inner: Repo<T>;
  #cache = new Map<number, T | null>();
  #hits = 0;
  #misses = 0;

  constructor(inner: Repo<T>) {
    this.#inner = inner;
  }

  get hits(): number {
    return this.#hits;
  }
  get misses(): number {
    return this.#misses;
  }

  async findById(id: number): Promise<T | null> {
    if (this.#cache.has(id)) {
      this.#hits++;
      const cached = this.#cache.get(id) ?? null;
      return cached ? { ...cached } : null;
    }
    this.#misses++;
    const item = await this.#inner.findById(id);
    this.#cache.set(id, item ? { ...item } : null);
    return item;
  }

  async save(item: T): Promise<void> {
    await this.#inner.save(item);
    this.#cache.set(item.id, { ...item });
  }
}

  async save() {},
};
```

```ts run save=solutions/P12-E19.ts
export class ValidationError extends Error {
  readonly field: string;
  constructor(field: string, message: string) {
    super(message);
    this.field = field;
  }
}

function parseIntStrict(text: string): number | null {
  return /^\d+$/.test(text.trim()) ? Number(text.trim()) : null;
}

export function parsePagination(query: Record<string, string | undefined>): { page: number; pageSize: number; offset: number } {
  let page = 1;
  if (query.page !== undefined) {
    const p = parseIntStrict(query.page);
    if (p === null || p < 1) {
      throw new ValidationError("page", "page חייב להיות מספר שלם חיובי");
    }
    page = p;
  }
  let pageSize = 20;
  if (query.pageSize !== undefined) {
    const s = parseIntStrict(query.pageSize);
    if (s === null || s < 1 || s > 100) {
      throw new ValidationError("pageSize", "pageSize חייב להיות בין 1 ל־100");
    }
    pageSize = s;
  }
  return { page, pageSize, offset: (page - 1) * pageSize };
}
```

```ts run save=solutions/P12-E24.ts
// P12-E24
export type Wallet = { id: string; balanceAgorot: number };
export interface WalletRepo {
  find(id: string): Promise<Wallet | null>;
  save(w: Wallet): Promise<void>;
}
export class NotFoundError extends Error {}
export class ValidationError extends Error {}
export class InsufficientFundsError extends Error {}

type TransferResult = { transferId: number; fromBalance: number; toBalance: number };

export class TransferService {
  #wallets: WalletRepo;
  #done = new Map<string, TransferResult>();
  #nextId = 1;

  constructor(wallets: WalletRepo) {
    this.#wallets = wallets;
  }

  async transfer(req: { from: string; to: string; amount: number; idempotencyKey: string }): Promise<TransferResult> {
    const previous = this.#done.get(req.idempotencyKey);
    if (previous) {
      return previous;
    }
    const agorot = Math.round(req.amount * 100);
    if (!(req.amount > 0) || Math.abs(req.amount * 100 - agorot) > 1e-9) {
      throw new ValidationError("סכום לא תקין");
    }
    if (req.from === req.to) {
      throw new ValidationError("לא ניתן להעביר לאותו ארנק");
    }
    const from = await this.#wallets.find(req.from);
    const to = await this.#wallets.find(req.to);
    if (!from || !to) {
      throw new NotFoundError("ארנק לא נמצא");
    }
    if (from.balanceAgorot < agorot) {
      throw new InsufficientFundsError("אין מספיק יתרה");
    }
    const newFrom = { ...from, balanceAgorot: from.balanceAgorot - agorot };
    const newTo = { ...to, balanceAgorot: to.balanceAgorot + agorot };
    await this.#wallets.save(newFrom);
    await this.#wallets.save(newTo);
    const result = { transferId: this.#nextId++, fromBalance: newFrom.balanceAgorot / 100, toBalance: newTo.balanceAgorot / 100 };
    this.#done.set(req.idempotencyKey, result);
    return result;
  }
}

const store = new Map<string, Wallet>([["a", { id: "a", balanceAgorot: 10000 }], ["b", { id: "b", balanceAgorot: 0 }]]);
const repo: WalletRepo = { find: async (id) => (store.has(id) ? { ...store.get(id)! } : null), save: async (w) => void store.set(w.id, { ...w }) };
```

```ts run save=solutions/P12-E25.ts
export class Container {
  #factories = new Map<string, (c: Container) => unknown>();
  #instances = new Map<string, unknown>();
  #resolving: string[] = [];

  register(name: string, factory: (c: Container) => unknown): void {
    if (this.#factories.has(name)) {
      throw new Error(`השירות ${name} כבר רשום`);
    }
    this.#factories.set(name, factory);
  }

  resolve<T = unknown>(name: string): T {
    if (this.#instances.has(name)) {
      return this.#instances.get(name) as T;
    }
    const factory = this.#factories.get(name);
    if (!factory) {
      throw new Error(`השירות ${name} לא רשום`);
    }
    if (this.#resolving.includes(name)) {
      const cycle = [...this.#resolving.slice(this.#resolving.indexOf(name)), name];
      throw new Error(`תלות מעגלית: ${cycle.join(" -> ")}`);
    }
    this.#resolving.push(name);
    try {
      const instance = factory(this);
      this.#instances.set(name, instance);
      return instance as T;
    } finally {
      this.#resolving.pop();
    }
  }
}

class Repo {}
class Service {
  repo: Repo;
  constructor(repo: Repo) {
    this.repo = repo;
  }
}
const c = new Container();
c.register("repo", () => new Repo());
c.register("service", (ct) => new Service(ct.resolve<Repo>("repo")));
const s = c.resolve<Service>("service");
console.log(s === c.resolve("service"), s.repo === c.resolve("repo"));

c.register("A", (ct) => ct.resolve("B"));
c.register("B", (ct) => ct.resolve("A"));
try {
  c.resolve("A");
} catch (err) {
  console.log((err as Error).message);
}
```

```ts run save=solutions/P12-E26.ts
// P12-E26: מחסן כמחלקה
export class Inventory {
  #items = new Map<string, number>();

  #checkQty(qty: number): void {
    if (!Number.isInteger(qty) || qty <= 0) {
      throw new RangeError("כמות חייבת להיות מספר שלם חיובי");
    }
  }

  add(sku: string, qty: number): void {
    this.#checkQty(qty);
    this.#items.set(sku, this.count(sku) + qty);
  }

  remove(sku: string, qty: number): void {
    this.#checkQty(qty);
    const have = this.count(sku);
    if (have < qty) {
      throw new Error(`אין מספיק ${sku}: יש ${have}, ביקשו ${qty}`);
    }
    if (have === qty) {
      this.#items.delete(sku);
    } else {
      this.#items.set(sku, have - qty);
    }
  }

  count(sku: string): number {
    return this.#items.get(sku) ?? 0;
  }

  totalItems(): number {
    return [...this.#items.values()].reduce((sum, n) => sum + n, 0);
  }

  skus(): string[] {
    return [...this.#items.keys()].sort();
  }
}
```

```ts run save=solutions/P12-E27.ts
// P12-E27: Money — אובייקט ערך
function withCommas(n: number): string {
  const digits = String(n);
  let out = "";
  for (let i = digits.length; i > 0; i -= 3) {
    const part = digits.slice(Math.max(0, i - 3), i);
    out = out === "" ? part : `${part},${out}`;
  }
  return out;
}

export class Money {
  readonly #agorot: number;

  private constructor(agorot: number) {
    this.#agorot = agorot;
  }

  static fromAgorot(agorot: number): Money {
    if (!Number.isInteger(agorot)) {
      throw new RangeError("סכום באגורות חייב להיות מספר שלם");
    }
    return new Money(agorot);
  }

  static fromShekels(shekels: number): Money {
    return Money.fromAgorot(Math.round(shekels * 100));
  }

  get agorot(): number {
    return this.#agorot;
  }

  plus(other: Money): Money {
    return new Money(this.#agorot + other.#agorot);
  }

  times(factor: number): Money {
    return Money.fromAgorot(Math.round(this.#agorot * factor));
  }

  equals(other: Money): boolean {
    return this.#agorot === other.#agorot;
  }

  toString(): string {
    const abs = Math.abs(this.#agorot);
    const shekels = withCommas(Math.floor(abs / 100));
    const cents = String(abs % 100).padStart(2, "0");
    return `${this.#agorot < 0 ? "-" : ""}₪${shekels}.${cents}`;
  }
}
```

```ts run save=solutions/P12-E28.ts
// P12-E28: דוח הלקוחות הגדולים
export type Order = { id: number; userId: number; total: number; status: "paid" | "cancelled" };
export type User = { id: number; name: string };

export interface OrderRepository {
  findAll(): Promise<Order[]>;
}
export interface UserRepository {
  findByIds(ids: number[]): Promise<User[]>;
}

export class ReportService {
  #orders: OrderRepository;
  #users: UserRepository;

  constructor(orders: OrderRepository, users: UserRepository) {
    this.#orders = orders;
    this.#users = users;
  }

  async topCustomers(n: number): Promise<{ name: string; total: number }[]> {
    const orders = await this.#orders.findAll();

    const totals = new Map<number, number>();
    for (const o of orders.filter((o) => o.status === "paid")) {
      totals.set(o.userId, (totals.get(o.userId) ?? 0) + o.total);
    }

    const top = [...totals].sort(([idA, a], [idB, b]) => b - a || idA - idB).slice(0, n);

    const users = await this.#users.findByIds(top.map(([id]) => id));
    const names = new Map(users.map((u) => [u.id, u.name]));

    return top.map(([id, total]) => ({ name: names.get(id) ?? "לא ידוע", total }));
  }
}

const orders: OrderRepository = {
  findAll: async () => [
    { id: 1, userId: 7, total: 100, status: "paid" },
    { id: 2, userId: 8, total: 300, status: "paid" },
    { id: 3, userId: 7, total: 250, status: "paid" },
    { id: 4, userId: 9, total: 999, status: "cancelled" },
  ],
};
const users: UserRepository = {
  findByIds: async (ids) => [{ id: 7, name: "דנה" }, { id: 8, name: "עומר" }].filter((u) => ids.includes(u.id)),
};

console.log(await new ReportService(orders, users).topCustomers(5));
```

```ts run save=solutions/P12-E34.ts
// P12-E34: טווח מספרים
export class Range {
  readonly start: number;
  readonly end: number;

  constructor(start: number, end: number) {
    if (end < start) throw new RangeError("end קטן מ־start");
    this.start = start;
    this.end = end;
  }

  static parse(text: string): Range {
    const parts = text.split("-").map((p) => p.trim());
    if (parts.length !== 2 || parts[0] === "" || parts[1] === "") throw new RangeError("טווח לא תקין");
    const start = Number(parts[0]);
    const end = Number(parts[1]);
    if (Number.isNaN(start) || Number.isNaN(end)) throw new RangeError("טווח לא תקין");
    return new Range(start, end);
  }

  get length(): number {
    return this.end - this.start;
  }

  includes(n: number): boolean {
    return n >= this.start && n <= this.end;
  }

  overlaps(other: Range): boolean {
    return this.start <= other.end && other.start <= this.end;
  }

  toString(): string {
    return `[${this.start}, ${this.end}]`;
  }
}

const r = Range.parse(" 3 - 7 ");
console.log(String(r), r.length, r.includes(7), r.overlaps(new Range(7, 10)), r.overlaps(new Range(8, 9)));
```


```ts run save=solutions/P12-E35.ts
// P12-E35: רשימת משימות
export type Todo = { id: number; title: string; done: boolean };

export class TodoList {
  #items: Todo[] = [];
  #nextId = 1;

  add(title: string): number {
    const t = title.trim();
    if (t === "") throw new Error("כותרת ריקה");
    const id = this.#nextId++;
    this.#items.push({ id, title: t, done: false });
    return id;
  }

  complete(id: number): void {
    const item = this.#items.find((t) => t.id === id);
    if (!item) throw new Error(`משימה ${id} לא קיימת`);
    item.done = true;
  }

  remove(id: number): boolean {
    const before = this.#items.length;
    this.#items = this.#items.filter((t) => t.id !== id);
    return this.#items.length < before;
  }

  get pending(): string[] {
    return this.#items.filter((t) => !t.done).map((t) => t.title);
  }

  all(): Todo[] {
    return this.#items.map((t) => ({ ...t }));
  }
}

const list = new TodoList();
const milk = list.add(" לקנות חלב ");
list.add("לשלם חשבון");
list.complete(milk);
console.log(list.pending, list.remove(milk), list.add("לצלצל"));
console.log(list.all());
```

```ts run save=solutions/P12-E39.ts
// P12-E39: יומן ציונים
export class Gradebook {
  #grades = new Map<string, number[]>();

  add(student: string, grade: number): void {
    const name = student.trim();
    if (name === "") throw new Error("שם ריק");
    if (!Number.isInteger(grade) || grade < 0 || grade > 100) throw new RangeError(`ציון לא תקין: ${grade}`);
    const list = this.#grades.get(name) ?? [];
    list.push(grade);
    this.#grades.set(name, list);
  }

  average(student: string): number {
    const list = this.#grades.get(student.trim());
    if (!list || list.length === 0) throw new Error(`אין ציונים ל־${student}`);
    const avg = list.reduce((s, g) => s + g, 0) / list.length;
    return Math.round(avg * 10) / 10;
  }

  top(n: number): { student: string; average: number }[] {
    return [...this.#grades.keys()]
      .map((student) => ({ student, average: this.average(student) }))
      .sort((a, b) => b.average - a.average || a.student.localeCompare(b.student))
      .slice(0, n);
  }

  get students(): number {
    return this.#grades.size;
  }
}

book.add("דנה", 90);
book.add("דנה", 85);
book.add("עומר", 95);
book.add(" נועה ", 70);
console.log(book.average("דנה"), book.students);
console.log(book.top(2));
```

```ts run save=solutions/P12-E40.ts
// P12-E40: תור משימות אסינכרוני
export class TaskQueue {
  #tail: Promise<unknown> = Promise.resolve();
  #pending = 0;

  add<T>(task: () => Promise<T>): Promise<T> {
    this.#pending++;
    const run = this.#tail.then(() => task()).finally(() => {
      this.#pending--;
    });
    this.#tail = run.catch(() => {});
    return run;
  }

  get pending(): number {
    return this.#pending;
  }
}

const sleep = (ms: number) => new Promise((r) => setTimeout(r, ms));
async function demo() {
  const q = new TaskQueue();
  const log: string[] = [];
  const a = q.add(async () => { log.push("א התחיל"); await sleep(30); log.push("א הסתיים"); return "א"; });
  const bad = q.add(async () => { throw new Error("ב נכשל"); });
  const c = q.add(async () => { log.push("ג התחיל"); return "ג"; });
  console.log(q.pending);
  console.log(await a, await bad.catch((e: Error) => e.message), await c, q.pending);
  console.log(log);
}
```

```ts run save=solutions/P12-E47.ts
  readonly label: string;
  constructor(label: string) {
    this.label = label;
  }
  area(): number {
    throw new Error("area לא מומשה");
  }
  describe(): string {

  }

  constructor(r: number) {
    super("עיגול");
    this.r = r;

  area(): number {
    return Math.PI * this.r ** 2;

  readonly w: number;
  readonly h: number;
  constructor(w: number, h: number) {
    super(w === h ? "ריבוע" : "מלבן");
    this.w = w;
    this.h = h;
  }
  area(): number {
    return this.w * this.h;
  }
}

export function totalArea(shapes: Shape[]): number {
  return shapes.reduce((sum, s) => sum + s.area(), 0);
```

```ts run save=solutions/P12-E48.ts
// P12-E48: שולח התראות — הרכבה ושגיאה עם cause
export type Sender = { send(phone: string, text: string): void };

export class NotifyError extends Error {
  readonly sent: number;
  constructor(message: string, sent: number, cause: unknown) {
    super(message, { cause });
    this.name = new.target.name;
    this.sent = sent;
  }
}

export class Notifier {
  #sender: Sender;

  constructor(sender: Sender) {
    this.#sender = sender;
  }

  notifyAll(users: { name: string; phone?: string }[], text: string): number {
    let sent = 0;
    for (const u of users) {
      if (!u.phone) continue;
      try {
        this.#sender.send(u.phone, text);
      } catch (err) {
        throw new NotifyError(`השליחה ל־${u.name} נכשלה`, sent, err);
      }
      sent++;
    }
    return sent;
  }
}

const log: string[] = [];
const fake: Sender = {
  send(phone, text) {
    if (phone === "000") throw new Error("מספר חסום");
    log.push(`${phone}: ${text}`);
  },
};
const n = new Notifier(fake);
console.log(n.notifyAll([{ name: "דנה", phone: "050" }, { name: "עומר" }], "שלום"), log);
try {
  n.notifyAll([{ name: "נועה", phone: "052" }, { name: "רון", phone: "000" }], "היי");
} catch (err) {
  if (err instanceof NotifyError) console.log(String(err), err.sent, String(err.cause));
}
```

```ts run save=solutions/P12-E52.ts
// P12-E52: כל השגיאות בבת אחת
export class LineError extends Error {
  readonly line: number;
  constructor(line: number, reason: string) {
    super(`שורה ${line}: ${reason}`);
    this.name = new.target.name;
    this.line = line;
  }
}

export class ImportError extends Error {
  readonly errors: LineError[];
  constructor(errors: LineError[]) {
    super(`${errors.length} שורות לא תקינות`);
    this.name = new.target.name;
    this.errors = errors;
  }
}

export function parseExpenses(text: string): { title: string; amount: number }[] {
  const rows: { title: string; amount: number }[] = [];
  const errors: LineError[] = [];
  const lines = text.split("\n");
  for (let i = 0; i < lines.length; i++) {
    if (lines[i].trim() === "") continue;
    const parts = lines[i].split(",").map((p) => p.trim());
    if (parts.length !== 2) {
      errors.push(new LineError(i + 1, "צריך שני שדות"));
      continue;
    }
    const [title, rawAmount] = parts;
    const amount = Number(rawAmount);
    if (title === "") {
      errors.push(new LineError(i + 1, "כותרת ריקה"));
    } else if (rawAmount === "" || !Number.isFinite(amount) || amount <= 0) {
      errors.push(new LineError(i + 1, "סכום לא תקין"));
    } else {
      rows.push({ title, amount });
    }
  }
  if (errors.length > 0) throw new ImportError(errors);
  return rows;
}

console.log(parseExpenses("קפה, 12\n\nספר,59.9\n"));
try {

} catch (err) {
  if (err instanceof ImportError) console.log(err.message, err.errors.map((e) => e.message));
}
```

```ts run save=solutions/P12-E53.ts
export class TemporaryError extends Error {
  constructor(message: string) {
    super(message);
    this.name = new.target.name;
  }

export async function retryTemporary<T>(fn: (attempt: number) => Promise<T>, attempts: number): Promise<T> {
  for (let i = 1; ; i++) {
    try {
      return await fn(i);
    } catch (err) {
      if (!(err instanceof TemporaryError) || i >= attempts) throw err;
    }
  }
}
```

```ts run save=solutions/P12-E54.ts
// P12-E54: בטל ובצע שוב
export class NothingToUndoError extends Error {
  constructor() {
    super("אין מה לבטל");
    this.name = new.target.name;
  }
}

export class NothingToRedoError extends Error {
  constructor() {
    super("אין מה לבצע שוב");
    this.name = new.target.name;
  }
}

export class History {
  #done: string[] = [];
  #undone: string[] = [];

  push(action: string): void {
    this.#done.push(action);
    this.#undone = [];
  }

  undo(): string {
    const action = this.#done.pop();
    if (action === undefined) throw new NothingToUndoError();
    this.#undone.push(action);
    return action;
  }

  redo(): string {
    const action = this.#undone.pop();
    if (action === undefined) throw new NothingToRedoError();
    this.#done.push(action);
    return action;
  }

  get done(): string[] {
    return [...this.#done];
  }
}

const h = new History();
h.push("כתוב א");
h.push("כתוב ב");
console.log(h.undo(), h.done);
console.log(h.redo(), h.done);
h.undo();
h.push("כתוב ג");
try {
  h.redo();
} catch (err) {
  console.log(String(err), h.done);
}
```

```ts run save=solutions/P12-E61.ts
export type User = { id: number; email: string; name: string };

export interface UserRepository {
  findById(id: number): Promise<User | null>;
  findByEmail(email: string): Promise<User | null>;
  save(user: User): Promise<void>;
  delete(id: number): Promise<boolean>;
}

export class InMemoryUserRepository implements UserRepository {
  #items = new Map<number, User>();

  async findById(id: number): Promise<User | null> {
    const u = this.#items.get(id);
    return u ? { ...u } : null;
  }

  async findByEmail(email: string): Promise<User | null> {
    const key = email.trim().toLowerCase();
    for (const u of this.#items.values()) {
      if (u.email.trim().toLowerCase() === key) return { ...u };
    }
    return null;
  }

  async save(user: User): Promise<void> {
    this.#items.set(user.id, { ...user });
  }

  async delete(id: number): Promise<boolean> {
    return this.#items.delete(id);
  }
}
```

```ts run save=solutions/P12-E62.ts
// P12-E62: שירות קופונים
export type Coupon = { code: string; percent: number; expiresAt: number; usesLeft: number; minTotal: number };

export interface CouponRepository {
  find(code: string): Promise<Coupon | null>;
  save(c: Coupon): Promise<void>;
}

export class CouponError extends Error {
  constructor(message: string) {
    super(message);
    this.name = new.target.name;
  }
}

export class CouponService {
  #coupons: CouponRepository;
  #now: () => number;

  constructor(coupons: CouponRepository, now: () => number) {
    this.#coupons = coupons;
    this.#now = now;
  }

  async apply(code: string, total: number): Promise<number> {
    const c = await this.#coupons.find(code.trim().toUpperCase());
    if (c === null) throw new CouponError("קופון לא קיים");
    if (this.#now() >= c.expiresAt) throw new CouponError("הקופון פג תוקף");
    if (c.usesLeft <= 0) throw new CouponError("הקופון נוצל");
    if (total < c.minTotal) throw new CouponError(`מינימום הזמנה: ${c.minTotal}`);
    await this.#coupons.save({ ...c, usesLeft: c.usesLeft - 1 });
    return Math.round(total * (1 - c.percent / 100) * 100) / 100;
  }
}

const store = new Map<string, Coupon>([["SALE10", { code: "SALE10", percent: 10, expiresAt: 5000, usesLeft: 1, minTotal: 50 }]]);
const repo: CouponRepository = {
  async find(code) {
    const c = store.get(code);
    return c ? { ...c } : null;
  },
  async save(c) {
    store.set(c.code, { ...c });
  },
};
let t = 1000;
const service = new CouponService(repo, () => t);
async function demo() {
  console.log(await service.apply(" sale10 ", 199.9));
  for (const [code, total] of [["SALE10", 100], ["NOPE", 100]] as const) {
    try {
      await service.apply(code, total);
    } catch (err) {
      console.log(String(err));
    }
  }
}
demo();
```

```ts run save=solutions/P12-E66.ts
export type Order = { id: number; total: number; status: "paid" | "refunded" | "pending"; createdAt: string };

export interface OrderRepository {
  findAll(): Promise<Order[]>;
}

export class StatsService {
  #orders: OrderRepository;

  constructor(orders: OrderRepository) {
    this.#orders = orders;
  }

  async revenueByMonth(year: number): Promise<Map<string, number>> {
    const months = new Map<string, number>();
    for (let m = 1; m <= 12; m++) months.set(`${year}-${String(m).padStart(2, "0")}`, 0);
    for (const o of await this.#orders.findAll()) {
      const key = o.createdAt.slice(0, 7);
      const current = months.get(key);
      if (current === undefined || o.status === "pending") continue;
      const sign = o.status === "paid" ? 1 : -1;
      months.set(key, Math.round((current + sign * o.total) * 100) / 100);
    }
    return months;
  }
}

const repo: OrderRepository = {
  async findAll() {
    return [
      { id: 1, total: 100.1, status: "paid", createdAt: "2026-01-05T10:00:00Z" },

      { id: 5, total: 70, status: "paid", createdAt: "2025-12-31T10:00:00Z" },
    ];
  },
};
```

```ts run save=solutions/P12-E67.ts
// P12-E67: חיפוש מוצרים עם עמודים
export type Product = { id: number; name: string; price: number };

export interface ProductRepository {
  findAll(): Promise<Product[]>;
}

type Query = { text?: string; minPrice?: number; maxPrice?: number; page?: number; pageSize?: number };

export class SearchService {
  #repo: ProductRepository;

  constructor(repo: ProductRepository) {
    this.#repo = repo;
  }

  async search(q: Query): Promise<{ items: Product[]; total: number }> {
    const { page = 1, pageSize = 10 } = q;
    const text = (q.text ?? "").trim().toLowerCase();
    const list = (await this.#repo.findAll())
      .filter((p) => text === "" || p.name.toLowerCase().includes(text))
      .filter((p) => q.minPrice === undefined || p.price >= q.minPrice)
      .filter((p) => q.maxPrice === undefined || p.price <= q.maxPrice)
      .sort((a, b) => a.price - b.price || a.name.localeCompare(b.name));
    return { items: list.slice((page - 1) * pageSize, page * pageSize), total: list.length };
  }
}

const repo: ProductRepository = {
  async findAll() {
    return [
      { id: 1, name: "Pen", price: 5 },
      { id: 2, name: "Pencil", price: 3 },
      { id: 3, name: "Notebook", price: 12 },
      { id: 4, name: "Fountain pen", price: 120 },
    ];
  },
};
const service = new SearchService(repo);
service.search({ text: " PEN", maxPrice: 100 }).then((r) => console.log(r));
service.search({ page: 2, pageSize: 3 }).then((r) => console.log(r.items.map((p) => p.name), r.total));
```

```ts run save=solutions/P12-E74.ts
// P12-E74: עדכון חלקי (PATCH)
export class ValidationError extends Error {
  readonly field: string;
  constructor(field: string, message: string) {
    super(message);
    this.name = new.target.name;
    this.field = field;
  }
}

export type ProductPatch = { name?: string; price?: number; tags?: string[] };

const ALLOWED = ["name", "price", "tags"];

export function parseProductPatch(input: unknown): ProductPatch {
  if (typeof input !== "object" || input === null || Array.isArray(input)) {
    throw new ValidationError("body", "גוף הבקשה חייב להיות אובייקט");
  }
  const keys = Object.keys(input);
  for (const key of keys) {
    if (!ALLOWED.includes(key)) throw new ValidationError(key, `שדה לא מוכר: ${key}`);
  }
  if (keys.length === 0) throw new ValidationError("body", "אין מה לעדכן");

  const patch: ProductPatch = {};
  if ("name" in input) {
    if (typeof input.name !== "string" || input.name.trim() === "") throw new ValidationError("name", "name לא תקין");
    patch.name = input.name.trim();
  }
  if ("price" in input) {
    const p = input.price;
    if (typeof p !== "number" || !Number.isFinite(p) || p <= 0 || Math.abs(Math.round(p * 100) - p * 100) > 1e-9) {
      throw new ValidationError("price", "price לא תקין");
    }
    patch.price = p;
  }
  if ("tags" in input) {
    const t = input.tags;
    if (!Array.isArray(t) || !t.every((x) => typeof x === "string" && x.trim() !== "")) {
      throw new ValidationError("tags", "tags לא תקין");
    }
    const tags = t.map((x: string) => x.trim().toLowerCase());
    if (new Set(tags).size !== tags.length) throw new ValidationError("tags", "tags לא תקין");
    patch.tags = tags;
  }
  return patch;
}

console.log(parseProductPatch({ name: "  עט כחול ", tags: ["New", " sale"] }));

  try {
    parseProductPatch(bad);
  } catch (err) {
    if (err instanceof ValidationError) console.log(err.field, "←", err.message);
  }
}
```

```ts run save=solutions/P12-E75.ts
// P12-E75: סינון מוצרים מה־URL
export class ValidationError extends Error {
  readonly field: string;
  constructor(field: string, message: string) {
    super(message);
    this.name = new.target.name;
    this.field = field;
  }
}

type Filters = { minPrice?: number; maxPrice?: number; tags: string[]; inStock: boolean };

export function parseFilters(query: Record<string, string | undefined>): Filters {
  const price = (field: string): number | undefined => {
    const raw = (query[field] ?? "").trim();
    if (raw === "") return undefined;
    if (!/^\d+(\.\d+)?$/.test(raw)) throw new ValidationError(field, `${field} לא תקין`);
    return Number(raw);
  };
  const min = price("minPrice");
  const max = price("maxPrice");
  if (min !== undefined && max !== undefined && min > max) {
    throw new ValidationError("minPrice", "minPrice גדול מ־maxPrice");
  }

  const tags: string[] = [];
  for (const t of (query.tags ?? "").split(",")) {
    const tag = t.trim().toLowerCase();
    if (tag !== "" && !tags.includes(tag)) tags.push(tag);
  }

  const rawStock = query.inStock;
  if (rawStock !== undefined && rawStock !== "true" && rawStock !== "false") {
    throw new ValidationError("inStock", "inStock לא תקין");
  }

  const result: Filters = { tags, inStock: rawStock === "true" };
  if (min !== undefined) result.minPrice = min;
  if (max !== undefined) result.maxPrice = max;
  return result;
}

} catch (err) {
  console.log(String(err));
}
```

```ts run save=solutions/P12-E76.ts
export class ValidationError extends Error {
  readonly field: string;
  constructor(field: string, message: string) {
    super(message);
    this.name = new.target.name;
    this.field = field;
  }
}

const isInt = (v: unknown, max: number): v is number => typeof v === "number" && Number.isInteger(v) && v > 0 && v <= max;

export function parseItems(input: unknown): { productId: number; qty: number }[] {
  if (!Array.isArray(input) || input.length < 1 || input.length > 50) {
    throw new ValidationError("items", "items חייב להיות מערך של 1–50 פריטים");
  }
  const totals = new Map<number, number>();
  input.forEach((item: unknown, i: number) => {
    const path = `items[${i}]`;
    if (typeof item !== "object" || item === null || Array.isArray(item)) throw new ValidationError(path, `${path} לא תקין`);
    const productId = "productId" in item ? item.productId : undefined;
    const qty = "qty" in item ? item.qty : undefined;
    if (!isInt(productId, Number.MAX_SAFE_INTEGER)) throw new ValidationError(`${path}.productId`, `${path}.productId לא תקין`);
    if (!isInt(qty, 100)) throw new ValidationError(`${path}.qty`, `${path}.qty לא תקין`);

    if (sum > 100) throw new ValidationError("items", `יותר מ־100 יחידות של מוצר ${productId}`);
    totals.set(productId, sum);
  });
  return [...totals].map(([productId, qty]) => ({ productId, qty }));
}
```

```ts run save=solutions/P12-E80.ts
// P12-E80: הרשמה מקצה לקצה
export class ValidationError extends Error {
  readonly field: string;
  constructor(field: string, message: string) {
    super(message);
    this.name = new.target.name;
    this.field = field;
  }
}

export class ConflictError extends Error {
  constructor(message: string) {
    super(message);
    this.name = new.target.name;
  }
}

type SignupDto = { email: string; name: string; password: string };
const EMAIL = /^[^@\s]+@[^@\s]+\.[^@\s]+$/;

export function parseSignup(input: unknown): SignupDto {
  if (typeof input !== "object" || input === null || Array.isArray(input)) throw new ValidationError("body", "body לא תקין");
  const email = "email" in input && typeof input.email === "string" ? input.email.trim().toLowerCase() : "";
  if (!EMAIL.test(email)) throw new ValidationError("email", "email לא תקין");
  const name = "name" in input && typeof input.name === "string" ? input.name.trim() : "";
  if (name === "") throw new ValidationError("name", "name לא תקין");
  const password = "password" in input && typeof input.password === "string" ? input.password : "";
  if (password.length < 8) throw new ValidationError("password", "password לא תקין");
  return { email, name, password };
}

export type User = { id: number; email: string; name: string; passwordHash: string };

export interface UserRepository {
  findByEmail(email: string): Promise<User | null>;
  insert(u: Omit<User, "id">): Promise<User>;
}

export class SignupService {
  #users: UserRepository;
  #hash: (password: string) => string;

  constructor(users: UserRepository, hash: (password: string) => string) {
    this.#users = users;
    this.#hash = hash;
  }

  async signup(input: unknown): Promise<{ id: number; email: string; name: string }> {
    const dto = parseSignup(input);
    if (await this.#users.findByEmail(dto.email)) throw new ConflictError("האימייל כבר רשום");
    const u = await this.#users.insert({ email: dto.email, name: dto.name, passwordHash: this.#hash(dto.password) });
    return { id: u.id, email: u.email, name: u.name };
  }
}

const rows: User[] = [];
const repo: UserRepository = {
  async findByEmail(email) {
    return rows.find((u) => u.email === email) ?? null;
  },
  async insert(u) {
    const user = { ...u, id: rows.length + 1 };
    rows.push(user);
    return { ...user };
  },
};
const service = new SignupService(repo, (p) => `hash(${p.length})`);
async function demo() {
  console.log(await service.signup({ email: " Dana@Mail.com ", name: "דנה", password: "12345678", isAdmin: true }));
  for (const body of [{ email: "dana@mail.com", name: "x", password: "abcdefgh" }, { email: "a@b", name: "x", password: "abcdefgh" }]) {
    try {
      await service.signup(body);
    } catch (err) {
      console.log(String(err));
    }
  }
  console.log(rows);
}
demo();
```
