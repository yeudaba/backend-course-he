// P13-E06
export function parseVersion(v) {
  const parts = String(v).split(".");
  if (parts.length !== 3 || !parts.every((p) => /^\d+$/.test(p))) {
    throw new Error(`גרסה לא תקינה: ${v}`);
  }
  return parts.map(Number);
}

function compare(a, b) {
  for (let i = 0; i < 3; i++) {
    if (a[i] !== b[i]) return a[i] - b[i];
  }
  return 0;
}

export function satisfiesCaret(version, range) {
  const v = parseVersion(version);
  const min = parseVersion(range.replace(/^\^/, ""));
  if (compare(v, min) < 0) return false;
  if (min[0] === 0) return v[0] === 0 && v[1] === min[1];
  return v[0] === min[0];
}

import fs from "node:fs/promises";
import path from "node:path";

export function depsReport(pkg) {
  const prod = Object.keys(pkg.dependencies ?? {}).sort();
  const dev = Object.keys(pkg.devDependencies ?? {}).sort();
  return { prod, dev, total: prod.length + dev.length };

    const run = ["start", "test"].includes(name) ? `npm ${name}` : `npm run ${name}`;

    if (err.code === "ENOENT") throw new Error("לא נמצא package.json");

  }
}

// P13-E08
export function parseArgs(argv) {
  const positional = [];
  const flags = {};
  for (let i = 0; i < argv.length; i++) {
    const arg = argv[i];
    if (!arg.startsWith("--")) {
      positional.push(arg);
      continue;
    }
    const body = arg.slice(2);
    const eq = body.indexOf("=");
    if (eq !== -1) {
      flags[body.slice(0, eq)] = body.slice(eq + 1);
    } else if (i + 1 < argv.length && !argv[i + 1].startsWith("--")) {
      flags[body] = argv[i + 1];
      i++;
    } else {
      flags[body] = true;
    }
  }
  return { positional, flags };
}

// P13-E09
export function envInt(env, name, fallback) {
  const raw = env[name]?.trim();
  if (raw === undefined || raw === "") return fallback;
  if (!/^-?\d+$/.test(raw)) throw new Error(`${name} חייב להיות מספר שלם`);
  return Number(raw);
}

export function envBool(env, name, fallback) {
  const raw = env[name]?.trim().toLowerCase();
  if (raw === undefined || raw === "") return fallback;
  if (["true", "1", "yes", "on"].includes(raw)) return true;
  if (["false", "0", "no", "off"].includes(raw)) return false;
  throw new Error(`${name} חייב להיות בוליאני`);
}

// P13-E10
export function parseDotenv(text) {
  const result = {};
  const lines = text.split("\n");
  for (let i = 0; i < lines.length; i++) {
    const line = lines[i].trim();
    if (line === "" || line.startsWith("#")) continue;
    const eq = line.indexOf("=");
    if (eq === -1) throw new Error(`שורה ${i + 1} לא תקינה`);
    const key = line.slice(0, eq).trim();
    let value = line.slice(eq + 1).trim();
    const q = value[0];
    if ((q === '"' || q === "'") && value.length >= 2 && value.at(-1) === q) {
      value = value.slice(1, -1);
    }
    result[key] = value;
  }
  return result;
}

import fs from "node:fs/promises";
import path from "node:path";

export async function dirStats(dir) {
  const entries = await fs.readdir(dir, { withFileTypes: true });
  const files = entries.filter((e) => e.isFile()).map((e) => e.name);
  const sizes = await Promise.all(files.map(async (name) => (await fs.stat(path.join(dir, name))).size));

// P13-E13
const parse = (v) => v.split(".").map(Number);

export function compareVersions(a, b) {
  const x = parse(a);
  const y = parse(b);
  for (let i = 0; i < 3; i++) {
    if (x[i] !== y[i]) return x[i] - y[i];
  }
  return 0;
}

export function satisfies(version, range) {
  const op = range[0];
  if (op !== "^" && op !== "~") return compareVersions(version, range) === 0;
  const min = range.slice(1);
  if (compareVersions(version, min) < 0) return false;
  const [vMaj, vMin] = parse(version);
  const [mMaj, mMin] = parse(min);
  return op === "^" ? vMaj === mMaj : vMaj === mMaj && vMin === mMin;
}

export function maxSatisfying(versions, range) {
  const ok = versions.filter((v) => satisfies(v, range)).sort(compareVersions);
  return ok.at(-1) ?? null;
}

const available = ["4.17.1", "4.18.2", "4.21.0", "5.0.0", "5.1.0"];
console.log(maxSatisfying(available, "^4.18.0"), maxSatisfying(available, "~4.18.0"), maxSatisfying(available, "^6.0.0"));

const LEVELS = ["debug", "info", "warn", "error"];

  const problems = [];

import { randomUUID } from "node:crypto";

const ALLOWED = [".png", ".jpg", ".jpeg", ".pdf"];

export async function storeUpload(baseDir, originalName, content) {
  const ext = path.extname(originalName).toLowerCase();

import fs from "node:fs";
import readline from "node:readline";

export async function countStatuses(file) {
  const result = { "2xx": 0, "3xx": 0, "4xx": 0, "5xx": 0, invalid: 0, slowest: null };
  const rl = readline.createInterface({ input: fs.createReadStream(file), crlfDelay: Infinity });
  for await (const raw of rl) {

      continue;

    if (result.slowest === null || ms > result.slowest.ms) {

}

// P13-E19
import fs from "node:fs";
import fsp from "node:fs/promises";
import path from "node:path";
import { createHash } from "node:crypto";

export async function fileHash(file) {
  const hash = createHash("sha256");
  for await (const chunk of fs.createReadStream(file)) {
    hash.update(chunk);
  }
  return hash.digest("hex");
}

export async function findDuplicates(dir) {
  const entries = await fsp.readdir(dir, { withFileTypes: true });
  const names = entries.filter((e) => e.isFile()).map((e) => e.name);
  const hashes = await Promise.all(names.map((n) => fileHash(path.join(dir, n))));
  const groups = new Map();
  names.forEach((name, i) => {
    const list = groups.get(hashes[i]) ?? [];
    list.push(name);
    groups.set(hashes[i], list);
  });
  return [...groups.values()]
    .filter((g) => g.length > 1)
    .map((g) => g.sort())
    .sort((a, b) => a[0].localeCompare(b[0]));
}

const tmp = await fsp.mkdtemp(path.join((await import("node:os")).tmpdir(), "dups-"));

  await fsp.writeFile(path.join(tmp, name), text);

// P13-E24: פותר תלויות שטוח
const parse = (v) => v.split(".").map(Number);
function compare(a, b) {
  const x = parse(a);
  const y = parse(b);
  for (let i = 0; i < 3; i++) if (x[i] !== y[i]) return x[i] - y[i];
  return 0;
}
function satisfies(version, range) {
  if (!range.startsWith("^")) return compare(version, range) === 0;
  const min = range.slice(1);
  return compare(version, min) >= 0 && parse(version)[0] === parse(min)[0];
}

export function resolveDeps(rootDeps, registry) {
  const chosen = {};
  const queue = Object.entries(rootDeps);
  while (queue.length > 0) {
    const [name, range] = queue.shift();
    if (chosen[name]) {
      if (!satisfies(chosen[name], range)) {
        throw new Error(`התנגשות: ${name}`);
      }
      continue;
    }
    const versions = Object.keys(registry[name] ?? {}).filter((v) => satisfies(v, range)).sort(compare);
    const best = versions.at(-1);
    if (!best) {
      throw new Error(`אין גרסה של ${name} שמתאימה ל־${range}`);
    }
    chosen[name] = best;
    queue.push(...Object.entries(registry[name][best].deps ?? {}));
  }
  return Object.fromEntries(Object.entries(chosen).sort(([a], [b]) => a.localeCompare(b)));
}

const registry = {
  express: { "5.0.0": { deps: { router: "^2.0.0", debug: "^4.0.0" } }, "5.1.0": { deps: { router: "^2.1.0", debug: "^4.3.0" } } },
  router: { "2.0.0": {}, "2.2.0": { deps: { debug: "^4.0.0" } } },
  debug: { "4.3.4": {}, "4.4.0": {} },
};
console.log(resolveDeps({ express: "^5.0.0" }, registry));
```

// P13-E25: log rotation
import fs from "node:fs/promises";
import path from "node:path";

async function exists(file) {
  try {
    await fs.access(file);
    return true;
  } catch {
    return false;
  }
}

export async function rotate(dir, name, maxFiles) {
  const actions = [];
  const file = (i) => path.join(dir, i === 0 ? name : `${name}.${i}`);
  const label = (i) => (i === 0 ? name : `${name}.${i}`);

  if (await exists(file(maxFiles))) {
    await fs.rm(file(maxFiles));
    actions.push(`delete ${label(maxFiles)}`);
  }
  for (let i = maxFiles - 1; i >= 0; i--) {
    if (await exists(file(i))) {
      await fs.rename(file(i), file(i + 1));
      actions.push(`${label(i)} -> ${label(i + 1)}`);
    }
  }
  await fs.writeFile(file(0), "");
  actions.push(`create ${name}`);
  return actions;
}

const tmp = await fs.mkdtemp(path.join((await import("node:os")).tmpdir(), "rot-"));
for (const f of ["app.log", "app.log.1", "app.log.3"]) await fs.writeFile(path.join(tmp, f), f);
console.log(await rotate(tmp, "app.log", 3));
console.log((await fs.readdir(tmp)).sort());

// P13-E26: פירוק שורת לוג עם regex
const LINE = /^(?<ip>\S+) \S+ \S+ \[(?<time>[^\]]+)\] "(?<method>[A-Z]+) (?<path>\S+) HTTP\/[\d.]+" (?<status>\d{3}) (?<bytes>\d+|-)$/;

export function parseAccessLine(line) {
  const m = line.trim().match(LINE);
  if (!m) {
    return null;
  }
  const { ip, time, method, path, status, bytes } = m.groups;
  return {
    ip,
    time,
    method,
    path,
    status: Number(status),
    bytes: bytes === "-" ? 0 : Number(bytes),
  };
}

console.log(parseAccessLine("משהו אחר לגמרי"));

// P13-E27: העלאת גרסה (semver) עם switch
export function bump(version, kind) {
  if (!/^\d+\.\d+\.\d+$/.test(version)) {
    throw new Error(`גרסה לא תקינה: ${version}`);
  }
  const [major, minor, patch] = version.split(".").map(Number);
  switch (kind) {
    case "major":
      return `${major + 1}.0.0`;
    case "minor":
      return `${major}.${minor + 1}.0`;
    case "patch":
      return `${major}.${minor}.${patch + 1}`;
    default:
      throw new Error(`סוג לא מוכר: ${kind}`);
  }
}

console.log(bump("1.4.2", "patch"), bump("1.4.2", "minor"), bump("1.4.2", "major"));
console.log(bump("0.9.9", "minor"));
try {
  bump("1.4", "patch");
} catch (err) {
  console.log(err.message);
}
```

0.10.0

<p>ה־<code>switch</code> מתאים כאן כי משווים ערך אחד לכמה מחרוזות קבועות, ולכל אחת יש פעולה אחרת. ‏<code>return</code> בתוך <code>case</code> חוסך את ה־<code>break</code>. ה־<code>default</code> הוא המקום הטבעי לשגיאה.</p>

// P13-E28: מיזוג קובצי הגדרות במקביל
import fs from "node:fs/promises";
import os from "node:os";
import path from "node:path";

async function readJson(file) {
  const text = await fs.readFile(file, "utf8");
  try {
    return JSON.parse(text);
  } catch {
    throw new Error(`${path.basename(file)}: JSON לא תקין`);
  }
}

export async function loadConfigs(files) {
  const configs = await Promise.all(files.map(readJson));
  return configs.reduce((merged, config) => ({ ...merged, ...config }), {});
}

const dir = await fs.mkdtemp(path.join(os.tmpdir(), "cfg-"));
await fs.writeFile(path.join(dir, "default.json"), '{ "port": 3000, "logLevel": "info" }');
await fs.writeFile(path.join(dir, "local.json"), '{ "logLevel": "debug" }');
await fs.writeFile(path.join(dir, "broken.json"), "{ port: 1 }");

console.log(await loadConfigs([path.join(dir, "default.json"), path.join(dir, "local.json")]));
try {
  await loadConfigs([path.join(dir, "default.json"), path.join(dir, "broken.json")]);
} catch (err) {
  console.log(err.message);
}
```

broken.json: JSON לא תקין
```
<p><code>Promise.all</code> שומר על הסדר של המערך, גם אם קובץ אחד נקרא מהר מהאחר. לכן ה־<code>reduce</code> ממזג בדיוק לפי הסדר שביקשו, והקובץ האחרון גובר.</p>
<p>ה־<code>try</code> הצר הוא העיקר כאן: הוא תופס רק את השגיאה שאנחנו יודעים להסביר (JSON שבור), ומוסיף לה את שם הקובץ. שגיאה אחרת, כמו קובץ חסר, עוברת כמו שהיא. ככה מי שקורא את ההודעה יודע מה באמת קרה.</p>
