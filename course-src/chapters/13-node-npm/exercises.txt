// P13-E06
export function parseVersion(v) {
  const parts = String(v).split(".");
  if (parts.length !== 3 || !parts.every((p) => /^\d+$/.test(p))) {
    throw new Error(`גרסה לא תקינה: ${v}`);
  }
  return parts.map(Number);
}

function compare(a, b) {
  for (let i = 0; i < 3; i++) {
    if (a[i] !== b[i]) return a[i] - b[i];
  }
  return 0;
}

export function satisfiesCaret(version, range) {
  const v = parseVersion(version);
  const min = parseVersion(range.replace(/^\^/, ""));
  if (compare(v, min) < 0) return false;
  if (min[0] === 0) return v[0] === 0 && v[1] === min[1];
  return v[0] === min[0];
}

import fs from "node:fs/promises";
import path from "node:path";

export function depsReport(pkg) {
  const prod = Object.keys(pkg.dependencies ?? {}).sort();
  const dev = Object.keys(pkg.devDependencies ?? {}).sort();
  return { prod, dev, total: prod.length + dev.length };

    const run = ["start", "test"].includes(name) ? `npm ${name}` : `npm run ${name}`;

    if (err.code === "ENOENT") throw new Error("לא נמצא package.json");

  }
}

// P13-E08
export function parseArgs(argv) {
  const positional = [];
  const flags = {};
  for (let i = 0; i < argv.length; i++) {
    const arg = argv[i];
    if (!arg.startsWith("--")) {
      positional.push(arg);
      continue;
    }
    const body = arg.slice(2);
    const eq = body.indexOf("=");
    if (eq !== -1) {
      flags[body.slice(0, eq)] = body.slice(eq + 1);
    } else if (i + 1 < argv.length && !argv[i + 1].startsWith("--")) {
      flags[body] = argv[i + 1];
      i++;
    } else {
      flags[body] = true;
    }
  }
  return { positional, flags };
}

// P13-E09
export function envInt(env, name, fallback) {
  const raw = env[name]?.trim();
  if (raw === undefined || raw === "") return fallback;
  if (!/^-?\d+$/.test(raw)) throw new Error(`${name} חייב להיות מספר שלם`);
  return Number(raw);
}

export function envBool(env, name, fallback) {
  const raw = env[name]?.trim().toLowerCase();
  if (raw === undefined || raw === "") return fallback;
  if (["true", "1", "yes", "on"].includes(raw)) return true;
  if (["false", "0", "no", "off"].includes(raw)) return false;
  throw new Error(`${name} חייב להיות בוליאני`);
}

// P13-E10
export function parseDotenv(text) {
  const result = {};
  const lines = text.split("\n");
  for (let i = 0; i < lines.length; i++) {
    const line = lines[i].trim();
    if (line === "" || line.startsWith("#")) continue;
    const eq = line.indexOf("=");
    if (eq === -1) throw new Error(`שורה ${i + 1} לא תקינה`);
    const key = line.slice(0, eq).trim();
    let value = line.slice(eq + 1).trim();
    const q = value[0];
    if ((q === '"' || q === "'") && value.length >= 2 && value.at(-1) === q) {
      value = value.slice(1, -1);
    }
    result[key] = value;
  }
  return result;
}

import fs from "node:fs/promises";
import path from "node:path";

export async function dirStats(dir) {
  const entries = await fs.readdir(dir, { withFileTypes: true });
  const files = entries.filter((e) => e.isFile()).map((e) => e.name);
  const sizes = await Promise.all(files.map(async (name) => (await fs.stat(path.join(dir, name))).size));

// P13-E13
const parse = (v) => v.split(".").map(Number);

export function compareVersions(a, b) {
  const x = parse(a);
  const y = parse(b);
  for (let i = 0; i < 3; i++) {
    if (x[i] !== y[i]) return x[i] - y[i];
  }
  return 0;
}

export function satisfies(version, range) {
  const op = range[0];
  if (op !== "^" && op !== "~") return compareVersions(version, range) === 0;
  const min = range.slice(1);
  if (compareVersions(version, min) < 0) return false;
  const [vMaj, vMin] = parse(version);
  const [mMaj, mMin] = parse(min);
  return op === "^" ? vMaj === mMaj : vMaj === mMaj && vMin === mMin;
}

export function maxSatisfying(versions, range) {
  const ok = versions.filter((v) => satisfies(v, range)).sort(compareVersions);
  return ok.at(-1) ?? null;
}

const available = ["4.17.1", "4.18.2", "4.21.0", "5.0.0", "5.1.0"];
console.log(maxSatisfying(available, "^4.18.0"), maxSatisfying(available, "~4.18.0"), maxSatisfying(available, "^6.0.0"));

const LEVELS = ["debug", "info", "warn", "error"];

  const problems = [];

import { randomUUID } from "node:crypto";

const ALLOWED = [".png", ".jpg", ".jpeg", ".pdf"];

export async function storeUpload(baseDir, originalName, content) {
  const ext = path.extname(originalName).toLowerCase();

import fs from "node:fs";
import readline from "node:readline";

export async function countStatuses(file) {
  const result = { "2xx": 0, "3xx": 0, "4xx": 0, "5xx": 0, invalid: 0, slowest: null };
  const rl = readline.createInterface({ input: fs.createReadStream(file), crlfDelay: Infinity });
  for await (const raw of rl) {

      continue;

    if (result.slowest === null || ms > result.slowest.ms) {

}

// P13-E19
import fs from "node:fs";
import fsp from "node:fs/promises";
import path from "node:path";
import { createHash } from "node:crypto";

export async function fileHash(file) {
  const hash = createHash("sha256");
  for await (const chunk of fs.createReadStream(file)) {
    hash.update(chunk);
  }
  return hash.digest("hex");
}

export async function findDuplicates(dir) {
  const entries = await fsp.readdir(dir, { withFileTypes: true });
  const names = entries.filter((e) => e.isFile()).map((e) => e.name);
  const hashes = await Promise.all(names.map((n) => fileHash(path.join(dir, n))));
  const groups = new Map();
  names.forEach((name, i) => {
    const list = groups.get(hashes[i]) ?? [];
    list.push(name);
    groups.set(hashes[i], list);
  });
  return [...groups.values()]
    .filter((g) => g.length > 1)
    .map((g) => g.sort())
    .sort((a, b) => a[0].localeCompare(b[0]));
}

const tmp = await fsp.mkdtemp(path.join((await import("node:os")).tmpdir(), "dups-"));

  await fsp.writeFile(path.join(tmp, name), text);

// P13-E24: פותר תלויות שטוח
const parse = (v) => v.split(".").map(Number);
function compare(a, b) {
  const x = parse(a);
  const y = parse(b);
  for (let i = 0; i < 3; i++) if (x[i] !== y[i]) return x[i] - y[i];
  return 0;
}
function satisfies(version, range) {
  if (!range.startsWith("^")) return compare(version, range) === 0;
  const min = range.slice(1);
  return compare(version, min) >= 0 && parse(version)[0] === parse(min)[0];
}

export function resolveDeps(rootDeps, registry) {
  const chosen = {};
  const queue = Object.entries(rootDeps);
  while (queue.length > 0) {
    const [name, range] = queue.shift();
    if (chosen[name]) {
      if (!satisfies(chosen[name], range)) {
        throw new Error(`התנגשות: ${name}`);
      }
      continue;
    }
    const versions = Object.keys(registry[name] ?? {}).filter((v) => satisfies(v, range)).sort(compare);
    const best = versions.at(-1);
    if (!best) {
      throw new Error(`אין גרסה של ${name} שמתאימה ל־${range}`);
    }
    chosen[name] = best;
    queue.push(...Object.entries(registry[name][best].deps ?? {}));
  }
  return Object.fromEntries(Object.entries(chosen).sort(([a], [b]) => a.localeCompare(b)));
}

const registry = {
  express: { "5.0.0": { deps: { router: "^2.0.0", debug: "^4.0.0" } }, "5.1.0": { deps: { router: "^2.1.0", debug: "^4.3.0" } } },
  router: { "2.0.0": {}, "2.2.0": { deps: { debug: "^4.0.0" } } },
  debug: { "4.3.4": {}, "4.4.0": {} },
};
console.log(resolveDeps({ express: "^5.0.0" }, registry));
```

// P13-E25: log rotation
import fs from "node:fs/promises";
import path from "node:path";

async function exists(file) {
  try {
    await fs.access(file);
    return true;
  } catch {
    return false;
  }
}

export async function rotate(dir, name, maxFiles) {
  const actions = [];
  const file = (i) => path.join(dir, i === 0 ? name : `${name}.${i}`);
  const label = (i) => (i === 0 ? name : `${name}.${i}`);

  if (await exists(file(maxFiles))) {
    await fs.rm(file(maxFiles));
    actions.push(`delete ${label(maxFiles)}`);
  }
  for (let i = maxFiles - 1; i >= 0; i--) {
    if (await exists(file(i))) {
      await fs.rename(file(i), file(i + 1));
      actions.push(`${label(i)} -> ${label(i + 1)}`);
    }
  }
  await fs.writeFile(file(0), "");
  actions.push(`create ${name}`);
  return actions;
}

const tmp = await fs.mkdtemp(path.join((await import("node:os")).tmpdir(), "rot-"));
for (const f of ["app.log", "app.log.1", "app.log.3"]) await fs.writeFile(path.join(tmp, f), f);
console.log(await rotate(tmp, "app.log", 3));
console.log((await fs.readdir(tmp)).sort());

// P13-E26: פירוק שורת לוג עם regex
const LINE = /^(?<ip>\S+) \S+ \S+ \[(?<time>[^\]]+)\] "(?<method>[A-Z]+) (?<path>\S+) HTTP\/[\d.]+" (?<status>\d{3}) (?<bytes>\d+|-)$/;

export function parseAccessLine(line) {
  const m = line.trim().match(LINE);
  if (!m) {
    return null;
  }
  const { ip, time, method, path, status, bytes } = m.groups;
  return {
    ip,
    time,
    method,
    path,
    status: Number(status),
    bytes: bytes === "-" ? 0 : Number(bytes),
  };
}

// P13-E27: העלאת גרסה (semver) עם switch
export function bump(version, kind) {
  if (!/^\d+\.\d+\.\d+$/.test(version)) {
    throw new Error(`גרסה לא תקינה: ${version}`);
  }
  const [major, minor, patch] = version.split(".").map(Number);
  switch (kind) {
    case "major":
      return `${major + 1}.0.0`;
    case "minor":
      return `${major}.${minor + 1}.0`;
    case "patch":
      return `${major}.${minor}.${patch + 1}`;
    default:
      throw new Error(`סוג לא מוכר: ${kind}`);
  }
}

console.log(bump("1.4.2", "patch"), bump("1.4.2", "minor"), bump("1.4.2", "major"));
console.log(bump("0.9.9", "minor"));
try {
  bump("1.4", "patch");
} catch (err) {
  console.log(err.message);
}

import fs from "node:fs/promises";
import os from "node:os";
import path from "node:path";

async function readJson(file) {
  const text = await fs.readFile(file, "utf8");
  try {
    return JSON.parse(text);
  } catch {
    throw new Error(`${path.basename(file)}: JSON לא תקין`);
  }
}

export async function loadConfigs(files) {
  const configs = await Promise.all(files.map(readJson));
  return configs.reduce((merged, config) => ({ ...merged, ...config }), {});
}

const dir = await fs.mkdtemp(path.join(os.tmpdir(), "cfg-"));
await fs.writeFile(path.join(dir, "default.json"), '{ "port": 3000, "logLevel": "info" }');
await fs.writeFile(path.join(dir, "local.json"), '{ "logLevel": "debug" }');
await fs.writeFile(path.join(dir, "broken.json"), "{ port: 1 }");

console.log(await loadConfigs([path.join(dir, "default.json"), path.join(dir, "local.json")]));
try {
  await loadConfigs([path.join(dir, "default.json"), path.join(dir, "broken.json")]);
} catch (err) {
  console.log(err.message);
}

export function rangeKind(range) {
  const m = String(range).trim().match(/^([~^]?)\d+\.\d+\.\d+$/);

const sortKeys = (obj) => Object.fromEntries(Object.entries(obj).sort(([a], [b]) => (a < b ? -1 : a > b ? 1 : 0)));

export function addDependency(pkg, name, version, { dev = false } = {}) {
  const target = dev ? "devDependencies" : "dependencies";
  const other = dev ? "dependencies" : "devDependencies";
  const result = { ...pkg, [target]: sortKeys({ ...pkg[target], [name]: `^${version}` }) };

    delete rest[name];
    if (Object.keys(rest).length === 0) delete result[other];
    else result[other] = rest;
  }
  return result;
}

// P13-E36: מה ישן?
const parts = (v) => v.replace(/^[~^]/, "").split(".").map(Number);
const TYPES = ["major", "minor", "patch"];

export function outdated(deps, latest) {
  const result = [];
  for (const [name, range] of Object.entries(deps)) {
    if (!(name in latest)) continue;
    const c = parts(range);
    const l = parts(latest[name]);
    for (let i = 0; i < 3; i++) {
      if (l[i] === c[i]) continue;
      if (l[i] > c[i]) {
        result.push({ name, current: c.join("."), latest: latest[name], type: TYPES[i] });
      }
      break;
    }
  }
  return result.sort((a, b) => (a.name < b.name ? -1 : 1));
}

const NM = "node_modules/";

export function lockSummary(lock) {
  let total = 0;
  let dev = 0;
  const sets = new Map();
  for (const [key, info] of Object.entries(lock.packages ?? {})) {
    if (key === "") continue;
    total++;
    if (info.dev === true) dev++;
    const name = key.slice(key.lastIndexOf(NM) + NM.length);
    if (!sets.has(name)) sets.set(name, new Set());
    sets.get(name).add(info.version);
  }
  const versions = new Map();
  for (const [name, set] of sets) versions.set(name, [...set].sort());
  const duplicates = [...versions].filter(([, v]) => v.length > 1).map(([n]) => n).sort();
  return { total, prod: total - dev, dev, versions, duplicates };
}

// P13-E42: תלויות לפי scope
const NONE = "(ללא)";

export function byScope(pkg) {
  const names = Object.keys({ ...pkg.dependencies, ...pkg.devDependencies });
  const groups = new Map();
  for (const name of names) {
    const scope = name.startsWith("@") ? name.slice(0, name.indexOf("/")) : NONE;
    if (!groups.has(scope)) groups.set(scope, []);
    groups.get(scope).push(name);
  }
  const result = new Map();
  const keys = [...groups.keys()].filter((k) => k !== NONE).sort();
  if (groups.has(NONE)) keys.unshift(NONE);
  for (const k of keys) result.set(k, groups.get(k).sort());
  return result;
}

console.log(byScope({
  dependencies: { express: "^5", "@prisma/client": "^6" },
  devDependencies: { "@types/node": "^24", "@types/express": "^5", eslint: "^9" },
}));
```

export function onShutdown(proc, cleanups, log) {
  let started = false;
  let finish;
  const done = new Promise((resolve) => (finish = resolve));

  async function handler(signal) {
    if (started) return;
    started = true;
    log(`התקבל ${signal}, סוגר...`);
    let failed = false;
    for (const fn of [...cleanups].reverse()) {
      try {
        await fn();
      } catch (err) {
        log(`ניקוי נכשל: ${err.message}`);
        failed = true;
      }
    }
    proc.exitCode = failed ? 1 : 0;
    log("סיום");
    finish();
  }

  proc.on("SIGINT", () => handler("SIGINT"));
  proc.on("SIGTERM", () => handler("SIGTERM"));
  return done;
}

const handlers = {};
const fakeProc = { exitCode: undefined, on: (sig, fn) => (handlers[sig] = fn) };

  async () => { throw new Error("השרת כבר סגור"); },
  async () => console.log("  מפסיק לקבל בקשות"),
], console.log);

// P13-E54: הגדרות מדגלים וממשתני סביבה
export function resolveSettings(args, env) {
  let portText;
  let verbose = env.VERBOSE === "1";
  let name;
  for (let i = 0; i < args.length; i++) {
    const a = args[i];
    if (a === "--port") {
      portText = args[i + 1];
      i++;
    } else if (a.startsWith("--port=")) {
      portText = a.slice("--port=".length);
    } else if (a === "--verbose") {
      verbose = true;
    } else if (a.startsWith("--")) {
      throw new Error(`דגל לא מוכר: ${a}`);
    } else if (name === undefined) {
      name = a;
    }
  }
  const text = portText ?? env.PORT ?? "3000";
  const port = Number(text);
  if (!/^\d+$/.test(text) || port < 1 || port > 65535) throw new Error(`port לא תקין: ${text}`);
  return { port, verbose, name: name ?? "app" };
}

console.log(resolveSettings(["--port=7000", "--verbose"], {}));
console.log(resolveSettings([], { PORT: "8080", VERBOSE: "1" }));
try {
  resolveSettings(["--prot", "1"], {});
} catch (err) {
  console.log(err.message);
}
```

export async function countItems(args, { log, error, readFile }) {
  if (args.length !== 1) {
    error("שימוש: count <file>");
    return 2;
  }

export function formatHelp(tool, commands) {
  const left = commands.map((c) => (c.args ? `${c.name} ${c.args}` : c.name));
  const width = Math.max(0, ...left.map((s) => s.length));
  const lines = [`שימוש: ${tool} <פקודה> [ארגומנטים]`, "", "פקודות:"];

  return lines.join("\n");
}

export function required(env, name) {
  const value = env[name]?.trim();
  if (value === undefined || value === "") throw new Error(`חסר משתנה סביבה: ${name}`);

// P13-E63: בודק לקובץ .env
export function lintEnv(text) {
  const warnings = [];
  const firstLine = new Map();
  const lines = text.split("\n");
  for (let i = 0; i < lines.length; i++) {
    const n = i + 1;
    const t = lines[i].trim();
    if (t === "" || t.startsWith("#")) continue;
    const eq = t.indexOf("=");
    if (eq === -1) {
      warnings.push(`שורה ${n}: חסר =`);
      continue;
    }
    const name = t.slice(0, eq).trim();
    const value = t.slice(eq + 1).trim();
    if (!/^[A-Z][A-Z0-9_]*$/.test(name)) {
      warnings.push(`שורה ${n}: שם לא תקין: ${name}`);
      continue;
    }
    if (firstLine.has(name)) {
      warnings.push(`שורה ${n}: ${name} מוגדר שוב (קודם בשורה ${firstLine.get(name)})`);
      continue;
    }
    firstLine.set(name, n);
    const q = value[0];
    if (q === '"' || q === "'") {
      if (value.length < 2 || !value.endsWith(q)) warnings.push(`שורה ${n}: מירכאות לא סגורות`);
    } else if (value.includes(" ")) {
      warnings.push(`שורה ${n}: ערך עם רווחים בלי מירכאות`);
    }
  }
  return warnings;
}

// P13-E64: מה חסר לי ב־.env?
function parse(text) {
  const map = new Map();
  for (const line of text.split("\n")) {
    const t = line.trim();
    if (t === "" || t.startsWith("#")) continue;
    const i = t.indexOf("=");
    if (i === -1) continue;
    const key = t.slice(0, i).trim();
    let value = t.slice(i + 1).trim();
    const q = value.match(/^(["'])(.*)\1$/);
    if (q) value = q[2];
    if (!map.has(key)) map.set(key, value);
  }
  return map;
}

export function envDiff(exampleText, envText) {
  const example = parse(exampleText);
  const env = parse(envText);
  const missing = [...example.keys()].filter((k) => !env.has(k));
  const empty = [...example.keys()].filter((k) => env.has(k) && env.get(k) === "");
  const extra = [...env.keys()].filter((k) => !example.has(k));
  return { missing, empty, extra };
}

// P13-E69: הפניות בין משתנים ב־.env
export function expand(vars) {
  const done = new Map();
  function resolve(key, stack) {
    if (done.has(key)) return done.get(key);
    if (stack.has(key)) throw new Error(`הפניה מעגלית: ${key}`);
    stack.add(key);
    const value = vars[key].replace(/\$\{(\w+)\}/g, (_, name) => {
      if (!(name in vars)) throw new Error(`משתנה לא מוגדר: ${name} (ב־${key})`);
      return resolve(name, stack);
    });
    stack.delete(key);
    done.set(key, value);
    return value;
  }
  const result = {};
  for (const key of Object.keys(vars)) result[key] = resolve(key, new Set());
  return result;
}

  CURRENCY: "ILS",
}));

  } catch (err) {

const camel = (name) => name.toLowerCase().replace(/_([a-z0-9])/g, (_, c) => c.toUpperCase());

function convert(rule, raw) {
  switch (rule.type) {
    case "string":
      return raw;
    case "int":
      return /^-?\d+$/.test(raw) ? Number(raw) : undefined;
    case "bool": {
      const v = raw.toLowerCase();
      if (v === "true" || v === "1") return true;
      if (v === "false" || v === "0") return false;
      return undefined;
    }
    case "enum":
      return rule.values.includes(raw) ? raw : undefined;
    default:
      throw new Error(`סוג לא מוכר: ${rule.type}`);
  }
}

export function parseEnv(env, schema) {
  const result = {};
  const problems = [];
  for (const [name, rule] of Object.entries(schema)) {
    const raw = env[name]?.trim() ?? "";
    let value;
    if (raw === "") {
      convert(rule, "");
      if (!("default" in rule)) {
        problems.push(`${name}: חסר`);
        continue;
      }
      value = rule.default;
    } else {
      value = convert(rule, raw);
      if (value === undefined) {
        problems.push(`${name}: לא תקין`);
        continue;
      }
    }
    result[camel(name)] = value;
  }
  if (problems.length > 0) throw new Error("הגדרות לא תקינות:\n" + problems.join("\n"));
  return result;
}

  DATABASE_URL: { type: "string" },
  DEBUG: { type: "bool", default: false },
  LOG_LEVEL: { type: "enum", values: ["debug", "info", "error"], default: "info" },
};

try {
  parseEnv({ PORT: "80a", LOG_LEVEL: "verbose" }, schema);
} catch (err) {
  console.log(err.message);
}

const SENSITIVE = /password|secret|token|key/i;

export function redact(value) {
  if (Array.isArray(value)) return value.map(redact);
  if (typeof value === "object" && value !== null) {

// P13-E77: שם קובץ פנוי
import fs from "node:fs/promises";
import path from "node:path";

async function exists(p) {
  try {
    await fs.stat(p);
    return true;
  } catch (err) {
    if (err.code === "ENOENT") return false;
    throw err;
  }
}

export async function freeName(dir, name) {
  const safe = path.basename(name);
  if (!(await exists(path.join(dir, safe)))) return safe;
  const ext = path.extname(safe);
  const base = safe.slice(0, safe.length - ext.length);
  for (let i = 1; ; i++) {
    const candidate = `${base} (${i})${ext}`;
    if (!(await exists(path.join(dir, candidate)))) return candidate;
  }
}

import os from "node:os";
const dir = await fs.mkdtemp(path.join(os.tmpdir(), "p13-e77-"));
await fs.writeFile(path.join(dir, "report.pdf"), "");
await fs.writeFile(path.join(dir, "report (1).pdf"), "");
console.log(await freeName(dir, "report.pdf"), "|", await freeName(dir, "../../etc/notes.txt"), "|", await freeName(dir, "archive.tar.gz"));
```
```out
report (2).pdf | notes.txt | archive.tar.gz
```
<p>ה־<code>basename</code> הוא שורת ההגנה: <code>"../../etc/notes.txt"</code> הופך ל־<code>"notes.txt"</code> לפני שבכלל נוגעים בדיסק. ושים לב: בין הבדיקה "הקובץ פנוי" לבין הכתיבה, תהליך אחר יכול לתפוס את השם (race condition, נדבר עליו בפרק 18). בקוד אמיתי כותבים עם <code>fs.writeFile(p, data, { flag: "wx" })</code> — שנכשל אם הקובץ כבר קיים — ומנסים את השם הבא.</p>

<ul><li><code>name.split(".")[0]</code> לבסיס — <code>my.report.pdf</code> הופך ל־<code>my (1).pdf</code>.</li><li><code>catch { return false; }</code> לכל שגיאה — "אין הרשאה" נחשב "פנוי", והכתיבה תיכשל אחר כך עם הודעה מבלבלת.</li></ul>
