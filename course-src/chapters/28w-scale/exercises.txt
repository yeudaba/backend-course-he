```ts save=solutions/P28.5-E06.ts
// P28.5-E06: percentiles — לא ממוצע
export function percentile(sorted: number[], p: number): number {
  if (sorted.length === 0) throw new Error("אין דגימות");
  if (!(p > 0 && p <= 100)) throw new Error("p בין 0 (לא כולל) ל־100");
  const rank = Math.ceil((p / 100) * sorted.length); // nearest-rank: 1..n
  return sorted[rank - 1];
}

export interface Summary {
  count: number;
  min: number;
  max: number;
  mean: number;
  p50: number;
  p95: number;
  p99: number;
}

export function summarize(samples: number[]): Summary {
  const s = [...samples].sort((a, b) => a - b); // עותק, ומיון מספרי
  const mean = s.reduce((a, b) => a + b, 0) / s.length;
  return {
    count: s.length,
    min: s[0],
    max: s[s.length - 1],
    mean: Math.round(mean * 100) / 100,
    p50: percentile(s, 50),
    p95: percentile(s, 95),
    p99: percentile(s, 99),
  };
}
```

```ts save=solutions/P28.5-E07.ts
// P28.5-E07: histogram שאפשר לאחד בין שרתים
// דליים לוגריתמיים: הגבול העליון של דלי i הוא 1.1^i מילישניות — שגיאה של עד 10%, בכל טווח
const GROWTH = 1.1;

export interface Histogram {
  counts: number[];
  total: number;
}

export const createHistogram = (): Histogram => ({ counts: [], total: 0 });

export const bucketOf = (ms: number): number => (ms <= 1 ? 0 : Math.ceil(Math.log(ms) / Math.log(GROWTH) - 1e-9));
export const upperBound = (i: number): number => GROWTH ** i;

export function record(h: Histogram, ms: number): void {
  if (!(ms >= 0)) throw new Error(`זמן לא תקין: ${ms}`);
  const i = bucketOf(ms);
  h.counts[i] = (h.counts[i] ?? 0) + 1;
  h.total++;
}

export function merge(...hs: Histogram[]): Histogram {
  const out = createHistogram();
  for (const h of hs) {
    h.counts.forEach((c, i) => {
      if (c) out.counts[i] = (out.counts[i] ?? 0) + c;
    });
    out.total += h.total;
  }
  return out;
}

export function histPercentile(h: Histogram, p: number): number {
  if (h.total === 0) throw new Error("histogram ריק");
  const rank = Math.ceil((p / 100) * h.total);
  let seen = 0;
  for (let i = 0; i < h.counts.length; i++) {
    seen += h.counts[i] ?? 0;
    if (seen >= rank) return Math.round(upperBound(i) * 100) / 100;
  }
  throw new Error("לא אמור לקרות");
}
```

```ts save=solutions/P28.5-E08.ts
// P28.5-E08: מחולל עומס קטן — concurrency קבוע, ו־percentiles של זמני תגובה
import { performance } from "node:perf_hooks";

export interface LoadResult {
  total: number;
  errors: number;
  durationMs: number;
  rps: number;
  latency: { p50: number; p95: number; p99: number; max: number };
}

const pct = (s: number[], p: number) => s[Math.max(0, Math.ceil((p / 100) * s.length) - 1)];

export async function runLoad(request: () => Promise<unknown>, { concurrency, total }: { concurrency: number; total: number }): Promise<LoadResult> {
  const latencies: number[] = [];
  let errors = 0;
  let next = 0;
  const started = performance.now();
  // concurrency "עובדים", כל אחד שולח בקשה, מחכה לתשובה, ושולח את הבאה — כמו concurrency משתמשים
  async function worker() {
    while (next < total) {
      next++;
      const t0 = performance.now();
      try {
        await request();
      } catch {
        errors++;
      }
      latencies.push(performance.now() - t0);
    }
  }
  await Promise.all(Array.from({ length: Math.min(concurrency, total) }, worker));
  const durationMs = performance.now() - started;
  latencies.sort((a, b) => a - b);
  const r = (x: number) => Math.round(x * 10) / 10;
  return {
    total,
    errors,
    durationMs: r(durationMs),
    rps: r((total / durationMs) * 1000),
    latency: { p50: r(pct(latencies, 50)), p95: r(pct(latencies, 95)), p99: r(pct(latencies, 99)), max: r(latencies.at(-1)!) },
  };
```

```ts save=solutions/P28.5-E09.ts
// L = λ × W : כמה בקשות "בתוך המערכת" בממוצע = קצב הגעה × זמן שהייה
export const inFlight = (rps: number, latencyMs: number): number => (rps * latencyMs) / 1000;

export function instancesNeeded({ peakRps, latencyMs, perInstance, headroom = 0.3, min = 2 }: { peakRps: number; latencyMs: number; perInstance: number; headroom?: number; min?: number }): number {
  if (!(headroom >= 0 && headroom < 1)) throw new Error("headroom בין 0 ל־1");
  const usable = perInstance * (1 - headroom); // לא מתכננים ל־100% — שם ה־latency מתפוצץ
  return Math.max(min, Math.ceil(inFlight(peakRps, latencyMs) / usable));
}

// כמה חיבורים למסד כל instance מחזיק, ואם כולם יחד חורגים מה־max_connections
export function checkDbConnections({ instances, poolPerInstance, maxConnections, reserved = 10 }: { instances: number; poolPerInstance: number; maxConnections: number; reserved?: number }) {
  const needed = instances * poolPerInstance;
  const available = maxConnections - reserved; // לאדמין, migrations, monitoring
  return { needed, available, ok: needed <= available, maxPoolPerInstance: Math.floor(available / instances) };
}
```

```ts save=solutions/P28.5-E10.ts
// P28.5-E10: Transform שמפרק NDJSON לאובייקטים — גם כש־chunk נחתך באמצע שורה, ובאמצע תו
import { Transform, type TransformCallback } from "node:stream";
import { StringDecoder } from "node:string_decoder";

export function ndjsonParser(): Transform {
  const decoder = new StringDecoder("utf8"); // מחזיק בתים של תו שנחתך עד שה־chunk הבא משלים אותו
  let buffer = "";
  let lineNo = 0;
  function pushLine(this: Transform, line: string): Error | null {
    lineNo++;
    if (line.trim() === "") return null;
    try {
      this.push(JSON.parse(line));
      return null;
    } catch {
      return new Error(`שורה ${lineNo}: JSON לא תקין`);
    }
  }
  return new Transform({
    readableObjectMode: true,
    transform(chunk: Buffer, _enc, cb: TransformCallback) {
      buffer += decoder.write(chunk);
      const lines = buffer.split("\n");
      buffer = lines.pop()!; // השורה האחרונה אולי עוד לא הסתיימה
      for (const l of lines) {
        const err = pushLine.call(this, l.replace(/\r$/, ""));
        if (err) return cb(err);
      }
      cb();
    },
    flush(cb: TransformCallback) {
      buffer += decoder.end();
      cb(pushLine.call(this, buffer.replace(/\r$/, "")) ?? undefined); // שורה אחרונה בלי \n
    },
  });
}
```

```ts save=solutions/P28.5-E13.ts
// P28.5-E13: עבודת CPU ב־worker threads — כדי שה־event loop ימשיך לענות
import { Worker } from "node:worker_threads";

// הקוד שרץ בכל worker (מוכן): מקבל [from, to] ומחזיר כמה ראשוניים יש בטווח
export const WORKER_CODE = `
const { parentPort } = require("node:worker_threads");
parentPort.on("message", ([from, to]) => {
  let count = 0;
  for (let n = Math.max(2, from); n < to; n++) {
    let prime = true;
    for (let d = 2; d * d <= n; d++) if (n % d === 0) { prime = false; break; }
    if (prime) count++;
  }
  parentPort.postMessage(count);
});`;

export async function countPrimesParallel(limit: number, { threads = 4, chunks = 16 } = {}): Promise<number> {
  const size = Math.ceil(limit / chunks);
  const ranges: [number, number][] = Array.from({ length: chunks }, (_, i) => [i * size, Math.min(limit, (i + 1) * size)]);
  let next = 0;
  let total = 0;
  const workers = Array.from({ length: Math.min(threads, ranges.length) }, () => new Worker(WORKER_CODE, { eval: true }));
  try {
    // כל worker לוקח את הטווח הבא כשהוא מסיים — כך אף אחד לא יושב בטל
    await Promise.all(
      workers.map(
        (w) =>
          new Promise<void>((resolve, reject) => {
            const giveNext = () => {
              if (next >= ranges.length) return resolve();
              w.postMessage(ranges[next++]);
            };
            w.on("message", (count: number) => {
              total += count;
              giveNext();
            });
            w.on("error", reject);
            giveNext();
          }),
      ),
    );
  } finally {
    await Promise.all(workers.map((w) => w.terminate()));
  }
  return total;
}
```

```ts save=solutions/P28.5-E14.ts
// P28.5-E14: מוניטור ל־event loop — מתריע כשמשהו חוסם
import { monitorEventLoopDelay } from "node:perf_hooks";

export interface LagReport {
  p99Ms: number;
  maxMs: number;
}

export function startLagMonitor({ thresholdMs, intervalMs = 1000, onLag }: { thresholdMs: number; intervalMs?: number; onLag: (r: LagReport) => void }): () => void {
  const h = monitorEventLoopDelay({ resolution: 10 });
  h.enable();
  const timer = setInterval(() => {
    const report = { p99Ms: Math.round(h.percentile(99) / 1e6), maxMs: Math.round(h.max / 1e6) }; // ננו־שניות ← מילי
    h.reset(); // כל חלון נמדד בנפרד
    if (report.maxMs >= thresholdMs) onLag(report);
  }, intervalMs);
  timer.unref(); // מוניטור לא אמור להשאיר את התהליך חי
  return () => {
    clearInterval(timer);
    h.disable();
  };
}
```

```ts save=solutions/P28.5-E16.ts
      SELECT *, strftime(timezone('Asia/Jerusalem', created_at), '%Y-%m-%d') AS day
```

```ts save=solutions/P28.5-E17.ts
export interface TableStats {
  rows: number;
  columns: Record<string, number>; // ממוצע בתים לערך, לכל עמודה
  partitionedBy?: string; // עמודת תאריך
  days?: number; // כמה ימים יש בטבלה (פרטישן לכל יום)
}

export interface QueryShape {
  columns: string[] | "*";
  days?: number; // WHERE על עמודת ה־partition: כמה ימים נבחרו
  limit?: number; // לא משנה את הסריקה!
}

const TiB = 2 ** 40;

export function estimateScan(t: TableStats, q: QueryShape, { pricePerTiB = 6.25 } = {}) {
  const cols = q.columns === "*" ? Object.keys(t.columns) : q.columns;
  for (const c of cols) if (!(c in t.columns)) throw new Error(`עמודה לא קיימת: ${c}`);
  const bytesPerRow = cols.reduce((s, c) => s + t.columns[c], 0); // עמודתי: רק העמודות שנבחרו
  const fraction = t.partitionedBy && t.days && q.days !== undefined ? Math.min(1, q.days / t.days) : 1; // partition pruning
  const bytes = Math.round(t.rows * fraction * bytesPerRow);
  return { bytes, costUsd: Math.round((bytes / TiB) * pricePerTiB * 100) / 100 };
}
```

```ts save=solutions/P28.5-E18.ts
// P28.5-E18: HyperLogLog — כמה ערכים שונים, ב־4KB זיכרון
import { createHash } from "node:crypto";

export interface HLL {
  p: number;
  registers: Uint8Array;
}

export const createHLL = (p = 12): HLL => ({ p, registers: new Uint8Array(1 << p) });

export const hash32 = (value: string | number): number => createHash("sha1").update(String(value)).digest().readUInt32BE(0);

export function add(h: HLL, value: string | number): void {
  const x = hash32(value);
  const idx = x >>> (32 - h.p); // p הביטים העליונים ← איזה רגיסטר
  const w = (x << h.p) >>> 0; // השאר
  const rho = w === 0 ? 32 - h.p + 1 : Math.clz32(w) + 1; // מיקום ה־1 הראשון
  if (rho > h.registers[idx]) h.registers[idx] = rho;
}

export function count(h: HLL): number {
  const m = h.registers.length;
  let sum = 0;
  let zeros = 0;
  for (const r of h.registers) {
    sum += 2 ** -r;
    if (r === 0) zeros++;
  }
  const alpha = 0.7213 / (1 + 1.079 / m);
  const e = (alpha * m * m) / sum;
  if (e <= 2.5 * m && zeros > 0) return Math.round(m * Math.log(m / zeros)); // מעט ערכים: linear counting
  return Math.round(e);
}

export function mergeHLL(a: HLL, b: HLL): HLL {
  if (a.p !== b.p) throw new Error("p שונה");
  const out = createHLL(a.p);
  for (let i = 0; i < out.registers.length; i++) out.registers[i] = Math.max(a.registers[i], b.registers[i]);
  return out;
}
```

```ts save=solutions/P28.5-E19.ts
// P28.5-E19: reservoir sampling — דגימה אחידה מזרם שלא יודעים את אורכו
export async function reservoirSample<T>(items: Iterable<T> | AsyncIterable<T>, k: number, random: () => number = Math.random): Promise<T[]> {
  if (!Number.isInteger(k) || k < 1) throw new Error("k חייב להיות שלם חיובי");
  const sample: T[] = [];
  let seen = 0;
  for await (const item of items) {
    seen++;
    if (sample.length < k) sample.push(item);
    else {
      const j = Math.floor(random() * seen); // 0..seen-1
      if (j < k) sample[j] = item; // הסיכוי של הפריט ה־n להיכנס: k/n
    }
  }
  return sample;
}
```

```ts save=solutions/P28.5-E24.ts
// P28.5-E24: API אנליטיקה מעל Parquet — DuckDB בתוך התהליך, cache, ו־p95 נמוך תחת עומס
import http from "node:http";
import { DuckDBInstance } from "@duckdb/node-api";

const DAY = /^\d{4}-\d{2}-\d{2}$/;
const isDay = (s: string | null): s is string => !!s && DAY.test(s) && !Number.isNaN(Date.parse(`${s}T00:00:00Z`)) && new Date(`${s}T00:00:00Z`).toISOString().startsWith(s);
const daysBetween = (a: string, b: string) => (Date.parse(`${b}T00:00:00Z`) - Date.parse(`${a}T00:00:00Z`)) / 86_400_000;

class TtlLru<V> {
  private map = new Map<string, { value: V; expires: number }>();
  private max: number;
  private ttlMs: number;
  private now: () => number;
  constructor(max: number, ttlMs: number, now: () => number) {
    this.max = max;
    this.ttlMs = ttlMs;
    this.now = now;
  }
  get(k: string): V | undefined {
    const e = this.map.get(k);
    if (!e) return undefined;
    if (e.expires <= this.now()) {
      this.map.delete(k);
      return undefined;
    }
    this.map.delete(k); // LRU: הכי חדש בסוף
    this.map.set(k, e);
    return e.value;
  }
  set(k: string, value: V) {
    this.map.delete(k);
    this.map.set(k, { value, expires: this.now() + this.ttlMs });
    if (this.map.size > this.max) this.map.delete(this.map.keys().next().value!);
  }
}

  await conn.run(
    `CREATE TABLE orders AS SELECT *, strftime(timezone('Asia/Jerusalem', created_at), '%Y-%m-%d') AS day FROM read_parquet($1)`,
    [parquetPath],
  );
  const cache = new TtlLru<string>(cacheSize, cacheTtlMs, now);

  async function revenue(from: string, to: string) {
    const r = await conn.runAndReadAll(
      `SELECT day, count(*)::INTEGER AS orders, round(sum(quantity * price_cents) / 100.0, 2)::DOUBLE AS revenue
       FROM orders WHERE day BETWEEN $1 AND $2 GROUP BY day ORDER BY day`,
      [from, to],
    );
    const days = r.getRowObjects().map((x) => ({ day: String(x.day), orders: Number(x.orders), revenue: Number(x.revenue) }));
    const total = { orders: days.reduce((s, d) => s + d.orders, 0), revenue: Math.round(days.reduce((s, d) => s + d.revenue * 100, 0)) / 100 };
    return { from, to, days, total };
  }
  async function topProducts(day: string, limit: number) {
    const r = await conn.runAndReadAll(
      `SELECT product, sum(quantity)::INTEGER AS units, round(sum(quantity * price_cents) / 100.0, 2)::DOUBLE AS revenue
       FROM orders WHERE day = $1 GROUP BY product ORDER BY revenue DESC, product LIMIT $2`,
      [day, limit],
    );
    return r.getRowObjects().map((x) => ({ product: String(x.product), units: Number(x.units), revenue: Number(x.revenue) }));
  }

  const send = (res: http.ServerResponse, status: number, body: unknown, cacheState?: string) => {
    res.statusCode = status;
    res.setHeader("content-type", "application/json");
    if (cacheState) res.setHeader("x-cache", cacheState);
    res.end(typeof body === "string" ? body : JSON.stringify(body));
  };
```

```ts save=solutions/P28.5-E25.ts
// P28.5-E25: סנכרון מצטבר מ־PostgreSQL ל־data lake (קובצי Parquet), ודוח שמתעלם מגרסאות ישנות
import fs from "node:fs/promises";
import path from "node:path";
import type { Pool } from "pg";
import type { DuckDBConnection } from "@duckdb/node-api";

interface State {
  seq: number;
  updatedAt: string | null; // ה־watermark: (updated_at, id) של השורה האחרונה שיוצאה
  id: number;
}

const q = (s: string) => `'${s.replaceAll("'", "''")}'`;

async function readState(dir: string): Promise<State> {
  try {
    return JSON.parse(await fs.readFile(path.join(dir, "_state.json"), "utf8"));
  } catch {
    return { seq: 0, updatedAt: null, id: 0 };
  }
}

export async function syncOrders({ pool, conn, lakeDir, pageSize = 1000 }: { pool: Pool; conn: DuckDBConnection; lakeDir: string; pageSize?: number }): Promise<{ exported: number; file: string | null }> {
  await fs.mkdir(lakeDir, { recursive: true });
  const state = await readState(lakeDir);
  const rows: any[] = [];
  let cursor = { updatedAt: state.updatedAt, id: state.id };
  for (;;) {
    // keyset על (updated_at, id): שורות עם אותו updated_at לא הולכות לאיבוד בין עמודים
    const { rows: page } = await pool.query(
      `SELECT id, customer_id, total_cents, status, created_at, updated_at, updated_at::text AS updated_at_text
       FROM orders
       WHERE $1::timestamptz IS NULL OR (updated_at, id) > ($1::timestamptz, $2::int)
       ORDER BY updated_at, id LIMIT $3`,
      [cursor.updatedAt, cursor.id, pageSize],
    );
    rows.push(...page);
    if (page.length < pageSize) break;
    const last = page.at(-1);
    cursor = { updatedAt: last.updated_at_text, id: last.id };
  }
  if (rows.length === 0) return { exported: 0, file: null }; // אין חדש ← אין קובץ ריק

  await conn.run(`CREATE OR REPLACE TEMP TABLE batch (id INTEGER, customer_id INTEGER, total_cents INTEGER, status VARCHAR, created_at TIMESTAMPTZ, updated_at TIMESTAMPTZ)`);
  const app = await conn.createAppender("batch");
  for (const r of rows) {
    app.appendInteger(r.id);
    app.appendInteger(r.customer_id);
    app.appendInteger(r.total_cents);
    app.appendVarchar(r.status);
    app.appendVarchar(r.created_at.toISOString());
    app.appendVarchar(r.updated_at_text);
    app.endRow();
  }
  app.closeSync();
  const seq = state.seq + 1;
  const file = path.join(lakeDir, `part-${String(seq).padStart(5, "0")}.parquet`);
  await conn.run(`COPY batch TO ${q(file)} (FORMAT parquet)`);
  // ה־state נכתב רק אחרי שהקובץ קיים; ובכתיבה אטומית (קובץ זמני ← rename)
  const last = rows.at(-1);
  const tmp = path.join(lakeDir, "_state.json.tmp");
  await fs.writeFile(tmp, JSON.stringify({ seq, updatedAt: last.updated_at_text, id: last.id }));
  await fs.rename(tmp, path.join(lakeDir, "_state.json"));
  return { exported: rows.length, file };
}

export async function lakeReport(conn: DuckDBConnection, lakeDir: string) {
  const r = await conn.runAndReadAll(`
    WITH latest AS (
      SELECT * FROM read_parquet(${q(path.join(lakeDir, "part-*.parquet"))})
      QUALIFY row_number() OVER (PARTITION BY id ORDER BY updated_at DESC) = 1 -- הגרסה האחרונה של כל הזמנה
    )
    SELECT strftime(timezone('Asia/Jerusalem', created_at), '%Y-%m-%d') AS day,
           count(*)::INTEGER AS orders, (sum(total_cents) / 100.0)::DOUBLE AS revenue
    FROM latest WHERE status <> 'cancelled'
    GROUP BY day ORDER BY day`);
  return r.getRowObjects().map((x) => ({ day: String(x.day), orders: Number(x.orders), revenue: Number(x.revenue) }));
}
```

```ts run save=solutions/P28.5-E26.ts
// P28.5-E26 — הנפוצים ביותר בזרם, בזיכרון קבוע (Space-Saving)
export type Counted<T> = { item: T; count: number; error: number };

export class TopK<T> {
  #k: number;
  #counters = new Map<T, { count: number; error: number }>();
  #seen = 0;

  constructor(k: number) {
    if (!Number.isInteger(k) || k < 1) throw new RangeError("k חייב להיות מספר שלם חיובי");
    this.#k = k;
  }

  add(item: T): void {
    this.#seen++;
    const counter = this.#counters.get(item);
    if (counter) {
      counter.count++;
      return;
    }
    if (this.#counters.size < this.#k) {
      this.#counters.set(item, { count: 1, error: 0 });
      return;
    }
    // מחליפים את המונה הקטן ביותר: החדש "יורש" את הספירה שלו (וזו גם הטעות האפשרית)
    let minItem: T | undefined;
    let min = Infinity;
    for (const [key, c] of this.#counters) {
      if (c.count < min) {
        min = c.count;
        minItem = key;
      }
    }
    this.#counters.delete(minItem as T);
    this.#counters.set(item, { count: min + 1, error: min });
  }

  get seen(): number {
    return this.#seen;
  }

  top(n = this.#k): Counted<T>[] {
    return [...this.#counters]
      .map(([item, c]) => ({ item, count: c.count, error: c.error }))
      .sort((a, b) => b.count - a.count || String(a.item).localeCompare(String(b.item)))
      .slice(0, n);
  }
}
```

```sql run db=28w-scale/files/views-days.sql save=solutions/P28.5-E27.sql
WITH daily AS (
  SELECT viewed_at::date AS day, COUNT(*) AS views, COUNT(DISTINCT user_id) AS users
  FROM page_views
  GROUP BY day
),
by_path AS (
  SELECT
    viewed_at::date AS day,
    path,
    COUNT(*) AS path_views,
    ROW_NUMBER() OVER (PARTITION BY viewed_at::date ORDER BY COUNT(*) DESC, path) AS rn
  FROM page_views
  GROUP BY viewed_at::date, path
)
SELECT d.day, d.views, d.users, p.path AS top_path, p.path_views AS top_path_views
FROM daily d
JOIN by_path p ON p.day = d.day AND p.rn = 1
ORDER BY d.day;
```

```ts run save=solutions/P28.5-E28.ts
// P28.5-E28 — דוח של בדיקת עומס
export type Result = { name: string; requests: number; seconds: number; p50: number; p99: number };

const fmt = new Intl.NumberFormat("en-US");

export function report(results: Result[]): string {
  if (results.length === 0) return "";
  const rows = results
    .map((r) => ({ ...r, rps: r.requests / r.seconds }))
    .sort((a, b) => b.rps - a.rps);
  const best = rows[0].rps;
  const cells = rows.map((r) => [
    r.name,
    fmt.format(Math.round(r.rps)),
    r.p50.toFixed(1),
    r.p99.toFixed(1),
    r.rps === best ? "fastest" : `${(best / r.rps).toFixed(2)}x slower`,
  ]);
  const header = ["name", "req/s", "p50 ms", "p99 ms", ""];
  const widths = header.map((h, i) => Math.max(h.length, ...cells.map((c) => c[i].length)));
  const line = (c: string[]) =>
    c
      .map((v, i) => (i === 0 || i === c.length - 1 ? v.padEnd(widths[i]) : v.padStart(widths[i])))
      .join("  ")
      .trimEnd();
  return [line(header), ...cells.map(line)].join("\n");
}

console.log(
  report([
    { name: "express + pg", requests: 182_400, seconds: 30, p50: 4.2, p99: 31.75 },
    { name: "fastify + pg", requests: 297_300, seconds: 30, p50: 2.61, p99: 18.4 },
    { name: "express + cache", requests: 1_204_500, seconds: 30, p50: 0.8, p99: 6.04 },
  ]),
);
```

<p>‏<code>trimEnd</code> בסוף כל שורה: העמודה האחרונה מרופדת לרוחב הארוך שלה (<code>6.60x slower</code>), ובשורת הכותרת היא ריקה. בלי <code>trimEnd</code> נשארים רווחים בסוף שורות, ו־git מסמן אותם באדום ב־diff.</p>
@mistakes
