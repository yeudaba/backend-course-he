```ts save=solutions/P19-E10.ts
// P19-E10
import type { Prisma, PrismaClient } from "../files/generated/prisma/client.ts";

export async function searchProducts(
  prisma: PrismaClient,
  { q, minPrice, maxPrice, page = 1, limit = 10 }: { q?: string; minPrice?: number; maxPrice?: number; page?: number; limit?: number } = {}
) {
  const where: Prisma.ProductWhereInput = {
    isActive: true,
    ...(q ? { name: { contains: q, mode: "insensitive" } } : {}),
    price: { gte: minPrice, lte: maxPrice },
  };
  const [rows, total] = await Promise.all([
    prisma.product.findMany({
      where,
      select: { id: true, name: true, price: true },
      orderBy: [{ price: "asc" }, { id: "asc" }],
      skip: (page - 1) * limit,
      take: limit,
    }),
    prisma.product.count({ where }),
  ]);
  return { data: rows.map((p) => ({ ...p, price: Number(p.price) })), total };
}
```

```ts save=solutions/P19-E12.ts
  constructor(code: string, message = code) {
    super(message);
    this.code = code;
```

```ts save=solutions/P19-E14.ts
export async function placeOrder(prisma: PrismaClient, customerId: number, items: { productId: number; quantity: number }[]) {
  return prisma.$transaction(async (tx) => {
    const lines = [];
    for (const { productId, quantity } of items) {
      const { count } = await tx.product.updateMany({
        where: { id: productId, stock: { gte: quantity } },
        data: { stock: { decrement: quantity } },
      });
      if (count === 0) throw new AppError("OUT_OF_STOCK");
      const { price } = await tx.product.findUniqueOrThrow({ where: { id: productId }, select: { price: true } });
      lines.push({ productId, quantity, unitPrice: price });

    const order = await tx.order.create({ data: { customerId, status: "pending", items: { create: lines } } });
```

```ts save=solutions/P19-E23.ts
import type { PrismaClient } from "../files/generated/prisma/client.ts";

export class AppError extends Error {
  code: string;
  constructor(code: string, message = code) {
    super(message);
    this.code = code;
  }
}

export async function transferStock(prisma: PrismaClient, fromProductId: number, toProductId: number, qty: number) {
  await prisma.$transaction(async (tx) => {
    await tx.product.update({ where: { id: toProductId }, data: { stock: { increment: qty } } });
    const { count } = await tx.product.updateMany({
      where: { id: fromProductId, stock: { gte: qty } },
      data: { stock: { decrement: qty } },
    });
    if (count === 0) throw new AppError("OUT_OF_STOCK");
  });
}
```

```ts save=solutions/P19-E24.ts
// P19-E24
import { Prisma, type PrismaClient } from "../files/generated/prisma/client.ts";

export class AppError extends Error {
  code: string;
  constructor(code: string, message = code) {
    super(message);
    this.code = code;
  }
}

export interface Product { id: number; name: string; category: string; price: number; stock: number; isActive: boolean }
export interface ListOptions { category?: string; q?: string; inStock?: boolean; sort?: "price" | "-price" | "name"; page?: number; limit?: number }

export interface ProductsRepository {
  list(options: ListOptions): Promise<{ data: Product[]; total: number }>;
  get(id: number): Promise<Product | undefined>;
  create(data: { name: string; category: string; price: number; stock: number }): Promise<Product>;
  update(id: number, changes: Partial<Omit<Product, "id">>): Promise<Product | undefined>;
  remove(id: number): Promise<boolean>;
}

const withCategory = { category: { select: { name: true } } } as const;
type Row = Prisma.ProductGetPayload<{ include: typeof withCategory }>;

const toProduct = (r: Row): Product => ({
  id: r.id,
  name: r.name,
  category: r.category.name,
  price: Number(r.price),
  stock: r.stock,
  isActive: r.isActive,
});

const code = (err: unknown) => (err instanceof Prisma.PrismaClientKnownRequestError ? err.code : undefined);

export class PrismaProductsRepository implements ProductsRepository {
  prisma: PrismaClient;
  constructor(prisma: PrismaClient) {
    this.prisma = prisma;
  }

  async list({ category, q, inStock, sort = "name", page = 1, limit = 10 }: ListOptions) {
    const where: Prisma.ProductWhereInput = {
      ...(category ? { category: { name: category } } : {}),
      ...(q ? { name: { contains: q, mode: "insensitive" } } : {}),
      ...(inStock ? { stock: { gt: 0 } } : {}),
    };
    const orderBy: Prisma.ProductOrderByWithRelationInput[] = [
      sort === "name" ? { name: "asc" } : { price: sort === "price" ? "asc" : "desc" },
      { id: "asc" },
    ];
    const [rows, total] = await Promise.all([
      this.prisma.product.findMany({ where, orderBy, include: withCategory, skip: (page - 1) * limit, take: limit }),
      this.prisma.product.count({ where }),
    ]);
    return { data: rows.map(toProduct), total };
  }

  async get(id: number) {
    const row = await this.prisma.product.findUnique({ where: { id }, include: withCategory });
    return row ? toProduct(row) : undefined;
  }

  async create({ name, category, price, stock }: { name: string; category: string; price: number; stock: number }) {
    try {
      const row = await this.prisma.product.create({
        data: { name, price, stock, category: { connect: { name: category } } },
        include: withCategory,
      });
      return toProduct(row);
    } catch (err) {
      if (code(err) === "P2025") throw new AppError("CATEGORY_NOT_FOUND");
      throw err;
    }
  }

  async update(id: number, { category, ...rest }: Partial<Omit<Product, "id">>) {
    const exists = await this.prisma.product.findUnique({ where: { id }, select: { id: true } });
    if (!exists) return undefined;
    try {
      const row = await this.prisma.product.update({
        where: { id },
        data: { ...rest, ...(category ? { category: { connect: { name: category } } } : {}) },
        include: withCategory,
      });
      return toProduct(row);
    } catch (err) {
      if (code(err) === "P2025") throw new AppError("CATEGORY_NOT_FOUND");
      throw err;
    }
  }

  async remove(id: number) {
    try {
      await this.prisma.product.delete({ where: { id } });
      return true;
    } catch (err) {
      if (code(err) === "P2025") return false;
      if (code(err) === "P2003") throw new AppError("PRODUCT_HAS_ORDERS");
      throw err;
    }
  }
}
```

```ts save=solutions/P19-E25.ts
// P19-E25
import type { PrismaClient } from "../files/generated/prisma/client.ts";

export async function monthlyReport(prisma: PrismaClient, year: number) {
  return prisma.$queryRaw<{ month: string; orders: number; revenue: number; runningTotal: number }[]>`
    WITH sales AS (
      SELECT EXTRACT(MONTH FROM o.created_at AT TIME ZONE 'UTC')::int AS m,
             COUNT(DISTINCT o.id) AS orders,
             SUM(oi.quantity * oi.unit_price) AS revenue
      FROM orders o
      JOIN order_items oi ON oi.order_id = o.id
      WHERE o.status <> 'cancelled'
        AND o.created_at >= make_timestamptz(${year}::int, 1, 1, 0, 0, 0, 'UTC')
        AND o.created_at <  make_timestamptz(${year}::int + 1, 1, 1, 0, 0, 0, 'UTC')
      GROUP BY 1
    )
    SELECT to_char(make_date(${year}::int, g.m, 1), 'YYYY-MM') AS month,
           COALESCE(s.orders, 0)::int AS orders,
           COALESCE(s.revenue, 0)::float AS revenue,
           (SUM(COALESCE(s.revenue, 0)) OVER (ORDER BY g.m))::float AS "runningTotal"
    FROM generate_series(1, 12) AS g(m)
    LEFT JOIN sales s ON s.m = g.m
    ORDER BY g.m`;
}
```

```prisma save=solutions/P19-E26.prisma
// P19-E26: הספרייה (P17-E26) כ־schema של Prisma
datasource db {
  provider = "postgresql"
}

model Author {
  id       Int          @id @default(autoincrement())
  fullName String       @map("full_name")
  books    BookAuthor[]

  @@map("authors")
}

model Book {
  id            Int          @id @default(autoincrement())
  title         String
  isbn          String       @unique
  publishedYear Int?         @map("published_year")
  authors       BookAuthor[]
  loans         Loan[]

  @@map("books")
}

model BookAuthor {
  bookId   Int    @map("book_id")
  book     Book   @relation(fields: [bookId], references: [id], onDelete: Cascade)
  authorId Int    @map("author_id")
  author   Author @relation(fields: [authorId], references: [id])

  @@id([bookId, authorId])
  @@map("book_authors")
}

model Member {
  id       Int    @id @default(autoincrement())
  fullName String @map("full_name")
  email    String @unique
  loans    Loan[]

  @@map("members")
}

model Loan {
  id         Int       @id @default(autoincrement())
  bookId     Int       @map("book_id")
  book       Book      @relation(fields: [bookId], references: [id])
  memberId   Int       @map("member_id")
  member     Member    @relation(fields: [memberId], references: [id])
  loanedOn   DateTime  @default(now()) @map("loaned_on") @db.Date
  returnedOn DateTime? @map("returned_on") @db.Date

  @@index([memberId])
  @@map("loans")
}
```
<p>שלוש שורות ב־schema שוות כאן הרבה SQL: ‏<code>@relation</code> הופך ל־<code>FOREIGN KEY</code>, ‏<code>@unique</code> ל־<code>UNIQUE INDEX</code>, ו־<code>@@id</code> ל־<code>PRIMARY KEY</code> על שתי עמודות. בלי <code>onDelete</code>, Prisma כותב <code>ON DELETE RESTRICT</code>, ולכן היסטוריית ההשאלות מוגנת כמו ב־P17-E26.</p>
<p>השדות <code>books BookAuthor[]</code> ו־<code>loans Loan[]</code> לא יוצרים עמודות. הם קיימים רק ב־Prisma Client, כדי שתוכל לכתוב <code>include: { loans: true }</code>. העמודה עצמה (<code>book_id</code>) נמצאת תמיד בצד ה"רבים".</p>
<p>מה <strong>לא</strong> עבר מ־P17-E26? ה־<code>CHECK (returned_on &gt;= loaned_on)</code>. ב־schema של Prisma אין דרך לכתוב CHECK. בפרויקט אמיתי יוצרים migration עם <code>--create-only</code>, ומוסיפים את ה־CHECK ביד לקובץ ה־SQL (שיעור 2). עוד הבדל: Prisma יוצר <code>SERIAL</code> ולא <code>GENERATED ALWAYS AS IDENTITY</code>.</p>
<p>ה־<code>@@index([memberId])</code> לא חובה כאן, אבל הוא הרגל טוב. PostgreSQL לא יוצר index על foreign key לבד (‏P18-E16), ו"כל ההשאלות של חבר" היא שאילתה נפוצה.</p>
@mistakes
<ul><li>לשכוח <code>@@map("books")</code> — הטבלה נקראת <code>"Book"</code> (באות גדולה ובמירכאות), וכל SQL שכתבת עד עכשיו לא מוצא אותה.</li><li><code>onDelete: Cascade</code> גם ב־<code>Loan</code> — מחיקת ספר מוחקת בשקט את ההיסטוריה שלו.</li><li><code>DateTime</code> בלי <code>@db.Date</code> — העמודה היא <code>timestamp(3)</code>, ו"השאלות של היום" צריכות טווח שעות.</li><li><code>id</code> נוסף ב־<code>BookAuthor</code> בלי <code>@@id([bookId, authorId])</code> — אותו קישור יכול להופיע פעמיים.</li></ul>

```ts save=solutions/P19-E27.ts
// P19-E27: עמוד מוצרים — רשימה וספירה במקביל
import type { PrismaClient } from "../files/generated/prisma/client.ts";

export type ProductPage = {
  items: { id: number; name: string; price: number }[];
  page: number;
  total: number;
  pages: number;
};

export async function getProductPage(
  prisma: PrismaClient,
  { page = 1, pageSize = 5, categoryId }: { page?: number; pageSize?: number; categoryId?: number } = {},
): Promise<ProductPage> {
  if (!Number.isInteger(page) || page < 1) throw new RangeError("page חייב להיות מספר שלם חיובי");
  if (!Number.isInteger(pageSize) || pageSize < 1 || pageSize > 50) throw new RangeError("pageSize חייב להיות 1–50");

  const where = { isActive: true, ...(categoryId !== undefined ? { categoryId } : {}) };
  const [rows, total] = await Promise.all([
    prisma.product.findMany({
      where,
      select: { id: true, name: true, price: true },
      orderBy: { id: "asc" },
      skip: (page - 1) * pageSize,
      take: pageSize,
    }),
    prisma.product.count({ where }),
  ]);

  return {
    items: rows.map((p) => ({ id: p.id, name: p.name, price: Number(p.price) })),
    page,
    total,
    pages: Math.ceil(total / pageSize),
  };
}
```
<p><code>Promise.all</code> שולח את שתי השאילתות יחד, וה־destructuring מחלק את התוצאות לפי המקום במערך: הראשונה היא הרשימה והשנייה הספירה. זמן התשובה הוא של השאילתה האיטית, ולא סכום שתיהן.</p>
<p>ה־<code>where</code> מוגדר פעם אחת ומשמש את שתיהן. אילו כתבת אותו פעמיים, יום אחד מישהו היה מוסיף תנאי רק לאחת, ו"עמוד 2 מתוך 3" היה מראה מספרים שלא מתאימים לרשימה.</p>
<p>אפשר גם <code>prisma.$transaction([findMany, count])</code>. שתי השאילתות רצות אז באותה transaction, ולכן רואות את אותם נתונים בדיוק, אבל אחת אחרי השנייה על אותו חיבור. לרשימת מוצרים, <code>Promise.all</code> מספיק.</p>
@mistakes
<ul><li><code>const rows = await findMany(...); const total = await count(...);</code> — עובד, אבל אחת אחרי השנייה.</li><li><code>pages: total / pageSize</code> — ‏<code>2.2</code> עמודים. ‏<code>Math.ceil</code>.</li><li><code>count()</code> בלי <code>where</code> — סופר גם מוצרים לא פעילים ומקטגוריות אחרות.</li><li><code>skip: page * pageSize</code> — עמוד 1 מדלג על חמשת הראשונים.</li></ul>

```ts save=solutions/P19-E28.ts
// P19-E28: repository לקטגוריות — interface, מחלקה ושגיאות משלך
import { Prisma, type PrismaClient } from "../files/generated/prisma/client.ts";

export type CategorySummary = { id: number; name: string; productCount: number };

export interface CategoryRepository {
  list(): Promise<CategorySummary[]>;
  create(name: string): Promise<CategorySummary>;
  rename(id: number, name: string): Promise<CategorySummary>;
}

export class ConflictError extends Error {
  constructor(message: string) {
    super(message);
    this.name = "ConflictError";
  }
}

export class NotFoundError extends Error {
  constructor(message: string) {
    super(message);
    this.name = "NotFoundError";
  }
}

const withCount = { id: true, name: true, _count: { select: { products: true } } } as const;
type Row = { id: number; name: string; _count: { products: number } };
const toSummary = (c: Row): CategorySummary => ({ id: c.id, name: c.name, productCount: c._count.products });

export class PrismaCategoryRepository implements CategoryRepository {
  #prisma: PrismaClient;

  constructor(prisma: PrismaClient) {
    this.#prisma = prisma;
  }

  async list(): Promise<CategorySummary[]> {
    const rows = await this.#prisma.category.findMany({ select: withCount, orderBy: { name: "asc" } });
    return rows.map(toSummary);
  }

  async create(name: string): Promise<CategorySummary> {
    const clean = this.#cleanName(name);
    try {
      return toSummary(await this.#prisma.category.create({ data: { name: clean }, select: withCount }));
    } catch (err) {
      throw this.#translate(err, clean);
    }
  }

  async rename(id: number, name: string): Promise<CategorySummary> {
    const clean = this.#cleanName(name);
    try {
      return toSummary(await this.#prisma.category.update({ where: { id }, data: { name: clean }, select: withCount }));
    } catch (err) {
      throw this.#translate(err, clean, id);
    }
  }

  #cleanName(name: string): string {
    const clean = name.trim();
    if (clean === "") throw new RangeError("שם קטגוריה ריק");
    return clean;
  }

  #translate(err: unknown, name: string, id?: number): unknown {
    if (!(err instanceof Prisma.PrismaClientKnownRequestError)) return err;
    switch (err.code) {
      case "P2002":
        return new ConflictError(`קטגוריה בשם ${name} כבר קיימת`);
      case "P2025":
        return new NotFoundError(`קטגוריה ${id} לא קיימת`);
      default:
        return err;
    }
  }
}
```
<p>מי שמשתמש ב־repository רואה רק את ה־<code>interface</code>: שלוש מתודות, ושתי שגיאות עם שמות שמובנים לכל אחד. ה־service יכול לתרגם <code>ConflictError</code> ל־409 ו־<code>NotFoundError</code> ל־404 בלי לדעת שקיים Prisma. בבדיקות אפשר להחליף אותו במחלקה מזויפת שמממשת אותו interface (פרק 22).</p>
<p><code>#translate</code> <em>מחזירה</em> שגיאה ולא זורקת. כך ה־<code>throw</code> נשאר גלוי במתודה שקראה לה, ו־TypeScript יודע שהקוד אחריו לא ממשיך. ה־<code>switch</code> מתאים כאן: ערך אחד (<code>err.code</code>) מול רשימת קודים, ו־<code>default</code> לכל השאר.</p>
<p><code>this.name = "ConflictError"</code>: בלי השורה הזו, בלוגים ובהודעת השגיאה כתוב <code>Error</code>. עם השורה רואים מיד מה קרה.</p>

<ul><li>לבדוק קודם אם השם קיים (<code>findUnique</code>) ורק אחר כך ליצור — שתי בקשות במקביל עוברות את הבדיקה, ואחת מהן נכשלת ב־P2002 בכל זאת. האילוץ במסד הוא הבדיקה האמיתית (פרק 18).</li><li><code>catch (err) { throw new ConflictError(...) }</code> בלי לבדוק את הקוד — כל תקלה במסד הופכת ל"כבר קיימת".</li><li><code>include: { products: true }</code> ואז <code>.length</code> — טוען את כל המוצרים רק כדי לספור אותם.</li><li><code>constructor(private prisma: PrismaClient)</code> — ‏Node נכשל ב־<code>ERR_UNSUPPORTED_TYPESCRIPT_SYNTAX</code>.</li></ul>

```ts run save=solutions/P19-E45.ts
// P19-E45: snake_case ל־camelCase
export function toCamel(name: string): string {
```

```ts run save=solutions/P19-E46.ts
export function describeDbUrl(url: string) {
  let u: URL;
  try {
    u = new URL(url);
  } catch {
    throw new Error("כתובת מסד לא תקינה");
  }
  const database = decodeURIComponent(u.pathname.slice(1));
  if (!["postgresql:", "postgres:"].includes(u.protocol) || database === "" || database.includes("/")) {
    throw new Error("כתובת מסד לא תקינה");
  }
  return {
    host: u.hostname,
    port: u.port === "" ? 5432 : Number(u.port),
    database,
    user: decodeURIComponent(u.username),
    password: u.password === "" ? null : "***",
  };
}
```

```ts save=solutions/P19-E62.ts
// P19-E62: "migrate diff" בזיכרון
export type Column = { table: string; name: string; type: string; nullable: boolean };

function byTable(cols: Column[]) {
  const m = new Map<string, Map<string, Column>>();
  for (const c of cols) {
    if (!m.has(c.table)) m.set(c.table, new Map());
    m.get(c.table)!.set(c.name, c);
  }
  return m;
}
const union = (a: Iterable<string>, b: Iterable<string>) => [...new Set([...a, ...b])].sort();

export function diffColumns(before: Column[], after: Column[]) {
  const b = byTable(before);
  const a = byTable(after);
  const steps: string[] = [];
  let destructive = false;
  for (const t of union(b.keys(), a.keys())) {
    const old = b.get(t);
    const cur = a.get(t);
    if (!old) { steps.push(`CREATE TABLE ${t}`); continue; }
    if (!cur) { steps.push(`DROP TABLE ${t}`); destructive = true; continue; }
    for (const name of union(old.keys(), cur.keys())) {
      const o = old.get(name);
      const c = cur.get(name);
      if (!o) steps.push(`ADD COLUMN ${t}.${name} ${c!.type}${c!.nullable ? "" : " NOT NULL"}`);
      else if (!c) { steps.push(`DROP COLUMN ${t}.${name}`); destructive = true; }
      else {
        if (o.type !== c.type) { steps.push(`ALTER COLUMN ${t}.${name} TYPE ${c.type}`); destructive = true; }
        if (o.nullable && !c.nullable) { steps.push(`ALTER COLUMN ${t}.${name} SET NOT NULL`); destructive = true; }
        if (!o.nullable && c.nullable) steps.push(`ALTER COLUMN ${t}.${name} DROP NOT NULL`);
      }
    }
  }
  return { steps, destructive };
}
```
<p>‏<code>DROP NOT NULL</code> לא הורס (רק מרשה יותר). ‏<code>ADD COLUMN ... NOT NULL</code> — גם הוא יכול להיכשל על טבלה מלאה (P19-E52); Prisma בודק את זה מול המסד האמיתי — כאן אין לנו את הנתונים.</p>

```ts
export function toApiCustomer(c: { id: number; fullName: string; email: string; city: string | null; joinedOn: Date }): ApiCustomer {
```

```ts save=solutions/P19-E63.ts
export function toCustomerData(body: { fullName: string; email: string; city?: string | null; joinedOn: string }) {
  const d = new Date(`${body.joinedOn}T00:00:00Z`);
```

```ts save=solutions/P19-E77.ts
export type ApiOrder = { id: number; customerId: number; status: string; createdAt: string; items: { productId: number; quantity: number; unitPrice: number }[]; total: number };

export function toApiOrder(o: { id: number; customerId: number; status: string; createdAt: Date; items: { productId: number; quantity: number; unitPrice: unknown }[] }): ApiOrder {
  const items = [...o.items]
    .sort((a, b) => a.productId - b.productId)
    .map((i) => ({ productId: i.productId, quantity: i.quantity, unitPrice: Number(i.unitPrice) }));

  return { id: o.id, customerId: o.customerId, status: o.status, createdAt: o.createdAt.toISOString(), items, total: cents / 100 };
}
```

```ts save=solutions/P19-E78.ts
import type { PrismaClient } from "../files/generated/prisma/client.ts";

export type ListQuery = { status?: "pending" | "paid" | "shipped" | "cancelled"; customerId?: number; sort?: "createdAt" | "-createdAt"; page?: number; limit?: number };

export async function listOrders(prisma: PrismaClient, q: ListQuery) {
  const { sort = "-createdAt", page = 1, limit = 5 } = q;
  const where = {
    ...(q.status !== undefined && { status: q.status }),
    ...(q.customerId !== undefined && { customerId: q.customerId }),
  };
  const dir = sort.startsWith("-") ? ("desc" as const) : ("asc" as const);
  const [data, total] = await Promise.all([
    prisma.order.findMany({
      where,
      select: { id: true, customerId: true, status: true },
      orderBy: [{ createdAt: dir }, { id: dir }],
      skip: (page - 1) * limit,
      take: limit,
    }),
    prisma.order.count({ where }),
  ]);
  return { data, meta: { page, limit, total } };
}
```

```ts save=solutions/P19-E83.ts
  constructor(initial: Customer[] = []) {

  }

    return c ? { ...c } : undefined;
```

```ts save=solutions/P19-E84.ts
// P19-E84: repository ללקוחות — עם Prisma
import { Prisma, type PrismaClient } from "../files/generated/prisma/client.ts";

export type Customer = { id: number; fullName: string; email: string; city: string | null };

export class AppError extends Error {
  code: string;
  constructor(code: string, message = code) {
    super(message);
    this.code = code;
  }
}

const SELECT = { id: true, fullName: true, email: true, city: true } as const;
const known = (err: unknown, code: string) => err instanceof Prisma.PrismaClientKnownRequestError && err.code === code;

export class PrismaCustomersRepository {
  private prisma: PrismaClient;
  constructor(prisma: PrismaClient) {
    this.prisma = prisma;
  }

  async findById(id: number): Promise<Customer | undefined> {
    return (await this.prisma.customer.findUnique({ where: { id }, select: SELECT })) ?? undefined;
  }

  async findByEmail(email: string): Promise<Customer | undefined> {
    return (await this.prisma.customer.findFirst({ where: { email: { equals: email, mode: "insensitive" } }, select: SELECT })) ?? undefined;
  }

  async insert(data: Omit<Customer, "id">): Promise<Customer> {
    const today = new Date(new Date().toISOString().slice(0, 10) + "T00:00:00Z");
    try {
      return await this.prisma.customer.create({ data: { ...data, joinedOn: today }, select: SELECT });
    } catch (err) {
      if (known(err, "P2002")) throw new AppError("EMAIL_TAKEN");
      throw err;
    }
  }

  async delete(id: number): Promise<boolean> {
    try {
      await this.prisma.customer.delete({ where: { id } });
      return true;
    } catch (err) {
      if (known(err, "P2025")) return false;
      throw err;
    }
  }
}
```

```ts save=solutions/P19-E88.ts
// P19-E88: router שמקבל repository
import express from "express";
import { z } from "zod";

type Customer = { id: number; fullName: string; email: string; city: string | null };
interface CustomersRepository {
  findById(id: number): Promise<Customer | undefined>;
  findByEmail(email: string): Promise<Customer | undefined>;
  insert(data: Omit<Customer, "id">): Promise<Customer>;
  delete(id: number): Promise<boolean>;
}

const CreateCustomer = z.object({
  fullName: z.string().trim().min(1),
  email: z.email().transform((e) => e.toLowerCase()),
  city: z.string().nullable().default(null),
});

const parseId = (raw: string) => {
  const id = Number(raw);
  return Number.isInteger(id) && id > 0 ? id : undefined;
};

export function createApp(repo: CustomersRepository) {
  const app = express();
  app.use(express.json());

  app.get("/customers/:id", async (req, res) => {
    const id = parseId(req.params.id);
    if (id === undefined) return res.status(400).json({ error: "VALIDATION" });
    const c = await repo.findById(id);
    if (!c) return res.status(404).json({ error: "NOT_FOUND" });
    res.json(c);
  });

  app.post("/customers", async (req, res) => {
    const parsed = CreateCustomer.safeParse(req.body);
    if (!parsed.success) return res.status(400).json({ error: "VALIDATION" });
    res.status(201).json(await repo.insert(parsed.data));
  });

  app.delete("/customers/:id", async (req, res) => {
    const id = parseId(req.params.id);
    if (id === undefined) return res.status(400).json({ error: "VALIDATION" });
    if (!(await repo.delete(id))) return res.status(404).json({ error: "NOT_FOUND" });
    res.status(204).end();
  });

  app.use((err: any, _req: express.Request, res: express.Response, _next: express.NextFunction) => {
    if (err?.code === "EMAIL_TAKEN") return res.status(409).json({ error: "EMAIL_TAKEN" });
    res.status(500).json({ error: "INTERNAL" });
  });
  return app;
}
```
