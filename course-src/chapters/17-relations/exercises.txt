```sql run db=17-relations/files/store.sql
  (SELECT COUNT(*) FROM customers c LEFT JOIN orders o ON o.customer_id = c.id) AS q2,
  (SELECT COUNT(*) FROM orders o LEFT JOIN customers c ON c.id = o.customer_id) AS q3,
  (SELECT COUNT(*) FROM customers c JOIN orders o ON true) AS q4;
```

```out
 q1 | q2 | q3 | q4
----+----+----+----

(1 row)
```

@@ P17-E03 | 1 | למה הסכום כפול? | 10 | שיעור 2
@task
<p>נניח שיש גם טבלת <code>payments(order_id, amount)</code>, ולהזמנה אחת יכולים להיות כמה תשלומים (חלוקה לכרטיסים). מישהו כתב:</p>
```sql
SELECT o.id, SUM(oi.quantity * oi.unit_price) AS total, SUM(p.amount) AS paid
FROM orders o
JOIN order_items oi ON oi.order_id = o.id
JOIN payments p     ON p.order_id = o.id
GROUP BY o.id;
```
<p>להזמנה עם 3 פריטים ו־2 תשלומים — מה יקרה ל־total ול־paid? למה? ואיך מתקנים?</p>
@hint1
<p>כמה שורות יש להזמנה הזו אחרי שני ה־JOIN־ים?</p>
@hint2
<p>לסכם כל צד בנפרד, ורק אז לחבר.</p>
@solution
<p>אחרי שני ה־JOIN־ים יש 3 × 2 = 6 שורות להזמנה (כל פריט עם כל תשלום). total יוצא פי 2 (כל פריט נספר פעמיים), ו־paid פי 3. זה ה־"fan-out" — אחד הבאגים הנפוצים בדוחות.</p>
```sql
WITH items AS (SELECT order_id, SUM(quantity * unit_price) AS total FROM order_items GROUP BY order_id),
     pays  AS (SELECT order_id, SUM(amount) AS paid FROM payments GROUP BY order_id)
SELECT o.id, i.total, COALESCE(p.paid, 0) AS paid
FROM orders o
JOIN items i     ON i.order_id = o.id
LEFT JOIN pays p ON p.order_id = o.id;
```
@mistakes
<ul><li>"לתקן" עם <code>SUM(DISTINCT ...)</code> — שני פריטים באותו מחיר ייספרו כאחד.</li></ul>

```sql
SELECT status, COUNT(*) FROM orders GROUP BY status HAVING COUNT(*) > 2;                   -- 2

```

```out
-------------

(2 rows)

 full_name
-----------
(0 rows)

    name
------------
 gift cards
(1 row)
```
<p>1: בלי Avi ו־Omer (עיר NULL). 2: <strong>כלום</strong> — בגלל ה־NULL ברשימה. 3: עובד כאן, כי <code>category_id</code> הוא NOT NULL. אבל אם יום אחד העמודה תהיה nullable ומוצר אחד יהיה בלי קטגוריה — השאילתה תחזיר כלום בשקט. לכן: <code>NOT EXISTS</code>.</p>

```sql run db=17-relations/files/store.sql save=solutions/P17-E27.sql
  SELECT o.id, o.customer_id, o.created_at, SUM(oi.quantity * oi.unit_price) AS total
```

```ts run save=solutions/P17-E28.ts
// P17-E28: JOIN ו־GROUP BY בקוד
export type Customer = { id: number; full_name: string; city: string | null };
export type Order = { id: number; customer_id: number; status: string };
export type OrderItem = { order_id: number; quantity: number; unit_price: number };
export type CityRevenue = { city: string; revenue: number };

export function revenueByCity(customers: Customer[], orders: Order[], items: OrderItem[]): CityRevenue[] {
  // "אינדקס" לפי id — כמו JOIN ... ON c.id = o.customer_id, בלי לחפש במערך כל פעם
  const customerById = new Map(customers.map((c) => [c.id, c]));
  const orderById = new Map(orders.map((o) => [o.id, o]));

  const agorotByCity = new Map<string, number>();
  for (const item of items) {
    const order = orderById.get(item.order_id);
    if (!order || order.status === "cancelled") continue;
    const city = customerById.get(order.customer_id)?.city ?? "unknown";
    const agorot = Math.round(item.quantity * item.unit_price * 100);
    agorotByCity.set(city, (agorotByCity.get(city) ?? 0) + agorot);
  }

  return [...agorotByCity]
    .map(([city, agorot]) => ({ city, revenue: agorot / 100 }))
    .sort((a, b) => b.revenue - a.revenue || a.city.localeCompare(b.city));
}

const customers = [
  { id: 1, full_name: "Dana", city: "Haifa" },
  { id: 2, full_name: "Avi", city: null },
];
const orders = [
  { id: 10, customer_id: 1, status: "paid" },
  { id: 11, customer_id: 2, status: "pending" },
  { id: 12, customer_id: 1, status: "cancelled" },
];
const items = [
  { order_id: 10, quantity: 3, unit_price: 19.9 },
  { order_id: 11, quantity: 1, unit_price: 89.9 },
  { order_id: 12, quantity: 5, unit_price: 100 },
];
console.log(revenueByCity(customers, orders, items));
```
```out
[
  { city: 'unknown', revenue: 89.9 },
  { city: 'Haifa', revenue: 59.7 }
]
```
<p>כל חלק בשאילתה של P17-E12 קיבל מקבילה בקוד. ‏<code>JOIN ... ON</code> הוא <code>Map.get</code> לפי id. ‏<code>WHERE status &lt;&gt; 'cancelled'</code> הוא <code>continue</code>. ‏<code>GROUP BY</code> עם <code>SUM</code> הוא <code>Map</code> של צוברים. ‏<code>ORDER BY</code> הוא <code>sort</code>. גם PostgreSQL עושה בדיוק את זה (נקרא <em>hash join</em>): בונה טבלת גיבוב מצד אחד, ועובר על הצד השני.</p>
<p>ההבדל: SQL עושה את זה ליד הנתונים, ומחזיר 4 שורות. בקוד היית צריך קודם להעביר את כל ההזמנות והפריטים מהמסד. לכן ברוב המקרים עדיף SQL, וזה תרגיל כדי לראות מה קורה "מתחת למכסה".</p>

```ts
export type Data = {
  customers: { id: number }[];
  products: { id: number }[];
  orders: { id: number; customer_id: number }[];
  order_items: { order_id: number; product_id: number }[];
```

```ts save=solutions/P17-E43.ts
const missing = (col: string, value: number, table: string) => `Key (${col})=(${value}) is not present in table "${table}".`;

export function checkForeignKeys(data: Data): string[] {
  const customerIds = new Set(data.customers.map((c) => c.id));
  const productIds = new Set(data.products.map((p) => p.id));
  const orderIds = new Set(data.orders.map((o) => o.id));
  const problems: string[] = [];
  for (const o of data.orders) {
    if (!customerIds.has(o.customer_id)) problems.push(missing("customer_id", o.customer_id, "customers"));
  }
  for (const i of data.order_items) {
    if (!orderIds.has(i.order_id)) problems.push(missing("order_id", i.order_id, "orders"));
    if (!productIds.has(i.product_id)) problems.push(missing("product_id", i.product_id, "products"));
  }
  return problems;
}
```

```ts save=solutions/P17-E44.ts
// P17-E44: הוספת פריט מ־Node, עם קודי השגיאה
export type Db = { query(text: string, values?: unknown[]): Promise<{ rows: any[] }> };
export type AddResult = { ok: true; unitPrice: number } | { ok: false; error: "not_found" | "duplicate" | "invalid" };

const ERRORS: Record<string, "not_found" | "duplicate" | "invalid"> = {
  "23503": "not_found", // foreign key: אין הזמנה כזו
  "23505": "duplicate", // המפתח הראשי (order_id, product_id)
  "23514": "invalid", // CHECK (quantity > 0)
};

export async function addOrderItem(client: Db, orderId: number, productId: number, quantity: number): Promise<AddResult> {
  const { rows } = await client.query("SELECT price FROM products WHERE id = $1 AND is_active", [productId]);
  if (rows.length === 0) return { ok: false, error: "not_found" };
  const unitPrice = Number(rows[0].price);
  try {
    await client.query("INSERT INTO order_items (order_id, product_id, quantity, unit_price) VALUES ($1, $2, $3, $4)", [orderId, productId, quantity, unitPrice]);
  } catch (err: any) {
    const error = ERRORS[err.code];
    if (error) return { ok: false, error };
    throw err;
  }
  return { ok: true, unitPrice };
}
```

```ts save=solutions/P17-E45.ts
// P17-E45: מ־JSON מקונן לטבלאות
export type FlatOrder = { orderId: number; email: string; fullName: string; items: { product: string; price: number; qty: number }[] };
export type Tables = {
  customers: { id: number; email: string; full_name: string }[];
  products: { id: number; name: string }[];
  orders: { id: number; customer_id: number }[];
  order_items: { order_id: number; product_id: number; quantity: number; unit_price: number }[];
};

export function normalize(input: FlatOrder[]): Tables {
  const t: Tables = { customers: [], products: [], orders: [], order_items: [] };
  const customerIds = new Map<string, number>();
  const productIds = new Map<string, number>();

  for (const o of input) {
    const email = o.email.toLowerCase();
    let customerId = customerIds.get(email);
    if (customerId === undefined) {
      customerId = customerIds.size + 1;
      customerIds.set(email, customerId);
      t.customers.push({ id: customerId, email, full_name: o.fullName });
    }
    t.orders.push({ id: o.orderId, customer_id: customerId });

    const lines = new Map<number, Tables["order_items"][number]>();
    for (const item of o.items) {
      let productId = productIds.get(item.product);
      if (productId === undefined) {
        productId = productIds.size + 1;
        productIds.set(item.product, productId);
        t.products.push({ id: productId, name: item.product });
      }
      const line = lines.get(productId);
      if (line) line.quantity += item.qty;
      else {
        const row = { order_id: o.orderId, product_id: productId, quantity: item.qty, unit_price: item.price };
        lines.set(productId, row);
        t.order_items.push(row);
      }
    }
  }
  return t;
}
```
<p>ה־<code>Map</code>ים הם ה"UNIQUE" שלך: לפני שמוסיפים שורה — בודקים אם כבר יש כזו. ‏<code>lines</code> מתאפס לכל הזמנה, כי המפתח הראשי הוא הזוג (הזמנה, מוצר) — אותו מוצר בהזמנה אחרת הוא שורה חדשה.</p>

```ts save=solutions/P17-E55.ts
// P17-E55: משורות שטוחות לאובייקטים
export type FlatRow = { order_id: number; full_name: string; status: string; product: string | null; quantity: number | null; unit_price: number | null };
export type Item = { product: string; quantity: number; unitPrice: number; lineTotal: number };
export type OrderView = { id: number; customer: string; status: string; items: Item[]; total: number };

export function nestOrders(rows: FlatRow[]): OrderView[] {
  const byId = new Map<number, OrderView>();
  const cents = new Map<number, number>();
  for (const r of rows) {
    let order = byId.get(r.order_id);
    if (!order) {
      order = { id: r.order_id, customer: r.full_name, status: r.status, items: [], total: 0 };
      byId.set(r.order_id, order);
      cents.set(r.order_id, 0);
    }
    if (r.product === null || r.quantity === null || r.unit_price === null) continue;
    const line = Math.round(r.unit_price * 100) * r.quantity;
    order.items.push({ product: r.product, quantity: r.quantity, unitPrice: r.unit_price, lineTotal: line / 100 });
    cents.set(r.order_id, cents.get(r.order_id)! + line);
  }
  for (const order of byId.values()) order.total = cents.get(order.id)! / 100;
  return [...byId.values()];
}
```
<p>ה־<code>Map</code> עושה את ה"GROUP BY" בקוד. בדיקת ה־null היא בדיוק המקרה של P17-E53 — ובזכות הטיפוס, TypeScript לא היה נותן לך לשכוח אותה.</p>

```ts save=solutions/P17-E56.ts
// P17-E56: פרטי הזמנה מהמסד, בשאילתה אחת
export type Db = { query(text: string, values?: unknown[]): Promise<{ rows: any[] }> };
export type OrderDetails = { id: number; status: string; customer: { id: number; name: string };
  items: { product: string; quantity: number; unitPrice: number }[] };

export async function getOrderDetails(client: Db, orderId: number): Promise<OrderDetails | null> {
  const { rows } = await client.query(
    `SELECT o.id AS order_id, o.status, c.id AS customer_id, c.full_name,
            p.name AS product, oi.quantity, oi.unit_price
     FROM orders o
     JOIN customers c        ON c.id = o.customer_id
     LEFT JOIN order_items oi ON oi.order_id = o.id
     LEFT JOIN products p     ON p.id = oi.product_id
     WHERE o.id = $1
     ORDER BY p.name`,
    [orderId],
  );
  if (rows.length === 0) return null;
  const first = rows[0];
  return {
    id: first.order_id,
    status: first.status,
    customer: { id: first.customer_id, name: first.full_name },
    items: rows
      .filter((r) => r.product !== null)
      .map((r) => ({ product: r.product, quantity: r.quantity, unitPrice: Number(r.unit_price) })),
  };
}
```

```ts run save=solutions/P17-E57.ts
// P17-E57: leftJoin גנרי
export function innerJoin<L, R>(left: L[], right: R[], on: (l: L, r: R) => boolean): { left: L; right: R }[] {
  const out: { left: L; right: R }[] = [];
  for (const l of left) for (const r of right) if (on(l, r)) out.push({ left: l, right: r });
  return out;
}

export function leftJoin<L, R>(left: L[], right: R[], on: (l: L, r: R) => boolean): { left: L; right: R | null }[] {
  const out: { left: L; right: R | null }[] = [];
  for (const l of left) {
    const matches = right.filter((r) => on(l, r));
    if (matches.length === 0) out.push({ left: l, right: null });
    else for (const r of matches) out.push({ left: l, right: r });
  }
  return out;
}

const customers = [{ id: 1, name: "Dana" }, { id: 4, name: "Avi" }];
const orders = [{ id: 1, customerId: 1 }, { id: 3, customerId: 1 }];

  console.log(left.name, right === null ? "-" : right.id);
}
```

```ts save=solutions/P17-E66.ts
export type Db = { query(text: string, values?: unknown[]): Promise<{ rows: any[] }> };
export type CategoryRevenue = { category: string; units: number; revenue: number };

export async function categoryRevenue(client: Db, { from, to, minRevenue = 0 }: { from: string; to: string; minRevenue?: number }): Promise<CategoryRevenue[]> {
  if (from >= to) throw new Error("טווח תאריכים לא תקין");
  const { rows } = await client.query(
    `SELECT cat.name AS category, SUM(oi.quantity) AS units, SUM(oi.quantity * oi.unit_price) AS revenue
     FROM order_items oi
     JOIN orders o       ON o.id = oi.order_id
     JOIN products p     ON p.id = oi.product_id
     JOIN categories cat ON cat.id = p.category_id
     WHERE o.status <> 'cancelled' AND o.created_at >= $1 AND o.created_at < $2
     GROUP BY cat.name
     HAVING SUM(oi.quantity * oi.unit_price) >= $3
     ORDER BY revenue DESC, category`,
    [from, to, minRevenue],
  );
  return rows.map((r) => ({ category: r.category, units: Number(r.units), revenue: Number(r.revenue) }));
}
```

<p>זה החישוב של "הלקוחות עם ממוצע ההזמנה הגבוה" (כאן — שניים) (הדוגמה עם <code>WITH order_totals</code> מהשיעור) — על מערכים בזיכרון. מה יודפס? איזו שורה ב־JS מקבילה לכל חלק ב־SQL (ה־CTE, ה־JOIN, ה־GROUP BY, ה־ORDER BY וה־LIMIT)?</p>
```js
const orders = [{ id: 1, customer: "Dana", status: "shipped" }, { id: 2, customer: "Yossi", status: "shipped" },
  { id: 3, customer: "Dana", status: "paid" }, { id: 4, customer: "Noa", status: "cancelled" }, { id: 5, customer: "Noa", status: "paid" }];
const items = [{ order: 1, total: 194.7 }, { order: 2, total: 278.7 }, { order: 3, total: 197 }, { order: 4, total: 649 }, { order: 5, total: 189 }];

const orderTotals = orders
  .filter((o) => o.status !== "cancelled")
  .map((o) => ({ customer: o.customer, total: items.filter((i) => i.order === o.id).reduce((s, i) => s + i.total, 0) }));

const byCustomer = new Map();
for (const t of orderTotals) {
  const g = byCustomer.get(t.customer) ?? { customer: t.customer, orders: 0, sum: 0 };
  g.orders += 1;
  g.sum += t.total;
  byCustomer.set(t.customer, g);
}
const result = [...byCustomer.values()]
  .map((g) => ({ customer: g.customer, orders: g.orders, avg: Math.round((g.sum / g.orders) * 100) / 100 }))
  .sort((a, b) => b.avg - a.avg)
  .slice(0, 2);
console.log(result);
```

```sql run db=17-relations/files/store.sql
WITH ranked AS (
  SELECT cat.name AS category, p.name, p.price,
         ROW_NUMBER() OVER (PARTITION BY cat.id ORDER BY p.price DESC, p.name) AS rn
  FROM products p
  JOIN categories cat ON cat.id = p.category_id
  WHERE p.is_active
)
SELECT category, name, price FROM ranked WHERE rn <= 2 ORDER BY category, rn;
```
