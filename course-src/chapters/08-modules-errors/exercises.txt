// P08-E08: זריקת שגיאות
function parsePositiveInt(input, fieldName = "ערך") {
  if (String(input).trim() === "") {
    throw new Error(`${fieldName}: שדה חובה`);
  }
  const n = Number(input);
  if (Number.isNaN(n)) {
    throw new Error(`${fieldName}: חייב להיות מספר`);
  }
  if (!Number.isInteger(n) || n < 1) {
    throw new Error(`${fieldName}: חייב להיות מספר שלם חיובי`);
  }
  return n;
}

try {
  console.log(parsePositiveInt("3", "כמות"));
  console.log(parsePositiveInt("2.5", "כמות"));
} catch (err) {
  console.log(err.message);

function safeParse(text, fallback = null) {
  try {
    return JSON.parse(text);
  } catch {
    return fallback;

  } catch {

// P08-E10: הפיכת שגיאות לערכים
function attempt(fn) {
  try {
    return { ok: true, value: fn() };
  } catch (err) {
    return { ok: false, error: err.message };
  }
}

function firstSuccessful(fns) {
  for (const fn of fns) {
    const result = attempt(fn);
    if (result.ok) {
      return result.value;
    }
  }
  throw new Error("כל הניסיונות נכשלו");
}

console.log(attempt(() => JSON.parse("[1]")), attempt(() => null.x).ok);

module.exports = { attempt, firstSuccessful };
```

// P08-E11: ממוצע — מתוקן
function averageScore(scores) {
  if (scores.length === 0) {
    return 0;
  }
  let sum = 0;
  for (let i = 0; i < scores.length; i++) {
    sum += scores[i];
  }

function retry(fn, times) {
  let lastError;
  for (let attempt = 1; attempt <= times; attempt++) {
    try {
      return fn(attempt);
    } catch (err) {
      lastError = err;
    }
  }
  throw lastError;

function parseLine(line) {
  const parts = line.split(",").map((p) => p.trim());
  if (parts.length < 3 || parts.some((p) => p === "")) {
    throw new Error("חסרים שדות");
  }

  const amount = Number(amountText);

}

const REQUIRED = ["DATABASE_URL", "JWT_SECRET"];

function loadConfig(env) {
  const missing = REQUIRED.filter((key) => !env[key]);

function longestStreak(days) {
  let best = 0;
  let current = 0;
  for (const visited of days) {
    current = visited ? current + 1 : 0;
    best = Math.max(best, current);
  }
  return best;
}

// P08-E24: מודול בנק
function bankError(code, message) {
  const err = new Error(message);
  err.code = code;
  return err;
}

function assertAmount(amount) {
  if (typeof amount !== "number" || !Number.isFinite(amount) || amount <= 0) {
    throw bankError("INVALID_AMOUNT", "סכום לא תקין");
  }
}

function deposit(account, amount) {
  assertAmount(amount);
  return { ...account, balance: account.balance + amount };
}

function withdraw(account, amount) {
  assertAmount(amount);
  if (account.balance < amount) {
    throw bankError("INSUFFICIENT_FUNDS", "אין מספיק יתרה");
  }
  return { ...account, balance: account.balance - amount };
}

function transfer(from, to, amount) {
  if (from.id === to.id) {
    throw bankError("SAME_ACCOUNT", "לא ניתן להעביר לאותו חשבון");
  }
  const newFrom = withdraw(from, amount);
  const newTo = deposit(to, amount);
  return [newFrom, newTo];
}

function safeTransfer(from, to, amount) {
  try {
    const [newFrom, newTo] = transfer(from, to, amount);
    return { ok: true, from: newFrom, to: newTo };
  } catch (err) {
    return { ok: false, code: err.code, message: err.message, from, to };
  }
}

const a = { id: 1, owner: "דנה", balance: 100 };
const b = { id: 2, owner: "עומר", balance: 20 };
console.log(safeTransfer(a, b, 30));
console.log(safeTransfer(a, b, 500).code);

module.exports = { bankError, deposit, withdraw, transfer, safeTransfer };
```

// P08-E25: מחשבון בלי eval
const OPERATORS = ["+", "-", "*", "/"];

function tokenize(expression) {
  const tokens = [];
  let current = "";
  const flush = () => {
    if (current !== "") {
      const n = Number(current);
      if (Number.isNaN(n)) throw new Error("ביטוי לא תקין");
      tokens.push(n);
      current = "";
    }
  };
  for (const ch of expression) {
    if ("0123456789.".includes(ch)) {
      current += ch;
    } else if (OPERATORS.includes(ch)) {
      flush();
      tokens.push(ch);
    } else if (ch === " ") {
      flush();
    } else {
      throw new Error(`תו לא חוקי: ${ch}`);
    }
  }
  flush();
  return tokens;
}

function validate(tokens) {
  const ok =
    tokens.length % 2 === 1 &&
    tokens.every((t, i) => (i % 2 === 0 ? typeof t === "number" : OPERATORS.includes(t)));
  if (!ok) throw new Error("ביטוי לא תקין");
}

function calculate(expression) {
  const tokens = tokenize(expression);
  validate(tokens);

  const stack = [tokens[0]];
  for (let i = 1; i < tokens.length; i += 2) {
    const op = tokens[i];
    const n = tokens[i + 1];
    if (op === "*") {
      stack.push(stack.pop() * n);
    } else if (op === "/") {
      if (n === 0) throw new Error("חילוק באפס");
      stack.push(stack.pop() / n);
    } else {
      stack.push(op, n);
    }
  }

  let result = stack[0];
  for (let i = 1; i < stack.length; i += 2) {
    result = stack[i] === "+" ? result + stack[i + 1] : result - stack[i + 1];
  }
  return result;
}

console.log(calculate("2 + 3 * 4"), calculate("10 / 4 - 1"), calculate("7"));
for (const bad of ["2 + x", "3 +", "4 / 0"]) {
  try {
    calculate(bad);
  } catch (err) {
    console.log(err.message);
  }
}

module.exports = { calculate, tokenize };
```

// P08-E27: כפילויות ברשימת הרשמה
function findDuplicates(emails) {
  if (!Array.isArray(emails)) {
    throw new TypeError("emails חייב להיות מערך");
  }
  const seen = new Set();
  const dups = new Set();
  for (const raw of emails) {
    const email = raw.trim().toLowerCase();
    if (seen.has(email)) {
      dups.add(email);
    } else {
      seen.add(email);
    }
  }
  return [...dups];
}

console.log(findDuplicates(["a@x.com", "b@x.com", " A@x.com", "c@x.com", "b@x.com", "a@x.com"]));
try {
  findDuplicates("a@x.com");
} catch (err) {
  console.log(`${err.name}: ${err.message}`);
}

module.exports = { findDuplicates };

// P08-E28: מפרש פקודות
function parseCommand(line) {
  const [word, ...rest] = line.trim().split(" ").filter((w) => w !== "");
  if (word === undefined) {
    throw new Error("לא הוקלדה פקודה");
  }

  switch (word.toLowerCase()) {
    case "add": {
      const amount = Number(rest[0]);
      if (rest.length === 0 || !Number.isFinite(amount) || amount <= 0) {
        throw new Error("סכום לא תקין");
      }
      return { cmd: "add", amount, note: rest.slice(1).join(" ") };
    }
    case "list":
      return { cmd: "list" };
    case "total":
      return { cmd: "total" };
    default:
      throw new Error(`פקודה לא מוכרת: "${word}"`);
  }
}

console.log(parseCommand("  add 25   קפה בבוקר "));
for (const line of ["add abc", "", "delete 3"]) {
  try {
    parseCommand(line);
  } catch (err) {
    console.log(err.message);
  }
}

module.exports = { parseCommand };
```

export function groupBy(list, keyFn) {
  const groups = new Map();
  for (const item of list) {
    const key = keyFn(item);
    if (!groups.has(key)) {
      groups.set(key, []);
    }
    groups.get(key).push(item);
  }
  return groups;
}

export default function countByStatus(orders) {
  const groups = groupBy(orders, (o) => o.status);
  return Object.fromEntries([...groups].map(([status, list]) => [status, list.length]));
}

function parseFrame(line) {
  const text = line.trim().slice(3);
  let fn = null;
  let location = text;
  if (text.endsWith(")")) {
    const open = text.indexOf(" (");
    fn = text.slice(0, open);
    location = text.slice(open + 2, -1);
  }
  const lastColon = location.lastIndexOf(":");
  const prevColon = location.lastIndexOf(":", lastColon - 1);
  const lineNo = Number(location.slice(prevColon + 1, lastColon));
  const colNo = Number(location.slice(lastColon + 1));
  if (lastColon === -1 || prevColon === -1 || Number.isNaN(lineNo) || Number.isNaN(colNo)) {
    return { fn, file: location, line: null, col: null };
  }
  return { fn, file: location.slice(0, prevColon), line: lineNo, col: colNo };

  const msg = err.message;
  switch (err.name) {
    case "TypeError": {
      const prefix = "Cannot read properties of ";
      if (msg.startsWith(prefix)) {
        const value = msg.slice(prefix.length, msg.indexOf(" ("));
        const field = msg.slice(msg.indexOf("'") + 1, msg.lastIndexOf("'"));
        return `ניסית לקרוא את "${field}" מתוך ${value} — מה שמשמאל לנקודה ריק`;

        return `"${msg.slice(0, -notFn.length)}" הוא לא פונקציה — בדוק את השם ואת הייבוא`;

      break;

      break;

@hint2
<p>אם הסרת את האיבר ה־<code>i</code>, <strong>אל תקדם</strong> את <code>i</code> — האיבר הבא זז למקומו.</p>
@solution
```js run file=08-modules-errors/starter/P08-E80.js save=solutions/P08-E80.js
function shrink(input, fails) {
  let current = [...input];
  let changed = true;
  while (changed) {
    changed = false;
    let i = 0;
    while (i < current.length) {
      const candidate = [...current.slice(0, i), ...current.slice(i + 1)];
      if (fails(candidate)) {
        current = candidate;
        changed = true;
      } else {
        i++;
      }
    }
  }
  return current;
}

const fails = (arr) => arr.includes(0) && arr.some((n) => n < 0);
const input = [5, 0, 8, -3, 2, 0, 7];
console.log(shrink(input, fails), input);

module.exports = { shrink };
```
