```ts run
  { description: "How many units of a product are in stock", inputSchema: { productId: z.number().int().positive() } },
```

```ts save=solutions/P29.5-E06.ts
// P29.5-E06: לולאת סוכן שמחזיקה מעמד — כלים במקביל, שגיאות ככלי, ותקרת צעדים
import type Anthropic from "@anthropic-ai/sdk";

export interface AgentTool {
  definition: Anthropic.Tool;
  run: (input: any) => Promise<string>;
}

export interface AgentResult {
  status: "done" | "max_steps" | "refused" | "truncated";
  text: string;
  steps: number;
  usage: { inputTokens: number; outputTokens: number };
}

export async function runAgent(
  client: Anthropic,
  { system, tools, prompt, maxSteps = 10, model = "claude-opus-5-5" }: { system: string; tools: Record<string, AgentTool>; prompt: string; maxSteps?: number; model?: string },
): Promise<AgentResult> {
  const messages: Anthropic.MessageParam[] = [{ role: "user", content: prompt }];
  const usage = { inputTokens: 0, outputTokens: 0 };
  const textOf = (m: Anthropic.Message) => m.content.flatMap((b) => (b.type === "text" ? [b.text] : [])).join("");

  for (let step = 1; step <= maxSteps; step++) {
    const res = await client.messages.create({
      model,
      max_tokens: 16000,
      system,
      tools: Object.values(tools).map((t) => t.definition),
      messages,
    });
    usage.inputTokens += res.usage.input_tokens;
    usage.outputTokens += res.usage.output_tokens;

    if (res.stop_reason === "refusal") return { status: "refused", text: "", steps: step, usage };
    if (res.stop_reason === "max_tokens") return { status: "truncated", text: textOf(res), steps: step, usage };
    messages.push({ role: "assistant", content: res.content }); // כל התוכן — כולל tool_use ו־thinking — בלי לשנות
    if (res.stop_reason !== "tool_use") return { status: "done", text: textOf(res), steps: step, usage };

    const calls = res.content.filter((b): b is Anthropic.ToolUseBlock => b.type === "tool_use");
    // כלים במקביל; כל תוצאה — באותו סדר, וכולן בהודעת user אחת
    const results = await Promise.all(
      calls.map(async (c): Promise<Anthropic.ToolResultBlockParam> => {
        const tool = tools[c.name];
        if (!tool) return { type: "tool_result", tool_use_id: c.id, content: `אין כלי בשם ${c.name}`, is_error: true };
        try {
          return { type: "tool_result", tool_use_id: c.id, content: await tool.run(c.input) };
        } catch (err) {
          // שגיאה של כלי היא מידע למודל (יכול לתקן ולנסות שוב), לא קריסה של הסוכן
          return { type: "tool_result", tool_use_id: c.id, content: `שגיאה: ${(err as Error).message}`, is_error: true };
        }
      }),
    );
    messages.push({ role: "user", content: results });
  }
  return { status: "max_steps", text: "", steps: maxSteps, usage };
}
```

```ts save=solutions/P29.5-E07.ts
import { z } from "zod";

export interface DefinedTool<T> {
  definition: Anthropic.Tool;
  execute: (input: unknown) => Promise<{ content: string; is_error?: true }>;
}

export function defineTool<T extends z.ZodObject<any>>({ name, description, schema, run }: { name: string; description: string; schema: T; run: (input: z.infer<T>) => Promise<string> }): DefinedTool<z.infer<T>> {
  if (!/^[a-zA-Z0-9_-]{1,64}$/.test(name)) throw new Error(`שם כלי לא תקין: ${name}`);
  const json = z.toJSONSchema(schema.strict()) as Record<string, unknown>; // strict ← additionalProperties: false
  delete json.$schema;
  return {
    definition: { name, description, input_schema: json as Anthropic.Tool.InputSchema, strict: true },
    async execute(input) {
      const parsed = schema.strict().safeParse(input);
      if (!parsed.success) {
        const msg = parsed.error.issues.map((i) => `${i.path.join(".") || "(root)"}: ${i.message}`).join("; ");
        return { content: `קלט לא תקין — ${msg}`, is_error: true };
      }
      try {
        return { content: await run(parsed.data) };
      } catch (err) {
        return { content: `שגיאה: ${(err as Error).message}`, is_error: true };
      }
    },
  };
}
```

```ts save=solutions/P29.5-E09.ts
import type Anthropic from "@anthropic-ai/sdk";
import type { Client } from "@modelcontextprotocol/sdk/client/index.js";

export async function mcpToolsForClaude(mcp: Client, { prefix = "" } = {}): Promise<Anthropic.Tool[]> {
  const out: Anthropic.Tool[] = [];
  let cursor: string | undefined;
  do {
    const page = await mcp.listTools(cursor ? { cursor } : {}); // שרת יכול להחזיר רשימה בעמודים

      const { $schema, ...schema } = t.inputSchema as Record<string, unknown>;
      out.push({ name: prefix + t.name, description: t.description ?? "", input_schema: schema as Anthropic.Tool.InputSchema });
    }
    cursor = page.nextCursor;
  } while (cursor);
  return out;
}

export async function callMcpTool(mcp: Client, call: Anthropic.ToolUseBlock, { prefix = "" } = {}): Promise<Anthropic.ToolResultBlockParam> {

    const res = await mcp.callTool({ name, arguments: call.input as Record<string, unknown> });
    const content = (res.content as { type: string; text?: string }[])

    // שרת שנפל / timeout — עדיין tool_result, כדי שהשיחה תישאר תקינה
```

```ts save=solutions/P29.5-E10.ts
  messages: Anthropic.MessageParam[];

  | { status: "needs_approval"; state: RunState; pending: { id: string; name: string; input: unknown }[] }
  | { status: "done"; state: RunState; text: string }
  | { status: "max_steps"; state: RunState };

export const startState = (prompt: string): RunState => ({ messages: [{ role: "user", content: prompt }], waiting: [], steps: 0 });

async function execute(cfg: ApprovalConfig, calls: RunState["waiting"], decisions: Record<string, "approve" | "deny">): Promise<Anthropic.ToolResultBlockParam[]> {
  return Promise.all(
    calls.map(async (c): Promise<Anthropic.ToolResultBlockParam> => {
      const tool = cfg.tools[c.name];
      if (!tool) return { type: "tool_result", tool_use_id: c.id, content: `אין כלי בשם ${c.name}`, is_error: true };
      if (cfg.needsApproval(c.name, c.input) && decisions[c.id] !== "approve") {
        return { type: "tool_result", tool_use_id: c.id, content: "המשתמש לא אישר את הפעולה. אל תנסה שוב — שאל אותו מה לעשות.", is_error: true };
      }
      try {
        return { type: "tool_result", tool_use_id: c.id, content: await tool.run(c.input) };
      } catch (err) {
        return { type: "tool_result", tool_use_id: c.id, content: `שגיאה: ${(err as Error).message}`, is_error: true };
      }
    }),
  );
}

export async function advance(client: Anthropic, cfg: ApprovalConfig, state: RunState, decisions: Record<string, "approve" | "deny"> = {}): Promise<AdvanceResult> {
  const s: RunState = structuredClone(state); // לא משנים את המצב שקיבלנו — הוא אולי שמור במסד
  const maxSteps = cfg.maxSteps ?? 10;
  if (s.waiting.length) {
    const needing = s.waiting.filter((c) => cfg.needsApproval(c.name, c.input));
    const missing = needing.filter((c) => !(c.id in decisions));
    if (missing.length) return { status: "needs_approval", state: s, pending: missing };
    s.messages.push({ role: "user", content: await execute(cfg, s.waiting, decisions) });
    s.waiting = [];
  }
  while (s.steps < maxSteps) {
    s.steps++;
    const res = await client.messages.create({
      model: cfg.model ?? "claude-opus-5-5",
      max_tokens: 16000,
      system: cfg.system,
      tools: Object.values(cfg.tools).map((t) => t.definition),
      messages: s.messages,
    });
    s.messages.push({ role: "assistant", content: res.content });
    if (res.stop_reason !== "tool_use") {
      return { status: "done", state: s, text: res.content.flatMap((b) => (b.type === "text" ? [b.text] : [])).join("") };
    }
    const calls = res.content.filter((b): b is Anthropic.ToolUseBlock => b.type === "tool_use").map((b) => ({ id: b.id, name: b.name, input: b.input }));
    const needing = calls.filter((c) => cfg.needsApproval(c.name, c.input));
    if (needing.length) {
      s.waiting = calls; // עוצרים לפני שמריצים משהו מהתור הזה — גם את הכלים שלא צריכים אישור
      return { status: "needs_approval", state: s, pending: needing };
    }
    s.messages.push({ role: "user", content: await execute(cfg, calls, {}) });
  }
  return { status: "max_steps", state: s };
}
```

```ts save=solutions/P29.5-E11.ts
// P29.5-E11: תוצאות כלי שלא מציפות את ה־context — קיצור, ועימוד
export function limitToolOutput(text: string, { maxChars = 4000 } = {}): string {
  if (text.length <= maxChars) return text;
  const head = Math.floor(maxChars * 0.7);
  const tail = maxChars - head;
  const omitted = text.length - head - tail;
  return `${text.slice(0, head)}\n…[הושמטו ${omitted} תווים מתוך ${text.length}. בקש טווח מצומצם יותר (offset/limit) כדי לראות את השאר]…\n${text.slice(text.length - tail)}`;
}

export function paginate<T>(items: T[], { offset = 0, limit = 20, maxLimit = 50 } = {}) {
  if (!Number.isInteger(offset) || offset < 0) throw new Error("offset לא תקין");
  if (!Number.isInteger(limit) || limit < 1) throw new Error("limit לא תקין");
  const n = Math.min(limit, maxLimit);
  const page = items.slice(offset, offset + n);
  const next = offset + page.length;
  return { items: page, total: items.length, nextOffset: next < items.length ? next : null };
}
```

```ts save=solutions/P29.5-E12.ts
// P29.5-E12: tracing לסוכן — כל קריאה למודל וכל כלי: זמן, טוקנים ועלות
import type Anthropic from "@anthropic-ai/sdk";

export interface Price {
  input: number; // $ למיליון טוקנים
  output: number;
  cacheRead: number;
}
export const PRICES: Record<string, Price> = {
  "claude-opus-5-5": { input: 4, output: 20, cacheRead: 0.2 },
  "claude-sonnet-5-5": { input: 2, output: 10, cacheRead: 0.2 },
  "claude-haiku-5-5": { input: 0.1, output: 0.5, cacheRead: 0.01 },
};

export interface ModelSpan {
  kind: "model";
  step: number;
  model: string;
  stopReason: string | null;
  inputTokens: number;
  outputTokens: number;
  cacheReadTokens: number;
  costUsd: number;
  ms: number;
}
export interface ToolSpan {
  kind: "tool";
  name: string;
  ms: number;
  isError: boolean;
}

export function createTracer(prices: Record<string, Price> = PRICES) {
  const spans: (ModelSpan | ToolSpan)[] = [];
  let step = 0;
  const round = (x: number) => Math.round(x * 1e6) / 1e6;
  return {
    spans,
    // עוטף את client.messages.create בלי לשנות את הקוד שקורא לו
    wrapClient(client: Anthropic): Anthropic {
      const create = client.messages.create.bind(client.messages);
      const messages = Object.create(client.messages);
      messages.create = async (params: any, opts?: any) => {
        const t0 = performance.now();
        const res = (await create(params, opts)) as Anthropic.Message;
        const p = prices[params.model];
        if (!p) throw new Error(`אין מחיר למודל ${params.model}`);
        const u = res.usage;
        const cacheRead = u.cache_read_input_tokens ?? 0;
        spans.push({
          kind: "model",
          step: ++step,
          model: params.model,
          stopReason: res.stop_reason,
          inputTokens: u.input_tokens,
          outputTokens: u.output_tokens,
          cacheReadTokens: cacheRead,
          costUsd: round((u.input_tokens * p.input + u.output_tokens * p.output + cacheRead * p.cacheRead) / 1e6),
          ms: Math.round(performance.now() - t0),
        });
        return res;
      };
      return new Proxy(client, { get: (t, k) => (k === "messages" ? messages : Reflect.get(t, k)) });
    },
    wrapTool<I>(name: string, run: (input: I) => Promise<string>) {
      return async (input: I) => {
        const t0 = performance.now();

          const out = await run(input);
```

```ts save=solutions/P29.5-E13.ts
// P29.5-E13: eval על המסלול — לא רק "מה ענה", אלא "איך הגיע לשם"
export interface Trajectory {
  toolCalls: { name: string; input: unknown }[];
  finalText: string;
  steps: number;
}

export interface Expectations {
  mustCall?: string[];
  mustNotCall?: string[];
  order?: [string, string][]; // [a, b]: הקריאה הראשונה ל־a לפני הקריאה הראשונה ל־b
  maxSteps?: number;
  maxToolCalls?: number;
  finalIncludes?: string[];
  finalExcludes?: string[];
}

export function evaluateTrajectory(t: Trajectory, e: Expectations): { pass: boolean; failures: string[] } {
  const failures: string[] = [];
  const names = t.toolCalls.map((c) => c.name);
  for (const n of e.mustCall ?? []) if (!names.includes(n)) failures.push(`לא קרא ל־${n}`);
  for (const n of e.mustNotCall ?? []) if (names.includes(n)) failures.push(`קרא ל־${n}, שאסור`);
  for (const [a, b] of e.order ?? []) {
    const ia = names.indexOf(a);
    const ib = names.indexOf(b);
    if (ib !== -1 && (ia === -1 || ia > ib)) failures.push(`קרא ל־${b} לפני ${a}`);
  }
  if (e.maxSteps !== undefined && t.steps > e.maxSteps) failures.push(`${t.steps} צעדים (מותר ${e.maxSteps})`);
  if (e.maxToolCalls !== undefined && names.length > e.maxToolCalls) failures.push(`${names.length} קריאות לכלים (מותר ${e.maxToolCalls})`);
  const text = t.finalText.toLowerCase();
  for (const s of e.finalIncludes ?? []) if (!text.includes(s.toLowerCase())) failures.push(`התשובה לא כוללת "${s}"`);
  for (const s of e.finalExcludes ?? []) if (text.includes(s.toLowerCase())) failures.push(`התשובה כוללת "${s}"`);
  return { pass: failures.length === 0, failures };
}
```

```ts save=solutions/P29.5-E16.ts
// P29.5-E16: guardrails — בדיקות בקוד, לפני כלי ואחרי תשובה
import type { Store } from "../files/store.ts";

const luhn = (digits: string) => {
  let sum = 0;
  for (let i = 0; i < digits.length; i++) {
    let d = Number(digits[digits.length - 1 - i]);
    if (i % 2 === 1) d = d * 2 > 9 ? d * 2 - 9 : d * 2;
    sum += d;
  }
  return sum % 10 === 0;
};
const israeliId = (id: string) => {
  const s = id.padStart(9, "0");
  let sum = 0;
  for (let i = 0; i < 9; i++) {
    const d = Number(s[i]) * ((i % 2) + 1);
    sum += d > 9 ? d - 9 : d;
  }
  return sum % 10 === 0;
};

export function redactPII(text: string): string {
  return text
    .replace(/\b(?:\d[ -]?){13,19}\b/g, (m) => (luhn(m.replace(/\D/g, "")) ? "[כרטיס]" : m))
    .replace(/(?<!\d)\d{9}(?!\d)/g, (m) => (israeliId(m) ? "[ת״ז]" : m))
    .replace(/(?<!\d)0(?:5\d|[2-489])-?\d{7}(?!\d)/g, "[טלפון]")
    .replace(/[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.[A-Za-z]{2,}/g, "[אימייל]");
}

export function checkRefund(input: { orderId: number; amountCents: number }, ctx: { store: Store; customerId: number; maxRefundCents: number }): { ok: true } | { ok: false; reason: string } {
  const o = ctx.store.orders.get(input.orderId);
  if (!o || o.customerId !== ctx.customerId) return { ok: false, reason: "ההזמנה לא נמצאה" };
  if (!Number.isInteger(input.amountCents) || input.amountCents <= 0) return { ok: false, reason: "סכום לא תקין" };
  if (o.status !== "paid" && o.status !== "shipped") return { ok: false, reason: `אי אפשר להחזיר הזמנה בסטטוס ${o.status}` };
  const already = ctx.store.refunds.filter((r) => r.orderId === o.id).reduce((s, r) => s + r.amountCents, 0);
  if (already + input.amountCents > o.totalCents) return { ok: false, reason: "הסכום גדול ממה שנשאר להחזיר" };
  if (input.amountCents > ctx.maxRefundCents) return { ok: false, reason: "מעל הסכום שסוכן רשאי להחזיר — דרוש נציג" };
  return { ok: true };
```

```ts save=solutions/P29.5-E18.ts
// P29.5-E18: הגבלת מקביליות לכל כלי — המודל ביקש 8 שאילתות במקביל; המסד מחזיק 2
export function createLimiter(limits: Record<string, number>, defaultLimit = 4) {
  const active = new Map<string, number>();
  const queues = new Map<string, (() => void)[]>();
  const limitOf = (name: string) => limits[name] ?? defaultLimit;

  async function acquire(name: string) {
    if ((active.get(name) ?? 0) < limitOf(name)) {
      active.set(name, (active.get(name) ?? 0) + 1);
      return;
    }
    await new Promise<void>((resolve) => {
      const q = queues.get(name) ?? [];
      q.push(resolve);
      queues.set(name, q);
    });
  }
  function release(name: string) {
    const next = queues.get(name)?.shift();
    if (next) next(); // המקום עובר ישר לממתין הבא — active לא משתנה
    else active.set(name, (active.get(name) ?? 1) - 1);
  }
  return {
    async run<T>(name: string, fn: () => Promise<T>): Promise<T> {
      await acquire(name);
      try {
        return await fn();
      } finally {
        release(name);
      }
    },
    active: (name: string) => active.get(name) ?? 0,
  };
}
```

```ts save=solutions/P29.5-E19.ts
import type Anthropic from "@anthropic-ai/sdk";

export type AgentEvent =
  | { type: "model"; step: number; stopReason: string | null }
  | { type: "tool_start"; id: string; name: string }
  | { type: "tool_end"; id: string; name: string; isError: boolean }
  | { type: "done"; text: string }
  | { type: "error"; message: string };

export async function* agentEvents(
  client: Anthropic,
  { system, tools, prompt, maxSteps = 10 }: { system: string; tools: Record<string, { definition: Anthropic.Tool; run: (input: any) => Promise<string> }>; prompt: string; maxSteps?: number },
): AsyncGenerator<AgentEvent> {
  const messages: Anthropic.MessageParam[] = [{ role: "user", content: prompt }];
  try {
    for (let step = 1; step <= maxSteps; step++) {
      const res = await client.messages.create({ model: "claude-opus-5-5", max_tokens: 16000, system, tools: Object.values(tools).map((t) => t.definition), messages });
      yield { type: "model", step, stopReason: res.stop_reason };
      messages.push({ role: "assistant", content: res.content });
      if (res.stop_reason !== "tool_use") {
        yield { type: "done", text: res.content.flatMap((b) => (b.type === "text" ? [b.text] : [])).join("") };
        return;
      }
      const results: Anthropic.ToolResultBlockParam[] = [];
      for (const c of res.content.filter((b): b is Anthropic.ToolUseBlock => b.type === "tool_use")) {
        yield { type: "tool_start", id: c.id, name: c.name };
        let r: Anthropic.ToolResultBlockParam;
        try {
          const t = tools[c.name];
          if (!t) throw new Error(`אין כלי בשם ${c.name}`);
          r = { type: "tool_result", tool_use_id: c.id, content: await t.run(c.input) };
        } catch (err) {
          r = { type: "tool_result", tool_use_id: c.id, content: `שגיאה: ${(err as Error).message}`, is_error: true };
        }
        results.push(r);
        yield { type: "tool_end", id: c.id, name: c.name, isError: !!r.is_error };
      }
      messages.push({ role: "user", content: results });
    }
    yield { type: "error", message: "עבר את תקרת הצעדים" };
  } catch (err) {
    yield { type: "error", message: (err as Error).message };
  }
}
```

```ts save=solutions/P29.5-E24.ts
const luhnOk = (d: string) => {
  let s = 0;
  for (let i = 0; i < d.length; i++) {
    let x = Number(d[d.length - 1 - i]);
    if (i % 2) x = x * 2 > 9 ? x * 2 - 9 : x * 2;
    s += x;
  }
  return s % 10 === 0;
};

  t
    .replace(/\b(?:\d[ -]?){13,19}\b/g, (m) => (luhnOk(m.replace(/\D/g, "")) ? "[כרטיס]" : m))
    .replace(/(?<!\d)0(?:5\d|[2-489])-?\d{7}(?!\d)/g, "[טלפון]")

Amount in agorot.", input_schema: { type: "object", properties: { orderId: { type: "integer" }, amountCents: { type: "integer" } }, required: ["orderId", "amountCents"], additionalProperties: false } },
      approval: true,
      run: async ({ orderId, amountCents }) => {

        const o = own(orderId);

      },

    for (let step = 0; step < 10; step++) {
      const res = await client.messages.create({ model: "claude-opus-5-5", max_tokens: 16000, system: SYSTEM, tools: Object.values(tools).map((t) => t.definition), messages: s.messages });
      s.messages.push({ role: "assistant", content: res.content });
      if (res.stop_reason !== "tool_use") return { status: "done", state: s, text: redact(res.content.flatMap((b) => (b.type === "text" ? [b.text] : [])).join("")) };
      const calls = res.content.filter((b): b is Anthropic.ToolUseBlock => b.type === "tool_use").map((b) => ({ id: b.id, name: b.name, input: b.input }));
      const pending = calls.filter((c) => tools[c.name]?.approval);
      if (pending.length) {
        s.waiting = calls;
        return { status: "needs_approval", state: s, pending };
      }
      s.messages.push({ role: "user", content: await runTools(calls, {}) });
    }
    throw new Error("יותר מדי צעדים");
  }

  return {
    start: (message: string) => loop({ messages: [{ role: "user", content: message }], waiting: [] }),
    async resume(state: State, decisions: Record<string, "approve" | "deny">) {
      const s = structuredClone(state);
      s.messages.push({ role: "user", content: await runTools(s.waiting, decisions) });
      s.waiting = [];
      return loop(s);
    },
  };
}
```

```ts save=solutions/P29.5-E25.ts
// P29.5-E25: חבילת evals לסוכן — כמה תרחישים, כמה ניסיונות לכל אחד, ודוח
import type Anthropic from "@anthropic-ai/sdk";
import { evaluateTrajectory, type Expectations, type Trajectory } from "./P29.5-E13.ts";

export interface Scenario {
  name: string;
  prompt: string;
  expect: Expectations;
}

export interface RunOutput extends Trajectory {
  costUsd: number;
}

export async function runEvalSuite({
  scenarios,
  makeClient,
  agent,
  trials = 3,
  concurrency = 4,
}: {
  scenarios: Scenario[];
  makeClient: (scenario: Scenario, trial: number) => Anthropic;
  agent: (client: Anthropic, prompt: string) => Promise<RunOutput>;
  trials?: number;
  concurrency?: number;
}) {
  const jobs = scenarios.flatMap((s) => Array.from({ length: trials }, (_, t) => ({ s, t })));
  const outcomes = new Map<string, { pass: boolean; failures: string[]; costUsd: number }[]>();
  let next = 0;
  async function worker() {
    while (next < jobs.length) {
      const { s, t } = jobs[next++];
      let r: { pass: boolean; failures: string[]; costUsd: number };
      try {
        const out = await agent(makeClient(s, t), s.prompt);
        r = { ...evaluateTrajectory(out, s.expect), costUsd: out.costUsd };
      } catch (err) {
        r = { pass: false, failures: [`הריצה נכשלה: ${(err as Error).message}`], costUsd: 0 }; // קריסה היא כישלון — לא מדלגים
      }
      outcomes.set(s.name, [...(outcomes.get(s.name) ?? []), r]);
    }
  }
  await Promise.all(Array.from({ length: Math.min(concurrency, jobs.length) }, worker));

  const results = scenarios.map((s) => {
    const rs = outcomes.get(s.name)!;
    const passes = rs.filter((r) => r.pass).length;
    return {
      name: s.name,
      trials: rs.length,
      passes,
      passRate: passes / rs.length,
      failures: [...new Set(rs.flatMap((r) => r.failures))],
      avgCostUsd: Math.round((rs.reduce((x, r) => x + r.costUsd, 0) / rs.length) * 1e6) / 1e6,
    };
  });
  const totalPasses = results.reduce((x, r) => x + r.passes, 0);
  return {
    results,
    passRate: totalPasses / jobs.length,
    flaky: results.filter((r) => r.passRate > 0 && r.passRate < 1).map((r) => r.name), // עובר לפעמים — הכי מסוכן
    failing: results.filter((r) => r.passRate === 0).map((r) => r.name),
  };
}
```
