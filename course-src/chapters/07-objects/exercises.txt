```js run save=solutions/P07-E12.js
// P07-E12: אובייקט ⇄ query string
function toQueryString(params) {
  return Object.entries(params)
    .filter(([, value]) => value !== null && value !== undefined)
    .map(([key, value]) => `${key}=${value}`)
    .join("&");
}

function parseQueryString(text) {
  const clean = text.startsWith("?") ? text.slice(1) : text;
  const result = {};
  if (clean === "") {
    return result;
  }
  for (const pair of clean.split("&")) {
    const [key, value = ""] = pair.split("=");
    result[key] = value;
  }
  return result;
}

console.log(toQueryString({ page: 2, sort: "price", q: null }));
console.log(parseQueryString("?page=2&sort=price"));

module.exports = { toQueryString, parseQueryString };
```
```out
page=2&sort=price
{ page: '2', sort: 'price' }
```
<p>בעולם האמיתי יש <code>URLSearchParams</code> מובנה, שגם מטפל בתווים מיוחדים ובעברית (קידוד <code>%D7%A9...</code>). נשתמש בו בפרק 14. וכאן רואים שוב: פרמטרים מה־URL הם <strong>תמיד מחרוזות</strong>.</p>

```js run save=solutions/P07-E18.js
// P07-E18: תדירות מילים עם Map
function words(text) {
  let clean = text.toLowerCase();
  for (const mark of [".", ",", "!", "?"]) {
    clean = clean.replaceAll(mark, " ");
  }
  return clean.split(" ").filter((w) => w !== "");
}

function wordFrequencies(text) {
  const counts = new Map();
  for (const w of words(text)) {
    counts.set(w, (counts.get(w) ?? 0) + 1);
  }
  return counts;
}

function topWords(text, n) {
  return [...wordFrequencies(text)]
    .sort((a, b) => b[1] - a[1] || a[0].localeCompare(b[0]))
    .slice(0, n);
}
```

```js run save=solutions/P07-E24.js
// P07-E24: מלאי והזמנות — הכול או כלום
function applyOrder(inventory, order) {
  const next = { ...inventory };
  for (const { code, qty } of order.items) {
    if (!(code in next)) {
      return { ok: false, inventory, error: `מוצר ${code} לא קיים` };
    }
    if (next[code] < qty) {
      return { ok: false, inventory, error: `אין מספיק מלאי של ${code} (נשארו ${next[code]})` };
    }
    next[code] -= qty;
  }
  return { ok: true, inventory: next };
}

function processOrders(inventory, orders) {
  let current = inventory;
  const accepted = [];
  const rejected = [];
  for (const order of orders) {
    const result = applyOrder(current, order);
    if (result.ok) {
      current = result.inventory;
      accepted.push(order.id);
    } else {
      rejected.push({ id: order.id, error: result.error });
    }
  }
  return { inventory: current, accepted, rejected };
}

const inventory = { A: 5, B: 1 };
const orders = [
  { id: 1, items: [{ code: "A", qty: 2 }, { code: "B", qty: 1 }] },
  { id: 2, items: [{ code: "A", qty: 2 }, { code: "B", qty: 1 }] },
  { id: 3, items: [{ code: "A", qty: 2 }, { code: "A", qty: 2 }] },
  { id: 4, items: [{ code: "Z", qty: 1 }] },
];
console.log(processOrders(inventory, orders), inventory);

module.exports = { applyOrder, processOrders };
```

```js run save=solutions/P07-E25.js
// P07-E25: twoSum ב־O(n) ואנגרמות
function twoSumFast(nums, target) {
  const seen = new Map();
  for (let j = 0; j < nums.length; j++) {
    const need = target - nums[j];
    if (seen.has(need)) {
      return [seen.get(need), j];
    }
    if (!seen.has(nums[j])) {
      seen.set(nums[j], j);
    }
  }
  return null;
}

function groupAnagrams(words) {
  const groups = new Map();
  for (const w of words) {
    const key = w.split("").sort().join("");
    if (!groups.has(key)) {
      groups.set(key, []);
    }
    groups.get(key).push(w);
  }
  return [...groups.values()];
}

console.log(twoSumFast([2, 7, 11, 15], 9), twoSumFast([3, 3], 6));
console.log(groupAnagrams(["eat", "tea", "tan", "ate", "nat", "bat"]));

module.exports = { twoSumFast, groupAnagrams };
```

```js run save=solutions/P07-E28.js
// P07-E28: יתרה מאירועים
const FEE = 5;

function balanceOf(events) {
  return events.reduce((balance, e) => {
    switch (e.type) {
      case "deposit":
        return balance + e.amount;
      case "withdraw":
        return e.amount <= balance ? balance - e.amount : balance;
      case "fee":
        return balance - FEE;
      default:
        return balance;
    }
  }, 0);
}

    { type: "withdraw", amount: 500 },
    { type: "fee" },

);
```

```js run file=07-objects/starter/P07-E44.js save=solutions/P07-E44.js
function parseSettings(text) {
  const result = {};
  for (const part of text.split(";")) {
    const eq = part.indexOf("=");
    if (eq === -1) {
      continue;
    }
    const key = part.slice(0, eq).trim();
    const value = part.slice(eq + 1).trim();
    if (key === "") {
      continue;
    }
    result[key] = value;
  }
  return result;
```

```js run file=07-objects/starter/P07-E56.js save=solutions/P07-E56.js
function productErrors(p) {
  const errors = [];
  if ((p.name ?? "").trim() === "") {
    errors.push("חסר שם");
  }
  if (typeof p.price !== "number" || Number.isNaN(p.price) || p.price <= 0) {
    errors.push("מחיר לא תקין");
  }
  if (!Number.isInteger(p.stock) || p.stock < 0) {
    errors.push("מלאי לא תקין");
  }
  return errors;
}

function validateAll(products) {

    }
  }
  return messages;
}
```

```js run file=07-objects/starter/P07-E67.js save=solutions/P07-E67.js
function toCamel(key) {
  return key
    .split("_")
    .map((part, i) => (i === 0 ? part : part[0].toUpperCase() + part.slice(1)))
    .join("");
}

  const result = {};

  return result;
}
```

```js run file=07-objects/starter/P07-E77.js save=solutions/P07-E77.js
function ordersByCustomer(orders) {
  const map = new Map();
  for (const o of orders) {
    if (!map.has(o.customerId)) {
      map.set(o.customerId, []);
    }
    map.get(o.customerId).push(o.id);
  }
  return map;
}

function customersWithoutOrders(customers, orders) {
  const withOrders = new Set(orders.map((o) => o.customerId));
  return customers.filter((c) => !withOrders.has(c.id)).map((c) => c.name);
}

function orphanOrders(customers, orders) {
  const ids = new Set(customers.map((c) => c.id));
  return orders.filter((o) => !ids.has(o.customerId)).map((o) => o.id);
}

const customers = [{ id: 1, name: "דנה" }, { id: 2, name: "עומר" }, { id: 3, name: "נועה" }];
const orders = [{ id: 10, customerId: 2 }, { id: 11, customerId: 9 }, { id: 12, customerId: 2 }];
console.log(ordersByCustomer(orders), customersWithoutOrders(customers, orders), orphanOrders(customers, orders));

module.exports = { ordersByCustomer, customersWithoutOrders, orphanOrders };
```
